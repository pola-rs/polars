41macro_rules! to_temporal_unit {
42    ($name: ident, $chrono_method: ident, $to_datetime_fn: expr,
43    $primitive_in: ty,
44    $primitive_out: ty,
45    $dtype_out:expr) => {
46        pub(crate) fn $name(arr: &PrimitiveArray<$primitive_in>) -> ArrayRef {
47            Box::new(PrimitiveArray::<$primitive_out>::from_trusted_len_iter(
48                arr.iter().map(|opt_value| {
49                    opt_value.and_then(|&value| {
50                        $to_datetime_fn(value).map(|dt| dt.$chrono_method() as $primitive_out)
51                    })
52                }),
53            )) as ArrayRef
54        }
55    };
56}

146#[cfg(any(feature = "dtype-date", feature = "dtype-datetime"))]
147/// The days since the Unix epoch of the first and the last date that chrono supports. The date
148/// parts of days outside of this range are null.
149fn supported_days() -> (i64, i64) {
150    let first = NaiveDate::MIN.num_days_from_ce() - EPOCH_DAYS_FROM_CE;
151    let last = NaiveDate::MAX.num_days_from_ce() - EPOCH_DAYS_FROM_CE;
152    (first as i64, last as i64)
153}

155#[cfg(any(feature = "dtype-date", feature = "dtype-datetime"))]
156/// The year, the month (1-12) and the day of the month (1-31) in the proleptic Gregorian calendar
157/// of `days` since the Unix epoch, for days in [`supported_days`].
158#[inline(always)]
159fn civil_from_days(days: i32) -> (i32, u32, u32) {
160    // Count from 0000-03-01, so that the leap day is the last day of a year.
161    let days = days + 719_468;
162    let era = days.div_euclid(146_097);
163    let day_of_era = (days - era * 146_097) as u32;
164    let year_of_era =
165        (day_of_era - day_of_era / 1_460 + day_of_era / 36_524 - day_of_era / 146_096) / 365;
166    let day_of_year = day_of_era - (365 * year_of_era + year_of_era / 4 - year_of_era / 100);
167    let month_from_march = (5 * day_of_year + 2) / 153;
168    let day = day_of_year - (153 * month_from_march + 2) / 5 + 1;
169    let month = if month_from_march < 10 {
170        month_from_march + 3
171    } else {
172        month_from_march - 9
173    };
174    let year = year_of_era as i32 + era * 400 + (month <= 2) as i32;
175    (year, month, day)
176}

178#[cfg(any(feature = "dtype-date", feature = "dtype-datetime"))]
179#[inline(always)]
180fn ordinal_from_days(days: i32) -> i16 {
181    let (year, month, day) = civil_from_days(days);
182    let february_short = 2 - is_leap_year(year) as u32;
183    (275 * month / 9 - february_short * ((month + 9) / 12) + day - 30) as i16
184}
185
186#[cfg(any(feature = "dtype-date", feature = "dtype-datetime"))]
187#[inline(always)]
188fn days_in_month_from_days(days: i32) -> i8 {
189    let (year, month, _) = civil_from_days(days);
190    if month == 2 {
191        28 + is_leap_year(year) as i8
192    } else {
193        30 + ((month ^ (month >> 3)) & 1) as i8
194    }
195}

197#[cfg(any(feature = "dtype-date", feature = "dtype-datetime"))]
198/// A temporal array as days since the Unix epoch: `to_days` maps its values to days and `bounds`
199/// are the values whose days are in [`supported_days`].
200struct DaysArray<'a, T: NativeType, F: Fn(T) -> i32> {
201    arr: &'a PrimitiveArray<T>,
202    to_days: F,
203    bounds: (T, T),
204}
205
206#[cfg(any(feature = "dtype-date", feature = "dtype-datetime"))]
207impl<T: NativeType + PartialOrd, F: Fn(T) -> i32> DaysArray<'_, T, F> {
208    fn all_supported(&self) -> bool {
209        let (first, last) = self.bounds;
210        self.arr
211            .values()
212            .iter()
213            .fold(true, |acc, &v| acc & (first <= v) & (v <= last))
214    }
215
216    fn map<O: NativeType>(&self, f: impl Fn(i32) -> O) -> ArrayRef {
217        let out = if self.all_supported() {
218            unary(self.arr, |v| f((self.to_days)(v)), O::PRIMITIVE.into())
219        } else {
220            let (first, last) = self.bounds;
221            PrimitiveArray::from_trusted_len_iter(self.arr.iter().map(|v| {
222                let v = *v?;
223                (first <= v && v <= last).then(|| f((self.to_days)(v)))
224            }))
225        };
226        Box::new(out)
227    }
228
229    fn map_bool(&self, f: impl Fn(i32) -> bool) -> ArrayRef {
230        let out = if self.all_supported() {
231            let values = self.arr.values();
232            let mut bits = BitmapBuilder::with_capacity(values.len());
233            // Fill a buffer first, so that `f` is vectorized.
234            let mut buf = [false; 64];
235            for chunk in values.chunks(64) {
236                for (b, &v) in buf.iter_mut().zip(chunk) {
237                    *b = f((self.to_days)(v));
238                }
239                let word = buf[..chunk.len()]
240                    .iter()
241                    .enumerate()
242                    .fold(0u64, |word, (i, &b)| word | ((b as u64) << i));
243                // SAFETY: the capacity is the number of values and a chunk has at most 64.
244                unsafe { bits.push_word_with_len_unchecked(word, chunk.len()) };
245            }
246            BooleanArray::new(
247                ArrowDataType::Boolean,
248                bits.freeze(),
249                self.arr.validity().cloned(),
250            )
251        } else {
252            let (first, last) = self.bounds;
253            BooleanArray::from_trusted_len_iter(self.arr.iter().map(|v| {
254                let v = *v?;
255                (first <= v && v <= last).then(|| f((self.to_days)(v)))
256            }))
257        };
258        Box::new(out)
259    }
260}

262#[cfg(feature = "dtype-date")]
263fn date_days(arr: &PrimitiveArray<i32>) -> DaysArray<'_, i32, impl Fn(i32) -> i32> {
264    let (first, last) = supported_days();
265    DaysArray {
266        arr,
267        to_days: |v| v,
268        bounds: (first as i32, last as i32),
269    }
270}

272#[cfg(feature = "dtype-datetime")]
273fn datetime_days(
274    arr: &PrimitiveArray<i64>,
275    time_unit: TimeUnit,
276) -> DaysArray<'_, i64, impl Fn(i64) -> i32> {
277    let per_day = match time_unit {
278        TimeUnit::Nanoseconds => NS_DAY,
279        TimeUnit::Microseconds => NS_DAY / 1_000,
280        TimeUnit::Milliseconds => NS_DAY / 1_000_000,
281    };
282    let (first, last) = supported_days();
283    let bounds = (
284        first.checked_mul(per_day).unwrap_or(i64::MIN),
285        (last + 1).checked_mul(per_day).map_or(i64::MAX, |v| v - 1),
286    );
287    DaysArray {
288        arr,
289        to_days: move |v: i64| v.div_euclid(per_day) as i32,
290        bounds,
291    }
292}

294macro_rules! date_parts {
295    ($date_fn:ident, $datetime_fn:ident, $method:ident, $f:expr) => {
296        #[cfg(feature = "dtype-date")]
297        pub(crate) fn $date_fn(arr: &PrimitiveArray<i32>) -> ArrayRef {
298            date_days(arr).$method($f)
299        }
300
301        #[cfg(feature = "dtype-datetime")]
302        pub(crate) fn $datetime_fn(arr: &PrimitiveArray<i64>, time_unit: TimeUnit) -> ArrayRef {
303            datetime_days(arr, time_unit).$method($f)
304        }
305    };
306}

332#[cfg(test)]
333#[cfg(any(feature = "dtype-date", feature = "dtype-datetime"))]
334mod test {
335    use polars_arrow::temporal_conversions::date32_to_date_opt;
336
337    use super::*;
338
339    #[test]
340    fn test_civil_from_days_matches_chrono() {
341        let (first, last) = supported_days();
342        let near_epoch = -800_000..=800_000;
343        let whole_range = (first..=last).step_by(997).chain([first, last]);
344        for days in near_epoch.chain(whole_range) {
345            let days = days as i32;
346            let date = date32_to_date_opt(days).unwrap();
347            assert_eq!(
348                civil_from_days(days),
349                (date.year(), date.month(), date.day()),
350                "{days}"
351            );
352            assert_eq!(ordinal_from_days(days), date.ordinal() as i16, "{days}");
353            assert_eq!(
354                days_in_month_from_days(days),
355                polars_utils::time::days_in_month(date.year(), date.month() as u8) as i8,
356                "{days}"
357            );
358            assert_eq!(is_leap_year(date.year()), date.leap_year(), "{days}");
359        }
360        assert!(date32_to_date_opt(first as i32 - 1).is_none());
361        assert!(date32_to_date_opt(last as i32 + 1).is_none());
362    }
