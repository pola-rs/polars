23pub(super) fn right_join_from_series(
24    left: &DataFrame,
25    right: DataFrame,
26    s_left: &Series,
27    s_right: &Series,
28    mut args: JoinArgs,
29    verbose: bool,
30    drop_names: Option<Vec<PlSmallStr>>,
31) -> PolarsResult<DataFrame> {
32    // Swap the order of tables to do a right join.
33    args.maintain_order = args.maintain_order.flip();
34    let (df_right, df_left) = materialize_left_join_from_series(
35        right, left, s_right, s_left, &args, verbose, drop_names,
36    )?;
37    _finish_join(df_left, df_right, args.suffix)
38}

40pub fn materialize_left_join_from_series(
41    mut left: DataFrame,
42    right_: &DataFrame,
43    s_left: &Series,
44    s_right: &Series,
45    args: &JoinArgs,
46    verbose: bool,
47    drop_names: Option<Vec<PlSmallStr>>,
48) -> PolarsResult<(DataFrame, DataFrame)> {
49    let mut s_left = s_left.clone();
50    // Eagerly limit left if possible.
51    if let Some(Slice::Positive { offset: 0, len }) = args.slice {
52        if !matches!(
53            args.maintain_order,
54            MaintainOrderJoin::Right | MaintainOrderJoin::RightLeft
55        ) {
56            left = left.slice(Slice::head(len));
57            s_left = s_left.slice(Slice::head(len));
58        }
59    }
60
61    // The join ids refer to the chunks of the keys, so the frames must have the same chunk
62    // boundaries as their keys, otherwise we go OOB.
63    let requires_ordering = matches!(
64        args.maintain_order,
65        MaintainOrderJoin::Right | MaintainOrderJoin::RightLeft
66    );
67    let chunks_match_key = |df: &DataFrame, s: &Series| {
68        // Dictionary columns are ignored by the frame's chunk-alignment checks, but their
69        // code chunks must also match the key before gathering by chunk-local ids (the codes are
70        // what gets gathered, so comparing them avoids materializing the values).
71        df.columns()
72            .iter()
73            .filter(|c| c.as_scalar_column().is_none())
74            .all(|c| match c.as_dictionary_column() {
75                Some(d) => d.codes().chunk_lengths().eq(s.chunk_lengths()),
76                None => c
77                    .as_materialized_series()
78                    .chunk_lengths()
79                    .eq(s.chunk_lengths()),
80            })
81    };
82
83    let mut right = Cow::Borrowed(right_);
84    let mut s_right = s_right.clone();
85    if requires_ordering || !chunks_match_key(&left, &s_left) {
86        left.rechunk_mut_par();
87        s_left = s_left.rechunk();
88    }
89    if requires_ordering || !chunks_match_key(right_, &s_right) {
90        let mut other = right_.clone();
91        other.rechunk_mut_par();
92        right = Cow::Owned(other);
93        s_right = s_right.rechunk();
94    }
95
96    let (left_idx, right_idx) = sort_or_hash_left(
97        &s_left,
98        &s_right,
99        verbose,
100        args.validation,
101        args.nulls_equal,
102    )?;
103
104    let right = if let Some(drop_names) = drop_names {
105        right.drop_many(drop_names)
106    } else {
107        right.drop(s_right.name()).unwrap()
108    };
109    try_raise_polars_abort();
110
111    #[cfg(feature = "chunked_ids")]
112    match (left_idx, right_idx) {
113        (ChunkJoinIds::Left(left_idx), ChunkJoinOptIds::Left(right_idx)) => {
114            if requires_ordering {
115                Ok(maintain_order_idx(
116                    &left,
117                    &right,
118                    left_idx.as_slice(),
119                    right_idx.as_slice(),
120                    args,
121                ))
122            } else {
123                Ok(RAYON.join(
124                    || materialize_left_join_idx_left(&left, left_idx.as_slice(), args),
125                    || materialize_left_join_idx_right(&right, right_idx.as_slice(), args),
126                ))
127            }
128        },
129        (ChunkJoinIds::Left(left_idx), ChunkJoinOptIds::Right(right_idx)) => Ok(RAYON.join(
130            || materialize_left_join_idx_left(&left, left_idx.as_slice(), args),
131            || materialize_left_join_chunked_right(&right, right_idx.as_slice(), args),
132        )),
133        (ChunkJoinIds::Right(left_idx), ChunkJoinOptIds::Right(right_idx)) => Ok(RAYON.join(
134            || materialize_left_join_chunked_left(&left, left_idx.as_slice(), args),
135            || materialize_left_join_chunked_right(&right, right_idx.as_slice(), args),
136        )),
137        (ChunkJoinIds::Right(left_idx), ChunkJoinOptIds::Left(right_idx)) => Ok(RAYON.join(
138            || materialize_left_join_chunked_left(&left, left_idx.as_slice(), args),
139            || materialize_left_join_idx_right(&right, right_idx.as_slice(), args),
140        )),
141    }
142
143    #[cfg(not(feature = "chunked_ids"))]
144    if requires_ordering {
145        Ok(maintain_order_idx(
146            &left,
147            &right,
148            left_idx.as_slice(),
149            right_idx.as_slice(),
150            args,
151        ))
152    } else {
153        Ok(RAYON.join(
154            || materialize_left_join_idx_left(&left, left_idx.as_slice(), args),
155            || materialize_left_join_idx_right(&right, right_idx.as_slice(), args),
156        ))
157    }
158}

160fn maintain_order_idx(
161    left: &DataFrame,
162    other: &DataFrame,
163    left_idx: &[IdxSize],
164    right_idx: &[NullableIdxSize],
165    args: &JoinArgs,
166) -> (DataFrame, DataFrame) {
167    let mut df = {
168        // SAFETY: left_idx and right_idx are contiguous memory that outlive the memory mapped slices
169        let left = unsafe { IdxCa::mmap_slice("a".into(), left_idx) };
170        let right = unsafe { IdxCa::mmap_slice("b".into(), bytemuck::cast_slice(right_idx)) };
171        unsafe {
172            DataFrame::new_unchecked(
173                left_idx.len(),
174                vec![left.into_series().into(), right.into_series().into()],
175            )
176        }
177    };
178
179    let options = SortMultipleOptions::new()
180        .with_order_descending(false)
181        .with_maintain_order(true);
182
183    let columns = match args.maintain_order {
184        // If the left order is preserved then there are no unsorted right rows
185        // So Left and LeftRight are equal
186        MaintainOrderJoin::Left | MaintainOrderJoin::LeftRight => vec!["a"],
187        MaintainOrderJoin::Right => vec!["b"],
188        MaintainOrderJoin::RightLeft => vec!["b", "a"],
189        _ => unreachable!(),
190    };
191
192    df.sort_in_place(columns, options).unwrap();
193    df.rechunk_mut();
194
195    let join_tuples_left = df
196        .column("a")
197        .unwrap()
198        .as_materialized_series()
199        .idx()
200        .unwrap()
201        .cont_slice()
202        .unwrap();
203
204    let join_tuples_right = df
205        .column("b")
206        .unwrap()
207        .as_materialized_series()
208        .idx()
209        .unwrap()
210        .cont_slice()
211        .unwrap();
212
213    RAYON.join(
214        || materialize_left_join_idx_left(left, join_tuples_left, args),
215        || materialize_left_join_idx_right(other, bytemuck::cast_slice(join_tuples_right), args),
216    )
217}

219fn materialize_left_join_idx_left(
220    left: &DataFrame,
221    left_idx: &[IdxSize],
222    args: &JoinArgs,
223) -> DataFrame {
224    let left_idx = if let Some(slice) = args.slice {
225        &left_idx[slice]
226    } else {
227        left_idx
228    };
229
230    unsafe {
231        left._create_left_df_from_slice(
232            left_idx,
233            true,
234            args.slice.is_some(),
235            matches!(
236                args.maintain_order,
237                MaintainOrderJoin::Left | MaintainOrderJoin::LeftRight
238            ) || args.how == JoinType::Left
239                && !matches!(
240                    args.maintain_order,
241                    MaintainOrderJoin::Right | MaintainOrderJoin::RightLeft,
242                ),
243        )
244    }
245}

247fn materialize_left_join_idx_right(
248    right: &DataFrame,
249    right_idx: &[NullableIdxSize],
250    args: &JoinArgs,
251) -> DataFrame {
252    let right_idx = if let Some(slice) = args.slice {
253        &right_idx[slice]
254    } else {
255        right_idx
256    };
257    unsafe { IdxCa::with_nullable_idx(right_idx, |idx| right.take_unchecked(idx)) }
258}
