6pub(super) fn left_join_from_series(

21pub(super) fn right_join_from_series(
22    left: &DataFrame,
23    right: DataFrame,
24    s_left: &Series,
25    s_right: &Series,
26    mut args: JoinArgs,
27    verbose: bool,
28    drop_names: Option<Vec<PlSmallStr>>,
29) -> PolarsResult<DataFrame> {
30    // Swap the order of tables to do a right join.
31    args.maintain_order = args.maintain_order.flip();
32    let (df_right, df_left) = materialize_left_join_from_series(
33        right, left, s_right, s_left, &args, verbose, drop_names,
34    )?;
35    _finish_join(df_left, df_right, args.suffix)
36}

38pub fn materialize_left_join_from_series(
39    mut left: DataFrame,
40    right_: &DataFrame,
41    s_left: &Series,
42    s_right: &Series,
43    args: &JoinArgs,
44    verbose: bool,
45    drop_names: Option<Vec<PlSmallStr>>,
46) -> PolarsResult<(DataFrame, DataFrame)> {
47    let mut s_left = s_left.clone();
48    // Eagerly limit left if possible.
49    if let Some((0, len)) = args.slice {
50        if !matches!(
51            args.maintain_order,
52            MaintainOrderJoin::Right | MaintainOrderJoin::RightLeft
53        ) {
54            left = left.slice(0, len);
55            s_left = s_left.slice(0, len);
56        }
57    }
58
59    // Ensure that the chunks are aligned otherwise we go OOB.
60    let requires_ordering = matches!(
61        args.maintain_order,
62        MaintainOrderJoin::Right | MaintainOrderJoin::RightLeft
63    );
64
65    let mut right = Cow::Borrowed(right_);
66    let mut s_right = s_right.clone();
67    if left.should_rechunk() || requires_ordering || left.n_chunks() != s_left.n_chunks() {
68        left.rechunk_mut_par();
69        s_left = s_left.rechunk();
70    }
71    if right.should_rechunk() || requires_ordering || right.n_chunks() != s_right.n_chunks() {
72        let mut other = right_.clone();
73        other.rechunk_mut_par();
74        right = Cow::Owned(other);
75        s_right = s_right.rechunk();
76    }
77
78    let (left_idx, right_idx) = sort_or_hash_left(
79        &s_left,
80        &s_right,
81        verbose,
82        args.validation,
83        args.nulls_equal,
84    )?;
85
86    let right = if let Some(drop_names) = drop_names {
87        right.drop_many(drop_names)
88    } else {
89        right.drop(s_right.name()).unwrap()
90    };
91    try_raise_polars_abort();
92
93    #[cfg(feature = "chunked_ids")]
94    match (left_idx, right_idx) {
95        (ChunkJoinIds::Left(left_idx), ChunkJoinOptIds::Left(right_idx)) => {
96            if requires_ordering {
97                Ok(maintain_order_idx(
98                    &left,
99                    &right,
100                    left_idx.as_slice(),
101                    right_idx.as_slice(),
102                    args,
103                ))
104            } else {
105                Ok(RAYON.join(
106                    || materialize_left_join_idx_left(&left, left_idx.as_slice(), args),
107                    || materialize_left_join_idx_right(&right, right_idx.as_slice(), args),
108                ))
109            }
110        },
111        (ChunkJoinIds::Left(left_idx), ChunkJoinOptIds::Right(right_idx)) => Ok(RAYON.join(
112            || materialize_left_join_idx_left(&left, left_idx.as_slice(), args),
113            || materialize_left_join_chunked_right(&right, right_idx.as_slice(), args),
114        )),
115        (ChunkJoinIds::Right(left_idx), ChunkJoinOptIds::Right(right_idx)) => Ok(RAYON.join(
116            || materialize_left_join_chunked_left(&left, left_idx.as_slice(), args),
117            || materialize_left_join_chunked_right(&right, right_idx.as_slice(), args),
118        )),
119        (ChunkJoinIds::Right(left_idx), ChunkJoinOptIds::Left(right_idx)) => Ok(RAYON.join(
120            || materialize_left_join_chunked_left(&left, left_idx.as_slice(), args),
121            || materialize_left_join_idx_right(&right, right_idx.as_slice(), args),
122        )),
123    }
124
125    #[cfg(not(feature = "chunked_ids"))]
126    if requires_ordering {
127        Ok(maintain_order_idx(
128            &left,
129            &right,
130            left_idx.as_slice(),
131            right_idx.as_slice(),
132            args,
133        ))
134    } else {
135        Ok(RAYON.join(
136            || materialize_left_join_idx_left(&left, left_idx.as_slice(), args),
137            || materialize_left_join_idx_right(&right, right_idx.as_slice(), args),
138        ))
139    }
140}

142fn maintain_order_idx(
143    left: &DataFrame,
144    other: &DataFrame,
145    left_idx: &[IdxSize],
146    right_idx: &[NullableIdxSize],
147    args: &JoinArgs,
148) -> (DataFrame, DataFrame) {
149    let mut df = {
150        // SAFETY: left_idx and right_idx are continuous memory that outlive the memory mapped slices
151        let left = unsafe { IdxCa::mmap_slice("a".into(), left_idx) };
152        let right = unsafe { IdxCa::mmap_slice("b".into(), bytemuck::cast_slice(right_idx)) };
153        unsafe {
154            DataFrame::new_unchecked(
155                left_idx.len(),
156                vec![left.into_series().into(), right.into_series().into()],
157            )
158        }
159    };
160
161    let options = SortMultipleOptions::new()
162        .with_order_descending(false)
163        .with_maintain_order(true);
164
165    let columns = match args.maintain_order {
166        // If the left order is preserved then there are no unsorted right rows
167        // So Left and LeftRight are equal
168        MaintainOrderJoin::Left | MaintainOrderJoin::LeftRight => vec!["a"],
169        MaintainOrderJoin::Right => vec!["b"],
170        MaintainOrderJoin::RightLeft => vec!["b", "a"],
171        _ => unreachable!(),
172    };
173
174    df.sort_in_place(columns, options).unwrap();
175    df.rechunk_mut();
176
177    let join_tuples_left = df
178        .column("a")
179        .unwrap()
180        .as_materialized_series()
181        .idx()
182        .unwrap()
183        .cont_slice()
184        .unwrap();
185
186    let join_tuples_right = df
187        .column("b")
188        .unwrap()
189        .as_materialized_series()
190        .idx()
191        .unwrap()
192        .cont_slice()
193        .unwrap();
194
195    RAYON.join(
196        || materialize_left_join_idx_left(left, join_tuples_left, args),
197        || materialize_left_join_idx_right(other, bytemuck::cast_slice(join_tuples_right), args),
198    )
199}

201fn materialize_left_join_idx_left(
202    left: &DataFrame,
203    left_idx: &[IdxSize],
204    args: &JoinArgs,
205) -> DataFrame {
206    let left_idx = if let Some((offset, len)) = args.slice {
207        slice_slice(left_idx, offset, len)
208    } else {
209        left_idx
210    };
211
212    unsafe {
213        left._create_left_df_from_slice(
214            left_idx,
215            true,
216            args.slice.is_some(),
217            matches!(
218                args.maintain_order,
219                MaintainOrderJoin::Left | MaintainOrderJoin::LeftRight
220            ) || args.how == JoinType::Left
221                && !matches!(
222                    args.maintain_order,
223                    MaintainOrderJoin::Right | MaintainOrderJoin::RightLeft,
224                ),
225        )
226    }
227}

229fn materialize_left_join_idx_right(
230    right: &DataFrame,
231    right_idx: &[NullableIdxSize],
232    args: &JoinArgs,
233) -> DataFrame {
234    let right_idx = if let Some((offset, len)) = args.slice {
235        slice_slice(right_idx, offset, len)
236    } else {
237        right_idx
238    };
239    unsafe { IdxCa::with_nullable_idx(right_idx, |idx| right.take_unchecked(idx)) }
240}
