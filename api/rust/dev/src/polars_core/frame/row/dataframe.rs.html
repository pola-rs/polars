4    /// Get a row from a [`DataFrame`]. Use of this is discouraged as it will likely be slow.
5    pub fn get_row(&self, idx: usize) -> PolarsResult<Row<'_>> {
6        let values = self
7            .materialized_column_iter()
8            .map(|s| s.get(idx))
9            .collect::<PolarsResult<Vec<_>>>()?;
10        Ok(Row(values))
11    }

13    /// Amortize allocations by reusing a row.
14    /// The caller is responsible to make sure that the row has at least the capacity for the number
15    /// of columns in the [`DataFrame`]
16    pub fn get_row_amortized<'a>(&'a self, idx: usize, row: &mut Row<'a>) -> PolarsResult<()> {
17        for (s, any_val) in self.materialized_column_iter().zip(&mut row.0) {
18            *any_val = s.get(idx)?;
19        }
20        Ok(())
21    }

26    ///
27    /// # Safety
28    /// Does not do any bounds checking.

46    /// Create a new [`DataFrame`] from an iterator over rows.
47    ///
48    /// This should only be used when you have row wise data, as this is a lot slower
49    /// than creating the [`Series`] in a columnar fashion.
50    pub fn from_rows_iter_and_schema<'a, I>(mut rows: I, schema: &Schema) -> PolarsResult<Self>
51    where
52        I: Iterator<Item = &'a Row<'a>>,
53    {
54        let capacity = rows.size_hint().0;
55
56        let mut buffers: Vec<_> = schema
57            .iter_values()
58            .map(|dtype| {
59                let buf: AnyValueBuffer = (dtype, capacity).into();
60                buf
61            })
62            .collect();
63
64        let mut expected_len = 0;
65        let mut width = None;
66        rows.try_for_each::<_, PolarsResult<()>>(|row| {
67            check_row_width(row, &mut width, buffers.len(), expected_len)?;
68            expected_len += 1;
69            for (value, buf) in row.0.iter().zip(&mut buffers) {
70                buf.add_fallible(value)?
71            }
72            Ok(())
73        })?;
74
75        let v = buffers
76            .into_iter()
77            .zip(schema.iter_names())
78            .map(|(b, name)| {
79                let mut c = b.into_series()?.into_column();
80                // if the schema adds a column not in the rows, we
81                // fill it with nulls
82                if c.is_empty() {
83                    Ok(Column::full_null(name.clone(), expected_len, c.dtype()))
84                } else {
85                    c.rename(name.clone());
86                    Ok(c)
87                }
88            })
89            .collect::<PolarsResult<Vec<_>>>()?;
90
91        DataFrame::new(expected_len, v)
92    }

140    /// Create a new [`DataFrame`] from rows. This should only be used when you have row-wise
141    /// data, as this is a lot slower than creating the [`Series`] in a columnar fashion
142    pub fn from_rows(rows: &[Row]) -> PolarsResult<Self> {
143        let schema = rows_to_schema_first_non_null(rows, Some(50))?;
144        let has_nulls = schema
145            .iter_values()
146            .any(|dtype| matches!(dtype, DataType::Null));
147        polars_ensure!(
148            !has_nulls, ComputeError: "unable to infer row types because of null values"
149        );
150        Self::from_rows_and_schema(rows, &schema)
151    }

154/// Every row must have the same width as the first, which can be narrower than
155/// the schema (the missing columns are filled with nulls) but not wider, as
156/// the extra values would otherwise be silently dropped.
157fn check_row_width(
158    row: &Row,
159    width: &mut Option<usize>,
160    n_columns: usize,
161    idx: usize,
162) -> PolarsResult<()> {
163    let expected = *width.get_or_insert(row.0.len().min(n_columns));
164    polars_ensure!(
165        row.0.len() == expected,
166        ShapeMismatch: "row at index {} has length {} (expected {})", idx, row.0.len(), expected
167    );
168    Ok(())
169}
