82    /// How integer values in this unit are rescaled to `to`.
83    pub const fn rescale_to(self, to: TimeUnit) -> TimeUnitRescale {
84        use TimeUnit::*;
85        match (self, to) {
86            (Nanoseconds, Microseconds) | (Microseconds, Milliseconds) => {
87                TimeUnitRescale::Divide(1_000)
88            },
89            (Nanoseconds, Milliseconds) => TimeUnitRescale::Divide(1_000_000),
90            (Microseconds, Nanoseconds) | (Milliseconds, Microseconds) => {
91                TimeUnitRescale::Multiply(1_000)
92            },
93            (Milliseconds, Nanoseconds) => TimeUnitRescale::Multiply(1_000_000),
94            (Nanoseconds, Nanoseconds)
95            | (Microseconds, Microseconds)
96            | (Milliseconds, Milliseconds) => TimeUnitRescale::Unchanged,
97        }
98    }
99}

101/// How integer values in one [`TimeUnit`] are rescaled to another.
102#[derive(Copy, Clone, Debug, PartialEq, Eq)]
103pub enum TimeUnitRescale {
104    Unchanged,
105    Multiply(i64),
106    Divide(i64),
107}
108
109impl TimeUnitRescale {
110    /// Rescales a datetime timestamp, rounding down.
111    #[inline]
112    pub fn datetime_value(self, v: i64) -> i64 {
113        use polars_utils::floor_divmod::FloorDivMod;
114        match self {
115            TimeUnitRescale::Unchanged => v,
116            TimeUnitRescale::Multiply(k) => v.wrapping_mul(k),
117            TimeUnitRescale::Divide(k) => v.wrapping_floor_div_mod(k).0,
118        }
119    }
120
121    /// Rescales a duration, rounding towards zero.
122    #[inline]
123    pub fn duration_value(self, v: i64) -> i64 {
124        match self {
125            TimeUnitRescale::Unchanged => v,
126            TimeUnitRescale::Multiply(k) => v.wrapping_mul(k),
127            TimeUnitRescale::Divide(k) => v.wrapping_div(k),
128        }
129    }
130}

157#[inline]
158pub fn convert_time_units<T>(v: T, tu_l: TimeUnit, tu_r: TimeUnit) -> T
159where
160    T: num_traits::Num + num_traits::NumCast,
161{
162    use TimeUnit::*;
163    match (tu_l, tu_r) {
164        (Nanoseconds, Microseconds) => v / T::from(1_000).unwrap(),
165        (Nanoseconds, Milliseconds) => v / T::from(1_000_000).unwrap(),
166        (Microseconds, Nanoseconds) => v * T::from(1_000).unwrap(),
167        (Microseconds, Milliseconds) => v / T::from(1_000).unwrap(),
168        (Milliseconds, Microseconds) => v * T::from(1_000).unwrap(),
169        (Milliseconds, Nanoseconds) => v * T::from(1_000_000).unwrap(),
170        _ => v,
171    }
