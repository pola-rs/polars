26impl From<&ArrowTimeUnit> for TimeUnit {
27    fn from(tu: &ArrowTimeUnit) -> Self {
28        match tu {
29            ArrowTimeUnit::Nanosecond => TimeUnit::Nanoseconds,
30            ArrowTimeUnit::Microsecond => TimeUnit::Microseconds,
31            ArrowTimeUnit::Millisecond => TimeUnit::Milliseconds,
32            // will be cast
33            ArrowTimeUnit::Second => TimeUnit::Milliseconds,
34        }
35    }
36}

72    /// `ns` nanoseconds expressed in this unit, truncated towards zero.
73    #[inline(always)]
74    pub const fn from_nsecs(self, ns: i64) -> i64 {
75        match self {
76            TimeUnit::Nanoseconds => ns,
77            TimeUnit::Microseconds => ns / 1_000,

108#[inline]
109pub fn convert_time_units<T>(v: T, tu_l: TimeUnit, tu_r: TimeUnit) -> T
110where
111    T: num_traits::Num + num_traits::NumCast,
112{
113    use TimeUnit::*;
114    match (tu_l, tu_r) {
115        (Nanoseconds, Microseconds) => v / T::from(1_000).unwrap(),
116        (Nanoseconds, Milliseconds) => v / T::from(1_000_000).unwrap(),
117        (Microseconds, Nanoseconds) => v * T::from(1_000).unwrap(),
118        (Microseconds, Milliseconds) => v / T::from(1_000).unwrap(),
119        (Milliseconds, Microseconds) => v * T::from(1_000).unwrap(),
120        (Milliseconds, Nanoseconds) => v * T::from(1_000_000).unwrap(),
121        _ => v,
122    }
