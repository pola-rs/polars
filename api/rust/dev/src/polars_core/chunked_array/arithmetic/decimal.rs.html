7use crate::prelude::arity::broadcast_try_binary_elementwise;

9impl DecimalChunked {
10    /// Applies `kernel(l, left_scale, r, right_scale, scale)` elementwise, producing a
11    /// `Decimal(38, scale)`. The kernel returns `None` if the result doesn't fit, or on
12    /// division by zero.
13    fn apply_scaled_kernel(
14        &self,
15        rhs: &Self,
16        scale: usize,
17        op: &str,
18        kernel: impl Fn(i128, usize, i128, usize, usize) -> Option<i128>,
19    ) -> PolarsResult<Self> {
20        let left_s = self.scale();
21        let right_s = rhs.scale();
22        let phys = broadcast_try_binary_elementwise(
23            self.physical(),
24            rhs.physical(),
25            |opt_l, opt_r| {
26                let (Some(l), Some(r)) = (opt_l, opt_r) else {
27                    return PolarsResult::Ok(None);
28                };
29                let ret = kernel(l, left_s, r, right_s, scale).ok_or_else(|| {
30                    // Only division and remainder can fail with a zero operand.
31                    if r == 0 {
32                        polars_err!(ComputeError: "division by zero Decimal")
33                    } else {
34                        polars_err!(
35                            ComputeError: "overflow in decimal {op}: result doesn't fit Decimal({DEC128_MAX_PREC}, {scale})"
36                        )
37                    }
38                })?;
39                Ok(Some(ret))
40            },
41        )?;
42        Ok(phys.into_decimal_unchecked(DEC128_MAX_PREC, scale))
43    }
44
45    /// Multiplies with the result rounded to `scale`.
46    pub fn mul_with_scale(&self, rhs: &Self, scale: usize) -> PolarsResult<Self> {
47        self.apply_scaled_kernel(rhs, scale, "multiplication", dec128_mul_scaled)
48    }
49
50    /// Divides with the result rounded to `scale`.
51    pub fn div_with_scale(&self, rhs: &Self, scale: usize) -> PolarsResult<Self> {
52        self.apply_scaled_kernel(rhs, scale, "division", dec128_div_scaled)
53    }
54
55    /// The exact remainder, with the sign of `rhs` if `floor` (as `%`), else of `self`.
56    pub fn rem_with(&self, rhs: &Self, floor: bool) -> PolarsResult<Self> {
57        let scale = self.scale().max(rhs.scale());
58        self.apply_scaled_kernel(rhs, scale, "remainder", |l, sl, r, sr, s| {
59            dec128_rem_scaled(l, sl, r, sr, s, floor)
60        })
61    }
62
63    /// The integer quotient, rounded down if `floor` (as `//`), else toward zero.
64    pub fn int_div(&self, rhs: &Self, floor: bool) -> PolarsResult<Self> {
65        self.int_div_with_scale(rhs, self.scale().max(rhs.scale()), floor)
66    }
67
68    /// [`Self::int_div`] with the quotient at `scale`.
69    pub fn int_div_with_scale(&self, rhs: &Self, scale: usize, floor: bool) -> PolarsResult<Self> {
70        self.apply_scaled_kernel(rhs, scale, "integer division", |l, sl, r, sr, s| {
71            dec128_int_div_scaled(l, sl, r, sr, s, floor)
72        })
73    }
74}
