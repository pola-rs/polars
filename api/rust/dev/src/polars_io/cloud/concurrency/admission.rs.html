9/// Order in which parked acquires of the in-flight byte budget are admitted.
10#[derive(Debug, Clone, Copy, PartialEq)]
11pub enum AdmissionOrder {
12    /// Lowest issue number first (see `in_issue_order`); acquires without one go first.
13    Issue,
14    /// Whichever waiter wins the race after a release; new acquires may pass parked waiters.
15    Arrival,
16}
17
18/// One issue number per fetch, in the order fetches are issued.
19static NEXT_ISSUE_SEQ: AtomicU64 = AtomicU64::new(1);
20
21tokio::task_local! {
22    static ISSUE_SEQ: u64;
23}
24
25/// Takes the next issue number now and uses it for the byte budget acquires of `fut`. Call it
26/// where fetches are issued in order, not inside their tasks.
27pub fn in_issue_order<F: Future>(fut: F) -> impl Future<Output = F::Output> {
28    ISSUE_SEQ.scope(NEXT_ISSUE_SEQ.fetch_add(1, Relaxed), fut)
29}
30
31/// Issue number of the current task, 0 if none.
32fn current_issue_seq() -> u64 {
33    ISSUE_SEQ.try_with(|seq| *seq).unwrap_or(0)
34}

37struct Waiter {
38    n_bytes: u64,
39    granted: oneshot::Sender<()>,
40}

42/// Parked acquires in admission order, keyed by (issue number, park number).
43#[derive(Debug, Default)]
44struct IssueQueue {
45    /// Next park number. It orders acquires with the same issue number (the ranges of one fetch,
46    /// or acquires without one) by when they parked, and keeps the keys unique.
47    next_park: u64,
48    waiters: BTreeMap<(u64, u64), Waiter>,
49}
50
51/// Byte-granularity budget with dynamic resize.
52///
53/// Uses atomic counters with a Notify (wake-up order) or a queue of grants (issue order) rather
54/// than tokio::Semaphore because we need instant shrink semantics (tokio::Semaphore requires
55/// "stealing" permits by acquiring them, which blocks under saturation).
56#[derive(Debug)]
57pub(super) struct ByteBudget {
58    // Current budget, measured in number of bytes, reflecting the maximum
59    // allowed in-flight volume.
60    current_budget: AtomicU64,
61    // Lowest allowed budget for the current_budget.
62    floor_budget: u64,
63    // Volume in use for in-flight traffic, as allowed by the current_budget.
64    inflight_in_use: AtomicU64,
65    // Gauge: acquires parked now.
66    acquires_parked: AtomicU64,
67    order: AdmissionOrder,
68    // `AdmissionOrder::Arrival` waiters.
69    waiters: Notify,
70    // `AdmissionOrder::Issue` waiters, and their count (read without the lock).
71    queue: Mutex<IssueQueue>,
72    queued: AtomicU64,
73}
74
75impl ByteBudget {
76    fn new(initial: u64, floor_budget: u64, order: AdmissionOrder) -> Self {
77        Self {
78            current_budget: AtomicU64::new(initial),
79            floor_budget,
80            inflight_in_use: AtomicU64::new(0),
81            acquires_parked: AtomicU64::new(0),
82            order,
83            waiters: Notify::new(),
84            queue: Mutex::default(),
85            queued: AtomicU64::new(0),
86        }
87    }
88
89    /// Acquire a bytes-based permit. The call site is responsible for capping the
90    /// request size to prevent deadlock. Returns whether the caller parked.
91    async fn acquire_strict(&self, n_bytes: u64, issue_seq: u64) -> bool {
92        // Pre-empt deadlock.
93        assert!(n_bytes <= self.floor_budget);
94
95        if self.order == AdmissionOrder::Issue {
96            return self.acquire_in_issue_order(n_bytes, issue_seq).await;
97        }
98
99        let mut parked = false;
100
101        // NOTE: Large waiters can starve under sustained small-request load.
102        // In practice, this may not be material issue.
103        loop {
104            let cap = self.current_budget.load(Ordering::Acquire);
105            let inflight = self.inflight_in_use.load(Ordering::Acquire);
106
107            if inflight + n_bytes <= cap {
108                if self
109                    .inflight_in_use
110                    .compare_exchange_weak(
111                        inflight,
112                        inflight + n_bytes,
113                        Ordering::AcqRel,
114                        Ordering::Relaxed,
115                    )
116                    .is_ok()
117                {
118                    // Fits. There may be leftover capacity for the next
119                    // waiter (e.g. one big release satisfying several small
120                    // acquires), so keep the wake chain alive — but only
121                    // because progress occurred.
122                    self.waiters.notify_one();
123                    return parked;
124                }
125                continue;
126            }
127
128            // Doesn't fit: register, re-check, park.
129            let notified = self.waiters.notified();
130            let cap = self.current_budget.load(Ordering::Acquire);
131            let inflight = self.inflight_in_use.load(Ordering::Acquire);
132            if inflight + n_bytes <= cap {
133                continue;
134            }
135            let _parked = Parked::new(&self.acquires_parked);
136            notified.await;
137            parked = true;
138        }
139    }
140
141    /// FIFO by issue number: parks while a lower issue number is parked, otherwise takes the bytes
142    /// if they fit. Equal issue numbers are admitted in the order they parked. Returns whether it
143    /// parked. A parked acquire always waits on the budget: the queue is only non-empty while its
144    /// front does not fit.
145    async fn acquire_in_issue_order(&self, n_bytes: u64, issue_seq: u64) -> bool {
146        if self.queued.load(SeqCst) == 0 && self.try_take(n_bytes) {
147            return false;
148        }
149
150        let (key, granted) = {
151            let mut queue = self.queue.lock().unwrap();
152            let lower_waits = queue
153                .waiters
154                .keys()
155                .next()
156                .is_some_and(|&(seq, _)| seq < issue_seq);
157            if !lower_waits && self.try_take(n_bytes) {
158                return false;
159            }
160            let key = (issue_seq, queue.next_park);
161            queue.next_park += 1;
162            let (tx, rx) = oneshot::channel();
163            queue.waiters.insert(
164                key,
165                Waiter {
166                    n_bytes,
167                    granted: tx,
168                },
169            );
170            // Count it before the grant pass below reads the bytes in use, so a release that saw
171            // `queued == 0` has already freed its bytes.
172            self.queued.fetch_add(1, SeqCst);
173            self.grant_locked(&mut queue);
174            if !queue.waiters.contains_key(&key) {
175                // Admitted by the grant pass above.
176                return false;
177            }
178            (key, rx)
179        };
180
181        let mut wait = QueuedAcquire {
182            budget: self,
183            key,
184            n_bytes,
185            granted: false,
186        };
187        let _parked = Parked::new(&self.acquires_parked);
188        // The sender is only dropped unsent when `wait` removes its own entry.
189        let _ = granted.await;
190        wait.granted = true;
191        true
192    }
193
194    /// Takes `n_bytes` if they fit.
195    fn try_take(&self, n_bytes: u64) -> bool {
196        let cap = self.current_budget.load(SeqCst);
197        let mut inflight = self.inflight_in_use.load(SeqCst);
198        while inflight + n_bytes <= cap {
199            match self.inflight_in_use.compare_exchange_weak(
200                inflight,
201                inflight + n_bytes,
202                SeqCst,
203                SeqCst,
204            ) {
205                Ok(_) => return true,
206                Err(current) => inflight = current,
207            }
208        }
209        false
210    }
211
212    /// Admits parked acquires from the front of the queue while they fit.
213    fn grant_locked(&self, queue: &mut IssueQueue) {
214        while let Some(front) = queue.waiters.first_entry() {
215            if !self.try_take(front.get().n_bytes) {
216                break;
217            }
218            self.queued.fetch_sub(1, SeqCst);
219            // A cancelled acquire removes its entry under the lock, so the receiver is alive.
220            let _ = front.remove().granted.send(());
221        }
222    }
223
224    fn grant_if_queued(&self) {
225        if self.queued.load(SeqCst) > 0 {
226            self.grant_locked(&mut self.queue.lock().unwrap());
227        }
228    }
229
230    fn release(&self, bytes: u64) {
231        self.inflight_in_use.fetch_sub(bytes, SeqCst);
232        match self.order {
233            AdmissionOrder::Issue => self.grant_if_queued(),
234            AdmissionOrder::Arrival => self.waiters.notify_one(),
235        }
236    }
237
238    fn resize(&self, new: u64) {
239        let new = new.max(self.floor_budget);
240        let old = self.current_budget.swap(new, SeqCst);
241        if new > old {
242            // Grow: maybe someone can now proceed.
243            match self.order {
244                AdmissionOrder::Issue => self.grant_if_queued(),
245                AdmissionOrder::Arrival => self.waiters.notify_waiters(),
246            }
247        }
248    }
249
250    fn current_budget(&self) -> u64 {
251        self.current_budget.load(Ordering::Relaxed)
252    }
253
254    fn floor_byte_budget(&self) -> u64 {
255        self.floor_budget
256    }
257
258    fn inflight_in_use(&self) -> u64 {
259        self.inflight_in_use.load(Ordering::Relaxed)
260    }
261}

263/// A parked issue-order acquire. If dropped before its grant, it leaves the queue; if dropped
264/// after a grant it did not see, it returns the bytes.
265struct QueuedAcquire<'a> {
266    budget: &'a ByteBudget,
267    key: (u64, u64),
268    n_bytes: u64,
269    granted: bool,
270}
271
272impl Drop for QueuedAcquire<'_> {
273    fn drop(&mut self) {
274        if self.granted {
275            return;
276        }
277        let mut queue = self.budget.queue.lock().unwrap();
278        if queue.waiters.remove(&self.key).is_some() {
279            self.budget.queued.fetch_sub(1, SeqCst);
280        } else {
281            self.budget.inflight_in_use.fetch_sub(self.n_bytes, SeqCst);
282        }
283        // It may have been the front of the queue.
284        self.budget.grant_locked(&mut queue);
285    }
286}

288/// Counts one parked acquire while alive (also when the acquire is cancelled).
289struct Parked<'a>(&'a AtomicU64);
290
291impl<'a> Parked<'a> {
292    fn new(n: &'a AtomicU64) -> Self {
293        n.fetch_add(1, Relaxed);
294        Self(n)
295    }
296}
297
298impl Drop for Parked<'_> {
299    fn drop(&mut self) {
300        self.0.fetch_sub(1, Relaxed);
301    }
302}

304#[derive(Debug)]
305pub(super) struct RequestBudget {
306    // Current budget, measured in count, reflecting the maximum allowed in-flight.
307    current_budget: AtomicU64,
308    // Lowest allowed budget for the current_budget.
309    floor_budget: u64,
310    // Count in use for in-flight traffic, as allowed by the current_budget.
311    inflight_in_use: AtomicU64,
312    waiters: Notify,
313}
314
315impl RequestBudget {
316    fn new(initial: u64, floor_budget: u64) -> Self {
317        Self {
318            current_budget: AtomicU64::new(initial),
319            floor_budget,
320            inflight_in_use: AtomicU64::new(0),
321            waiters: Notify::new(),
322        }
323    }
324
325    /// Acquire a count-based permit. The call site is responsible for capping the
326    /// request size to prevent deadlock.
327    async fn acquire_one(&self) {
328        let n = 1;
329
330        loop {
331            let cap = self.current_budget.load(Ordering::Acquire);
332            let inflight = self.inflight_in_use.load(Ordering::Acquire);
333
334            if inflight + n <= cap {
335                if self
336                    .inflight_in_use
337                    .compare_exchange_weak(
338                        inflight,
339                        inflight + n,
340                        Ordering::AcqRel,
341                        Ordering::Relaxed,
342                    )
343                    .is_ok()
344                {
345                    // Fits.
346                    self.waiters.notify_one();
347                    return;
348                }
349                continue;
350            }
351
352            // Doesn't fit: register, re-check, park.
353            let notified = self.waiters.notified();
354            let cap = self.current_budget.load(Ordering::Acquire);
355            let inflight = self.inflight_in_use.load(Ordering::Acquire);
356            if inflight + n <= cap {
357                continue;
358            }
359            notified.await;
360        }
361    }
362
363    fn release_one(&self) {
364        let n = 1;
365
366        self.inflight_in_use.fetch_sub(n, Ordering::AcqRel);
367        self.waiters.notify_one();
368    }
369
370    fn resize(&self, new: u64) {
371        let new = new.max(self.floor_budget);
372        let old = self.current_budget.swap(new, Ordering::AcqRel);
373        if new > old {
374            // Grow: maybe someone can now proceed.
375            self.waiters.notify_waiters();
376        }
377    }
378
379    fn inflight_in_use(&self) -> u64 {
380        self.inflight_in_use.load(Ordering::Relaxed)
381    }
382}

385pub struct InFlightStats {
386    pub bytes_budget: u64,
387    pub bytes_in_use: u64,
388    // May exceed 1.0 transiently after a budget shrink, while
389    // previously-admitted traffic drains. Expected, not a bug.
390    pub bytes_saturation: f64,
391    // Gauge: byte acquires parked at the time of `stats()`.
392    pub byte_acquires_parked: u64,
393    pub request_budget: u64,
394    pub requests_in_use: u64,
395    pub requests_saturation: f64,
396}

398#[derive(Debug)]
399pub struct InFlightBudget {
400    byte_budget: Arc<ByteBudget>,
401    request_budget: Arc<RequestBudget>,
402    // Flow: byte acquires admitted after parking, since the last `take_byte_acquires_unparked()`.
403    byte_acquires_unparked: AtomicU64,
404    // Best sustained delivered bandwidth in bytes/s, set by the controller.
405    delivered_bandwidth: AtomicU64,
406}
407
408impl InFlightBudget {
409    pub fn new(
410        initial_byte_budget: u64,
411        floor_byte_budget: u64,
412        initial_request_budget: u64,
413        floor_request_budget: u64,
414        admission_order: AdmissionOrder,
415    ) -> Self {
416        Self {
417            byte_budget: Arc::new(ByteBudget::new(
418                initial_byte_budget,
419                floor_byte_budget,
420                admission_order,
421            )),
422            request_budget: Arc::new(RequestBudget::new(
423                initial_request_budget,
424                floor_request_budget,
425            )),
426            byte_acquires_unparked: AtomicU64::new(0),
427            delivered_bandwidth: AtomicU64::new(0),
428        }
429    }
430
431    pub async fn acquire(self: &Arc<Self>, n_bytes: u64) -> InFlightPermit {
432        // NOTE: since chunk_size is a target, merge_ranges and split_ranges may overshoot
433        // the floor. Cap'ing prevents deadlock at the expense of memory management precision.
434        let n_bytes = n_bytes.min(self.byte_budget.floor_byte_budget());
435
436        // Byte budget (may wait). Cancel-safe internally.
437        if self
438            .byte_budget
439            .acquire_strict(n_bytes, current_issue_seq())
440            .await
441        {
442            self.byte_acquires_unparked.fetch_add(1, Relaxed);
443        }
444
445        // Guard immediately — synchronous, so there's no cancellation
446        // window between reservation and guard.
447        let bytes = BytesReservation {
448            budget: self.byte_budget.clone(),
449            n_bytes,
450        };
451
452        self.request_budget.acquire_one().await;
453
454        let request = RequestReservation {
455            budget: self.request_budget.clone(),
456        };
457
458        InFlightPermit {
459            _bytes_reservation: bytes,
460            _req_permit: request,
461        }
462    }
463
464    pub fn current_byte_budget(&self) -> u64 {
465        self.byte_budget.current_budget()
466    }
467
468    pub fn floor_byte_budget(&self) -> u64 {
469        self.byte_budget.floor_byte_budget()
470    }
471
472    pub fn resize_byte_budget(&self, new: u64) {
473        self.byte_budget.resize(new);
474    }
475
476    pub fn resize_request_budget(&self, new: u64) {
477        self.request_budget.resize(new);
478    }
479
480    /// Best sustained delivered bandwidth of the store in bytes/s; 0 until measured.
481    pub fn delivered_bandwidth(&self) -> u64 {
482        self.delivered_bandwidth.load(Relaxed)
483    }
484
485    pub(super) fn set_delivered_bandwidth(&self, bytes_per_sec: u64) {
486        self.delivered_bandwidth.store(bytes_per_sec, Relaxed);
487    }
488
489    /// Flow: byte acquires admitted after parking, since the last call. Read and reset.
490    pub fn take_byte_acquires_unparked(&self) -> u64 {
491        self.byte_acquires_unparked.swap(0, Relaxed)
492    }
493
494    pub fn stats(&self) -> InFlightStats {
495        let bytes_budget = self.byte_budget.current_budget();
496        let bytes_in_use = self.byte_budget.inflight_in_use();
497        let bytes_saturation = if bytes_budget > 0 {
498            bytes_in_use as f64 / bytes_budget as f64
499        } else {
500            0.0
501        };
502
503        let request_budget = self.request_budget.current_budget.load(Relaxed);
504        let requests_in_use = self.request_budget.inflight_in_use();
505        let requests_saturation = if request_budget > 0 {
506            requests_in_use as f64 / request_budget as f64
507        } else {
508            0.0
509        };
510
511        InFlightStats {
512            bytes_budget,
513            bytes_in_use,
514            bytes_saturation,
515            byte_acquires_parked: self.byte_budget.acquires_parked.load(Relaxed),
516            request_budget,
517            requests_in_use,
518            requests_saturation,
519        }
520    }
521}

549#[cfg(test)]
550mod tests {
551    use std::pin::Pin;
552    use std::task::{Context, Poll, Waker};
553
554    use super::*;
555
556    fn poll<F: Future>(f: &mut Pin<Box<F>>) -> Poll<F::Output> {
557        f.as_mut().poll(&mut Context::from_waker(Waker::noop()))
558    }
559
560    #[test]
561    fn issue_order_admission() {
562        let b = ByteBudget::new(10, 10, AdmissionOrder::Issue);
563        assert_eq!(
564            poll(&mut Box::pin(b.acquire_strict(10, 5))),
565            Poll::Ready(false)
566        );
567
568        let mut a3 = Box::pin(b.acquire_strict(4, 3));
569        let mut a2 = Box::pin(b.acquire_strict(4, 2));
570        let mut a1 = Box::pin(b.acquire_strict(4, 1));
571        assert!(poll(&mut a3).is_pending() && poll(&mut a2).is_pending());
572        assert!(poll(&mut a1).is_pending());
573        // A 0-byte acquire without an issue number does not wait.
574        assert_eq!(
575            poll(&mut Box::pin(b.acquire_strict(0, 0))),
576            Poll::Ready(false)
577        );
578        // A cancelled front leaves the queue without taking bytes.
579        drop(a1);
580        assert_eq!(b.queued.load(SeqCst), 2);
581
582        b.release(2);
583        // Fits, but a lower issue number waits: parks, counted.
584        let mut a9 = Box::pin(b.acquire_strict(2, 9));
585        assert!(poll(&mut a9).is_pending());
586        assert_eq!(b.acquires_parked.load(Relaxed), 3);
587        // No issue number: passes the queue when it fits.
588        assert_eq!(
589            poll(&mut Box::pin(b.acquire_strict(2, 0))),
590            Poll::Ready(false)
591        );
592
593        b.release(6);
594        assert_eq!(poll(&mut a2), Poll::Ready(true));
595        assert!(poll(&mut a3).is_pending() && poll(&mut a9).is_pending());
596
597        b.release(4);
598        assert_eq!(poll(&mut a3), Poll::Ready(true));
599        assert_eq!(poll(&mut a9), Poll::Ready(true));
600        assert_eq!(b.queued.load(SeqCst), 0);
601        assert_eq!(b.inflight_in_use(), 10);
602    }
