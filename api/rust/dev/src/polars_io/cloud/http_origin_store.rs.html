16/// [`ObjectStore`] for all URLs on one HTTP(S) origin, sharing one connection pool.
17/// Locations are the raw URL path and query; each op uses a per-URL [`HttpStore`].
18#[derive(Debug)]
19pub(crate) struct HttpOriginStore {
20    /// `scheme://authority`, no trailing slash.
21    base_url: String,
22    client_options: ClientOptions,
23    client: HttpClient,
24}
25
26impl HttpOriginStore {
27    pub(crate) fn new(base_url: &str, client_options: ClientOptions) -> Result<Self> {
28        let client = ReqwestConnector::default().connect(&client_options)?;
29
30        Ok(Self {
31            base_url: base_url.to_string(),
32            client_options,
33            client,
34        })
35    }
36
37    fn store_for(&self, location: &Path) -> Result<HttpStore> {
38        HttpBuilder::new()
39            .with_url(format!("{}/{}", self.base_url, location))
40            .with_client_options(self.client_options.clone())
41            .with_http_connector(SharedHttpConnector(self.client.clone()))
42            .build()
43    }
44
45    fn not_implemented(&self, operation: &str) -> Error {
46        Error::NotImplemented {
47            operation: operation.to_string(),
48            implementer: self.to_string(),
49        }
50    }
51}

53/// Sets the location on an error of a per-URL store, which only knows an empty one.
54fn with_location(err: Error, location: &Path) -> Error {
55    let wrap = |source| {
56        Box::new(LocationError {
57            location: location.to_string(),
58            source,
59        }) as _
60    };
61
62    match err {
63        Error::Generic { store, source } => Error::Generic {
64            store,
65            source: wrap(source),
66        },
67        Error::NotSupported { source } => Error::NotSupported {
68            source: wrap(source),
69        },
70        mut err => {
71            if let Error::NotFound { path, .. }
72            | Error::AlreadyExists { path, .. }
73            | Error::Precondition { path, .. }
74            | Error::NotModified { path, .. }
75            | Error::PermissionDenied { path, .. }
76            | Error::Unauthenticated { path, .. } = &mut err
77            {
78                *path = location.to_string();
79            }
80            err
81        },
82    }
83}

85#[derive(Debug)]
86struct LocationError {
87    location: String,
88    source: Box<dyn std::error::Error + Send + Sync>,
89}
90
91impl std::fmt::Display for LocationError {
92    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
93        write!(f, "{} (location: {})", self.source, self.location)
94    }
95}
96
97impl std::error::Error for LocationError {
98    fn source(&self) -> Option<&(dyn std::error::Error + 'static)> {
99        Some(self.source.as_ref())
100    }
101}

119#[async_trait]
120impl ObjectStore for HttpOriginStore {
121    async fn put_opts(
122        &self,
123        location: &Path,
124        payload: PutPayload,
125        opts: PutOptions,
126    ) -> Result<PutResult> {
127        self.store_for(location)?
128            .put_opts(&Path::default(), payload, opts)
129            .await
130            .map_err(|e| with_location(e, location))
131    }
132
133    async fn put_multipart_opts(
134        &self,
135        location: &Path,
136        opts: PutMultipartOptions,
137    ) -> Result<Box<dyn MultipartUpload>> {
138        self.store_for(location)?
139            .put_multipart_opts(&Path::default(), opts)
140            .await
141            .map_err(|e| with_location(e, location))
142    }
143
144    async fn get_opts(&self, location: &Path, options: GetOptions) -> Result<GetResult> {
145        let mut out = self
146            .store_for(location)?
147            .get_opts(&Path::default(), options)
148            .await
149            .map_err(|e| with_location(e, location))?;
150        out.meta.location = location.clone();
151        out.payload = match out.payload {
152            GetResultPayload::Stream(stream) => {
153                let location = location.clone();
154                let stream = stream.map_err(move |e| with_location(e, &location));
155                GetResultPayload::Stream(stream.boxed())
156            },
157            payload => payload,
158        };
159        Ok(out)
160    }
161
162    async fn get_ranges(&self, location: &Path, ranges: &[Range<u64>]) -> Result<Vec<Bytes>> {
163        self.store_for(location)?
164            .get_ranges(&Path::default(), ranges)
165            .await
166            .map_err(|e| with_location(e, location))
167    }
168
169    fn delete_stream(
170        &self,
171        _locations: BoxStream<'static, Result<Path>>,
172    ) -> BoxStream<'static, Result<Path>> {
173        futures::stream::once(std::future::ready(Err(self.not_implemented("delete")))).boxed()
174    }
175
176    fn list(&self, _prefix: Option<&Path>) -> BoxStream<'static, Result<ObjectMeta>> {
177        futures::stream::once(std::future::ready(Err(self.not_implemented("list")))).boxed()
178    }
179
180    async fn list_with_delimiter(&self, _prefix: Option<&Path>) -> Result<ListResult> {
181        Err(self.not_implemented("list_with_delimiter"))
182    }
183
184    async fn copy_opts(&self, _from: &Path, _to: &Path, _options: CopyOptions) -> Result<()> {
185        Err(self.not_implemented("copy"))
186    }
187}

189#[cfg(test)]
190mod tests {
191    use super::*;
192
193    #[test]
194    fn test_with_location() {
195        let location = Path::from("dir/a.parquet");
196
197        let err = Error::NotFound {
198            path: String::new(),
199            source: "404".into(),
200        };
201        let err = with_location(err, &location);
202        assert!(matches!(&err, Error::NotFound { path, .. } if path == "dir/a.parquet"));
203
204        let err = Error::Generic {
205            store: "HTTP",
206            source: "reset".into(),
207        };
208        let err = with_location(err, &location);
209        assert!(matches!(err, Error::Generic { .. }));
210        assert!(err.to_string().ends_with("reset (location: dir/a.parquet)"));
211    }
