62    /// Accepts a function that returns (credential, expiry time as seconds since UNIX_EPOCH)
63    ///
64    /// This functionality is unstable.
65    pub fn from_func(
66        // Internal notes
67        // * This function is exposed as the Rust API for `PlCredentialProvider`
68        func: impl Fn() -> Pin<
69            Box<dyn Future<Output = PolarsResult<(ObjectStoreCredential, u64)>> + Send + Sync>,
70        > + Send
71        + Sync
72        + 'static,
73    ) -> Self {
74        Self::Function(CredentialProviderFunction(Arc::new(func)))
75    }
76
77    /// Intended to be called with an internal `CredentialProviderBuilder` from
78    /// py-polars.
79    #[cfg(feature = "python")]
80    pub fn from_python_builder(func: pyo3::Py<pyo3::PyAny>) -> Self {
81        Self::Python(python_impl::PythonCredentialProvider::Builder(Arc::new(
82            PythonObject(func),
83        )))
84    }
85
86    #[cfg(feature = "python")]
87    pub fn from_auto_init_json(json: &str) -> PolarsResult<Self> {
88        serde_json::from_str(json).map(Self::AutoInit).map_err(
89            |e| polars_error::polars_err!(ComputeError: "invalid auto-init credential config: {e}"),
90        )
91    }
92
93    /// Python passes a `CredentialProviderBuilder`, this calls the builder to build the final
94    /// credential provider.
95    ///
96    /// This returns `Option` as the auto-initialization case is fallible and falls back to None.
97    pub(crate) fn try_into_initialized(
98        self,
99        clear_cached_credentials: bool,
100    ) -> PolarsResult<Option<Self>> {
101        match self {
102            Self::Function(_) => Ok(Some(self)),
103            #[cfg(feature = "python")]
104            Self::Python(v) => Ok(v
105                .try_into_initialized(clear_cached_credentials)?
106                .map(Self::Python)),
107            Self::AutoInit(config) => config.try_into_initialized(clear_cached_credentials),
108        }
109    }
110
111    pub fn stable_cache_key(&self) -> PolarsResult<Vec<u8>> {
112        match self {
113            Self::Function(CredentialProviderFunction(v)) => Ok((Arc::as_ptr(v) as *const ()
114                as usize)
115                .to_ne_bytes()
116                .to_vec()),
117            #[cfg(feature = "python")]
118            Self::Python(v) => v.stable_cache_key(),
119            Self::AutoInit(config) => {
120                // Hashed, as the config can contain a token.
121                let json = serde_json::to_vec(config).map_err(to_compute_err)?;
122                let mut out = b"auto_init:".to_vec();
123                out.extend_from_slice(blake3::hash(&json).as_bytes());
124                Ok(out)
125            },
126        }
127    }

130impl AutoInitConfig {
131    fn try_into_initialized(
132        self,
133        clear_cached_credentials: bool,
134    ) -> PolarsResult<Option<PlCredentialProvider>> {
135        // A token given in `storage_options` needs no Python.
136        #[cfg(feature = "gcp")]
137        if let Self::GcpToken { token } = &self {
138            let credential = Arc::new(object_store::gcp::GcpCredential {
139                bearer: token.to_string(),
140            });
141
142            return Ok(Some(PlCredentialProvider::from_func(move || {
143                let credential = credential.clone();
144                Box::pin(async move { Ok((ObjectStoreCredential::Gcp(credential), u64::MAX)) })
145            })));
146        }
147
148        #[cfg(feature = "python")]
149        {
150            Ok(
151                PythonCredentialProvider::build_auto_init(&self, clear_cached_credentials)?
152                    .map(PlCredentialProvider::Python),
153            )
154        }
155
156        // Without Python, fall back to the object_store's own environment-based credentials, as
157        // when the Python provider's dependencies are not installed. A requested profile cannot
158        // be loaded that way.
159        #[cfg(not(feature = "python"))]
160        {
161            let _ = clear_cached_credentials;
162
163            if let Self::Aws {
164                profile_name: Some(profile_name),
165                ..
166            } = &self
167            {
168                polars_bail!(
169                    ComputeError:
170                    "cannot load requested aws_profile '{profile_name}': requires Python"
171                )
172            }
173
174            Ok(None)
175        }
176    }
177}

578/// Avoids calling the credential provider function if we have not yet passed the expiry time.
579#[derive(Debug)]
580struct FetchedCredentialsCache<C>(tokio::sync::Mutex<(C, u64, bool)>);
581
582impl<C: Clone> FetchedCredentialsCache<C> {
583    fn new(init_creds: C) -> Self {
584        Self(tokio::sync::Mutex::new((init_creds, 0, true)))
585    }
586
587    async fn get_maybe_update(
588        &self,
589        // Taking an `impl Future` here allows us to potentially avoid a `Box::pin` allocation from
590        // a `Fn() -> Pin<Box<dyn Future>>` by having it wrapped in an `async { f() }` block. We
591        // will not poll that block if the credentials have not yet expired.
592        update_func: impl Future<Output = PolarsResult<(C, u64)>>,
593    ) -> PolarsResult<C> {
594        let verbose = config::verbose();
595
596        fn expiry_msg(last_fetched_expiry: u64, now: u64) -> String {
597            if last_fetched_expiry == u64::MAX {
598                "expiry = (never expires)".into()
599            } else {
600                format!(
601                    "expiry = {} (in {} seconds)",
602                    last_fetched_expiry,
603                    last_fetched_expiry.saturating_sub(now)
604                )
605            }
606        }
607
608        let mut inner = self.0.lock().await;
609        let (last_fetched_credentials, last_fetched_expiry, log_use_cached) = &mut *inner;
610
611        let current_time = SystemTime::now()
612            .duration_since(UNIX_EPOCH)
613            .unwrap()
614            .as_secs();
615
616        if *last_fetched_expiry <= current_time {
617            if verbose {
618                eprintln!(
619                    "[FetchedCredentialsCache]: \
620                    Call update_func: current_time = {}, \
621                    last_fetched_expiry = {}",
622                    current_time, *last_fetched_expiry
623                )
624            }
625
626            let (credentials, expiry) = update_func.await?;
627
628            *last_fetched_credentials = credentials;
629            *last_fetched_expiry = expiry;
630            *log_use_cached = true;
631
632            if expiry < current_time && expiry != 0 {
633                polars_bail!(
634                    ComputeError:
635                    "credential expiry time {} is older than system time {} \
636                     by {} seconds",
637                    expiry,
638                    current_time,
639                    current_time - expiry
640                )
641            }
642
643            if verbose {
644                eprintln!(
645                    "[FetchedCredentialsCache]: Finish update_func: new {}",
646                    expiry_msg(
647                        *last_fetched_expiry,
648                        SystemTime::now()
649                            .duration_since(UNIX_EPOCH)
650                            .unwrap()
651                            .as_secs()
652                    )
653                )
654            }
655        } else if verbose && *log_use_cached {
656            *log_use_cached = false;
657            let now = SystemTime::now()
658                .duration_since(UNIX_EPOCH)
659                .unwrap()
660                .as_secs();
661            eprintln!(
662                "[FetchedCredentialsCache]: Using cached credentials: \
663                current_time = {}, {}",
664                now,
665                expiry_msg(*last_fetched_expiry, now)
666            )
667        }
668
669        Ok(last_fetched_credentials.clone())
670    }
671}

688    #[derive(Clone, Debug)]
689    pub enum PythonCredentialProvider {
690        /// Indicates `py_object` is a `CredentialProviderBuilder`.
691        Builder(Arc<PythonObject>),
692        /// Indicates `py_object` is an instantiated credential provider
693        Provider(Arc<PythonObject>),
694    }
695
696    impl PythonCredentialProvider {
697        /// Builds the Python credential provider for an auto-init config, using the Python
698        /// that executes the query. Returns `None` if it could not be auto-initialized.
699        pub(super) fn build_auto_init(
700            config: &AutoInitConfig,
701            clear_cached_credentials: bool,
702        ) -> PolarsResult<Option<Self>> {
703            let config_json = serde_json::to_string(config).map_err(to_compute_err)?;
704
705            let opt_initialized_py_object = Python::attach(|py| {
706                let v = py
707                    .import(intern!(py, "polars.io.cloud.credential_provider._builder"))?
708                    .call_method1(
709                        intern!(py, "_build_auto_init_credential_provider"),
710                        (config_json, clear_cached_credentials),
711                    )?;
712
713                pyo3::PyResult::Ok((!v.is_none()).then(|| v.unbind()))
714            })?;
715
716            Ok(opt_initialized_py_object
717                .map(PythonObject)
718                .map(Arc::new)
719                .map(Self::Provider))
720        }
721
722        /// Performs initialization if necessary.
723        ///
724        /// This exists as a separate step that must be called beforehand. This approach is easier
725        /// as the alternative is to refactor the `IntoCredentialProvider` trait to return
726        /// `PolarsResult<Option<T>>` for every single function.
727        pub(super) fn try_into_initialized(
728            self,
729            clear_cached_credentials: bool,
730        ) -> PolarsResult<Option<Self>> {
731            match self {
732                Self::Builder(py_object) => {
733                    let opt_initialized_py_object = Python::attach(|py| {
734                        let build_fn =
735                            py_object.getattr(py, intern!(py, "build_credential_provider"))?;
736
737                        let v = build_fn.call1(py, (clear_cached_credentials,))?;
738                        let v = (!v.is_none(py)).then_some(v);
739
740                        pyo3::PyResult::Ok(v)
741                    })?;
742
743                    Ok(opt_initialized_py_object
744                        .map(PythonObject)
745                        .map(Arc::new)
746                        .map(Self::Provider))
747                },
748                Self::Provider(_) => {
749                    // Note: We don't expect to hit here.
750                    Ok(Some(self))
751                },
752            }
753        }
754
755        fn unwrap_as_provider(self) -> Arc<PythonObject> {
756            match self {
757                Self::Builder(_) => panic!(),
758                Self::Provider(v) => v,
759            }
760        }
761
762        pub(crate) fn unwrap_as_provider_ref(&self) -> &Arc<PythonObject> {
763            match self {
764                Self::Builder(_) => panic!(),
765                Self::Provider(v) => v,
766            }
767        }
768
769        pub(super) fn func_addr(&self) -> usize {
770            (match self {
771                Self::Builder(v) => Arc::as_ptr(v),
772                Self::Provider(v) => Arc::as_ptr(v),
773            }) as *const () as usize
774        }
775
776        pub fn stable_cache_key(&self) -> PolarsResult<Vec<u8>> {
777            let obj = match self {
778                Self::Builder(obj) | Self::Provider(obj) => obj,
779            };
780            let err = |e| {
781                polars_err!(ComputeError:
782                "failed to extract stable_cache_key: {e}")
783            };
784            Python::attach(|py| {
785                obj.call_method0(py, "stable_cache_key")
786                    .and_then(|r| r.extract::<Vec<u8>>(py))
787            })
788            .map_err(err)
789        }
790    }

793        #[cfg(feature = "aws")]
794        fn into_aws_provider(self) -> object_store::aws::AwsCredentialProvider {
795            use polars_error::PolarsResult;
796
797            use crate::cloud::credential_provider::{
798                CredentialProviderFunction, ObjectStoreCredential,
799            };
800
801            let func = self.unwrap_as_provider();
802
803            CredentialProviderFunction(Arc::new(move || {
804                let func = func.clone();
805                Box::pin(async move {
806                    let mut credentials = object_store::aws::AwsCredential {
807                        key_id: String::new(),
808                        secret_key: String::new(),
809                        token: None,
810                    };
811
812                    let expiry = Python::attach(|py| {
813                        let v = func.0.call0(py)?.into_bound(py);
814                        let (storage_options, expiry) =
815                            v.extract::<(pyo3::Bound<'_, PyDict>, Option<u64>)>()?;
816
817                        for (k, v) in storage_options.iter() {
818                            let k = k.extract::<PyBackedStr>()?;
819                            let v = v.extract::<Option<String>>()?;
820
821                            match k.as_ref() {
822                                "aws_access_key_id" => {
823                                    credentials.key_id = v.ok_or_else(|| {
824                                        PyValueError::new_err("aws_access_key_id was None")
825                                    })?;
826                                },
827                                "aws_secret_access_key" => {
828                                    credentials.secret_key = v.ok_or_else(|| {
829                                        PyValueError::new_err("aws_secret_access_key was None")
830                                    })?
831                                },
832                                "aws_session_token" => credentials.token = v,
833                                v => {
834                                    return pyo3::PyResult::Err(PyValueError::new_err(format!(
835                                        "unknown configuration key for aws: {}, \
836                                    valid configuration keys are: \
837                                    {}, {}, {}",
838                                        v,
839                                        "aws_access_key_id",
840                                        "aws_secret_access_key",
841                                        "aws_session_token"
842                                    )));
843                                },
844                            }
845                        }
846
847                        pyo3::PyResult::Ok(expiry.unwrap_or(u64::MAX))
848                    })?;
849
850                    if credentials.key_id.is_empty() {
851                        return Err(PolarsError::ComputeError(
852                            "aws_access_key_id was empty or not given".into(),
853                        ));
854                    }
855
856                    if credentials.secret_key.is_empty() {
857                        return Err(PolarsError::ComputeError(
858                            "aws_secret_access_key was empty or not given".into(),
859                        ));
860                    }
861
862                    PolarsResult::Ok((ObjectStoreCredential::Aws(Arc::new(credentials)), expiry))
863                })
864            }))
865            .into_aws_provider()
866        }

868        #[cfg(feature = "azure")]
869        fn into_azure_provider(self) -> object_store::azure::AzureCredentialProvider {
870            use object_store::azure::AzureAccessKey;
871            use percent_encoding::percent_decode_str;
872            use polars_core::config::verbose_print_sensitive;
873            use polars_error::PolarsResult;
874
875            use crate::cloud::credential_provider::{
876                CredentialProviderFunction, ObjectStoreCredential,
877            };
878
879            let func = self.unwrap_as_provider();
880
881            return CredentialProviderFunction(Arc::new(move || {
882                let func = func.clone();
883                Box::pin(async move {
884                    let mut credentials = None;
885
886                    static VALID_KEYS_MSG: &str =
887                        "valid configuration keys are: ('account_key', 'bearer_token', 'sas_token')";
888
889                    let expiry = Python::attach(|py| {
890                        let v = func.0.call0(py)?.into_bound(py);
891                        let (storage_options, expiry) =
892                            v.extract::<(pyo3::Bound<'_, PyDict>, Option<u64>)>()?;
893
894                        for (k, v) in storage_options.iter() {
895                            let k = k.extract::<PyBackedStr>()?;
896                            let v = v.extract::<String>()?;
897
898                            match k.as_ref() {
899                                "account_key" => {
900                                    credentials =
901                                        Some(object_store::azure::AzureCredential::AccessKey(
902                                            AzureAccessKey::try_new(v.as_str()).map_err(|e| {
903                                                PyValueError::new_err(e.to_string())
904                                            })?,
905                                        ))
906                                },
907                                "bearer_token" => {
908                                    credentials =
909                                        Some(object_store::azure::AzureCredential::BearerToken(v))
910                                },
911                                "sas_token" => {
912                                    credentials =
913                                        Some(object_store::azure::AzureCredential::SASToken(
914                                            split_sas(&v).map_err(|err_msg| {
915                                                verbose_print_sensitive(|| {
916                                                    format!("error decoding SAS token: {err_msg} (token: {v})")
917                                                });
918
919                                                PyValueError::new_err(format!(
920                                                    "error decoding SAS token: {err_msg}. \
921                                                    Set POLARS_VERBOSE_SENSITIVE=1 to print the value"
922                                                ))
923                                            })?,
924                                        ))
925                                },
926                                v => {
927                                    return pyo3::PyResult::Err(PyValueError::new_err(format!(
928                                        "unknown configuration key for azure: {v}, {VALID_KEYS_MSG}"
929                                    )));
930                                },
931                            }
932                        }
933
934                        pyo3::PyResult::Ok(expiry.unwrap_or(u64::MAX))
935                    })?;
936
937                    let Some(credentials) = credentials else {
938                        return Err(PolarsError::ComputeError(
939                            format!(
940                                "did not find a valid configuration key for azure, {VALID_KEYS_MSG}"
941                            )
942                            .into(),
943                        ));
944                    };
945
946                    PolarsResult::Ok((ObjectStoreCredential::Azure(Arc::new(credentials)), expiry))
947                })
948            }))
949            .into_azure_provider();
950
951            /// Copied and adjusted from object-store.
952            ///
953            /// https://github.com/apache/arrow-rs-object-store/blob/7a0504b4924fcecee17d768fd7190b8f71b0877f/src/azure/builder.rs#L1072-L1089
954            fn split_sas(sas: &str) -> Result<Vec<(String, String)>, &'static str> {
955                let sas = percent_decode_str(sas)
956                    .decode_utf8()
957                    .map_err(|_| "UTF-8 decode error")?;
958
959                let kv_str_pairs = sas
960                    .trim_start_matches('?')
961                    .split('&')
962                    .filter(|s| !s.chars().all(char::is_whitespace));
963
964                let mut pairs = Vec::new();
965
966                for kv_pair_str in kv_str_pairs {
967                    let (k, v) = kv_pair_str
968                        .trim()
969                        .split_once('=')
970                        .ok_or("missing SAS component")?;
971                    pairs.push((k.into(), v.into()))
972                }
973
974                Ok(pairs)
975            }
976        }

980            use polars_error::PolarsResult;
981
982            use crate::cloud::credential_provider::{
983                CredentialProviderFunction, ObjectStoreCredential,
984            };
985
986            let func = self.unwrap_as_provider();
987
988            CredentialProviderFunction(Arc::new(move || {
989                let func = func.clone();
990                Box::pin(async move {
991                    let mut credentials = object_store::gcp::GcpCredential {
992                        bearer: String::new(),
993                    };
994
995                    let expiry = Python::attach(|py| {
996                        let v = func.0.call0(py)?.into_bound(py);
997                        let (storage_options, expiry) =
998                            v.extract::<(pyo3::Bound<'_, PyDict>, Option<u64>)>()?;
999
1000                        for (k, v) in storage_options.iter() {
1001                            let k = k.extract::<PyBackedStr>()?;
1002                            let v = v.extract::<String>()?;
1003
1004                            match k.as_ref() {
1005                                "bearer_token" => credentials.bearer = v,
1006                                v => {
1007                                    return pyo3::PyResult::Err(PyValueError::new_err(format!(
1008                                        "unknown configuration key for gcp: {}, \
1009                                    valid configuration keys are: {}",
1010                                        v, "bearer_token",
1011                                    )));
1012                                },
1013                            }
1014                        }
1015
1016                        pyo3::PyResult::Ok(expiry.unwrap_or(u64::MAX))
1017                    })?;
1018
1019                    if credentials.bearer.is_empty() {
1020                        return Err(PolarsError::ComputeError(
1021                            "bearer was empty or not given".into(),
1022                        ));
1023                    }
1024
1025                    PolarsResult::Ok((ObjectStoreCredential::Gcp(Arc::new(credentials)), expiry))
1026                })
1027            }))

1031        /// # Panics
1032        /// Panics if `self` is not an initialized provider.
1033        fn storage_update_options(&self) -> PolarsResult<Vec<(PlSmallStr, PlSmallStr)>> {
1034            let py_object = self.unwrap_as_provider_ref();
1035
1036            Python::attach(|py| {
1037                py_object
1038                    .getattr(py, "_storage_update_options")
1039                    .map_or(Ok(vec![]), |f| {
1040                        let v = f
1041                            .call0(py)?
1042                            .extract::<pyo3::Bound<'_, PyDict>>(py)
1043                            .map_err(pyo3::PyErr::from)?;
1044
1045                        let mut out = Vec::with_capacity(v.len());
1046
1047                        for dict_item in v.call_method0("items")?.try_iter()? {
1048                            let (key, value) =
1049                                dict_item?.extract::<(PyBackedStr, PyBackedStr)>()?;
1050
1051                            out.push(((&*key).into(), (&*value).into()))
1052                        }
1053
1054                        Ok(out)
1055                    })
1056            })
1057        }
