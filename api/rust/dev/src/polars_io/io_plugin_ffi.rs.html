1//! Host side of the generic I/O plugin contracts of `polars_io_ext_ffi`:
2//! `polars.io_plugin.file_reader.v1` (see [`crate::external_reader::ffi`]) and
3//! `polars.io_plugin.lazyframe_resolver.v1` (the `Ffi` DSL resolver of `polars-plan`).
4//!
5//! Plugins are found by Python package name through `polars_io_ext_ffi::registry`, whose locator
6//! the Python bindings install.
7use polars_arrow::datatypes::ArrowDataType;
8use polars_arrow::ffi;
9use polars_core::prelude::*;
10use polars_error::{PolarsError, PolarsResult, polars_bail, polars_ensure, polars_err};
11pub use polars_io_ext_ffi::common::{ArrowSchema, FfiError, FfiErrorKind};
12use polars_io_ext_ffi::{file_reader_v1, lazyframe_resolver_v1, registry};

14/// The `file_reader.v1` contract of `package`.
15pub fn file_reader_v1(package: &str) -> PolarsResult<&'static file_reader_v1::Plugin> {
16    // SAFETY: the plugin ID fixes the struct type.
17    unsafe { registry::lookup(package, file_reader_v1::ID) }
18        .map_err(|e| polars_err!(ComputeError: "{e}"))
19}

21/// The `lazyframe_resolver.v1` contract of `package`.
22pub fn lazyframe_resolver_v1(
23    package: &str,
24) -> PolarsResult<&'static lazyframe_resolver_v1::Plugin> {
25    // SAFETY: the plugin ID fixes the struct type.
26    unsafe { registry::lookup(package, lazyframe_resolver_v1::ID) }
27        .map_err(|e| polars_err!(ComputeError: "{e}"))
28}

30/// A plugin error as a Polars error of the same kind. Plugin errors of other kinds are compute
31/// errors that name the plugin.
32///
33/// `INVALID_INPUT` is a Python `ValueError` with the plugin's message, as the contract promises
34/// (an invalid operation error without the Python bindings).
35pub fn ffi_err_to_polars(e: FfiError, package: &str) -> PolarsError {
36    let msg = e.message();
37    let kind = e.kind();
38    match kind {
39        FfiErrorKind::COLUMN_NOT_FOUND => PolarsError::ColumnNotFound(msg.into()),
40        FfiErrorKind::COMPUTE => PolarsError::ComputeError(msg.into()),
41        FfiErrorKind::INVALID_OPERATION => PolarsError::InvalidOperation(msg.into()),
42        #[cfg(feature = "python")]
43        FfiErrorKind::INVALID_INPUT => {
44            pyo3::exceptions::PyValueError::new_err(msg.to_string()).into()
45        },
46        #[cfg(not(feature = "python"))]
47        FfiErrorKind::INVALID_INPUT => PolarsError::InvalidOperation(msg.into()),
48        FfiErrorKind::SCHEMA_MISMATCH => PolarsError::SchemaMismatch(msg.into()),
49        FfiErrorKind::SHAPE_MISMATCH => PolarsError::ShapeMismatch(msg.into()),
50        FfiErrorKind::NO_DATA => PolarsError::NoData(msg.into()),
51        FfiErrorKind::OUT_OF_BOUNDS => PolarsError::OutOfBounds(msg.into()),
52        FfiErrorKind::DUPLICATE => PolarsError::Duplicate(msg.into()),
53        FfiErrorKind::SQL_INTERFACE => PolarsError::SQLInterface(msg.into()),
54        FfiErrorKind::SQL_SYNTAX => PolarsError::SQLSyntax(msg.into()),
55        FfiErrorKind::IO => PolarsError::IO {
56            error: std::sync::Arc::new(std::io::Error::other(msg)),
57            msg: None,
58        },
59        kind => polars_err!(ComputeError: "{package} ({}): {msg}", kind.name()),
60    }
61}

63/// Imports an Arrow struct schema from a plugin as a schema.
64pub fn import_struct_schema(schema: ArrowSchema, package: &str) -> PolarsResult<Schema> {
65    polars_ensure!(
66        !schema.is_released(),
67        ComputeError: "{package} returned a released schema"
68    );
69    // SAFETY: the contract's struct has the C data interface layout.
70    let schema: ffi::ArrowSchema = unsafe { schema.transmute_into() };
71    let field = unsafe { ffi::import_field_from_c(&schema)? };
72    let ArrowDataType::Struct(fields) = field.dtype else {
73        polars_bail!(
74            ComputeError:
75            "{package} returned a non-struct schema: {:?}", field.dtype
76        )
77    };
78    // (`Field::from` unwraps a time zone it can't read: an error for a plugin's)
79    for f in fields.iter() {
80        DataType::check_arrow_time_zones(&f.dtype)?;
81    }
82    Ok(fields.iter().map(Field::from).collect())
83}
