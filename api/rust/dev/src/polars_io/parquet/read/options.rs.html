34/// Properties for reading files encrypted with Parquet modular encryption.
35///
36/// These hold secret keys, so can't be serialized, and are compared and hashed
37/// by pointer.
38#[derive(Clone)]
39pub struct PlFileDecryptionProperties(pub Arc<FileDecryptionProperties>);
40
41impl Debug for PlFileDecryptionProperties {
42    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
43        // FileDecryptionProperties doesn't output keys.
44        self.0.fmt(f)
45    }
46}
47
48impl Eq for PlFileDecryptionProperties {}
49
50impl PartialEq for PlFileDecryptionProperties {
51    fn eq(&self, other: &Self) -> bool {
52        Arc::ptr_eq(&self.0, &other.0)
53    }
54}
55
56impl Hash for PlFileDecryptionProperties {
57    fn hash<H: std::hash::Hasher>(&self, state: &mut H) {
58        state.write_usize(Arc::as_ptr(&self.0) as usize)
59    }
60}

106pub enum ParallelStrategy {
107    /// Don't parallelize
108    None,
109    /// Parallelize over the columns
110    Columns,
111    /// Parallelize over the row groups
112    RowGroups,
113    /// First evaluates the pushed-down predicates in parallel and determines a mask of which rows
114    /// to read. Then, it parallelizes over both the columns and the row groups while filtering out
115    /// rows that do not need to be read. This can provide significant speedups for large files
116    /// (i.e. many row-groups) with a predicate that filters clustered rows or filters heavily. In
117    /// other cases, this may slow down the scan compared other strategies.
118    ///
119    /// If no predicate is given, this falls back to back to [`ParallelStrategy::Auto`].
120    Prefiltered,
121    /// Automatically determine over which unit to parallelize
122    /// This will choose the most occurring unit.
123    #[default]
124    Auto,
125}
