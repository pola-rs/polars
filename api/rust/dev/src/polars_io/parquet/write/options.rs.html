44/// Properties for writing files encrypted with Parquet modular encryption.
45///
46/// These hold secret keys, so can't be serialized, and are compared and hashed
47/// by pointer.
48#[derive(Clone)]
49pub struct PlFileEncryptionProperties(pub Arc<FileEncryptionProperties>);
50
51impl Debug for PlFileEncryptionProperties {
52    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
53        // FileEncryptionProperties doesn't output keys.
54        self.0.fmt(f)
55    }
56}
57
58impl Eq for PlFileEncryptionProperties {}
59
60impl PartialEq for PlFileEncryptionProperties {
61    fn eq(&self, other: &Self) -> bool {
62        Arc::ptr_eq(&self.0, &other.0)
63    }
64}
65
66impl Hash for PlFileEncryptionProperties {
67    fn hash<H: std::hash::Hasher>(&self, state: &mut H) {
68        state.write_usize(Arc::as_ptr(&self.0) as usize)
69    }
70}

113/// The compression strategy to use for writing Parquet files.
114#[derive(Debug, Eq, PartialEq, Hash, Clone, Copy)]
115#[cfg_attr(feature = "serde", derive(Serialize, Deserialize))]
116#[cfg_attr(feature = "dsl-schema", derive(schemars::JsonSchema))]
117pub enum ParquetCompression {
118    Uncompressed,
119    Snappy,
120    Gzip(Option<GzipLevel>),
121    Brotli(Option<BrotliLevel>),
122    Zstd(Option<ZstdLevel>),
123    Lz4Raw,
124}
125
126impl Default for ParquetCompression {
127    fn default() -> Self {
128        Self::Zstd(None)
129    }
130}
131
132impl From<ParquetCompression> for CompressionOptions {
133    fn from(value: ParquetCompression) -> Self {
134        use ParquetCompression::*;
135        match value {
136            Uncompressed => CompressionOptions::Uncompressed,
137            Snappy => CompressionOptions::Snappy,
138            Gzip(level) => CompressionOptions::Gzip(level),
139            Brotli(level) => CompressionOptions::Brotli(level),
140            Lz4Raw => CompressionOptions::Lz4Raw,
141            Zstd(level) => CompressionOptions::Zstd(level),
142        }
143    }
144}
