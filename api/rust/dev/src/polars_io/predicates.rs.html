11pub trait PhysicalIoExpr: Send + Sync {
12    /// Take a [`DataFrame`] and produces a boolean [`Series`] that serves
13    /// as a predicate mask
14    fn evaluate_io(&self, df: &DataFrame) -> PolarsResult<Series>;
15}

17#[derive(Debug, Clone)]
18pub enum SpecializedColumnPredicate {
19    Equal(Scalar),
20    /// A closed (inclusive) range.
21    Between(Scalar, Scalar),
22    EqualOneOf(Box<[Scalar]>),
23    StartsWith(Box<[u8]>),
24    EndsWith(Box<[u8]>),
25    RegexMatch(regex::bytes::Regex),
26}

28#[derive(Clone)]
29pub struct ColumnPredicateExpr {
30    column_name: PlSmallStr,
31    dtype: DataType,
32    source_arrow_dtype: ArrowDataType,
33    #[cfg(feature = "parquet")]
34    specialized: Option<SpecializedParquetColumnExpr>,
35    expr: Arc<dyn PhysicalIoExpr>,
36}
37
38impl ColumnPredicateExpr {
39    pub fn new(
40        column_name: PlSmallStr,
41        dtype: DataType,
42        source_arrow_dtype: ArrowDataType,
43        expr: Arc<dyn PhysicalIoExpr>,
44        specialized: Option<SpecializedColumnPredicate>,
45    ) -> Self {
46        use SpecializedColumnPredicate as S;
47        #[cfg(feature = "parquet")]
48        use SpecializedParquetColumnExpr as P;
49        #[cfg(feature = "parquet")]
50        let specialized = specialized.and_then(|s| {
51            Some(match s {
52                S::Equal(s) => P::Equal(cast_to_parquet_scalar(s)?),
53                S::Between(low, high) => {
54                    P::Between(cast_to_parquet_scalar(low)?, cast_to_parquet_scalar(high)?)
55                },
56                S::EqualOneOf(scalars) => P::EqualOneOf(
57                    scalars
58                        .into_iter()
59                        .map(|s| cast_to_parquet_scalar(s).ok_or(()))
60                        .collect::<Result<Box<_>, ()>>()
61                        .ok()?,
62                ),
63                S::StartsWith(s) => P::StartsWith(s),
64                S::EndsWith(s) => P::EndsWith(s),
65                S::RegexMatch(s) => P::RegexMatch(s),
66            })
67        });
68
69        Self {
70            column_name,
71            dtype,
72            source_arrow_dtype,
73            #[cfg(feature = "parquet")]
74            specialized,
75            expr,
76        }
77    }
78}

81impl ParquetColumnExpr for ColumnPredicateExpr {
82    fn evaluate_mut(&self, values: &dyn Array, bm: &mut BitmapBuilder) {
83        // We should never evaluate nulls with this.
84        assert!(values.validity().is_none_or(|v| v.set_bits() == 0));
85
86        // @TODO: Probably these unwraps should be removed.
87        let series = predicate_values_to_series(
88            self.column_name.clone(),
89            values,
90            &self.dtype,
91            &self.source_arrow_dtype,
92        )
93        .unwrap();
94        let column = series.into_column();
95        let df = unsafe { DataFrame::new_unchecked(values.len(), vec![column]) };
96
97        // @TODO: Probably these unwraps should be removed.
98        let true_mask = self.expr.evaluate_io(&df).unwrap();
99        let true_mask = true_mask.bool().unwrap();
100
101        bm.reserve(true_mask.len());
102        for chunk in true_mask.downcast_iter() {
103            match chunk.validity() {
104                None => bm.extend_from_bitmap(chunk.values()),
105                Some(v) => bm.extend_from_bitmap(&(chunk.values() & v)),
106            }
107        }
108    }
109    fn evaluate_null(&self) -> bool {
110        let column = Column::full_null(self.column_name.clone(), 1, &self.dtype);
111        let df = unsafe { DataFrame::new_unchecked(1, vec![column]) };
112
113        // @TODO: Probably these unwraps should be removed.
114        let true_mask = self.expr.evaluate_io(&df).unwrap();
115        let true_mask = true_mask.bool().unwrap();
116
117        true_mask.get(0).unwrap_or(false)
118    }
119
120    fn as_specialized(&self) -> Option<&SpecializedParquetColumnExpr> {
121        self.specialized.as_ref()
122    }
123}

125#[cfg(feature = "parquet")]
126fn predicate_values_to_series(
127    name: PlSmallStr,
128    values: &dyn Array,
129    dtype: &DataType,
130    source_arrow_dtype: &ArrowDataType,
131) -> PolarsResult<Series> {
132    // For example, Arrow seconds will be stored as Polars milliseconds, so we can not just
133    // zero-copy construct the predicate series
134    let timestamp_units_differ = matches!(
135        (source_arrow_dtype, dtype),
136        (
137            ArrowDataType::Timestamp(source_unit, _),
138            DataType::Datetime(target_unit, _),
139        ) if source_unit != &target_unit.to_arrow()
140    );
141
142    if timestamp_units_differ {
143        let values = polars_compute::cast::cast(
144            values,
145            source_arrow_dtype,
146            polars_compute::cast::CastOptionsImpl::default(),
147        )?;
148        Series::try_from((name, values))
149    } else {
150        Series::from_chunk_and_dtype(name, values.to_boxed(), dtype)
151    }
152}

154#[cfg(feature = "parquet")]
155fn cast_to_parquet_scalar(scalar: Scalar) -> Option<ParquetScalar> {
156    use AnyValue as A;
157    use ParquetScalar as P;
158
159    Some(match scalar.into_value() {
160        A::Null => P::Null,
161        A::Boolean(v) => P::Boolean(v),
162
163        A::UInt8(v) => P::UInt8(v),
164        A::UInt16(v) => P::UInt16(v),
165        A::UInt32(v) => P::UInt32(v),
166        A::UInt64(v) => P::UInt64(v),
167
168        A::Int8(v) => P::Int8(v),
169        A::Int16(v) => P::Int16(v),
170        A::Int32(v) => P::Int32(v),
171        A::Int64(v) => P::Int64(v),
172
173        #[cfg(feature = "dtype-date")]
174        A::Date(v) => P::Int32(v),
175        #[cfg(feature = "dtype-datetime")]
176        A::Datetime(v, _, _) | A::DatetimeOwned(v, _, _) => P::Int64(v),
177        #[cfg(feature = "dtype-duration")]
178        A::Duration(v, _) => P::Int64(v),
179        #[cfg(feature = "dtype-time")]
180        A::Time(v) => P::Int64(v),
181
182        A::Float32(v) => P::Float32(v),
183        A::Float64(v) => P::Float64(v),
184
185        // @TODO: Cast to string
186        #[cfg(feature = "dtype-categorical")]
187        A::Categorical(_, _) | A::CategoricalOwned(_, _) | A::Enum(_, _) | A::EnumOwned(_, _) => {
188            return None;
189        },
190
191        A::String(v) => P::String(v.into()),
192        A::StringOwned(v) => P::String(v.as_str().into()),
193        A::Binary(v) => P::Binary(v.into()),
194        A::BinaryOwned(v) => P::Binary(v.into()),
195        _ => return None,
196    })
197}

199#[cfg(any(feature = "parquet", feature = "ipc"))]
200pub fn apply_predicate(
201    df: &mut DataFrame,
202    predicate: Option<&dyn PhysicalIoExpr>,
203    parallel: bool,
204) -> PolarsResult<()> {
205    if let (Some(predicate), false) = (&predicate, df.columns().is_empty()) {
206        let s = predicate.evaluate_io(df)?;
207        let mask = s.bool().expect("filter predicates was not of type boolean");
208
209        if parallel {
210            *df = df.filter(mask)?;
211        } else {
212            *df = df.filter_seq(mask)?;
213        }
214    }
215    Ok(())
216}

225pub trait SkipBatchPredicate: Send + Sync {
226    fn schema(&self) -> &SchemaRef;
227
228    fn can_skip_batch(
229        &self,
230        batch_size: IdxSize,
231        live_columns: &PlIndexSet<PlSmallStr>,
232        mut statistics: PlIndexMap<PlSmallStr, ColumnStatistics>,
233    ) -> PolarsResult<bool> {
234        let mut columns = Vec::with_capacity(1 + live_columns.len() * 3);
235
236        columns.push(Column::new_scalar(
237            PlSmallStr::from_static("len"),
238            Scalar::new(IDX_DTYPE, batch_size.into()),
239            1,
240        ));
241
242        for col in live_columns.iter() {
243            let dtype = self.schema().get(col).unwrap();
244            let (min, max, nc) = match statistics.swap_remove(col) {
245                None => (
246                    Scalar::null(dtype.clone()),
247                    Scalar::null(dtype.clone()),
248                    Scalar::null(IDX_DTYPE),
249                ),
250                Some(stat) => (
251                    Scalar::new(dtype.clone(), stat.min),
252                    Scalar::new(dtype.clone(), stat.max),
253                    Scalar::new(
254                        IDX_DTYPE,
255                        stat.null_count.map_or(AnyValue::Null, |nc| nc.into()),
256                    ),
257                ),
258            };
259            columns.extend([
260                Column::new_scalar(format_pl_smallstr!("{col}_min"), min, 1),
261                Column::new_scalar(format_pl_smallstr!("{col}_max"), max, 1),
262                Column::new_scalar(format_pl_smallstr!("{col}_nc"), nc, 1),
263            ]);
264        }
265
266        // SAFETY:
267        // * Each column is length = 1
268        // * We have an IndexSet, so each column name is unique
269        let df = unsafe { DataFrame::new_unchecked(1, columns) };
270        Ok(self.evaluate_with_stat_df(&df)?.get_bit(0))
271    }
272    fn evaluate_with_stat_df(&self, df: &DataFrame) -> PolarsResult<Bitmap>;
273}

293pub struct PhysicalExprWithConstCols<T> {
294    constants: Vec<(PlSmallStr, Scalar)>,
295    child: T,
296}
297
298impl SkipBatchPredicate for PhysicalExprWithConstCols<Arc<dyn SkipBatchPredicate>> {
299    fn schema(&self) -> &SchemaRef {
300        self.child.schema()
301    }
302
303    fn evaluate_with_stat_df(&self, df: &DataFrame) -> PolarsResult<Bitmap> {
304        let mut df = df.clone();
305        for (name, scalar) in self.constants.iter() {
306            df.with_column(Column::new_scalar(
307                name.clone(),
308                scalar.clone(),
309                df.height(),
310            ))?;
311        }
312        self.child.evaluate_with_stat_df(&df)
313    }
314}

316impl PhysicalIoExpr for PhysicalExprWithConstCols<Arc<dyn PhysicalIoExpr>> {
317    fn evaluate_io(&self, df: &DataFrame) -> PolarsResult<Series> {
318        let mut df = df.clone();
319        for (name, scalar) in self.constants.iter() {
320            df.with_column(Column::new_scalar(
321                name.clone(),
322                scalar.clone(),
323                df.height(),
324            ))?;
325        }
326
327        self.child.evaluate_io(&df)
328    }

335    /// Column names that are used in the predicate.

342    pub column_predicates: Arc<ColumnPredicates>,

345    pub hive_predicate: Option<Arc<dyn PhysicalIoExpr>>,
346
347    pub hive_predicate_is_full_predicate: bool,
348}

350impl ScanIOPredicate {
351    pub fn set_external_constant_columns(&mut self, constant_columns: Vec<(PlSmallStr, Scalar)>) {
352        if constant_columns.is_empty() {
353            return;
354        }
355
356        let mut live_columns = self.live_columns.as_ref().clone();
357        for (c, _) in constant_columns.iter() {
358            live_columns.swap_remove(c);
359        }
360        self.live_columns = Arc::new(live_columns);
361
362        if let Some(skip_batch_predicate) = self.skip_batch_predicate.take() {
363            let mut sbp_constant_columns = Vec::with_capacity(constant_columns.len() * 3);
364            for (c, v) in constant_columns.iter() {
365                sbp_constant_columns.push((format_pl_smallstr!("{c}_min"), v.clone()));
366                sbp_constant_columns.push((format_pl_smallstr!("{c}_max"), v.clone()));
367                let nc = if v.is_null() {
368                    AnyValue::Null
369                } else {
370                    (0 as IdxSize).into()
371                };
372                sbp_constant_columns
373                    .push((format_pl_smallstr!("{c}_nc"), Scalar::new(IDX_DTYPE, nc)));
374            }
375            self.skip_batch_predicate = Some(Arc::new(PhysicalExprWithConstCols {
376                constants: sbp_constant_columns,
377                child: skip_batch_predicate,
378            }));
379        }
380
381        let mut column_predicates = self.column_predicates.as_ref().clone();
382        for (c, _) in constant_columns.iter() {
383            column_predicates.predicates.remove(c);
384        }
385        self.column_predicates = Arc::new(column_predicates);
386
387        self.predicate = Arc::new(PhysicalExprWithConstCols {
388            constants: constant_columns,
389            child: self.predicate.clone(),
390        });
391    }
392}
