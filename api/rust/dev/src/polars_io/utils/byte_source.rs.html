31#[allow(async_fn_in_trait)]
32pub trait ByteSource: Send + Sync {
33    async fn get_size(&self) -> PolarsResult<usize>;
34    /// Fetch the last `n` bytes and the total size of the source, in a single request. Returns
35    /// fewer than `n` bytes if the source is smaller than `n`.
36    async fn get_suffix(&self, n: usize) -> PolarsResult<(Buffer<u8>, usize)>;
37    /// # Panics
38    /// Panics if `range` is not in bounds.
39    async fn get_range(&self, range: Range<usize>) -> PolarsResult<Buffer<u8>>;
40    /// Note: This will mutably sort ranges for coalescing.
41    async fn get_ranges(
42        &self,
43        ranges: &mut [Range<usize>],
44    ) -> PolarsResult<PlHashMap<usize, Buffer<u8>>>;
45}

69impl ByteSource for BufferByteSource {

71        Ok(self.0.as_ref().len())

74    async fn get_suffix(&self, n: usize) -> PolarsResult<(Buffer<u8>, usize)> {
75        let len = self.0.as_ref().len();

77    }
78
79    async fn get_range(&self, range: Range<usize>) -> PolarsResult<Buffer<u8>> {
80        let out = self.0.clone().sliced(range);
81        Ok(out)

85        &self,
86        ranges: &mut [Range<usize>],
87    ) -> PolarsResult<PlHashMap<usize, Buffer<u8>>> {
88        Ok(ranges
89            .iter()
90            .map(|x| (x.start, self.0.clone().sliced(x.clone())))
91            .collect())
92    }

139#[cfg(target_os = "linux")]
140fn _fadvise(file: &std::fs::File, advice: FileAdvice) {
141    let advice = match advice {
142        // The OS default is already POSIX_FADV_NORMAL.
143        FileAdvice::Normal => return,
144        FileAdvice::Sequential => libc::POSIX_FADV_SEQUENTIAL,
145        FileAdvice::Random => libc::POSIX_FADV_RANDOM,
146        FileAdvice::WillNeed => libc::POSIX_FADV_WILLNEED,
147    };
148
149    unsafe {
150        libc::posix_fadvise(file.as_raw_fd(), 0, 0, advice);
151    }
152}

154#[cfg(all(unix, not(feature = "nightly")))]
155fn pread_exact(file: &std::fs::File, buf: &mut [u8], offset: u64) -> std::io::Result<()> {
156    use std::os::unix::fs::FileExt;
157    file.read_exact_at(buf, offset)
158}

160#[cfg(all(windows, not(feature = "nightly")))]
161fn pread_exact(file: &std::fs::File, buf: &mut [u8], offset: u64) -> std::io::Result<()> {
162    use std::os::windows::fs::FileExt;
163    let mut filled = 0;
164    while filled < buf.len() {
165        match file.seek_read(&mut buf[filled..], offset + filled as u64)? {
166            0 => return Err(std::io::ErrorKind::UnexpectedEof.into()),
167            n => filled += n,
168        }
169    }
170    Ok(())
171}

173/// `pread_exact` into uninitialized memory, so that `read_buffered` does not
174/// have to zero the buffer the read is about to overwrite anyway.
175#[cfg(all(feature = "nightly", unix))]
176fn pread_exact_uninit(
177    file: &std::fs::File,
178    buf: std::io::BorrowedCursor<'_, u8>,
179    offset: u64,
180) -> std::io::Result<()> {
181    use std::os::unix::fs::FileExt;
182    file.read_buf_exact_at(buf, offset)
183}
184
185#[cfg(all(feature = "nightly", windows))]
186fn pread_exact_uninit(
187    file: &std::fs::File,
188    mut buf: std::io::BorrowedCursor<'_, u8>,
189    mut offset: u64,
190) -> std::io::Result<()> {
191    use std::os::windows::fs::FileExt;
192    // `seek_read_buf` is the short-read variant; loop it to fill the cursor.
193    while buf.capacity() > 0 {
194        let written = buf.written();
195        file.seek_read_buf(buf.reborrow(), offset)?;
196        match buf.written() - written {
197            0 => return Err(std::io::ErrorKind::UnexpectedEof.into()),
198            n => offset += n as u64,
199        }
200    }
201    Ok(())
202}

204/// Read `len` bytes at `offset` through the page cache.
205///
206/// Under `nightly` the destination is left uninitialized and filled by the read
207/// itself; on stable it must be zeroed first, which memsets the whole range
208/// before the kernel overwrites it.
209#[cfg(feature = "nightly")]
210fn read_buffered(file: &std::fs::File, offset: u64, len: usize) -> PolarsResult<Buffer<u8>> {
211    let mut v: Vec<u8> = Vec::with_capacity(len);
212
213    let filled = {
214        let mut buf = std::io::BorrowedBuf::from(&mut v.spare_capacity_mut()[..len]);
215        pread_exact_uninit(file, buf.unfilled(), offset)?;
216        buf.len()
217    };
218
219    debug_assert_eq!(filled, len);
220    // Safety: `pread_exact_uninit` filled the cursor, so the first `filled`
221    // bytes of the allocation are initialized.
222    unsafe { v.set_len(filled) };
223
224    Ok(Buffer::from(v))
225}
226
227#[cfg(not(feature = "nightly"))]
228fn read_buffered(file: &std::fs::File, offset: u64, len: usize) -> PolarsResult<Buffer<u8>> {
229    let mut buf = vec![0u8; len];
230    pread_exact(file, &mut buf, offset)?;
231    Ok(Buffer::from(buf))
232}

234/// Read `len` bytes at `offset`, through direct I/O when `align` says the file
235/// supports it and through the page cache otherwise.
236#[cfg(target_os = "linux")]
237fn read_at(
238    file: &std::fs::File,
239    align: Option<DioAlign>,
240    offset: u64,
241    len: usize,
242    size: u64,
243) -> PolarsResult<Buffer<u8>> {
244    match align {
245        Some(align) => direct_io::read_aligned(file, align, offset, len, size),
246        None => read_buffered(file, offset, len),
247    }
248}

254    offset: u64,
255    len: usize,
256    _size: u64,

265        io_metrics: Option<Arc<IOMetrics>>,

267        // The open path is `open`, `fcntl`, `statx` and `fstat` - all blocking.
268        ASYNC

270            .await

274    fn open_blocking(
275        path: PlRefPath,
276        read_context: FileReadContext,
277        io_metrics: Option<Arc<IOMetrics>>,
278    ) -> PolarsResult<Self> {
279        let path = path.as_std_path();
280        let enable_o_direct = read_context.enable_o_direct;
281
282        let file = {
283            #[cfg(target_os = "linux")]
284            let f = if enable_o_direct {
285                direct_io::open_o_direct(path).or_else(|e| match e.raw_os_error() {
286                    // The filesystem does not support O_DIRECT.
287                    Some(libc::EINVAL) => std::fs::File::open(path),
288                    _ => Err(e),
289                })
290            } else {
291                std::fs::File::open(path)
292            };
293            #[cfg(not(target_os = "linux"))]
294            let f = std::fs::File::open(path);
295
296            f.map_err(|e| _limit_path_len_io_err(path, e))?
297        };
298
299        let file = Arc::new(file);
300
301        #[cfg(target_os = "linux")]
302        let o_direct_align =
303            direct_io::probe_or_disable(&file).map_err(|e| _limit_path_len_io_err(path, e))?;
304
305        #[cfg(not(target_os = "linux"))]
306        let o_direct_align: Option<DioAlign> = None;
307
308        #[cfg(target_os = "linux")]
309        if o_direct_align.is_none() {
310            // Inert under O_DIRECT: there is no page cache to advise.
311            _fadvise(&file, read_context.advice);
312        }
313
314        let concurrency = read_context.concurrency;
315        let permits = read_context.permits;
316
317        let size = file
318            .metadata()
319            .map_err(|e| _limit_path_len_io_err(path, e))?
320            .len();
321
322        if polars_config::config().verbose() {
323            let name = if polars_config::config().verbose_sensitive() {
324                path.display().to_string()
325            } else {
326                "<file>".to_string()
327            };
328            match (enable_o_direct, o_direct_align) {
329                (true, Some(a)) => eprintln!(
330                    "[FileByteSource]: {name}: direct IO active, \
331                        alignment offset: {}, memory: {}",
332                    a.offset, a.memory
333                ),
334                (true, None) => eprintln!(
335                    "[FileByteSource]: {name}: direct IO requested but not active, \
336                        using buffered reads"
337                ),
338                (false, _) => {},
339            }
340        }
341
342        Ok(FileByteSource {
343            file,
344            o_direct_align,
345            concurrency,
346            permits,
347            size,
348            io_metrics: OptIOMetrics(io_metrics),
349        })
350    }

408    /// Whether every page of `ranges` is in the page cache. `false` when the OS cannot tell
409    /// (non-Linux, kernels before 6.5) and under direct I/O.
410    pub fn is_cached(&self, ranges: &[Range<usize>]) -> bool {
411        #[cfg(all(
412            target_os = "linux",
413            any(target_arch = "x86_64", target_arch = "aarch64")
414        ))]
415        if self.o_direct_align.is_none() {
416            return cachestat::all_cached(&self.file, ranges);
417        }
418
419        let _ = ranges;
420        false
421    }

440/// `cachestat(2)`, Linux 6.5+.
441#[cfg(all(
442    target_os = "linux",
443    any(target_arch = "x86_64", target_arch = "aarch64")
444))]
445mod cachestat {
446    use std::ops::Range;
447    use std::os::fd::AsRawFd;
448    use std::sync::atomic::{AtomicBool, Ordering};
449
450    use polars_utils::mem::PAGE_SIZE;
451
452    #[repr(C)]
453    struct CachestatRange {
454        off: u64,
455        len: u64,
456    }
457
458    #[repr(C)]
459    #[derive(Default)]
460    struct Cachestat {
461        nr_cache: u64,
462        nr_dirty: u64,
463        nr_writeback: u64,
464        nr_evicted: u64,
465        nr_recently_evicted: u64,
466    }
467
468    /// Not in `libc` for these targets yet; the same number on both.
469    const SYS_CACHESTAT: libc::c_long = 451;
470
471    /// Set once the kernel has refused the call, so later checks skip the syscall.
472    static UNSUPPORTED: AtomicBool = AtomicBool::new(false);
473
474    /// Whether every page overlapping `ranges` is in the page cache.
475    pub(super) fn all_cached(file: &std::fs::File, ranges: &[Range<usize>]) -> bool {
476        if UNSUPPORTED.load(Ordering::Relaxed) {
477            return false;
478        }
479
480        // One call per run of adjacent or overlapping ranges, e.g. the column chunks of a row
481        // group. Gaps are not bridged: their pages need not be cached.
482        let mut spans: Vec<Range<usize>> =
483            ranges.iter().filter(|r| !r.is_empty()).cloned().collect();
484        spans.sort_unstable_by_key(|r| r.start);
485        spans.dedup_by(|next, cur| {
486            let merge = next.start <= cur.end;
487            if merge {
488                cur.end = cur.end.max(next.end);
489            }
490            merge
491        });
492
493        let page = *PAGE_SIZE;
494        for r in &spans {
495            let mut range = CachestatRange {
496                off: r.start as u64,
497                len: r.len() as u64,
498            };
499            let mut cs = Cachestat::default();
500            // Safety: both structs match the kernel's layout and outlive the call.
501            let ret = unsafe {
502                libc::syscall(
503                    SYS_CACHESTAT,
504                    file.as_raw_fd(),
505                    &mut range as *mut CachestatRange,
506                    &mut cs as *mut Cachestat,
507                    0,
508                )
509            };
510            if ret != 0 {
511                // ENOSYS before 6.5 and EPERM under some seccomp filters hold for every file.
512                // Others, like EOPNOTSUPP on hugetlbfs, are about this file only.
513                if let Some(libc::ENOSYS | libc::EPERM) =
514                    std::io::Error::last_os_error().raw_os_error()
515                {
516                    UNSUPPORTED.store(true, Ordering::Relaxed);
517                }
518                return false;
519            }
520            // The kernel counts the pages that overlap the byte range.
521            let pages = (r.end - 1) / page - r.start / page + 1;
522            if cs.nr_cache < pages as u64 {
523                return false;
524            }
525        }
526        true
527    }
528}

530impl ByteSource for FileByteSource {
531    async fn get_size(&self) -> PolarsResult<usize> {
532        usize::try_from(self.size)
533            .map_err(|_| polars_err!(ComputeError: "file size {} does not fit in usize", self.size))
534    }
535
536    async fn get_suffix(&self, n: usize) -> PolarsResult<(Buffer<u8>, usize)> {
537        let size = self.get_size().await?;
538        let bytes = self.get_range(size.saturating_sub(n)..size).await?;
539        Ok((bytes, size))
540    }
541
542    async fn get_range(&self, range: Range<usize>) -> PolarsResult<Buffer<u8>> {
543        assert!(range.end as u64 <= self.size);
544
545        let file = self.file.clone();
546        let offset = range.start as u64;
547        let len = range.len();
548        let size = self.size;
549        let o_direct = self.o_direct_align;
550
551        let permit = self.permits.clone().acquire_owned().await.unwrap();
552
553        self.io_metrics()
554            .record_io_read(len as u64, async move {
555                ASYNC
556                    .spawn_blocking(move || {
557                        let _permit = permit;
558                        read_at(&file, o_direct, offset, len, size)
559                    })
560                    .await
561            })
562            .await
563            .expect("blocking task panicked")
564    }
565
566    async fn get_ranges(
567        &self,
568        ranges: &mut [Range<usize>],
569    ) -> PolarsResult<PlHashMap<usize, Buffer<u8>>> {
570        if let [range] = ranges {
571            let mut out = PlHashMap::with_capacity(1);
572            out.insert(range.start, self.get_range(range.clone()).await?);
573            return Ok(out);
574        }
575
576        ranges.sort_unstable_by_key(|r| r.start);
577
578        let mut spans: Vec<Range<usize>> = Vec::with_capacity(ranges.len());
579
580        // Threshold for coalescing. Note, individual ranges may exceed MAX_SPAN.
581        const MAX_SPAN: usize = 8 << 20;
582        // Tolerate small gaps. We match typical page size for now; this could be tuned.
583        const MAX_GAP: usize = 4096;
584
585        for r in ranges.iter() {
586            match spans.last_mut() {
587                Some(last)
588                    if r.start.saturating_sub(last.end) <= MAX_GAP
589                        && r.end.saturating_sub(last.start) <= MAX_SPAN =>
590                {
591                    last.end = last.end.max(r.end)
592                },
593                _ => spans.push(r.clone()),
594            }
595        }
596
597        let mut fetched: Vec<(Range<usize>, Buffer<u8>)> = futures::stream::iter(spans)
598            .map(|span| async move {
599                let buf = self.get_range(span.clone()).await?;
600                PolarsResult::Ok((span, buf))
601            })
602            .buffer_unordered(self.concurrency)
603            .try_collect()
604            .await?;
605
606        // Slice out of the containing span into the original ranges.
607        fetched.sort_unstable_by_key(|(s, _)| s.start);
608
609        if let Some(w) = ranges.windows(2).find(|w| w[0].start == w[1].start) {
610            polars_bail!(
611                ComputeError:
612                "duplicate range start {} in read request ({:?} and {:?})",
613                w[0].start, w[0], w[1]
614            );
615        }
616
617        let mut out = PlHashMap::with_capacity(ranges.len());
618        for r in ranges.iter() {
619            // No span available.
620            if r.is_empty() {
621                out.insert(r.start, Buffer::new());
622                continue;
623            }
624
625            // Spans are sorted by start and each range lies within one, so the
626            // containing span is the last one starting at or before `r.start`.
627            let idx = fetched.partition_point(|(s, _)| s.start <= r.start);
628            let (span, buf) = &fetched[idx - 1];
629            debug_assert!(span.start <= r.start && r.end <= span.end);
630
631            let off = r.start - span.start;
632            out.insert(r.start, buf.clone().sliced(off..off + r.len()));
633        }
634
635        debug_assert_eq!(out.len(), ranges.len());
636        Ok(out)
637    }
638}

656            build_object_store(path, cloud_options, false).await?;
657        let path = object_path_from_str(&prefix)?;
658
659        store.set_io_metrics(io_metrics);
660
661        Ok(Self {
662            store,
663            path,
664            config,
665        })
666    }

669    fn chunk_size(&self) -> usize {
670        self.config.chunk_size
671    }
672
673    fn concurrency_strategy(&self) -> ConcurrencyStrategy {
674        self.config.strategy
675    }

706/// Dynamic dispatch to async functions.
707pub enum DynByteSource {
708    Buffer(BufferByteSource),
709    File(FileByteSource),
710    #[cfg(feature = "cloud")]
711    Cloud(ObjectStoreByteSource),
712}
713
714impl DynByteSource {
715    pub fn variant_name(&self) -> &str {
716        match self {
717            Self::Buffer(_) => "Buffer",
718            Self::File(_) => "File",
719            #[cfg(feature = "cloud")]
720            Self::Cloud(_) => "Cloud",
721        }
722    }
723
724    pub fn is_cloud(&self) -> bool {
725        match self {
726            Self::Buffer(_) => false,
727            Self::File(_) => false,
728            #[cfg(feature = "cloud")]
729            Self::Cloud(_) => true,
730        }
731    }
732
733    pub fn chunk_size(&self) -> Option<usize> {
734        match self {
735            Self::Buffer(_) => None,
736            Self::File(_) => None,
737            #[cfg(feature = "cloud")]
738            Self::Cloud(source) => Some(source.config.chunk_size),
739        }
740    }
741
742    pub fn concurrency_strategy(&self) -> Option<ConcurrencyStrategy> {
743        match self {
744            Self::Buffer(_) => None,
745            // Concurrency is handled directly inside FileByteSource.
746            Self::File(_) => None,
747            #[cfg(feature = "cloud")]
748            Self::Cloud(source) => Some(source.concurrency_strategy()),
749        }
750    }
751}

826pub enum DynByteSourceBuilder {
827    Mmap,
828    /// Use std::fs::File positional read (pread).
829    FilePread(FileReadContext),
830    /// Supports both cloud and local files, requires cloud feature.
831    ObjectStore(FetchConfig),
832}
833
834impl DynByteSourceBuilder {
835    pub async fn try_build_from_path(
836        &self,
837        path: PlRefPath,
838        cloud_options: Option<&CloudOptions>,
839        io_metrics: Option<Arc<IOMetrics>>,
840    ) -> PolarsResult<DynByteSource> {
841        Ok(match self {
842            Self::Mmap => {
843                BufferByteSource::try_new_mmap_from_path(path.as_std_path(), cloud_options)
844                    .await?
845                    .into()
846            },
847            Self::FilePread(read_context) => {
848                FileByteSource::try_new_from_path(path, read_context.clone(), io_metrics)
849                    .await?
850                    .into()
851            },
852            Self::ObjectStore(fetch_config) => feature_gated!("cloud", {
853                ObjectStoreByteSource::try_new_from_path(
854                    path,
855                    cloud_options,
856                    io_metrics,
857                    *fetch_config,
858                )
859                .await?
860                .into()
861            }),
862        })
863    }
864
865    pub fn chunk_size(&self) -> Option<usize> {
866        match self {
867            Self::Mmap => None,
868            Self::FilePread(_) => None,
869            Self::ObjectStore(fetch_config) => Some(fetch_config.chunk_size),
870        }
871    }
872
873    pub fn concurrency_strategy(&self) -> Option<&ConcurrencyStrategy> {
874        match self {
875            Self::Mmap => None,
876            Self::FilePread(_) => None,
877            Self::ObjectStore(fetch_config) => Some(&fetch_config.strategy),
878        }
879    }

889    fn source(dir: &std::path::Path, len: usize) -> (FileByteSource, Vec<u8>) {
890        let path = dir.join("f.bin");
891        let contents: Vec<u8> = (0..len).map(|i| (i % 251) as u8).collect();
892        std::fs::write(&path, &contents).unwrap();
893
894        let read_context = FileReadContext {
895            enable_o_direct: false,
896            concurrency: 4,
897            permits: Arc::new(Semaphore::new(4)),
898            advice: FileAdvice::Normal,
899        };
900        let source = FileByteSource::try_new_from_std(
901            std::fs::File::open(&path).unwrap(),
902            read_context,
903            None,
904        )
905        .unwrap();
906        (source, contents)
907    }

917    #[test]
918    fn get_ranges_resolves_every_range() {
919        let dir = tempfile::tempdir().unwrap();
920        let (source, contents) = source(dir.path(), 17 * MIB);
921        let rt = runtime();
922
923        let cases: [(&str, &[Range<usize>]); 6] = [
924            ("disjoint", &[0..100, 500..600, 900..1000]),
925            ("unsorted", &[900..1000, 0..100, 500..600]),
926            ("partial overlap", &[0..100, 50..150]),
927            ("contained", &[0..200, 50..100]),
928            // A zero-length column chunk is declarable in file metadata.
929            ("empty range at a span end", &[0..100, 100..100]),
930            // Spans become [0..4096, 5M..14M, 13M..16M]: the last two overlap,
931            // and both contain 13.5M, but only the third covers 16M.
932            (
933                "overlapping spans",
934                &[
935                    0..4096,
936                    5_000_000..14_000_000,
937                    13_000_000..13_500_000,
938                    13_500_000..16_000_000,
939                ],
940            ),
941        ];
942
943        for (name, ranges) in cases {
944            let mut ranges = ranges.to_vec();
945            let expect = ranges.clone();
946            let out = rt.block_on(source.get_ranges(&mut ranges)).unwrap();
947
948            assert_eq!(out.len(), expect.len(), "{name}");
949            for r in &expect {
950                let got = out
951                    .get(&r.start)
952                    .unwrap_or_else(|| panic!("{name}: {r:?} missing"));
953                assert_eq!(got.as_ref(), &contents[r.clone()], "{name}: {r:?}");
954            }
955        }

973    #[test]
974    fn is_cached_follows_the_page_cache() {
975        let dir = tempfile::tempdir().unwrap();
976        let (source, _) = source(dir.path(), 4 * MIB);
977        let ranges = [0..100, MIB + 1..2 * MIB + 1, 4 * MIB - 1..4 * MIB, 5..5];
978
979        // Just written, so cached; `false` means the kernel lacks cachestat(2).
980        if !source.is_cached(&ranges) {
981            return;
982        }
983        assert!(source.is_cached(&[]));
984
985        // Drop the pages. tmpfs keeps them, so only check where they can go.
986        source.file.sync_all().unwrap();
987        unsafe { libc::posix_fadvise(source.file.as_raw_fd(), 0, 0, libc::POSIX_FADV_DONTNEED) };
988        let mut st: libc::statfs = unsafe { std::mem::zeroed() };
989        let path = std::ffi::CString::new(dir.path().as_os_str().as_encoded_bytes()).unwrap();
990        assert_eq!(unsafe { libc::statfs(path.as_ptr(), &mut st) }, 0);
991        if st.f_type == libc::TMPFS_MAGIC {
992            return;
993        }
994        assert!(!source.is_cached(&ranges));
995
996        let read = MIB + 1..2 * MIB + 1;
997        source.read_blocking(read.clone()).unwrap();
998        assert!(source.is_cached(std::slice::from_ref(&read)));
999        assert!(!source.is_cached(&ranges));
1000    }
