36/// A value's path in a document, for errors.
37#[derive(Clone, Copy)]
38pub(crate) struct KeyPath<'a> {
39    parent: Option<&'a KeyPath<'a>>,
40    /// `None` for a list element.
41    key: Option<&'a str>,
42}
43
44impl<'a> KeyPath<'a> {
45    pub(crate) fn root(key: &'a str) -> Self {
46        Self {
47            parent: None,
48            key: Some(key),
49        }
50    }
51
52    fn child(&'a self, key: Option<&'a str>) -> Self {
53        Self {
54            parent: Some(self),
55            key,
56        }
57    }
58}

60impl Display for KeyPath<'_> {
61    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
62        if let Some(parent) = self.parent {
63            parent.fmt(f)?;
64        }
65        match self.key {
66            Some(key) if self.parent.is_some() => write!(f, ".{key}"),
67            Some(key) => f.write_str(key),
68            None => f.write_str("[]"),
69        }
70    }
71}

81    #[inline]
82    pub(crate) fn add(&mut self, value: &Value) -> PolarsResult<()> {
83        use AnyValueBuffer::*;
84        match &mut self.buf {
85            _ if value.is_null() => {
86                self.buf.add(AnyValue::Null);
87                Ok(())
88            },
89            Boolean(buf) => {
90                match value {
91                    Value::Static(StaticNode::Bool(v)) => buf.append_value(*v),
92                    Value::Static(StaticNode::Null) => buf.append_null(),
93                    _ if self.ignore_errors => buf.append_null(),
94                    v => {
95                        polars_bail!(ComputeError: "cannot parse '{}' ({}) as Boolean", v, v.value_type())
96                    },
97                }
98                Ok(())
99            },
100            #[cfg(feature = "dtype-i8")]
101            Int8(buf) => {
102                let v = deserialize_numeric::<Int8Type>(value, value.as_i8(), self.ignore_errors)?;
103                buf.append_option(v);
104                Ok(())
105            },
106            #[cfg(feature = "dtype-i16")]
107            Int16(buf) => {
108                let v =
109                    deserialize_numeric::<Int16Type>(value, value.as_i16(), self.ignore_errors)?;
110                buf.append_option(v);
111                Ok(())
112            },
113            #[cfg(feature = "dtype-u8")]
114            UInt8(buf) => {
115                let v = deserialize_numeric::<UInt8Type>(value, value.as_u8(), self.ignore_errors)?;
116                buf.append_option(v);
117                Ok(())
118            },
119            #[cfg(feature = "dtype-u16")]
120            UInt16(buf) => {
121                let v =
122                    deserialize_numeric::<UInt16Type>(value, value.as_u16(), self.ignore_errors)?;
123                buf.append_option(v);
124                Ok(())
125            },
126            Int32(buf) => {
127                let v =
128                    deserialize_numeric::<Int32Type>(value, value.as_i32(), self.ignore_errors)?;
129                buf.append_option(v);
130                Ok(())
131            },
132            Int64(buf) => {
133                let v =
134                    deserialize_numeric::<Int64Type>(value, value.as_i64(), self.ignore_errors)?;
135                buf.append_option(v);
136                Ok(())
137            },
138            UInt64(buf) => {
139                let v =
140                    deserialize_numeric::<UInt64Type>(value, value.as_u64(), self.ignore_errors)?;
141                buf.append_option(v);
142                Ok(())
143            },
144            UInt32(buf) => {
145                let v =
146                    deserialize_numeric::<UInt32Type>(value, value.as_u32(), self.ignore_errors)?;
147                buf.append_option(v);
148                Ok(())
149            },
150            Float32(buf) => {
151                let v = deserialize_numeric::<Float32Type>(
152                    value,
153                    value.cast_f64().map(|f| f as f32),
154                    self.ignore_errors,
155                )?;
156                buf.append_option(v);
157                Ok(())
158            },
159            Float64(buf) => {
160                let v = deserialize_numeric::<Float64Type>(
161                    value,
162                    value.cast_f64(),
163                    self.ignore_errors,
164                )?;
165                buf.append_option(v);
166                Ok(())
167            },
168            String(buf) => {
169                match value {
170                    Value::String(v) => buf.append_value(v),
171                    // Forcibly convert to String using the Display impl.
172                    v => buf.append_value(format_pl_smallstr!("{}", ValueDisplay(v))),
173                }
174                Ok(())
175            },
176            #[cfg(feature = "dtype-datetime")]
177            Datetime(buf, tu, _) => {
178                let v =
179                    deserialize_datetime::<Int64Type>(value, "Datetime", self.ignore_errors, *tu)?;
180                buf.append_option(v);
181                Ok(())
182            },
183            #[cfg(feature = "dtype-date")]
184            Date(buf) => {
185                let v = deserialize_datetime::<Int32Type>(
186                    value,
187                    "Date",
188                    self.ignore_errors,
189                    TimeUnit::Microseconds, // ignored
190                )?;
191                buf.append_option(v);
192                Ok(())
193            },
194            All(dtype, buf) => {
195                let path = KeyPath::root(self.name);
196                let av = deserialize_all(value, dtype, self.ignore_errors, self.extra_keys, path)?;
197                buf.push(av);
198                Ok(())
199            },
200            Null(builder) => {
201                if !(matches!(value, Value::Static(StaticNode::Null)) || self.ignore_errors) {
202                    polars_bail!(ComputeError: "got non-null value for NULL-typed column: {}", value)
203                };
204
205                builder.append_null();
206                Ok(())
207            },
208            #[allow(unreachable_patterns)]
209            _ => polars_bail!(
210                ComputeError: "unsupported dtype for column '{}' when reading NDJSON",
211                self.name
212            ),
213        }
214    }

217        self.buf.add(AnyValue::Null).expect("should not fail");
218    }

220pub(crate) fn init_buffers<'a>(
221    schema: &'a Schema,
222    capacity: usize,
223    ignore_errors: bool,
224    extra_keys: &'a ExtraKeysCheck,
225) -> PolarsResult<PlIndexMap<BufferKey<'a>, Buffer<'a>>> {
226    schema
227        .iter()
228        .enumerate()
229        .map(|(i, (name, dtype))| {
230            let av_buf = (dtype, capacity).into();
231            let key = KnownKey::from(name.as_str());
232            let check_nested = extra_keys.nested.get(i).copied().unwrap_or(false);
233            Ok((
234                BufferKey(key),
235                Buffer {
236                    name,
237                    buf: av_buf,
238                    ignore_errors,
239                    extra_keys: check_nested.then_some(extra_keys),
240                },
241            ))
242        })
243        .collect()
244}

246fn deserialize_numeric<T: PolarsNumericType>(
247    value: &Value,
248    n: Option<T::Native>,
249    ignore_errors: bool,
250) -> PolarsResult<Option<T::Native>> {
251    match n {
252        Some(v) => Ok(Some(v)),
253        None if ignore_errors => Ok(None),
254        None => Err(
255            polars_err!(ComputeError: "cannot parse '{}' ({}) as {:?}", value, value.value_type(), T::get_static_dtype()),
256        ),
257    }
258}

260#[cfg(feature = "dtype-datetime")]
261fn deserialize_datetime<T>(
262    value: &Value,
263    type_name: &str,
264    ignore_errors: bool,
265    tu: TimeUnit,
266) -> PolarsResult<Option<T::Native>>
267where
268    T: PolarsNumericType,
269    DatetimeInfer<T>: TryFromWithUnit<Pattern>,
270{
271    match value {
272        Value::String(val) => {
273            if let Some(pattern) = infer_pattern_single(val) {
274                if let Ok(mut infer) = DatetimeInfer::try_from_with_unit(pattern, Some(tu)) {
275                    if let Some(v) = infer.parse(val) {
276                        return Ok(Some(v));
277                    }
278                }
279            }
280        },
281        Value::Static(StaticNode::Null) => return Ok(None),
282        _ => {},
283    };
284
285    if ignore_errors {
286        return Ok(None);
287    }
288
289    polars_bail!(ComputeError: "cannot parse '{}' ({}) as {}", value, value.value_type(), type_name)
290}

292fn deserialize_all<'a>(
293    json: &Value,
294    dtype: &DataType,
295    ignore_errors: bool,
296    extra_keys: Option<&ExtraKeysCheck>,
297    path: KeyPath<'_>,
298) -> PolarsResult<AnyValue<'a>> {
299    if let Value::Static(StaticNode::Null) = json {
300        return Ok(AnyValue::Null);
301    }
302    match dtype {
303        #[cfg(feature = "dtype-datetime")]
304        DataType::Date => {
305            let value = deserialize_datetime::<Int32Type>(
306                json,
307                "Date",
308                ignore_errors,
309                TimeUnit::Microseconds, // ignored
310            )?;
311            return Ok(if let Some(value) = value {
312                AnyValue::Date(value)
313            } else {
314                AnyValue::Null
315            });
316        },
317        #[cfg(feature = "dtype-datetime")]
318        DataType::Datetime(tu, tz) => {
319            let value = deserialize_datetime::<Int64Type>(json, "Datetime", ignore_errors, *tu)?;
320            return Ok(if let Some(value) = value {
321                AnyValue::DatetimeOwned(value, *tu, tz.as_ref().map(|s| Arc::from(s.clone())))
322            } else {
323                AnyValue::Null
324            });
325        },
326        #[cfg(feature = "dtype-f16")]
327        dt @ DataType::Float16 => {
328            use num_traits::AsPrimitive;
329            use polars_utils::float16::pf16;
330
331            return match json.cast_f64() {
332                Some(v) => Ok(AnyValue::Float16(AsPrimitive::<pf16>::as_(v))),
333                None if ignore_errors => Ok(AnyValue::Null),
334                None => {
335                    polars_bail!(ComputeError: "cannot parse '{}' ({}) as {}", json, json.value_type(), dt)
336                },
337            };
338        },
339        dt @ DataType::Float32 => {
340            return match json.cast_f64() {
341                Some(v) => Ok(AnyValue::Float32(v as f32)),
342                None if ignore_errors => Ok(AnyValue::Null),
343                None => {
344                    polars_bail!(ComputeError: "cannot parse '{}' ({}) as {}", json, json.value_type(), dt)
345                },
346            };
347        },
348        dt @ DataType::Float64 => {
349            return match json.cast_f64() {
350                Some(v) => Ok(AnyValue::Float64(v)),
351                None if ignore_errors => Ok(AnyValue::Null),
352                None => {
353                    polars_bail!(ComputeError: "cannot parse '{}' ({}) as {}", json, json.value_type(), dt)
354                },
355            };
356        },
357        DataType::String => {
358            return Ok(match json {
359                Value::String(s) => AnyValue::StringOwned(s.as_ref().into()),
360                v => AnyValue::StringOwned(format_pl_smallstr!("{}", ValueDisplay(v))),
361            });
362        },
363        dt if dt.is_primitive_numeric() => {
364            return match json.as_i128() {
365                Some(v) => Ok(AnyValue::Int128(v).into_static()),
366                None if ignore_errors => Ok(AnyValue::Null),
367                None => {
368                    polars_bail!(ComputeError: "cannot parse '{}' ({}) as {}", json, json.value_type(), dt)
369                },
370            };
371        },
372        _ => {},
373    }
374
375    let out = match json {
376        Value::Static(StaticNode::Bool(b)) => AnyValue::Boolean(*b),
377        Value::Static(StaticNode::I64(i)) => AnyValue::Int64(*i),
378        Value::Static(StaticNode::U64(u)) => AnyValue::UInt64(*u),
379        Value::Static(StaticNode::F64(f)) => AnyValue::Float64(*f),
380        Value::String(s) => AnyValue::StringOwned(s.as_ref().into()),
381        Value::Array(arr) => {
382            let Some(inner_dtype) = dtype.inner_dtype() else {
383                if ignore_errors {
384                    return Ok(AnyValue::Null);
385                }
386                polars_bail!(ComputeError: "expected dtype '{}' in JSON value, got dtype: Array\n\nEncountered value: {}", dtype, json);
387            };
388            let path = path.child(None);
389            let vals: Vec<AnyValue> = arr
390                .iter()
391                .map(|val| deserialize_all(val, inner_dtype, ignore_errors, extra_keys, path))
392                .collect::<PolarsResult<_>>()?;
393            let strict = !ignore_errors;
394            let s =
395                Series::from_any_values_and_dtype(PlSmallStr::EMPTY, &vals, inner_dtype, strict)?;
396            AnyValue::List(s)
397        },
398        #[cfg(feature = "dtype-struct")]
399        Value::Object(doc) => {
400            if let DataType::Struct(fields) = dtype {
401                let document = &**doc;
402
403                let mut n_found = 0;
404                let mut vals = Vec::with_capacity(fields.len());
405                for field in fields {
406                    vals.push(match document.get(field.name.as_str()) {
407                        Some(value) => {
408                            n_found += 1;
409                            let path = path.child(Some(field.name.as_str()));
410                            deserialize_all(value, &field.dtype, ignore_errors, extra_keys, path)?
411                        },
412                        None => AnyValue::Null,
413                    });
414                }
415                if let Some(extra_keys) = extra_keys
416                    && n_found < document.len()
417                    && let Some(key) = document
418                        .keys()
419                        .find(|k| !fields.iter().any(|f| f.name.as_str() == k.as_ref()))
420                {
421                    return Err(extra_keys.err(path.child(Some(key.as_ref()))));
422                }
423                AnyValue::StructOwned(Box::new((vals, fields.clone())))
424            } else {
425                if ignore_errors {
426                    return Ok(AnyValue::Null);
427                }
428                polars_bail!(
429                    ComputeError: "expected {} in json value, got object", dtype,
430                );
431            }
432        },
433        val => AnyValue::StringOwned(format!("{val:#?}").into()),
434    };
435    Ok(out)
436}

438/// Wrapper for serde_json's `Value` with a human-friendly Display impl for nested types:
439///
440/// * Default: `{"x": Static(U64(1))}`
441/// * ValueDisplay: `{x: 1}`
442///
443/// This is intended for reading in arbitrary `Value` types into a String type. Note that the output
444/// is not guaranteed to be valid JSON as we don't do any escaping of e.g. quote/newline values.
445struct ValueDisplay<'a>(&'a Value<'a>);
446
447impl std::fmt::Display for ValueDisplay<'_> {
448    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
449        use std::fmt::Display;
450
451        use Value::*;
452
453        match self.0 {
454            Static(s) => Display::fmt(s, f),
455            String(s) => {
456                f.write_char('"')?;
457
458                let s: &mut &[u8] = &mut s.as_bytes();
459
460                while !s.is_empty() {
461                    f.write_str({
462                        let i = memchr::memchr3(b'"', b'\n', b'\r', s);
463
464                        // Safety: If `i` is `Some(_)`, it points to an ASCII char.
465                        unsafe {
466                            str::from_utf8_unchecked(s.split_off(..i.unwrap_or(s.len())).unwrap())
467                        }
468                    })?;
469
470                    if let Some(&[c]) = s.split_off(..1) {
471                        match c {
472                            b'"' => f.write_str(r#"\""#)?,
473                            b'\n' => f.write_str(r#"\n"#)?,
474                            b'\r' => f.write_str(r#"\r"#)?,
475                            _ => unreachable!(),
476                        }
477                    }
478                }
479
480                f.write_char('"')?;
481
482                Ok(())
483            },
484            Array(a) => {
485                f.write_char('[')?;
486
487                let mut iter = a.iter();
488
489                for v in (&mut iter).take(1) {
490                    ValueDisplay(v).fmt(f)?;
491                }
492
493                for v in iter {
494                    f.write_str(", ")?;
495                    ValueDisplay(v).fmt(f)?;
496                }
497
498                f.write_char(']')
499            },
500            Object(o) => {
501                f.write_char('{')?;
502
503                let mut iter = o.iter();
504
505                for (k, v) in (&mut iter).take(1) {
506                    f.write_char('"')?;
507
508                    f.write_str(k)?;
509                    f.write_str(r#"": "#)?;
510                    ValueDisplay(v).fmt(f)?;
511                }
512
513                for (k, v) in iter {
514                    f.write_str(r#", ""#)?;
515
516                    f.write_str(k)?;
517                    f.write_str(r#"": "#)?;
518                    ValueDisplay(v).fmt(f)?;
519                }
520
521                f.write_char('}')
522            },
523        }
524    }
