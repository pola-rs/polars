10/// Materializes hive partitions.
11/// We have a special num_rows arg, as df can be empty when a projection contains
12/// only hive partition columns.
13///
14/// The `hive_partition_columns` must be ordered by their position in the `reader_schema`. The
15/// columns will be materialized by their positions in the file schema if they exist, or otherwise
16/// at the end.
17///
18/// # Safety
19///
20/// num_rows equals the height of the df when the df height is non-zero.
21pub(crate) fn materialize_hive_partitions<F, M>(
22    df: &mut DataFrame,
23    reader_schema: &polars_schema::Schema<F, M>,
24    hive_partition_columns: Option<&[Series]>,
25) {
26    let num_rows = df.height();
27
28    if let Some(hive_columns) = hive_partition_columns {
29        // Insert these hive columns in the order they are stored in the file.
30        if hive_columns.is_empty() {
31            return;
32        }
33
34        let hive_columns = hive_columns
35            .iter()
36            .map(|s| ScalarColumn::new(s.name().clone(), s.first(), num_rows).into())
37            .collect::<Vec<Column>>();
38
39        if reader_schema.index_of(hive_columns[0].name()).is_none() || df.width() == 0 {
40            // Fast-path - all hive columns are at the end
41            if df.width() == 0 {
42                unsafe { df.set_height(num_rows) };
43            }
44            unsafe { df.hstack_mut_unchecked(&hive_columns) };
45            return;
46        }
47
48        let mut merged = Vec::with_capacity(df.width() + hive_columns.len());
49
50        // `hive_partitions_from_paths()` guarantees `hive_columns` is sorted by their appearance in `reader_schema`.
51        merge_sorted_to_schema_order(
52            &mut unsafe { df.columns_mut() }.drain(..),
53            &mut hive_columns.into_iter(),
54            reader_schema,
55            &mut merged,
56        );
57
58        *df = unsafe { DataFrame::new_unchecked(num_rows, merged) };
59    }
60}

62/// Merge 2 lists of columns into one, where each list contains columns ordered such that their indices
63/// in the `schema` are in ascending order.
64///
65/// Layouts:
66/// * `cols_lhs`: `[row_index?, ..schema_columns?, ..other_left?]`
67///   * If the first item in `cols_lhs` is not found in the schema, it will be assumed to be a
68///     `row_index` column and placed first into the result.
69/// * `cols_rhs`: `[..schema_columns? ..other_right?]`
70///
71/// Output:
72/// * `[..schema_columns?, ..other_left?, ..other_right?]`
73///
74/// Note: The `row_index` column should be handled before calling this function.
75///
76/// # Panics
77/// Panics if either `cols_lhs` or `cols_rhs` is empty.
78pub fn merge_sorted_to_schema_order<'a, F, M>(
79    cols_lhs: &'a mut dyn Iterator<Item = Column>,
80    cols_rhs: &'a mut dyn Iterator<Item = Column>,
81    schema: &polars_schema::Schema<F, M>,
82    output: &'a mut Vec<Column>,
83) {
84    merge_sorted_to_schema_order_impl(cols_lhs, cols_rhs, output, &|v| schema.index_of(v.name()))
85}

87pub fn merge_sorted_to_schema_order_impl<'a, T, O>(
88    cols_lhs: &'a mut dyn Iterator<Item = T>,
89    cols_rhs: &'a mut dyn Iterator<Item = T>,
90    output: &mut O,
91    get_opt_index: &dyn for<'b> Fn(&'b T) -> Option<usize>,
92) where
93    O: Extend<T>,
94{
95    let mut series_arr = [cols_lhs.peekable(), cols_rhs.peekable()];
96
97    (|| {
98        let (Some(a), Some(b)) = (
99            series_arr[0]
100                .peek()
101                .and_then(|x| get_opt_index(x).or(Some(0))),
102            series_arr[1].peek().and_then(get_opt_index),
103        ) else {
104            return;
105        };
106
107        let mut schema_idx_arr = [a, b];
108
109        loop {
110            // Take from the side whose next column appears earlier in the `schema`.
111            let arg_min = if schema_idx_arr[1] < schema_idx_arr[0] {
112                1
113            } else {
114                0
115            };
116
117            output.extend([series_arr[arg_min].next().unwrap()]);
118
119            let Some(v) = series_arr[arg_min].peek() else {
120                return;
121            };
122
123            let Some(i) = get_opt_index(v) else {
124                // All columns in `cols_lhs` should be present in `schema` except for a row_index column.
125                // We assume that if a row_index column exists it is always the first column and handle that at
126                // initialization.
127                debug_assert_eq!(arg_min, 1);
128                break;
129            };
130
131            schema_idx_arr[arg_min] = i;
132        }
133    })();
134
135    let [a, b] = series_arr;
136    output.extend(a);
137    output.extend(b);
138}

140/// # Panics
141/// The `Display` impl of this will panic if a column has non-unit length.
142pub struct HivePathFormatter<'a> {
143    keys: &'a [Column],
144}
145
146impl<'a> HivePathFormatter<'a> {
147    pub fn new(keys: &'a [Column]) -> Self {
148        Self { keys }
149    }
150}
151
152impl std::fmt::Display for HivePathFormatter<'_> {
153    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
154        for column in self.keys {
155            assert_eq!(column.len(), 1);
156            let column = match column.dtype() {
157                // The `String` cast drops fractional seconds, which would make distinct keys
158                // collide.
159                #[cfg(feature = "dtype-time")]
160                DataType::Time => column
161                    .as_materialized_series()
162                    .time()
163                    .unwrap()
164                    .to_string("%T%.f")
165                    .unwrap()
166                    .into_column(),
167                _ => column.cast(&DataType::String).unwrap(),
168            };
169
170            let key = column.name();
171            let value = percent_encoding::percent_encode(
172                column
173                    .str()
174                    .unwrap()
175                    .get(0)
176                    .unwrap_or("__HIVE_DEFAULT_PARTITION__")
177                    .as_bytes(),
178                HIVE_VALUE_ENCODE_CHARSET,
179            );
180
181            write!(f, "{key}={value}/")?
182        }
183
184        Ok(())
185    }
186}
