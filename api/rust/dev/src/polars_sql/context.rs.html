94/// Disambiguate qualified wildcard columns that conflict with each other or other projections.
95fn disambiguate_projection_cols(
96    items: Vec<ProjectionItem>,
97    schema: &Schema,
98) -> PolarsResult<Vec<(Expr, bool)>> {
99    // Establish qualified wildcard names (with counts), and other expression names
100    let mut qualified_wildcard_names: PlHashMap<PlSmallStr, usize> = PlHashMap::new();
101    let mut other_names: PlHashSet<PlSmallStr> = PlHashSet::new();
102    for item in &items {
103        match item {
104            ProjectionItem::QualifiedExprs(_, exprs) => {
105                for expr in exprs {
106                    if let Some(name) = expr_output_name(expr) {
107                        *qualified_wildcard_names.entry(name.clone()).or_insert(0) += 1;
108                    }
109                }
110            },
111            ProjectionItem::Exprs(exprs, _) => {
112                for expr in exprs {
113                    if let Some(name) = expr_output_name(expr) {
114                        other_names.insert(name.clone());
115                    }
116                }
117            },
118        }
119    }
120
121    // Names requiring disambiguation (duplicates across wildcards, eg: `tbl1.*`,`tbl2.*`)
122    let needs_suffix: PlHashSet<PlSmallStr> = qualified_wildcard_names
123        .into_iter()
124        .filter(|(name, count)| *count > 1 || other_names.contains(name))
125        .map(|(name, _)| name)
126        .collect();
127
128    // Output, applying suffixes where needed
129    let mut result: Vec<(Expr, bool)> = Vec::new();
130    for item in items {
131        match item {
132            ProjectionItem::QualifiedExprs(tbl_name, exprs) if !needs_suffix.is_empty() => {
133                for expr in exprs {
134                    if let Some(name) = expr_output_name(&expr) {
135                        if needs_suffix.contains(name) {
136                            let suffixed = format_pl_smallstr!("{}:{}", name, tbl_name);
137                            if schema.contains(suffixed.as_str()) {
138                                // RENAME can give a column the name of another one.
139                                let expr = match strip_outer_alias(&expr) {
140                                    Expr::Column(name) if name == suffixed => Expr::Column(name),
141                                    e => e.alias(suffixed),
142                                };
143                                result.push((expr, false));
144                                continue;
145                            }
146                            if other_names.contains(name) {
147                                polars_bail!(
148                                    SQLInterface:
149                                    "column '{}' is duplicated in the SELECT (explicitly, and via the `*` wildcard)", name
150                                );
151                            }
152                        }
153                    }
154                    result.push((expr, false));
155                }
156            },
157            ProjectionItem::QualifiedExprs(_, exprs) => {
158                result.extend(exprs.into_iter().map(|e| (e, false)));
159            },
160            ProjectionItem::Exprs(exprs, is_explicit_alias) => {
161                result.extend(exprs.into_iter().map(|e| (e, is_explicit_alias)));
162            },
163        }
164    }
165    Ok(result)
166}

168/// Assign unique output names to SELECT-list projections that would otherwise collide.
169///
170/// SQL allows repeated and derived output names in a SELECT list; Polars frames do not.
171/// The first projection to claim a name keeps it, and a later unaliased projection that
172/// collides gets a `name:n` suffix. Two cases are deliberately left to collide: explicit
173/// `AS` aliases, which are never renamed (`SELECT 1 AS x, 2 AS x`), and an unaliased
174/// expression that verbatim repeats the one already holding the name (`SELECT a, a`),
175/// which is a duplicate selection rather than an incidental clash.
176fn disambiguate_output_names(exprs: Vec<(Expr, bool)>, schema: &Schema) -> PolarsResult<Vec<Expr>> {
177    let protected: PlHashSet<PlSmallStr> = exprs
178        .iter()
179        .filter(|(_, is_explicit_alias)| *is_explicit_alias)
180        .map(|(e, _)| Ok(e.to_field(schema)?.name))
181        .collect::<PolarsResult<_>>()?;
182
183    let mut used: PlHashSet<PlSmallStr> = PlHashSet::new();
184    let mut claimed: PlHashMap<PlSmallStr, Expr> = PlHashMap::new();
185    let mut result = Vec::with_capacity(exprs.len());
186    for (expr, is_explicit_alias) in exprs {
187        let name = expr.to_field(schema)?.name;
188        let verbatim_repeat = claimed.get(&name).is_some_and(|claimant| claimant == &expr);
189        if is_explicit_alias
190            || verbatim_repeat
191            || (!used.contains(&name) && !protected.contains(&name))
192        {
193            if !is_explicit_alias {
194                claimed.entry(name.clone()).or_insert_with(|| expr.clone());
195            }
196            used.insert(name);
197            result.push(expr);
198        } else {
199            let mut n = 1usize;
200            let unique_name = loop {
201                let candidate = format_pl_smallstr!("{}:{}", name, n);
202                if !used.contains(&candidate) && !protected.contains(&candidate) {
203                    break candidate;
204                }
205                n += 1;
206            };
207            used.insert(unique_name.clone());
208            claimed.insert(unique_name.clone(), expr.clone());
209            result.push(expr.alias(unique_name));
210        }
211    }
212    Ok(result)
213}

424    /// The catalog DDL of `query`, if any, for the caller to perform: [`SQLContext::execute`]
425    /// rejects it.
426    pub fn catalog_ddl(&self, query: &str) -> PolarsResult<Option<CatalogDdl>> {
427        let stmt = parse_single_statement(query)?;
428        match &stmt {
429            Statement::Drop {
430                object_type: ObjectType::Table,
431                ..
432            } => {},
433            Statement::CreateTable(create_table) if !create_table.temporary => {
434                return self.catalog_create_table(create_table);
435            },
436            _ => return Ok(None),
437        }
438        let (names, if_exists, purge) = drop_table_parts(&stmt)?;
439        Ok(self
440            .catalog_drop_target(names, purge)?
441            .map(|(name, table)| CatalogDdl::DropTable {
442                name,
443                table,
444                if_exists,
445                purge,
446            }))
447    }

449    /// The catalog table that a `CREATE TABLE` of `name` creates, resolved as in a read; `None`
450    /// for a registered table. A name that a catalog applies to but gives no namespace is an
451    /// error, so it doesn't silently create a registered table.
452    fn catalog_create_target(&self, name: &ObjectName) -> PolarsResult<Option<CatalogTable>> {
453        let Some(parts) = name_parts(name) else {
454            return Ok(None);
455        };
456        let named = self.catalogs.named_catalog(&parts);
457        let Some(catalog) = named.or(self.catalogs.default_catalog()) else {
458            return Ok(None);
459        };
460        if let Some(table) = self.catalogs.resolve(&parts) {
461            return Ok(Some(table));
462        }
463        let temp_hint = match named {
464            Some(_) => "",
465            None => ", or use `CREATE TEMP TABLE` for a local table",
466        };
467        polars_bail!(
468            SQLInterface: "table '{}' has no namespace: catalog '{}' has no default namespace; write '{}.<namespace>.{}'{}",
469            display_name(&parts), catalog, catalog, parts.last().unwrap(), temp_hint
470        )
471    }

473    /// The catalog DDL of a `CREATE TABLE`, if its name resolves to a catalog table.
474    fn catalog_create_table(&self, create_table: &CreateTable) -> PolarsResult<Option<CatalogDdl>> {
475        let Some(table) = self.catalog_create_target(&create_table.name)? else {
476            return Ok(None);
477        };
478        self.validate_create_table(create_table, true)?;
479        let CreateTable {
480            name,
481            if_not_exists,
482            columns,
483            query,
484            like,
485            table_options,
486            hive_formats,
487            ..
488        } = create_table;
489        polars_ensure!(like.is_none(), SQLInterface: "`CREATE TABLE ... LIKE` is not supported for catalog tables");
490        polars_ensure!(query.is_none(), SQLInterface: "`CREATE TABLE ... AS SELECT` is not yet supported for catalog tables");
491        polars_ensure!(!columns.is_empty(), SQLInterface: "CREATE TABLE expected a query, column definitions, or LIKE clause");
492        let (schema, required) = create_table_columns(columns)?;
493        let location = hive_formats.as_ref().and_then(|f| f.location.as_deref());
494        let (location, properties) = create_table_options(table_options, location)?;
495
496        // A registered table of the written or the full name would hide the new table from reads.
497        let tbl_name = object_name_to_string(name);
498        let full_name = table.full_name();
499        if let Some(registered) = self.registered_table_key(&tbl_name)? {
500            let hint = match name.0.len() == full_name.len() {
501                true => "unregister it".to_string(),
502                false => format!(
503                    "write the full name '{}', or unregister it",
504                    display_name(&full_name)
505                ),
506            };
507            polars_bail!(
508                SQLInterface: "table '{}' would be hidden by registered table '{}'; {}",
509                tbl_name, registered, hint
510            );
511        }
512        if let Some(registered) = self.registered_table_key(&full_name.join("."))? {
513            polars_bail!(
514                SQLInterface: "table '{}' would be hidden by registered table '{}'; unregister it",
515                tbl_name, registered
516            );
517        }
518        Ok(Some(CatalogDdl::CreateTable {
519            name: tbl_name,
520            table,
521            if_not_exists: *if_not_exists,
522            schema,
523            required,
524            location,
525            properties,
526        }))
527    }

547    /// Rewrite the catalog table names of `stmt` to their full names (see
548    /// [`rewrite_catalog_names`]): by the attached catalogs, else by the names a deferred
549    /// statement recorded.
550    fn rewrite_catalog_names(
551        &self,
552        stmt: &mut Statement,
553    ) -> PolarsResult<(Vec<RewrittenName>, CatalogSpellings)> {
554        if self.catalogs.is_empty() && self.catalog_names.is_empty() {
555            return Ok(Default::default());
556        }
557        let table_map = self.table_map.read().unwrap();
558        // A name matching several tables ignoring case is left for the lookup to report.
559        let is_local = |name: &str| !matches!(key_ignoring_case(&table_map, name), Ok(None));
560        if self.catalogs.is_empty() {
561            let recorded = |parts: &[String]| {
562                let (_, full) = self.catalog_names.iter().find(|(w, _)| w == parts)?;
563                Some(full.clone())
564            };
565            rewrite_catalog_names(stmt, recorded, is_local)
566        } else {
567            let resolve = |parts: &[String]| self.catalogs.resolve(parts).map(|t| t.full_name());
568            rewrite_catalog_names(stmt, resolve, is_local)
569        }
570    }

573    /// The name resolves as in a read, so a table of the default catalog is also an error.
574    /// With `local_first`, a registered table with the whole name is changed instead, and a
575    /// name starting with an attached catalog that resolves to no table is left to it. With
576    /// `if_exists`, only a name starting with an attached catalog is, so the statement's no-op
577    /// on a missing table applies.
578    fn ensure_not_catalog_table(
579        &self,
580        name: &ObjectName,
581        statement: &str,
582        local_first: bool,
583        if_exists: bool,
584    ) -> PolarsResult<()> {
585        let Some(parts) = name_parts(name) else {
586            return Ok(());
587        };
588        let named = self.catalogs.named_catalog(&parts);
589        let table = match (named, if_exists) {
590            (None, true) => None,
591            _ => self.catalogs.resolve(&parts),
592        };
593        let Some(table) = table else {
594            return match named {
595                Some(catalog) if !local_first => polars_bail!(
596                    SQLInterface: "{} on a table of attached catalog '{}' is not supported{}",
597                    statement, catalog, self.catalogs.unresolved_hint(&parts).unwrap_or_default()
598                ),
599                _ => Ok(()),
600            };
601        };
602        if local_first
603            && self
604                .registered_table_key(&object_name_to_string(name))?
605                .is_some()
606        {
607            return Ok(());
608        }
609        let resolved = match named {
610            Some(_) => String::new(),
611            None => format!(
612                " ('{}' is '{}')",
613                display_name(&parts),
614                display_name(&table.full_name())
615            ),
616        };
617        polars_bail!(
618            SQLInterface: "{} on a table of attached catalog '{}' is not supported{}",
619            statement, table.catalog, resolved
620        )
621    }

623    /// Execute a SQL query, returning a [`LazyFrame`].
624    /// ```rust
625    /// # use polars_sql::SQLContext;
626    /// # use polars_core::prelude::*;
627    /// # use polars_lazy::prelude::*;
628    /// # fn main() {
629    ///
630    /// let mut ctx = SQLContext::new();
631    /// let df = df! {
632    ///    "a" =>  [1, 2, 3],
633    /// }
634    /// .unwrap();
635    ///
636    /// ctx.register("df", df.clone().lazy());
637    /// let sql_df = ctx.execute("SELECT * FROM df").unwrap().collect().unwrap();
638    /// assert!(sql_df.equals(&df));
639    /// # }
640    ///```
641    pub fn execute(&mut self, query: &str) -> PolarsResult<LazyFrame> {
642        let mut stmt = parse_single_statement(query)?;
643
644        if self.can_defer(&stmt) {
645            // The plan keeps the statement as written: translating it rewrites the names of its
646            // catalog tables again, also where it was written each.
647            let catalog_names = match self.catalogs.is_empty() {
648                true => Vec::new(),
649                false => self.rewrite_catalog_names(&mut stmt.clone())?.0,
650            };
651            return Ok(LazyFrame::from(self.defer_query(
652                query,
653                stmt,
654                &catalog_names,
655            )));
656        }
657
658        let (_, catalog_spellings) = self.rewrite_catalog_names(&mut stmt)?;
659        desugar_quantified_subqueries(&mut stmt);
660        self.catalog_spellings = catalog_spellings;
661        self.statement_time = Default::default();
662        let res = self.execute_statement(&stmt).inspect_err(|_| {
663            self.clear_statement_state();
664        })?;
665
666        // Ensure the result uses the proper arenas.
667        // This will instantiate new arenas with a new version.
668        let lp_arena = std::mem::take(&mut self.lp_arena);
669        let expr_arena = std::mem::take(&mut self.expr_arena);
670        res.set_cached_arena(lp_arena, expr_arena);
671
672        self.clear_statement_state();
673        Ok(res)
674    }

683    /// Build a [`DslPlan::SQL`] holding `query` and a snapshot of the relations it references.
684    ///
685    /// `stmt` is kept alongside so that resolving does not parse `query` again.
686    fn defer_query(
687        &self,
688        query: &str,
689        mut stmt: Statement,
690        catalog_names: &[RewrittenName],
691    ) -> DslPlan {
692        // Only registered tables: catalog tables travel separately, with their full names.
693        let table_map = self.table_map.read().unwrap();
694        let relations = statement_table_identifiers(&stmt, true, true)
695            .into_iter()
696            .flat_map(|name| {
697                if let Ok(Some(lf)) = get_ignoring_case(&table_map, &name) {
698                    return vec![(PlSmallStr::from_string(name), lf.logical_plan.clone())];
699                }
700                // Tables matching only ignoring case are all kept, so resolving reports them.
701                table_map
702                    .iter()
703                    .filter(|(key, _)| key.eq_ignore_ascii_case(&name))
704                    .map(|(key, lf)| (PlSmallStr::from_str(key), lf.logical_plan.clone()))
705                    .collect()
706            })
707            .collect();
708
709        // Catalog tables keep their full names, so parsing `query` again can rewrite it.
710        let to_parts = |parts: &[String]| parts.iter().map(|p| PlSmallStr::from_str(p)).collect();
711        let catalog_tables = catalog_names
712            .iter()
713            .filter_map(|(written, full)| {
714                let lf = self.catalog_tables.get(&full.join("."))?;
715                Some((to_parts(written), to_parts(full), lf.logical_plan.clone()))
716            })
717            .collect();
718
719        desugar_quantified_subqueries(&mut stmt);
720        DslPlan::SQL {
721            query: Arc::new(query.to_owned()),
722            relations,
723            catalog_tables,
724            cached_stmt: CachedSqlStatement::new(Arc::new(stmt)),
725        }
726    }

728    /// Execute `query` eagerly against the given arenas, returning the resulting DSL.
729    ///
730    /// Resolves a [`DslPlan::SQL`] node. The caller's arenas are used so that input schemas
731    /// resolve into the arenas of the ongoing conversion. `cached` is the desugared statement
732    /// the node was built with, and `query` is only parsed when it is absent.
733    pub(crate) fn execute_with_arenas(
734        &mut self,
735        query: &str,
736        cached: Option<&Statement>,
737        lp_arena: &mut Arena<IR>,
738        expr_arena: &mut Arena<AExpr>,
739    ) -> PolarsResult<DslPlan> {
740        let rewritten;
741        let stmt = match cached {
742            Some(stmt) if self.catalog_names.is_empty() => stmt,
743            cached => {
744                let mut stmt = match cached {
745                    Some(stmt) => stmt.clone(),
746                    None => {
747                        let mut stmt = parse_single_statement(query)?;
748                        desugar_quantified_subqueries(&mut stmt);
749                        stmt
750                    },
751                };
752                self.catalog_spellings = self.rewrite_catalog_names(&mut stmt)?.1;
753                rewritten = stmt;
754                &rewritten
755            },
756        };
757
758        self.statement_time = Default::default();
759        std::mem::swap(&mut self.lp_arena, lp_arena);
760        std::mem::swap(&mut self.expr_arena, expr_arena);
761        let res = self.execute_statement(stmt);
762        std::mem::swap(&mut self.lp_arena, lp_arena);
763        std::mem::swap(&mut self.expr_arena, expr_arena);
764
765        self.clear_statement_state();
766        Ok(res?.logical_plan)
767    }

864    /// `expr` with each subquery, and each read of its result, replaced by a null of its dtype,
865    /// so that it can be typed before the subqueries are resolved.
866    pub(crate) fn with_typed_subqueries(&mut self, expr: &Expr) -> PolarsResult<Expr> {
867        let mut dtypes = PlHashMap::default();
868        for e in expr {
869            if let Expr::SubPlan(lp, names) = e {
870                for (name, select_expr) in names.iter() {
871                    let mut lf = LazyFrame::from((***lp).clone()).select([select_expr.clone()]);
872                    let schema = self.get_frame_schema(&mut lf)?;
873                    dtypes.insert(name.clone(), schema.get_at_index(0).unwrap().1.clone());
874                }
875            }
876        }
877        if dtypes.is_empty() {
878            return Ok(expr.clone());
879        }
880        let typed_null = |name: &PlSmallStr| lit(Scalar::null(dtypes[name].clone()));
881        Ok(expr.clone().map_expr(|e| match &e {
882            Expr::SubPlan(_, names) => typed_null(&names[0].0),
883            Expr::Column(name) if dtypes.contains_key(name) => typed_null(name),
884            _ => e,
885        }))
886    }

888    /// The relation qualifying a compound identifier (see `match_qualifier`) and the qualifier's
889    /// length. Before a `FROM` is processed, any leading run naming a table qualifies.
890    pub(crate) fn resolve_qualifier(
891        &self,
892        idents: &[Ident],
893    ) -> PolarsResult<Option<(String, usize)>> {
894        if self.active_relations.is_empty() {
895            for len in (1..idents.len()).rev() {
896                let qualifier = join_idents(&idents[..len]);
897                if self.get_relation_frame(&qualifier)?.is_some() {
898                    return Ok(Some((qualifier, len)));
899                }
900            }
901            return Ok(None);
902        }
903        match match_qualifier(idents, self.active_relations.iter()) {
904            None => Ok(None),
905            Some((matches, len)) => match matches.as_slice() {
906                [relation] => {
907                    let key = relation.key();
908                    // Relations are looked up by their dotted name, which `s.t` and an alias
909                    // `"s.t"` share.
910                    polars_ensure!(
911                        !self.active_relations.iter().any(|r| r.key() == key && r != *relation),
912                        SQLInterface: "ambiguous reference to table \"{}\" (another relation in FROM has the same dotted name; use a different alias)",
913                        key
914                    );
915                    Ok(Some((key, len)))
916                },
917                _ => Err(ambiguous_relation_err(
918                    &join_idents(&idents[..len]),
919                    &matches,
920                )),
921            },
922        }
923    }

1011    fn expr_or_ordinal(
1012        &mut self,
1013        e: &SQLExpr,
1014        exprs: &[Expr],
1015        selected: Option<&[Expr]>,
1016        schema: Option<&Schema>,
1017        clause: &str,
1018    ) -> PolarsResult<Expr> {
1019        match e {
1020            SQLExpr::UnaryOp {
1021                op: SQLUnaryOperator::Minus,
1022                expr,
1023            } if matches!(
1024                **expr,
1025                SQLExpr::Value(ValueWithSpan {
1026                    value: SQLValue::Number(_, _),
1027                    ..
1028                })
1029            ) =>
1030            {
1031                if let SQLExpr::Value(ValueWithSpan {
1032                    value: SQLValue::Number(ref idx, _),
1033                    ..
1034                }) = **expr
1035                {
1036                    Err(polars_err!(
1037                    SQLSyntax:
1038                    "negative ordinal values are invalid for {}; found -{}",
1039                    clause,
1040                    idx
1041                    ))
1042                } else {
1043                    unreachable!()
1044                }
1045            },
1046            SQLExpr::Value(ValueWithSpan {
1047                value: SQLValue::Number(idx, _),
1048                ..
1049            }) => {
1050                // note: sql queries are 1-indexed
1051                let idx = idx.parse::<usize>().map_err(|_| {
1052                    polars_err!(
1053                        SQLSyntax:
1054                        "negative ordinal values are invalid for {}; found {}",
1055                        clause,
1056                        idx
1057                    )
1058                })?;
1059                // note: "selected" cols represent final projection order, so we use those for
1060                // ordinal resolution. "exprs" may include cols that are subsequently dropped.
1061                let cols = if let Some(cols) = selected {
1062                    cols
1063                } else {
1064                    exprs
1065                };
1066                Ok(cols
1067                    .get(idx - 1)
1068                    .ok_or_else(|| {
1069                        polars_err!(
1070                            SQLInterface:
1071                            "{} ordinal value must refer to a valid column; found {}",
1072                            clause,
1073                            idx
1074                        )
1075                    })?
1076                    .clone())
1077            },
1078            SQLExpr::Value(v) => Err(polars_err!(
1079                SQLSyntax:
1080                "{} requires a valid expression or positive ordinal; found {}", clause, v,
1081            )),
1082            _ => {
1083                // Handle qualified cross-aliasing in ORDER BY clauses
1084                // (eg: `SELECT a AS b, -b AS a ... ORDER BY self.a`)
1085                let mut expr = parse_sql_expr(e, self, schema)?;
1086                if matches!(e, SQLExpr::CompoundIdentifier(_)) {
1087                    if let Some(schema) = schema {
1088                        expr = expr.map_expr(|ex| match &ex {
1089                            Expr::Column(name) => {
1090                                let prefixed = format!("__POLARS_ORIG_{}", name.as_str());
1091                                if schema.contains(prefixed.as_str()) {
1092                                    col(prefixed)
1093                                } else {
1094                                    ex
1095                                }
1096                            },
1097                            _ => ex,
1098                        });
1099                    }
1100                }
1101                Ok(expr)
1102            },
1103        }
1104    }

1106    /// Track the columns of the joined relation `r_name` that also exist on the left
1107    /// side, and so are stored under an aliased name in the joined frame. The other
1108    /// relations of the right input (`right_relations`) refer to these columns too.
1109    pub(crate) fn record_joined_aliases(
1110        &mut self,
1111        r_name: &str,
1112        right_relations: &[String],
1113        left_schema: &Schema,
1114        right_schema: &Schema,
1115        joined_schema: &Schema,
1116    ) -> PolarsResult<()> {
1117        let renamed: PlHashMap<String, String> = right_schema
1118            .iter_names()
1119            .filter(|name| left_schema.contains(name))
1120            .filter_map(|name| {
1121                let aliased_name = format!("{name}:{r_name}");
1122                joined_schema
1123                    .contains(aliased_name.as_str())
1124                    .then(|| (name.to_string(), aliased_name))
1125            })
1126            .collect();
1127        for relation in right_relations {
1128            if relation == r_name {
1129                continue;
1130            }
1131            let Some(mut relation_lf) = self.get_relation_frame(relation)? else {
1132                continue;
1133            };
1134            let relation_schema = self.get_frame_schema(&mut relation_lf)?;
1135            let aliases = self.joined_aliases.entry(relation.clone()).or_default();
1136            for name in aliases.values_mut() {
1137                if let Some(aliased_name) = renamed.get(name.as_str()) {
1138                    *name = aliased_name.clone();
1139                }
1140            }
1141            for (name, aliased_name) in &renamed {
1142                if relation_schema.contains(name) && !aliases.contains_key(name) {
1143                    aliases.insert(name.clone(), aliased_name.clone());
1144                }
1145            }
1146        }
1147        self.joined_aliases.insert(r_name.to_string(), renamed);
1148        Ok(())
1149    }

1235    fn process_query(&mut self, expr: &SetExpr, query: &Query) -> PolarsResult<LazyFrame> {
1236        match expr {
1237            SetExpr::Select(select_stmt) => self.execute_select(select_stmt, query),
1238            SetExpr::Query(nested_query) => {
1239                // The CTEs of a parenthesized query stay visible to the ORDER BY wrapping it.
1240                let lf = self.execute_query(nested_query)?;
1241                self.process_order_by(lf, &query.order_by, None)
1242            },
1243            SetExpr::SetOperation {
1244                op: SetOperator::Union,
1245                set_quantifier,
1246                left,
1247                right,
1248            } => self.process_union(left, right, set_quantifier, query),
1249
1250            #[cfg(feature = "semi_anti_join")]
1251            SetExpr::SetOperation {
1252                op: op @ (SetOperator::Intersect | SetOperator::Except),
1253                set_quantifier,
1254                left,
1255                right,
1256            } => self.process_except_intersect(left, right, op, set_quantifier, query),
1257
1258            SetExpr::Values(Values {
1259                explicit_row: _,
1260                rows,
1261                value_keyword: _,
1262            }) => self.process_values(rows.iter().map(|p| &p.content)),
1263
1264            SetExpr::Table(tbl) => {
1265                if let Some(table_name) = tbl.table_name.as_ref() {
1266                    let name = match &tbl.schema_name {
1267                        Some(schema_name) => format!("{schema_name}.{table_name}"),
1268                        None => table_name.clone(),
1269                    };
1270                    self.get_table_from_current_scope(&name)?.ok_or_else(|| {
1271                        polars_err!(
1272                            SQLInterface: "no table or alias named '{}' found",
1273                            tbl
1274                        )
1275                    })
1276                } else {
1277                    polars_bail!(SQLInterface: "'TABLE' requires valid table name")
1278                }
1279            },
1280            op => {
1281                polars_bail!(SQLInterface: "'{}' operation is currently unsupported", op)
1282            },
1283        }
1284    }

1286    #[cfg(feature = "semi_anti_join")]
1287    fn process_except_intersect(
1288        &mut self,
1289        left: &SetExpr,
1290        right: &SetExpr,
1291        op: &SetOperator,
1292        quantifier: &SetQuantifier,
1293        query: &Query,
1294    ) -> PolarsResult<LazyFrame> {
1295        let (join_type, op_name) = match op {
1296            SetOperator::Except => (JoinType::Anti, "EXCEPT"),
1297            SetOperator::Intersect => (JoinType::Semi, "INTERSECT"),
1298            _ => unreachable!("process_except_intersect is only reached for EXCEPT/INTERSECT"),
1299        };
1300
1301        // Note: each side of the EXCEPT/INTERSECT operation should execute
1302        // in isolation to prevent context state leakage between them
1303        let operand_query = Self::set_op_operand_query(query);
1304        let mut lf = self.execute_isolated(|ctx| ctx.process_query(left, &operand_query))?;
1305        let mut rf = self.execute_isolated(|ctx| ctx.process_query(right, &operand_query))?;
1306        let lf_schema = self.get_frame_schema(&mut lf)?;
1307
1308        let lf_cols: Vec<_> = lf_schema.iter_names_cloned().map(col).collect();
1309        let bag_semantics = matches!(quantifier, SetQuantifier::All);
1310        let rf_cols = match quantifier {
1311            SetQuantifier::ByName => None,
1312            SetQuantifier::Distinct | SetQuantifier::None | SetQuantifier::All => {
1313                let rf_schema = self.get_frame_schema(&mut rf)?;
1314                let rf_cols: Vec<_> = rf_schema.iter_names_cloned().map(col).collect();
1315                if lf_cols.len() != rf_cols.len() {
1316                    polars_bail!(SQLInterface: "{} requires equal number of columns in each table (use '{} BY NAME' to combine mismatched tables)", op_name, op_name)
1317                }
1318                Some(rf_cols)
1319            },
1320            _ => {
1321                polars_bail!(SQLInterface: "'{} {}' is not supported", op_name, quantifier.to_string())
1322            },
1323        };
1324
1325        let lf = if bag_semantics {
1326            let occurrence_col = unique_column_name();
1327            let mut lf_on = lf_cols.clone();
1328            let mut rf_on =
1329                rf_cols.expect("ALL quantifier always resolves the right-hand columns above");
1330
1331            lf = lf.with_column(
1332                int_range(lit(0), len(), 1, DataType::Int64)
1333                    .over(lf_cols.clone())?
1334                    .alias(occurrence_col.clone()),
1335            );
1336            rf = rf.with_column(
1337                int_range(lit(0), len(), 1, DataType::Int64)
1338                    .over(rf_on.clone())?
1339                    .alias(occurrence_col.clone()),
1340            );
1341            lf_on.push(col(occurrence_col.clone()));
1342            rf_on.push(col(occurrence_col.clone()));
1343
1344            let joined_tbl = lf
1345                .join_builder()
1346                .with(rf)
1347                .how(join_type)
1348                .join_nulls(true)
1349                .left_on(lf_on)
1350                .right_on(rf_on)
1351                .finish()?;
1352            joined_tbl.drop(cols([occurrence_col]))
1353        } else {
1354            let join = lf.join_builder().with(rf).how(join_type).join_nulls(true);
1355            let joined_tbl = match rf_cols {
1356                Some(rf_cols) => join.left_on(lf_cols).right_on(rf_cols).finish(),
1357                None => join.on(lf_cols).finish(),
1358            }?;
1359            joined_tbl.unique(None, UniqueKeepStrategy::Any)
1360        };
1361        self.process_order_by(lf, &query.order_by, None)
1362    }

1364    fn process_union(
1365        &mut self,
1366        left: &SetExpr,
1367        right: &SetExpr,
1368        quantifier: &SetQuantifier,
1369        query: &Query,
1370    ) -> PolarsResult<LazyFrame> {
1371        let quantifier = *quantifier;
1372
1373        // Note: each side of the UNION operation should execute
1374        // in isolation to prevent context state leakage between them
1375        let operand_query = Self::set_op_operand_query(query);
1376        let mut lf = self.execute_isolated(|ctx| ctx.process_query(left, &operand_query))?;
1377        let mut rf = self.execute_isolated(|ctx| ctx.process_query(right, &operand_query))?;
1378
1379        let opts = UnionArgs {
1380            parallel: true,
1381            to_supertypes: true,
1382            maintain_order: false,
1383            ..Default::default()
1384        };
1385        let lf = match quantifier {
1386            // UNION [ALL | DISTINCT]
1387            SetQuantifier::All | SetQuantifier::Distinct | SetQuantifier::None => {
1388                let lf_schema = self.get_frame_schema(&mut lf)?;
1389                let rf_schema = self.get_frame_schema(&mut rf)?;
1390                if lf_schema.len() != rf_schema.len() {
1391                    polars_bail!(SQLInterface: "UNION requires equal number of columns in each table (use 'UNION BY NAME' to combine mismatched tables)")
1392                }
1393                // rename `rf` columns to match `lf` if they differ; SQL behaves
1394                // positionally on UNION ops (unless using the "BY NAME" qualifier)
1395                if lf_schema.iter_names().ne(rf_schema.iter_names()) {
1396                    rf = rf.rename(rf_schema.iter_names(), lf_schema.iter_names(), true);
1397                }
1398                let concatenated = concat(vec![lf, rf], opts);
1399                match quantifier {
1400                    SetQuantifier::Distinct | SetQuantifier::None => {
1401                        concatenated.map(|lf| lf.unique(None, UniqueKeepStrategy::Any))
1402                    },
1403                    _ => concatenated,
1404                }
1405            },
1406            // UNION ALL BY NAME
1407            #[cfg(feature = "diagonal_concat")]
1408            SetQuantifier::AllByName => concat_lf_diagonal(vec![lf, rf], opts),
1409            // UNION [DISTINCT] BY NAME
1410            #[cfg(feature = "diagonal_concat")]
1411            SetQuantifier::ByName | SetQuantifier::DistinctByName => {
1412                let concatenated = concat_lf_diagonal(vec![lf, rf], opts);
1413                concatenated.map(|lf| lf.unique(None, UniqueKeepStrategy::Any))
1414            },
1415            #[allow(unreachable_patterns)]
1416            _ => {
1417                polars_bail!(SQLInterface: "'UNION {}' is not currently supported", quantifier)
1418            },
1419        }?;
1420
1421        self.process_order_by(lf, &query.order_by, None)
1422    }

1424    /// Process UNNEST as a lateral operation when it contains column references
1425    /// (handles `CROSS JOIN UNNEST(col) AS name` by exploding the referenced col).
1426    fn process_unnest_lateral(
1427        &self,
1428        lf: LazyFrame,
1429        alias: &Option<TableAlias>,
1430        array_exprs: &[SQLExpr],
1431        with_offset: bool,
1432    ) -> PolarsResult<LazyFrame> {
1433        let alias = alias
1434            .as_ref()
1435            .ok_or_else(|| polars_err!(SQLSyntax: "UNNEST table must have an alias"))?;
1436        polars_ensure!(!with_offset, SQLInterface: "UNNEST tables do not (yet) support WITH ORDINALITY|OFFSET");
1437
1438        let (mut explode_cols, mut rename_from, mut rename_to) = (
1439            Vec::with_capacity(array_exprs.len()),
1440            Vec::with_capacity(array_exprs.len()),
1441            Vec::with_capacity(array_exprs.len()),
1442        );
1443        let is_single_col = array_exprs.len() == 1;
1444
1445        for (i, arr_expr) in array_exprs.iter().enumerate() {
1446            let col_name = match arr_expr {
1447                SQLExpr::Identifier(ident) => PlSmallStr::from_str(&ident.value),
1448                SQLExpr::CompoundIdentifier(parts) => {
1449                    PlSmallStr::from_str(&parts.last().unwrap().value)
1450                },
1451                SQLExpr::Array(_) => polars_bail!(
1452                    SQLInterface: "CROSS JOIN UNNEST with both literal arrays and column references is not supported"
1453                ),
1454                other => polars_bail!(
1455                    SQLSyntax: "UNNEST expects column references or array literals, found {:?}", other
1456                ),
1457            };
1458            // alias: column name from "AS t(col)", or table alias
1459            if let Some(name) = alias
1460                .columns
1461                .get(i)
1462                .map(|c| c.name.value.as_str())
1463                .or_else(|| is_single_col.then_some(alias.name.value.as_str()))
1464                .filter(|name| !name.is_empty() && *name != col_name.as_str())
1465            {
1466                rename_from.push(col_name.clone());
1467                rename_to.push(PlSmallStr::from_str(name));
1468            }
1469            explode_cols.push(col_name);
1470        }
1471
1472        let mut lf = lf.explode(
1473            Selector::ByName {
1474                names: Arc::from(explode_cols),
1475                strict: true,
1476            },
1477            ExplodeOptions {
1478                empty_as_null: true,
1479                keep_nulls: true,
1480            },
1481        );
1482        if !rename_from.is_empty() {
1483            lf = lf.rename(rename_from, rename_to, true);
1484        }
1485        Ok(lf)
1486    }

1488    fn process_values<'a>(
1489        &mut self,
1490        values: impl Iterator<Item = &'a Vec<SQLExpr>>,
1491    ) -> PolarsResult<LazyFrame> {
1492        let frame_rows: Vec<Vec<AnyValue>> = values.map(|row| {
1493            row.iter().map(|expr| {
1494                let expr = parse_sql_expr(expr, self, None)?;
1495                match expr {
1496                    Expr::Literal(value) => {
1497                        value.to_any_value()
1498                            .ok_or_else(|| polars_err!(SQLInterface: "invalid literal value: {:?}", value))
1499                            .map(|av| av.into_static())
1500                    },
1501                    _ => polars_bail!(SQLInterface: "VALUES clause expects literals; found {}", expr),
1502                }
1503            }).collect()
1504        }).collect::<PolarsResult<_>>()?;
1505
1506        // Each column takes the supertype of all its rows, not the first row's type.
1507        let width = frame_rows.first().map_or(0, Vec::len);
1508        polars_ensure!(
1509            frame_rows.iter().all(|row| row.len() == width),
1510            SQLSyntax: "VALUES rows must all have the same number of values"
1511        );
1512        let columns = (0..width)
1513            .map(|i| {
1514                let values: Vec<AnyValue> = frame_rows.iter().map(|row| row[i].clone()).collect();
1515                series_from_literals(format_pl_smallstr!("column_{i}"), &values).map(Column::from)
1516            })
1517            .collect::<PolarsResult<Vec<_>>>()?;
1518        Ok(DataFrame::new_infer_height(columns)?.lazy())
1519    }

1546    // DROP TABLE <tbl>
1547    fn execute_drop_table(&mut self, stmt: &Statement) -> PolarsResult<LazyFrame> {
1548        let (names, if_exists, purge) = drop_table_parts(stmt)?;
1549        if let Some((name, table)) = self.catalog_drop_target(names, purge)? {
1550            polars_bail!(
1551                SQLInterface: "DROP TABLE of catalog table '{}' ('{}') must be performed on the catalog (see `SQLContext::catalog_ddl`)",
1552                name, display_name(&table.full_name())
1553            );
1554        }
1555
1556        let mut table_map = self.table_map.write().unwrap();
1557        // Resolve every name before dropping any, so an error drops nothing.
1558        let mut keys = Vec::with_capacity(names.len());
1559        for name in names {
1560            let tbl = object_name_to_string(name);
1561            // `DROP TABLE IF EXISTS <tbl>` is a no-op on a missing table;
1562            // otherwise dropping a table that doesn't exist is an error.
1563            match key_ignoring_case(&table_map, &tbl)?.cloned() {
1564                Some(key) => keys.push(key),
1565                None if if_exists => {},
1566                None => polars_bail!(SQLInterface: "table '{}' does not exist", tbl),
1567            }
1568        }
1569        for key in keys {
1570            table_map.remove(&key);
1571        }
1572        Ok(DataFrame::empty().lazy())
1573    }

1575    /// The catalog table a `DROP TABLE` targets, with its name as written. A registered table

1736                ((0..n).collect(), vec![false; n])
1737            };
1738            let mut reference_counts = PlHashMap::<String, usize>::new();
1739            let mut collector = TableIdentifierCollector::default();
1740            let _ = dyn_visit::visit(query, &mut collector);
1741            for name in collector.tables {
1742                *reference_counts.entry(name).or_insert(0) += 1;
1743            }
1744            // The step of a recursive CTE reads the working table, not the CTE itself.
1745            for (cte, _) in with
1746                .cte_tables
1747                .iter()
1748                .zip(&reads_itself)
1749                .filter(|(_, r)| **r)
1750            {
1751                let name = &cte.alias.name.value;
1752                let mut collector = TableIdentifierCollector::default();
1753                let _ = dyn_visit::visit(&*cte.query, &mut collector);
1754                let n_self = collector.tables.iter().filter(|t| *t == name).count();
1755                if let Some(count) = reference_counts.get_mut(name) {
1756                    *count = count.saturating_sub(n_self);
1757                }
1758            }
1759
1760            for idx in order {
1761                let cte = &with.cte_tables[idx];
1762                let cte_name = cte.alias.name.value.clone();
1763                let mut lf = if reads_itself[idx] {
1764                    self.execute_recursive_cte(cte)?
1765                } else {
1766                    // Note: isolate CTE execution to prevent context state leakage
1767                    let lf = self.execute_isolated(|ctx| ctx.execute_query(&cte.query))?;
1768                    self.rename_columns_from_table_alias(lf, &cte.alias)?
1769                };
1770                if reference_counts.get(&cte_name).copied().unwrap_or(0) > 1 {
1771                    lf = lf.cache();
1772                }
1773                self.register_cte(&cte_name, lf);
1774            }
1775        }
1776        Ok(())
1777    }

1779    /// Plans a CTE that reads itself: `anchor UNION ALL step`, where only `step` reads the CTE.
1780    /// The step reads the rows of the round before.
1781    fn execute_recursive_cte(&mut self, cte: &Cte) -> PolarsResult<LazyFrame> {
1782        let name = &cte.alias.name.value;
1783        let query = &cte.query;
1784        let SetExpr::SetOperation {
1785            op: SetOperator::Union,
1786            set_quantifier,
1787            left,
1788            right,
1789        } = &*query.body
1790        else {
1791            polars_bail!(
1792                SQLInterface: "recursive CTE '{}' must have the form `anchor UNION ALL step`", name
1793            )
1794        };
1795        polars_ensure!(
1796            matches!(set_quantifier, SetQuantifier::All),
1797            SQLInterface: "recursive CTE '{}' only supports UNION ALL", name
1798        );
1799        polars_ensure!(
1800            query.with.is_none()
1801                && query.order_by.is_none()
1802                && query.limit_clause.is_none()
1803                && query.fetch.is_none(),
1804            SQLInterface: "recursive CTE '{}' does not support WITH, ORDER BY, LIMIT or FETCH around its UNION ALL", name
1805        );
1806        let term_query = |body: &SetExpr| Query {
1807            body: Box::new(body.clone()),
1808            ..query.as_ref().clone()
1809        };
1810        let anchor_query = term_query(left);
1811        let step_query = term_query(right);
1812        polars_ensure!(
1813            referenced_ctes(&anchor_query, std::slice::from_ref(name)).is_empty(),
1814            SQLInterface: "the anchor of recursive CTE '{}' may not read '{}'", name, name
1815        );
1816
1817        let anchor = self.execute_isolated(|ctx| ctx.execute_query(&anchor_query))?;
1818        let mut anchor = self.rename_columns_from_table_alias(anchor, &cte.alias)?;
1819        let mut schema = self.get_frame_schema(&mut anchor)?;
1820
1821        // The step runs once per round, so a CTE it reads is cached to compute it only once.
1822        let mut collector = TableIdentifierCollector::default();
1823        let _ = dyn_visit::visit(&step_query, &mut collector);
1824        for (cte_name, lf) in self.cte_map.iter_mut() {
1825            let read_by_step = collector
1826                .tables
1827                .iter()
1828                .any(|t| t.eq_ignore_ascii_case(cte_name) && !t.eq_ignore_ascii_case(name));
1829            if read_by_step && !matches!(lf.logical_plan, DslPlan::Cache { .. }) {
1830                *lf = lf.clone().cache();
1831            }
1832        }
1833        let known_caches = self.cache_ids();
1834        let id = UniqueId::new();
1835        // A column with a NULL anchor takes the type of the step. Each pass can type one more
1836        // column, as a column may get its type from another column of the round before.
1837        let mut step;
1838        let mut passes = 0;
1839        loop {
1840            self.register_cte(
1841                name,
1842                DslPlan::WorkingTable {
1843                    id,
1844                    schema: schema.clone(),
1845                }
1846                .into(),
1847            );
1848            step = self.execute_isolated(|ctx| ctx.execute_query(&step_query))?;
1849            let step_schema = self.get_frame_schema(&mut step)?;
1850            polars_ensure!(
1851                step_schema.len() == schema.len(),
1852                SQLInterface: "the anchor and the step of recursive CTE '{}' have a different number of columns ({} and {})",
1853                name, schema.len(), step_schema.len()
1854            );
1855            let typed_schema: Schema = schema
1856                .iter()
1857                .zip(step_schema.iter_values())
1858                .map(|((col_name, dtype), step_dtype)| {
1859                    let dtype = if dtype.is_null() { step_dtype } else { dtype };
1860                    (col_name.clone(), dtype.clone())
1861                })
1862                .collect();
1863            passes += 1;
1864            if typed_schema == *schema || passes > schema.len() {
1865                break;
1866            }
1867            schema = Arc::new(typed_schema);
1868        }
1869
1870        let cast_to_schema = |lf: LazyFrame, lf_schema: &Schema| {
1871            let exprs: Vec<Expr> = lf_schema
1872                .iter()
1873                .zip(schema.iter())
1874                .map(|((lf_name, lf_dtype), (col_name, dtype))| {
1875                    let e = col(lf_name.clone());
1876                    let e = if lf_dtype == dtype {
1877                        e
1878                    } else {
1879                        e.strict_cast(dtype.clone())
1880                    };
1881                    e.alias(col_name.clone())
1882                })
1883                .collect();
1884            lf.select(exprs)
1885        };
1886        let anchor_schema = self.get_frame_schema(&mut anchor)?;
1887        let anchor = cast_to_schema(anchor, &anchor_schema);
1888        let step_schema = self.get_frame_schema(&mut step)?;
1889        let step = cast_to_schema(step, &step_schema);
1890
1891        let round_local_caches: Arc<[UniqueId]> = (&step.logical_plan)
1892            .into_iter()
1893            .filter_map(|plan| match plan {
1894                DslPlan::Cache { id, .. } if !known_caches.contains(id) => Some(*id),
1895                _ => None,
1896            })
1897            .collect::<PlIndexSet<_>>()
1898            .into_iter()
1899            .collect();
1900
1901        Ok(DslPlan::Recursive {
1902            anchor: Arc::new(anchor.logical_plan),
1903            step: Arc::new(step.logical_plan),
1904            id,
1905            round_local_caches,
1906        }
1907        .into())
1908    }

1964    /// `(w ORDER BY x)` takes the PARTITION BY of `w`, and its ORDER BY if it has one.
1965    /// As in PostgreSQL, it may not replace either of them, and `w` may not have a frame.
1966    fn resolve_window_spec(&self, spec: &WindowSpec) -> PolarsResult<WindowSpec> {
1967        let Some(base_name) = &spec.window_name else {
1968            return Ok(spec.clone());
1969        };
1970        let base = self.named_window(base_name)?;
1971        polars_ensure!(
1972            spec.partition_by.is_empty(),
1973            SQLSyntax: "cannot override PARTITION BY of window '{}'", base_name.value
1974        );
1975        polars_ensure!(
1976            spec.order_by.is_empty() || base.order_by.is_empty(),
1977            SQLSyntax: "cannot override ORDER BY of window '{}'", base_name.value
1978        );
1979        polars_ensure!(
1980            base.window_frame.is_none(),
1981            SQLSyntax: "cannot copy window '{}' because it has a frame clause", base_name.value
1982        );
1983        Ok(WindowSpec {
1984            window_name: None,
1985            partition_by: base.partition_by,
1986            order_by: if spec.order_by.is_empty() {
1987                base.order_by
1988            } else {
1989                spec.order_by.clone()
1990            },
1991            window_frame: spec.window_frame.clone(),
1992        })
1993    }

2007    fn execute_joins(&mut self, tbl_expr: &TableWithJoins) -> PolarsResult<LazyFrame> {
2008        let (l_name, mut lf) = self.get_table(&tbl_expr.relation)?;
2009        let relation_keys = |names: Vec<RelationName>| -> Vec<String> {
2010            names.iter().map(RelationName::key).collect()
2011        };
2012        // The factors whose columns the joins so far return.
2013        let mut kept = vec![&tbl_expr.relation];
2014        if !tbl_expr.joins.is_empty() {
2015            for join in &tbl_expr.joins {
2016                // A join condition only sees the relations joined so far.
2017                self.active_relations = factor_relations(
2018                    kept.iter().copied().chain(std::iter::once(&join.relation)),
2019                    &self.catalog_spellings,
2020                );
2021                let left_relations = relation_keys(factor_relations(
2022                    kept.iter().copied(),
2023                    &self.catalog_spellings,
2024                ));
2025                let right_relations = relation_keys(factor_relations(
2026                    std::iter::once(&join.relation),
2027                    &self.catalog_spellings,
2028                ));
2029                let (keep_left, keep_right) = join_keeps_inputs(&join.join_operator);
2030                if !keep_left {
2031                    kept.clear();
2032                }
2033                if keep_right {
2034                    kept.push(&join.relation);
2035                }
2036                // Handle "CROSS JOIN UNNEST(col)" as a lateral join op
2037                if let (
2038                    JoinOperator::CrossJoin(JoinConstraint::None),
2039                    TableFactor::UNNEST {
2040                        alias,
2041                        array_exprs,
2042                        with_offset,
2043                        ..
2044                    },
2045                ) = (&join.join_operator, &join.relation)
2046                {
2047                    if array_exprs.iter().any(|e| !matches!(e, SQLExpr::Array(_))) {
2048                        lf = self.process_unnest_lateral(lf, alias, array_exprs, *with_offset)?;
2049                        continue;
2050                    }
2051                }
2052
2053                let (r_name, mut rf) = self.get_table(&join.relation)?;
2054                if r_name.is_empty() {
2055                    // Require non-empty to avoid duplicate column errors from nested self-joins.
2056                    polars_bail!(
2057                        SQLInterface:
2058                        "cannot JOIN on unnamed relation; please provide an alias"
2059                    )
2060                }
2061                let left_schema = self.get_frame_schema(&mut lf)?;
2062                let right_schema = self.get_frame_schema(&mut rf)?;
2063
2064                lf = match &join.join_operator {
2065                    op @ (JoinOperator::Join(constraint)  // note: bare "join" is inner
2066                    | JoinOperator::FullOuter(constraint)
2067                    | JoinOperator::Left(constraint)
2068                    | JoinOperator::LeftOuter(constraint)
2069                    | JoinOperator::Right(constraint)
2070                    | JoinOperator::RightOuter(constraint)
2071                    | JoinOperator::Inner(constraint)
2072                    | JoinOperator::Anti(constraint)
2073                    | JoinOperator::Semi(constraint)
2074                    | JoinOperator::LeftAnti(constraint)
2075                    | JoinOperator::LeftSemi(constraint)
2076                    | JoinOperator::RightAnti(constraint)
2077                    | JoinOperator::RightSemi(constraint)) => {
2078                        let keep_right = matches!(
2079                            op,
2080                            JoinOperator::RightAnti(_) | JoinOperator::RightSemi(_)
2081                        );
2082                        self.process_join(
2083                            &TableInfo {
2084                                frame: lf,
2085                                name: (&l_name).into(),
2086                                schema: left_schema.clone(),
2087                                relations: left_relations,
2088                            },
2089                            &TableInfo {
2090                                frame: rf,
2091                                name: (&r_name).into(),
2092                                schema: right_schema.clone(),
2093                                relations: right_relations,
2094                            },
2095                            constraint,
2096                            match op {
2097                                JoinOperator::Join(_) | JoinOperator::Inner(_) => JoinType::Inner,
2098                                JoinOperator::Left(_) | JoinOperator::LeftOuter(_) => {
2099                                    JoinType::Left
2100                                },
2101                                JoinOperator::Right(_) | JoinOperator::RightOuter(_) => {
2102                                    JoinType::Right
2103                                },
2104                                JoinOperator::FullOuter(_) => JoinType::Full,
2105                                #[cfg(feature = "semi_anti_join")]
2106                                JoinOperator::Anti(_)
2107                                | JoinOperator::LeftAnti(_)
2108                                | JoinOperator::RightAnti(_) => JoinType::Anti,
2109                                #[cfg(feature = "semi_anti_join")]
2110                                JoinOperator::Semi(_)
2111                                | JoinOperator::LeftSemi(_)
2112                                | JoinOperator::RightSemi(_) => JoinType::Semi,
2113                                join_type => polars_bail!(
2114                                    SQLInterface:
2115                                    "join type '{:?}' not currently supported",
2116                                    join_type
2117                                ),
2118                            },
2119                            keep_right,
2120                        )?
2121                    },
2122                    #[cfg(feature = "asof_join")]
2123                    JoinOperator::AsOf {
2124                        match_condition,
2125                        constraint,
2126                    } => self.process_asof_join(
2127                        &TableInfo {
2128                            frame: lf,
2129                            name: (&l_name).into(),
2130                            schema: left_schema.clone(),
2131                            relations: left_relations,
2132                        },
2133                        &TableInfo {
2134                            frame: rf,
2135                            name: (&r_name).into(),
2136                            schema: right_schema.clone(),
2137                            relations: right_relations,
2138                        },
2139                        match_condition,
2140                        constraint,
2141                    )?,
2142                    JoinOperator::CrossJoin(JoinConstraint::None) => {
2143                        lf.cross_join(rf, Some(format_pl_smallstr!(":{}", r_name)))
2144                    },
2145                    JoinOperator::CrossJoin(constraint) => {
2146                        polars_bail!(
2147                            SQLInterface:
2148                            "CROSS JOIN does not support {:?} constraint; consider INNER JOIN instead",
2149                            constraint
2150                        )
2151                    },
2152                    join_type => {
2153                        polars_bail!(SQLInterface: "join type '{:?}' not currently supported", join_type)
2154                    },

2166        };

2168    }

2443    /// Register a `GROUPING()` call of the current query block, returning the
2444    /// placeholder column that stands for its value. Repeating a call yields the
2445    /// same placeholder, so every clause of the block refers to one column.

2449            Some(call) => call.placeholder.clone(),

2453                    args,
2454                    placeholder: placeholder.clone(),

2457            },

2542    fn execute_select(&mut self, select_stmt: &Select, query: &Query) -> PolarsResult<LazyFrame> {
2543        let has_group_by = match &select_stmt.group_by {
2544            GroupByExpr::Expressions(exprs, _) => !exprs.is_empty(),
2545            GroupByExpr::All(_) => true,
2546        };
2547        let scope = GroupScope {
2548            mark_whole_frame_windows: has_group_by,
2549            ..Default::default()
2550        };

2557    fn execute_select_block(
2558        &mut self,
2559        select_stmt: &Select,
2560        query: &Query,
2561    ) -> PolarsResult<LazyFrame> {
2562        // Check that the statement doesn't contain unsupported SELECT clauses
2563        self.validate_select(select_stmt)?;
2564
2565        // Parse named windows first, as they may be referenced in the SELECT clause
2566        self.register_named_windows(&select_stmt.named_window)?;
2567
2568        // Get `FROM` table/data
2569        let mut implicit_join_filter: Option<SQLExpr> = None;
2570        let mut lf = if select_stmt.from.is_empty() {
2571            self.active_relations.clear();
2572            DataFrame::new(1, vec![])?.lazy()
2573        } else {
2574            let from = &select_stmt.from;
2575            let first = from.first().unwrap();
2576            // Set before the relations are resolved, so a join constraint
2577            // resolves against them.
2578            self.active_relations = declared_relations(from, &self.catalog_spellings);
2579            let mut lf = self.execute_from_statement(first)?;
2580            if from.len() > 1 {
2581                implicit_join_filter =
2582                    self.process_implicit_joins(&mut lf, from, &select_stmt.selection)?;
2583            }
2584            lf
2585        };
2586
2587        self.set_join_base(&select_stmt.from);
2588        self.check_select_column_refs(select_stmt)?;
2589
2590        // Apply `WHERE` constraint (using residual filter for implicit joins)
2591        let where_expr = if implicit_join_filter.is_some() {
2592            implicit_join_filter.as_ref()
2593        } else {
2594            select_stmt.selection.as_ref()
2595        };
2596        let mut schema = self.get_frame_schema(&mut lf)?;
2597
2598        // Shared by every pass over this frame: WHERE, the projection list and HAVING.
2599        let mut bindings = SubqueryBindings::new();
2600
2601        // A correlated scalar subquery is lowered against the frame as it stands, so
2602        // the conjuncts that don't need it are applied first: the lowering can then
2603        // restrict its aggregate to the rows that survive them.
2604        let mut join_tree = None;
2605        let where_expr: Option<Cow<'_, SQLExpr>> = match where_expr {
2606            Some(where_expr) if expr_contains_scalar_subquery(where_expr) => {
2607                let n_grouping_calls = self.group_scope.grouping_calls.len();
2608                let residual;
2609                (lf, residual) = self.rewrite_subquery_conjuncts(
2610                    lf,
2611                    where_expr,
2612                    FilterMode::KeepTrue,
2613                    &schema,
2614                    RewriteStage::BeforeScalarLowering,
2615                )?;
2616                self.reject_grouping_in_where(n_grouping_calls)?;
2617                join_tree = self.outer_join_tree(select_stmt, where_expr, &residual)?;
2618                schema = self.get_frame_schema(&mut lf)?;
2619                combine_conditions(
2620                    residual.into_iter().cloned().collect(),
2621                    SQLBinaryOperator::And,
2622                )
2623                .map(Cow::Owned)
2624            },
2625            other => other.map(Cow::Borrowed),
2626        };
2627
2628        // Lower correlated scalar subqueries in the WHERE clause before the filter is
2629        // parsed, leaving `[NOT] EXISTS` to `process_where`.
2630        let effective_where = match where_expr.as_deref() {
2631            Some(where_expr) => {
2632                let lowered;
2633                (lf, lowered) = self.lower_correlated_subqueries(
2634                    lf,
2635                    &schema,
2636                    where_expr,
2637                    LowerScope::ScalarOnly,
2638                    &mut bindings,
2639                    join_tree.as_ref(),
2640                )?;
2641                if matches!(lowered, Cow::Owned(_)) {
2642                    schema = self.get_frame_schema(&mut lf)?;
2643                }
2644                Some(lowered)
2645            },
2646            None => None,
2647        };
2648
2649        lf = self.process_where(
2650            lf,
2651            effective_where.as_deref(),
2652            FilterMode::KeepTrue,
2653            Some(schema.clone()),
2654        )?;
2655
2656        // HAVING is parsed further below but lowered here, against the pre-aggregation
2657        // frame.
2658        let mut lowered_projection: Option<Vec<SelectItem>> = None;
2659        let mut lowered_having: Option<Cow<'_, SQLExpr>> = None;
2660        {
2661            let mut changed = false;
2662            for (idx, item) in select_stmt.projection.iter().enumerate() {
2663                let (SelectItem::UnnamedExpr(e) | SelectItem::ExprWithAlias { expr: e, .. }) = item
2664                else {
2665                    continue;
2666                };
2667                let lowered;
2668                (lf, lowered) = self.lower_correlated_subqueries(
2669                    lf,
2670                    &schema,
2671                    e,
2672                    LowerScope::ScalarAndPredicates,
2673                    &mut bindings,
2674                    None,
2675                )?;
2676                if let Cow::Owned(lowered) = lowered {
2677                    changed = true;
2678                    let items =
2679                        lowered_projection.get_or_insert_with(|| select_stmt.projection.clone());
2680                    match &mut items[idx] {
2681                        SelectItem::UnnamedExpr(slot)
2682                        | SelectItem::ExprWithAlias { expr: slot, .. } => *slot = lowered,
2683                        _ => unreachable!("only expression items are lowered"),
2684                    }
2685                }
2686            }
2687            if let Some(having) = &select_stmt.having {
2688                let lowered;
2689                (lf, lowered) = self.lower_correlated_subqueries(
2690                    lf,
2691                    &schema,
2692                    having,
2693                    LowerScope::ScalarAndPredicates,
2694                    &mut bindings,
2695                    None,
2696                )?;
2697                changed |= matches!(lowered, Cow::Owned(_));
2698                lowered_having = Some(lowered);
2699            }
2700            if changed {
2701                schema = self.get_frame_schema(&mut lf)?;
2702            }
2703        }
2704        let projection: &[SelectItem] = lowered_projection
2705            .as_deref()
2706            .unwrap_or(&select_stmt.projection);
2707
2708        if let Some(qualify) = &select_stmt.qualify {
2709            check_qualify(qualify, &select_stmt.projection)?;
2710        }
2711
2712        let mark_whole_frame_windows =
2713            std::mem::replace(&mut self.group_scope.mark_whole_frame_windows, true);
2714        let mut projections_with_flags = self.marking_aggregates(|ctx| {
2715            ctx.column_projections(projection, select_stmt.flavor, &schema)
2716        })?;
2717        self.group_scope.mark_whole_frame_windows = mark_whole_frame_windows;
2718        let mut subquery_names;
2719        (lf, subquery_names) = self.process_subqueries(
2720            lf,
2721            projections_with_flags.iter_mut().map(|(e, _)| e).collect(),
2722            SubqueryShape::Scalar,
2723        )?;
2724        for (expr, _) in projections_with_flags.iter_mut() {
2725            *expr = broadcast_subqueries_in_inputs(
2726                expr.clone(),
2727                &subquery_names,
2728                &self.group_scope.subqueries_read_per_row,
2729            );
2730        }
2731        schema = self.get_frame_schema(&mut lf)?;
2732
2733        // Placeholder names are generated per execution, so an unaliased projection
2734        // named after one gets the name given to any other unnamed scalar expression.
2735        // Matching on the output name rather than the expression shape also catches
2736        // projections that merely derive their name from a placeholder.
2737        if !subquery_names.is_empty() {
2738            for (expr, is_explicit_alias) in projections_with_flags.iter_mut() {
2739                if *is_explicit_alias {
2740                    continue;
2741                }
2742                let is_placeholder_name = expr
2743                    .to_field(&schema)
2744                    .is_ok_and(|f| subquery_names.contains(&f.name));
2745                if is_placeholder_name {
2746                    *expr = expr.clone().alias(PlSmallStr::from_static("literal"));
2747                }
2748            }
2749        }
2750
2751        // QUALIFY filters on windows over the rows the SELECT list reads, so it is computed
2752        // with the SELECT list, as a hidden column. It can name SELECT aliases, which read the
2753        // subqueries of the SELECT list resolved above.
2754        let qualify_column = format_pl_smallstr!("__POLARS_QUALIFY_{}", unique_column_name());
2755        let mut qualify = match &select_stmt.qualify {
2756            Some(qualify) => {
2757                let projections: Vec<Expr> = projections_with_flags
2758                    .iter()
2759                    .map(|(e, _)| e.clone())
2760                    .collect();
2761                let output_names = OutputNames::new(
2762                    &projections,
2763                    &schema,
2764                    &self.with_placeholder_columns(&schema),
2765                );
2766                let output_schema = output_names.extend_schema(&schema);
2767                let mark_whole_frame_windows =
2768                    std::mem::replace(&mut self.group_scope.mark_whole_frame_windows, true);
2769                let parsed = self
2770                    .marking_aggregates(|ctx| parse_sql_expr(qualify, ctx, Some(&output_schema)));
2771                self.group_scope.mark_whole_frame_windows = mark_whole_frame_windows;
2772                let mut qualify = output_names.resolve(parsed?);
2773                if qualify.clone().meta().has_multiple_outputs() {
2774                    qualify = all_horizontal([qualify])?;
2775                }
2776                let qualify_subquery_names;
2777                (lf, qualify_subquery_names) =
2778                    self.process_subqueries(lf, vec![&mut qualify], SubqueryShape::Scalar)?;
2779                if !qualify_subquery_names.is_empty() {
2780                    subquery_names.extend(qualify_subquery_names);
2781                    schema = self.get_frame_schema(&mut lf)?;
2782                }
2783                let qualify = broadcast_subqueries_in_inputs(
2784                    qualify,
2785                    &subquery_names,
2786                    &self.group_scope.subqueries_read_per_row,
2787                );
2788                Some(qualify.alias(qualify_column.clone()))
2789            },
2790            None => None,
2791        };
2792
2793        let (mut projections, mut explicit_aliases): (Vec<Expr>, Vec<bool>) =
2794            projections_with_flags.into_iter().unzip();
2795
2796        // Apply `UNNEST` expressions
2797        let mut explode_names = Vec::new();
2798        let mut explode_exprs = Vec::new();
2799        let mut explode_lookup = PlHashMap::new();
2800
2801        for expr in &projections {
2802            for e in expr {
2803                if let Expr::Explode { input, .. } = e {
2804                    match input.as_ref() {
2805                        Expr::Column(name) => explode_names.push(name.clone()),
2806                        other_expr => {
2807                            // Note: skip aggregate expressions; those are handled in the GROUP BY phase
2808                            if !has_expr(other_expr, |e| matches!(e, Expr::Agg(_) | Expr::Len)) {
2809                                let temp_name =
2810                                    format_pl_smallstr!("__POLARS_UNNEST_{}", explode_exprs.len());
2811                                explode_exprs.push(other_expr.clone().alias(temp_name.as_str()));
2812                                explode_lookup.insert(other_expr.clone(), temp_name.clone());
2813                                explode_names.push(temp_name);
2814                            }
2815                        },
2816                    }
2817                }
2818            }
2819        }
2820        if !explode_names.is_empty() {
2821            if !explode_exprs.is_empty() {
2822                lf = lf.with_columns(explode_exprs);
2823            }
2824            lf = lf.explode(
2825                Selector::ByName {
2826                    names: Arc::from(explode_names),
2827                    strict: true,
2828                },
2829                ExplodeOptions {
2830                    empty_as_null: true,
2831                    keep_nulls: true,
2832                },
2833            );
2834            projections = projections
2835                .into_iter()
2836                .map(|p| {
2837                    // Update "projections" with column refs to the now-exploded expressions
2838                    p.map_expr(|e| match e {
2839                        Expr::Explode { input, .. } => explode_lookup
2840                            .get(input.as_ref())
2841                            .map(|name| Expr::Column(name.clone()))
2842                            .unwrap_or_else(|| input.as_ref().clone()),
2843                        _ => e,

3421        self.join_inputs(
3422            tbl_left,

3427            keep_right,

3431    /// Join on `constraint` and on the key pairs `extra_keys`. Without either, every row
3432    /// pairs with every row.
3433    fn join_inputs(
3434        &mut self,
3435        tbl_left: &TableInfo,
3436        tbl_right: &TableInfo,
3437        constraint: Option<&JoinConstraint>,
3438        extra_keys: (Vec<Expr>, Vec<Expr>),
3439        join_type: JoinType,
3440        keep_right: bool,
3441    ) -> PolarsResult<LazyFrame> {
3442        let (mut left_on, mut right_on, predicates) = match constraint {
3443            // A condition that reads no input (eg: `ON TRUE`, `ON 1 = 1`) pairs every row with
3444            // every row, or none: join on the condition itself as a boolean key (a null key
3445            // matches nothing, like an unknown condition).
3446            Some(JoinConstraint::On(expr))
3447                if let Some(predicate) = self.input_independent_predicate(expr)? =>
3448            {
3449                (
3450                    vec![strip_join_aliases(predicate).cast(DataType::Boolean)],
3451                    vec![lit(true)],
3452                    vec![],
3453                )
3454            },
3455            Some(constraint) => {
3456                let keys = process_join_constraint(constraint, tbl_left, tbl_right, self)?;
3457                for e in keys.0.iter().chain(&keys.1).chain(&keys.2) {
3458                    reject_unresolved_subquery(e, "JOIN ... ON")?;
3459                }
3460                keys
3461            },
3462            None => (vec![], vec![], vec![]),
3463        };
3464        left_on.extend(extra_keys.0);
3465        right_on.extend(extra_keys.1);
3466        if left_on.is_empty() && predicates.is_empty() {
3467            left_on.push(lit(true));
3468            right_on.push(lit(true));
3469        }
3470        if keep_right {
3471            polars_ensure!(
3472                predicates.is_empty(),
3473                InvalidOperation: "'{}' join is not supported with non-equi join conditions",
3474                join_type
3475            );
3476            return tbl_right
3477                .frame
3478                .clone()
3479                .join_builder()
3480                .with(tbl_left.frame.clone())
3481                .left_on(right_on)
3482                .right_on(left_on)
3483                .how(join_type)
3484                .finish();
3485        }
3486        let coalesce_type = match constraint {
3487            // "NATURAL" joins should coalesce; otherwise we disambiguate
3488            Some(JoinConstraint::Natural) => JoinCoalesce::CoalesceColumns,
3489            _ => JoinCoalesce::KeepColumns,
3490        };
3491        let suffix = format!(":{}", tbl_right.name);
3492
3493        let joined = if predicates.is_empty() {
3494            // Equi-join: standard left_on/right_on path
3495            tbl_left
3496                .frame
3497                .clone()
3498                .join_builder()
3499                .with(tbl_right.frame.clone())
3500                .left_on(left_on)
3501                .right_on(right_on)
3502                .how(join_type)
3503                .suffix(suffix)
3504                .coalesce(coalesce_type)
3505                .finish()?
3506        } else {
3507            // Non-equi conditions: convert to predicates for `join_where`.
3508            // Any equi-conditions become equality predicates with right-side
3509            // columns suffixed to match their merged-schema names.
3510            let mut all_predicates = predicates;
3511            for (l, r) in left_on.into_iter().zip(right_on) {
3512                let r_suffixed = suffix_conflicting_columns(r, tbl_left, tbl_right, &suffix);
3513                all_predicates.push(l.eq(r_suffixed));
3514            }
3515            let all_predicates: Vec<Expr> =
3516                all_predicates.into_iter().map(strip_join_aliases).collect();
3517            tbl_left
3518                .frame
3519                .clone()
3520                .join_builder()
3521                .with(tbl_right.frame.clone())
3522                .how(join_type)
3523                .suffix(suffix)
3524                .coalesce(coalesce_type)
3525                .join_where(all_predicates)
3526        };
3527
3528        Ok(joined)
3529    }

3547    /// Which input of a join the column `expr` reads: `Some(true)` for the left input,
3548    /// `Some(false)` for the right one, and `None` if it is not a column or is in both.
3549    pub(crate) fn column_join_input(
3550        &self,
3551        expr: &SQLExpr,
3552        tbl_left: &TableInfo,
3553        tbl_right: &TableInfo,
3554    ) -> Option<bool> {
3555        let column_side = |name: &str| match (
3556            tbl_left.schema.contains(name),
3557            tbl_right.schema.contains(name),
3558        ) {
3559            (true, false) => Some(true),
3560            (false, true) => Some(false),
3561            _ => None,
3562        };

3564            SQLExpr::Identifier(ident) => column_side(&ident.value),
3565            // A table qualifier takes precedence over a struct column of the same name.
3566            SQLExpr::CompoundIdentifier(idents) => match self.resolve_qualifier(idents) {
3567                Ok(Some((name, _))) => Some(!self.is_right_relation(&name, tbl_right)),
3568                Ok(None) => column_side(&idents[0].value),
3569                // An ambiguous qualifier is reported when the condition is parsed.
3570                Err(_) => None,
3571            },
3572            _ => None,
3573        }

3596    /// Join on an `ON` condition that holds subqueries. A conjunct whose own columns read
3597    /// one input is applied to that input first: as a filter if the join drops the rows it
3598    /// rejects, else as a flag column joined against `TRUE`. Its subqueries must not read
3599    /// the joined inputs. Any other conjunct stays in the condition, which rejects
3600    /// subqueries.
3601    fn process_join_on_subqueries(
3602        &mut self,
3603        tbl_left: &TableInfo,
3604        tbl_right: &TableInfo,
3605        expr: &SQLExpr,
3606        join_type: JoinType,
3607        keep_right: bool,
3608    ) -> PolarsResult<LazyFrame> {
3609        // Whether the join drops the rows of an input that match no row of the other.
3610        let (filter_left, filter_right) = match (&join_type, keep_right) {
3611            (JoinType::Inner, _) => (true, true),
3612            (JoinType::Left, _) => (false, true),
3613            (JoinType::Right, _) => (true, false),
3614            #[cfg(feature = "semi_anti_join")]
3615            (JoinType::Semi, _) => (true, true),
3616            #[cfg(feature = "semi_anti_join")]
3617            (JoinType::Anti, false) => (false, true),
3618            #[cfg(feature = "semi_anti_join")]
3619            (JoinType::Anti, true) => (true, false),
3620            _ => (false, false),
3621        };
3622        let mut left = tbl_left.clone();
3623        let mut right = tbl_right.clone();
3624        let (mut left_on, mut right_on) = (vec![], vec![]);
3625        let mut flags = vec![];
3626        let mut residual = vec![];
3627        for conjunct in flatten_and_conditions(expr) {
3628            if !expr_contains_subquery(conjunct) {
3629                residual.push(conjunct.clone());
3630                continue;
3631            }
3632            let is_left = match self.conjunct_join_inputs(conjunct, tbl_left, tbl_right) {
3633                JoinInputs::Left => true,
3634                JoinInputs::Right => false,
3635                JoinInputs::Neither => !filter_right && filter_left,
3636                JoinInputs::Both | JoinInputs::Unknown => {
3637                    residual.push(conjunct.clone());
3638                    continue;
3639                },
3640            };
3641            let (tbl, filter) = if is_left {
3642                (&mut left, filter_left)
3643            } else {
3644                (&mut right, filter_right)
3645            };
3646            // Planning the subqueries here also rejects those that read the other input.
3647            let mut condition =
3648                strip_join_aliases(parse_sql_expr(conjunct, self, Some(&tbl.schema))?);
3649            let placeholders;
3650            (tbl.frame, placeholders) = self.process_subqueries(
3651                tbl.frame.clone(),
3652                vec![&mut condition],
3653                SubqueryShape::Scalar,
3654            )?;
3655            let condition = condition.cast(DataType::Boolean);
3656            if filter {
3657                tbl.frame =
3658                    drop_subquery_placeholders(tbl.frame.clone().filter(condition), placeholders);
3659                continue;
3660            }
3661            let name = unique_column_name();
3662            tbl.frame = drop_subquery_placeholders(
3663                tbl.frame.clone().with_column(condition.alias(name.clone())),
3664                placeholders,
3665            );
3666            tbl.schema = self.get_frame_schema(&mut tbl.frame)?;
3667            let (own_keys, other_keys) = if is_left {
3668                (&mut left_on, &mut right_on)
3669            } else {
3670                (&mut right_on, &mut left_on)
3671            };
3672            own_keys.push(col(name.clone()));
3673            other_keys.push(lit(true));
3674            flags.push(name);
3675        }
3676        let residual = combine_conditions(residual, SQLBinaryOperator::And).map(JoinConstraint::On);
3677        let joined = self.join_inputs(
3678            &left,
3679            &right,
3680            residual.as_ref(),
3681            (left_on, right_on),
3682            join_type,
3683            keep_right,
3684        )?;
3685        Ok(if flags.is_empty() {
3686            joined
3687        } else {
3688            joined.drop(Selector::ByName {
3689                names: flags.into(),
3690                strict: false,
3691            })
3692        })
3693    }

3695    /// Which inputs of a join the conjunct `expr` reads, leaving out the inside of its
3696    /// subqueries. `Unknown` if a column is in neither input or, unqualified, in both.
3697    fn conjunct_join_inputs(
3698        &self,
3699        expr: &SQLExpr,
3700        tbl_left: &TableInfo,
3701        tbl_right: &TableInfo,
3702    ) -> JoinInputs {
3703        struct Reads<'a> {
3704            ctx: &'a SQLContext,
3705            left: &'a TableInfo,
3706            right: &'a TableInfo,
3707            subquery_depth: usize,
3708            reads_left: bool,
3709            reads_right: bool,
3710            unknown: bool,
3711        }
3712
3713        impl Reads<'_> {
3714            fn column(&mut self, name: &str) {
3715                match (
3716                    self.left.schema.contains(name),
3717                    self.right.schema.contains(name),
3718                ) {
3719                    (true, false) => self.reads_left = true,
3720                    (false, true) => self.reads_right = true,
3721                    _ => self.unknown = true,
3722                }
3723            }
3724        }

3726        impl sqlparser::ast::Visitor for Reads<'_> {
3727            type Break = ();
3728
3729            fn pre_visit_query(&mut self, _query: &Query) -> ControlFlow<()> {
3730                self.subquery_depth += 1;
3731                ControlFlow::Continue(())
3732            }
3733
3734            fn post_visit_query(&mut self, _query: &Query) -> ControlFlow<()> {
3735                self.subquery_depth -= 1;
3736                ControlFlow::Continue(())
3737            }
3738
3739            fn pre_visit_expr(&mut self, expr: &SQLExpr) -> ControlFlow<()> {
3740                if self.subquery_depth > 0 {
3741                    return ControlFlow::Continue(());
3742                }
3743                match expr {
3744                    SQLExpr::Identifier(ident) => self.column(&ident.value),
3745                    SQLExpr::CompoundIdentifier(idents) => match self.ctx.resolve_qualifier(idents)
3746                    {
3747                        Ok(Some((name, _))) => match (
3748                            self.left.relations.contains(&name),
3749                            self.right.relations.contains(&name),
3750                        ) {
3751                            (true, false) => self.reads_left = true,
3752                            (false, true) => self.reads_right = true,
3753                            _ => self.unknown = true,
3754                        },
3755                        Ok(None) => self.column(&idents[0].value),
3756                        Err(_) => self.unknown = true,
3757                    },
3758                    _ => {},
3759                }
3760                ControlFlow::Continue(())
3761            }
3762        }
3763
3764        let mut reads = Reads {
3765            ctx: self,
3766            left: tbl_left,
3767            right: tbl_right,
3768            subquery_depth: 0,
3769            reads_left: false,
3770            reads_right: false,
3771            unknown: false,
3772        };
3773        let _ = dyn_visit::visit(expr, &mut reads);
3774        match (reads.unknown, reads.reads_left, reads.reads_right) {
3775            (true, _, _) => JoinInputs::Unknown,
3776            (false, true, true) => JoinInputs::Both,
3777            (false, true, false) => JoinInputs::Left,
3778            (false, false, true) => JoinInputs::Right,
3779            (false, false, false) => JoinInputs::Neither,
3780        }
3781    }

3783    /// Which input an `ASOF JOIN` match operand reads: `Some(true)` for the left input,
3784    /// `Some(false)` for the right one, and `None` if all its columns are in both.
3785    #[cfg(feature = "asof_join")]
3786    fn asof_operand_is_left(
3787        &self,
3788        expr: &SQLExpr,
3789        tbl_left: &TableInfo,
3790        tbl_right: &TableInfo,
3791    ) -> PolarsResult<Option<bool>> {
3792        let (reads_left, reads_right) = self.join_inputs_read(expr, tbl_left, tbl_right);
3793        polars_ensure!(
3794            !(reads_left && reads_right),
3795            SQLSyntax: "ASOF JOIN MATCH_CONDITION operand {} reads both tables", expr
3796        );
3797        Ok(match (reads_left, reads_right) {
3798            (true, _) => Some(true),
3799            (_, true) => Some(false),
3800            _ => None,
3801        })
3802    }

3804    /// `ASOF JOIN r MATCH_CONDITION (l.t >= r.t) [ON l.k = r.k AND ... | USING (k, ...)]`.
3805    ///
3806    /// Every left row is kept, paired with the closest right row that satisfies the match
3807    /// condition and has equal keys, or with nulls if there is none.
3808    #[cfg(feature = "asof_join")]
3809    fn process_asof_join(
3810        &mut self,
3811        tbl_left: &TableInfo,
3812        tbl_right: &TableInfo,
3813        match_condition: &SQLExpr,
3814        constraint: &JoinConstraint,
3815    ) -> PolarsResult<LazyFrame> {
3816        let mut condition = match_condition;
3817        while let SQLExpr::Nested(inner) = condition {
3818            condition = inner;
3819        }
3820        let SQLExpr::BinaryOp { left, op, right } = condition else {
3821            polars_bail!(
3822                SQLSyntax: "ASOF JOIN MATCH_CONDITION must be a single `>=`, `>`, `<=` or `<` comparison, found {}",
3823                match_condition
3824            )
3825        };
3826        polars_ensure!(
3827            matches!(
3828                op,
3829                SQLBinaryOperator::GtEq
3830                    | SQLBinaryOperator::Gt
3831                    | SQLBinaryOperator::LtEq
3832                    | SQLBinaryOperator::Lt
3833            ),
3834            SQLSyntax: "ASOF JOIN MATCH_CONDITION must be a single `>=`, `>`, `<=` or `<` comparison, found {}",
3835            match_condition
3836        );
3837        polars_ensure!(
3838            expr_references_any_column(left) && expr_references_any_column(right),
3839            SQLSyntax: "ASOF JOIN MATCH_CONDITION must compare a column of each table, found {}",
3840            match_condition
3841        );
3842        let left_is_left = match (
3843            self.asof_operand_is_left(left, tbl_left, tbl_right)?,
3844            self.asof_operand_is_left(right, tbl_left, tbl_right)?,
3845        ) {
3846            (Some(l), Some(r)) if l != r => l,
3847            (Some(l), None) => l,
3848            (None, Some(r)) => !r,
3849            (Some(_), Some(_)) => polars_bail!(
3850                SQLSyntax: "ASOF JOIN MATCH_CONDITION must compare a column of each table, found {}",
3851                match_condition
3852            ),
3853            (None, None) => polars_bail!(
3854                SQLSyntax: "ASOF JOIN MATCH_CONDITION is ambiguous, qualify its columns with their table name: {}",
3855                match_condition
3856            ),
3857        };
3858        let swapped = !left_is_left;
3859        let (left_operand, right_operand) = if swapped {
3860            (right, left)
3861        } else {
3862            (left, right)
3863        };
3864        let left_on =
3865            strip_join_aliases(parse_sql_expr(left_operand, self, Some(&tbl_left.schema))?);
3866        let right_on = strip_join_aliases(parse_sql_expr(
3867            right_operand,
3868            self,
3869            Some(&tbl_right.schema),
3870        )?);
3871        let (strategy, allow_eq) = match (op, swapped) {
3872            (SQLBinaryOperator::GtEq, false) | (SQLBinaryOperator::LtEq, true) => {
3873                (AsofStrategy::Backward, true)
3874            },
3875            (SQLBinaryOperator::Gt, false) | (SQLBinaryOperator::Lt, true) => {
3876                (AsofStrategy::Backward, false)
3877            },
3878            (SQLBinaryOperator::LtEq, false) | (SQLBinaryOperator::GtEq, true) => {
3879                (AsofStrategy::Forward, true)
3880            },
3881            _ => (AsofStrategy::Forward, false),
3882        };
3883
3884        let (left_by, right_by) = match constraint {
3885            JoinConstraint::None => (vec![], vec![]),
3886            _ => {
3887                let (left_by, right_by, predicates) =
3888                    process_join_constraint(constraint, tbl_left, tbl_right, self)?;
3889                polars_ensure!(
3890                    predicates.is_empty(),
3891                    SQLSyntax: "ASOF JOIN only supports `=` conditions in ON; the inequality goes in MATCH_CONDITION"
3892                );
3893                (left_by, right_by)
3894            },
3895        };
3896        for e in [&left_on, &right_on]
3897            .into_iter()
3898            .chain(&left_by)
3899            .chain(&right_by)
3900        {
3901            reject_unresolved_subquery(e, "ASOF JOIN")?;
3902        }
3903
3904        let to_supertype = |l: Expr, r: Expr| -> PolarsResult<(Expr, Expr)> {
3905            let l_dtype = l.to_field(&tbl_left.schema)?.dtype;
3906            let r_dtype = r.to_field(&tbl_right.schema)?.dtype;
3907            asof_key_pair(l, &l_dtype, r, &r_dtype)
3908        };
3909
3910        // `by` keys are column names, so each key pair is computed into a temporary column
3911        // that is dropped after the join.
3912        let mut left = tbl_left.frame.clone();
3913        let mut right = tbl_right.frame.clone();
3914        let mut by_names = Vec::with_capacity(left_by.len());
3915        for (l, r) in left_by.into_iter().zip(right_by) {
3916            let (l, r) = to_supertype(l, r)?;
3917            let name = loop {
3918                let name = unique_column_name();
3919                if !tbl_left.schema.contains(&name) && !tbl_right.schema.contains(&name) {
3920                    break name;
3921                }
3922            };
3923            left = left.with_column(l.alias(name.clone()));
3924            right = right.with_column(r.alias(name.clone()));
3925            by_names.push(name);
3926        }
3927
3928        // Both inputs must be sorted on the match key.
3929        let (left_on, right_on) = to_supertype(left_on, right_on)?;
3930        let left = left.sort_by_exprs([left_on.clone()], SortMultipleOptions::default());
3931        let right = right.sort_by_exprs([right_on.clone()], SortMultipleOptions::default());
3932
3933        let has_by = !by_names.is_empty();
3934        let joined = left
3935            .join_builder()
3936            .with(right)
3937            .left_on([left_on])
3938            .right_on([right_on])
3939            .how(JoinType::AsOf(Box::new(AsOfOptions {
3940                strategy,
3941                tolerance: None,
3942                tolerance_str: None,
3943                left_by: has_by.then(|| by_names.clone()),
3944                right_by: has_by.then(|| by_names.clone()),
3945                allow_eq,
3946                check_sortedness: false,
3947            })))
3948            .suffix(format!(":{}", tbl_right.name))
3949            .coalesce(JoinCoalesce::KeepColumns)
3950            .finish()?;
3951        Ok(if has_by {
3952            joined.drop(Selector::ByName {
3953                names: Arc::from(by_names),
3954                strict: true,
3955            })
3956        } else {
3957            joined
3958        })
3959    }

3961    /// Process implicit (comma-separated) joins from `FROM t1, t2, ...` syntax.
3962    ///
3963    /// Extracts join predicates from the WHERE clause, joining each additional table with
3964    /// either an INNER JOIN (cross-table predicates found) or a CROSS JOIN (no predicates).
3965    /// Tables are joined in the order they become connectable rather than source order.
3966    /// Returns the residual WHERE conditions not consumed as join predicates.
3967    fn process_implicit_joins(
3968        &mut self,
3969        lf: &mut LazyFrame,
3970        from: &[TableWithJoins],
3971        where_clause: &Option<SQLExpr>,
3972    ) -> PolarsResult<Option<SQLExpr>> {
3973        let first = from.first().unwrap();
3974        let mut remaining_where = where_clause.clone();
3975        let mut joined_table_names: Vec<String> = kept_factors(first)
3976            .into_iter()
3977            .filter_map(get_table_name)
3978            .collect();
3979        // Resolving is `&mut self`, so all relations must be resolved up-front rather than
3980        // mid-probe while picking a join order below.
3981        let mut pending = Vec::with_capacity(from.len() - 1);
3982        for tbl_expr in from.iter().skip(1) {
3983            let mut rf = self.execute_from_statement(tbl_expr)?;
3984            let r_name = kept_factors(tbl_expr)
3985                .into_iter()
3986                .next()
3987                .and_then(get_table_name)
3988                .unwrap_or_default();
3989            polars_ensure!(
3990                !r_name.is_empty(),
3991                SQLInterface: "implicit joins require named tables; please provide an alias"
3992            );
3993            let right_schema = self.get_frame_schema(&mut rf)?;
3994            pending.push((tbl_expr, rf, r_name, right_schema));
3995        }
3996
3997        // Each round takes the first pending relation with a join predicate against the
3998        // tables joined so far; relations with no predicate yet are deferred. Genuinely

4053                            &joined_table_names.first().cloned().unwrap_or_default(),

4597            SQLExpr::Identifier(ident) => {
4598                match resolve_select_alias(&ident.value, projections, schema) {
4599                    Some(aliased) => Ok(aliased),
4600                    None => resolve(self, e),
4601                }
4602            },
4603            // Drop the alias that restores the unqualified name, so two
4604            // relations sharing a column name give distinct keys.
4605            SQLExpr::CompoundIdentifier(_) => resolve(self, e).map(|e| strip_outer_alias(&e)),
4606            _ => resolve(self, e),
4607        }
4608    }

4610    /// Bind the grouping sets and every `GROUPING()` call of the block to the
4611    /// resolved keys. A plain `GROUP BY` that uses `GROUPING()` is a single set
4612    /// holding every key.
4613    fn bind_grouping_sets(
4614        &mut self,
4615        group_by_keys: &[Expr],
4616        grouping_sets: Option<Vec<Vec<usize>>>,
4617        projections: &[Expr],
4618        schema: &Schema,
4619        order_by: &Option<OrderBy>,
4620    ) -> PolarsResult<Option<GroupingSets>> {
4621        // Calls in clauses parsed after aggregation must be registered before the
4622        // branches are built; their validation happens when those clauses are parsed.
4623        let order_by_exprs = match order_by {
4624            Some(OrderBy {
4625                kind: OrderByKind::Expressions(exprs),
4626                ..
4627            }) => exprs.iter().map(|ob| &ob.expr).collect(),
4628            _ => vec![],
4629        };
4630        for expr in order_by_exprs {
4631            for args in collect_grouping_calls(expr) {
4632                self.register_grouping_call(args);
4633            }
4634        }
4635        let sets = match grouping_sets {
4636            Some(sets) => sets,
4637            None if self.group_scope.grouping_calls.is_empty() => return Ok(None),
4638            None => vec![(0..group_by_keys.len()).collect()],
4639        };
4640        // Later parses of the same calls must still find these registrations.
4641        let calls = std::mem::take(&mut self.group_scope.grouping_calls);
4642        let resolved_args = calls
4643            .iter()
4644            .map(|call| {
4645                call.args
4646                    .iter()
4647                    .map(|arg| self.resolve_group_by_key(arg, projections, schema, false))
4648                    .collect::<PolarsResult<Vec<_>>>()
4649            })
4650            .collect::<PolarsResult<Vec<_>>>();
4651        self.group_scope.grouping_calls = calls;
4652        let grouping = GroupingSets::new(
4653            group_by_keys.to_vec(),
4654            sets,
4655            &self.group_scope.grouping_calls,
4656            resolved_args?,
4657        )?;
4658        Ok(Some(grouping))
4659    }

4661    #[allow(clippy::too_many_arguments)]
4662    fn process_group_by(
4663        &mut self,
4664        mut lf: LazyFrame,
4665        group_by_keys: &[Expr],
4666        projections: Vec<Expr>,
4667        explicit_aliases: &[bool],
4668        having: Option<Expr>,
4669        qualify: Option<(PlSmallStr, Expr)>,
4670        order_by: &Option<OrderBy>,
4671        grouping: Option<&GroupingSets>,
4672        subquery_names: &PlHashSet<PlSmallStr>,
4673    ) -> PolarsResult<(LazyFrame, Vec<PlSmallStr>, Option<OrderBy>)> {
4674        // Ordinary grouping stores each computed key in a hidden column. The other
4675        // expressions read that column, as the key's output name can be taken by
4676        // an input column or by another key.
4677        let mut stored_keys: Vec<(Expr, PlSmallStr)> = Vec::new();
4678        let column_keys: Vec<Expr>;
4679        let group_by_keys = if grouping.is_none() {
4680            let mut prepared = Vec::new();
4681            column_keys = group_by_keys
4682                .iter()
4683                .map(|key| {
4684                    let stripped = strip_outer_alias(key);
4685                    if matches!(stripped, Expr::Column(_)) || is_constant(&stripped) {
4686                        return key.clone();
4687                    }
4688                    let name = unique_column_name();
4689                    prepared.push(stripped.clone().alias(name.clone()));
4690                    stored_keys.push((stripped, name.clone()));
4691                    col(name)
4692                })
4693                .collect();
4694            if !prepared.is_empty() {
4695                lf = lf.with_columns(prepared);
4696            }
4697            &column_keys[..]
4698        } else {
4699            group_by_keys
4700        };
4701        let mut schema_before = self.get_frame_schema(&mut lf)?;
4702        let key_schema =
4703            expressions_to_schema(group_by_keys, &schema_before, |duplicate_name: &str| {
4704                format!("group_by keys contained duplicate output name '{duplicate_name}'")
4705            })?;
4706        if let Some(grouping) = grouping {
4707            stored_keys = grouping.stored_keys(&key_schema);
4708        }
4709        let mut group_by_keys_schema = key_schema.clone();
4710        // Columns the parsed expressions may name before the aggregation resolves them:
4711        // the whole-frame window partition, and `GROUPING()` values, which are
4712        // per-set constants the projections treat like group keys.
4713        let whole_frame_partition = self.group_scope.whole_frame_partition.clone();
4714        let placeholders: Vec<PlSmallStr> = grouping
4715            .map(|g| g.placeholders().cloned().collect())
4716            .unwrap_or_default();
4717        if whole_frame_partition.is_some() || !placeholders.is_empty() {
4718            let mut extended = Arc::unwrap_or_clone(schema_before);
4719            if let Some(partition) = &whole_frame_partition {
4720                extended.with_column(partition.clone(), DataType::Int32);
4721            }
4722            for placeholder in &placeholders {
4723                extended.with_column(placeholder.clone(), DataType::Int64);
4724                group_by_keys_schema.with_column(placeholder.clone(), DataType::Int64);
4725            }
4726            schema_before = Arc::new(extended);
4727        }
4728
4729        // Disambiguate SELECT-list output-name collisions.
4730        let projections = disambiguate_output_names(
4731            projections
4732                .into_iter()
4733                .zip(explicit_aliases.iter().copied())
4734                .collect(),
4735            &schema_before,
4736        )?;
4737        // Hidden output columns: the ORDER BY items with aggregates or computed keys, and
4738        // the QUALIFY predicate that the block filters on after the aggregation.
4739        let (order_by, mut extra_projections) =
4740            self.resolve_order_by_aggregates(order_by, &projections, &schema_before, &stored_keys)?;
4741        let projections = if grouping.is_none() && !stored_keys.is_empty() {
4742            projections
4743                .into_iter()
4744                .map(|p| {
4745                    let stripped = strip_outer_alias(&p);
4746                    let bound = bind_stored_keys(stripped.clone(), &stored_keys);
4747                    Ok(if bound == stripped {
4748                        p
4749                    } else {
4750                        bound.alias(p.to_field(&schema_before)?.name)
4751                    })
4752                })
4753                .collect::<PolarsResult<Vec<_>>>()?
4754        } else {
4755            projections
4756        };
4757
4758        let projections: Vec<Expr> = projections
4759            .into_iter()
4760            .map(reduce_correlated_cols_in_group_context)
4761            .collect();
4762
4763        // Note: HAVING is evaluated in the group context (`group_by().having(...)` for
4764        // ordinary grouping, a post-union filter for grouping sets), so any reference
4765        // to a SELECT alias is resolved to the aggregate it names.
4766        let mut having = having.map(|having_expr| {
4767            let having_expr = having_expr.map_expr(|e| match &e {
4768                Expr::Column(name) => resolve_select_alias(name, &projections, &schema_before)
4769                    .map_or(e, |resolved| strip_outer_alias(&resolved)),
4770                _ => e,
4771            });
4772            let having_expr = match grouping {
4773                // Grouping sets bind the keys once HAVING runs on the combined rows.
4774                Some(_) => having_expr,
4775                None => bind_stored_keys(having_expr, &stored_keys),
4776            };
4777            reduce_correlated_cols_in_group_context(having_expr)
4778        });
4779
4780        // Resolve scalar subqueries in HAVING and the ORDER BY aggregates against the
4781        // pre-aggregation frame.
4782        let new_subquery_names;
4783        let mut exprs: Vec<&mut Expr> = extra_projections.iter_mut().map(|(_, e)| e).collect();
4784        exprs.extend(having.as_mut());
4785        (lf, new_subquery_names) = self.process_subqueries(lf, exprs, SubqueryShape::Scalar)?;
4786        if !new_subquery_names.is_empty() {
4787            let schema = self.get_frame_schema(&mut lf)?;
4788            let schema_before = Arc::make_mut(&mut schema_before);
4789            for name in &new_subquery_names {
4790                schema_before.with_column(name.clone(), schema.get(name).unwrap().clone());
4791            }
4792        }
4793        for e in extra_projections
4794            .iter_mut()
4795            .map(|(_, e)| e)
4796            .chain(having.as_mut())
4797        {
4798            *e = broadcast_subqueries_in_inputs(
4799                e.clone(),
4800                &new_subquery_names,
4801                &self.group_scope.subqueries_read_per_row,
4802            );
4803        }
4804        let mut subquery_names = subquery_names.clone();
4805        subquery_names.extend(new_subquery_names);
4806
4807        extra_projections
4808            .extend(qualify.map(|(name, e)| (name, reduce_correlated_cols_in_group_context(e))));
4809
4810        // Note: remove the `group_by` keys as Polars adds those implicitly.
4811        let mut aliased_aggregations: PlHashMap<PlSmallStr, PlSmallStr> = PlHashMap::new();
4812        let mut post_agg_projection: Vec<Expr> = Vec::new();
4813        let mut projection_overrides = PlHashMap::with_capacity(projections.len());
4814        let mut projection_aliases = PlHashSet::new();
4815        let mut group_key_aliases = PlHashSet::new();
4816
4817        // Pre-compute group key data (alias-stripped expression + aggregated output
4818        // name) to avoid repeated work matching projections against the group keys
4819        let group_key_data: Vec<_> = group_by_keys
4820            .iter()
4821            .map(|gk| {
4822                (
4823                    strip_outer_alias(gk),
4824                    gk.to_field(&schema_before).ok().map(|f| f.name),
4825                )
4826            })
4827            .collect();
4828
4829        // ORDER BY extras were parsed above and may have registered the partition.
4830        let whole_frame_partition = self.group_scope.whole_frame_partition.clone();
4831        let mut splitter = GroupContextSplitter {
4832            keys: &group_by_keys_schema,
4833            key_exprs: group_key_data
4834                .iter()
4835                .filter_map(|(key, name)| Some((key.clone(), name.clone()?)))
4836                .chain(match grouping {
4837                    Some(_) => vec![],
4838                    None => stored_keys.clone(),
4839                })
4840                .collect(),
4841            whole_frame_partition: whole_frame_partition.as_ref(),
4842            subquery_names: &subquery_names,
4843            subqueries_after_aggregation: grouping.is_none() && group_by_keys.is_empty(),
4844            subqueries_read_after: PlIndexSet::new(),
4845            aggregates: AggregateOutputs::with_capacity(projections.len()),
4846        };
4847        if let Some(having) = &having {
4848            splitter.check_columns_per_group(having)?;
4849        }
4850        // Post-aggregation expressions read computed keys from their stored columns.
4851        let bind_keys = |e: Expr| bind_stored_keys(e, &stored_keys);
4852
4853        let projection_group_key: Vec<Option<PlSmallStr>> = projections
4854            .iter()
4855            .map(|p| {
4856                let p_stripped = strip_outer_alias(p);
4857                group_key_data.iter().find_map(|(gk_stripped, gk_name)| {
4858                    (*gk_stripped == p_stripped)
4859                        .then(|| gk_name.clone())
4860                        .flatten()
4861                })
4862            })
4863            .collect();
4864
4865        for (e, group_key) in projections.iter().zip(&projection_group_key) {
4866            let matches_group_key = group_key.is_some();
4867            let is_non_group_key_expr = !matches_group_key && splitter.requires_group_processing(e);
4868
4869            // Note: if simple aliased expression we defer aliasing until after the group_by.
4870            // Use `e_inner` to track the potentially unwrapped expression for field lookup.
4871            let mut e_inner = e;
4872            if let Expr::Alias(expr, alias) = e {
4873                if e.clone().meta().is_simple_projection(Some(&schema_before)) {
4874                    group_key_aliases.insert(alias.as_ref());
4875                    e_inner = expr
4876                } else if let Expr::Function {
4877                    function: FunctionExpr::StructExpr(StructFunction::FieldByName(name)),
4878                    ..
4879                } = expr.deref()
4880                {
4881                    projection_overrides
4882                        .insert(alias.as_ref(), col(name.clone()).alias(alias.clone()));
4883                } else if !is_non_group_key_expr && !group_by_keys_schema.contains(alias) {
4884                    projection_aliases.insert(alias.as_ref());
4885                }
4886            }
4887            let field = e_inner.to_field(&schema_before)?;
4888            if is_non_group_key_expr {
4889                splitter.check_columns_per_group(e)?;
4890                // Window functions run on the aggregated frame; only the aggregates
4891                // inside them run in the group context. The same holds for anything
4892                // combining aggregates with grouped keys or `GROUPING()` values.
4893                let bound = bind_keys(strip_outer_alias(e));
4894                if splitter.needs_post_aggregation(&bound) {
4895                    let post_agg_expr = splitter.hoist(bound)?;
4896                    post_agg_projection.push(post_agg_expr.alias(field.name.clone()));
4897                    continue;
4898                }
4899                let mut e = e.clone();
4900                // If aggregation colname conflicts with a group key,
4901                // alias it to avoid duplicate/mis-tracked columns
4902                if group_by_keys_schema.get(&field.name).is_some() {
4903                    let alias_name = format_pl_smallstr!("__POLARS_AGG_{}", field.name);
4904                    e = e.alias(alias_name.clone());
4905                    aliased_aggregations.insert(field.name.clone(), alias_name);
4906                }
4907                splitter.aggregates.push(e);
4908            } else if !matches_group_key {
4909                // Non-aggregated columns must be part of the GROUP BY clause
4910                if let Expr::Column(_)
4911                | Expr::Function {
4912                    function: FunctionExpr::StructExpr(StructFunction::FieldByName(_)),
4913                    ..
4914                } = e_inner
4915                {
4916                    if !group_by_keys_schema.contains(&field.name) {
4917                        polars_bail!(SQLSyntax: "'{}' should participate in the GROUP BY clause or an aggregate function", &field.name);
4918                    }
4919                } else {
4920                    splitter.check_columns_per_group(e_inner)?;
4921                }
4922            }
4923        }
4924
4925        for (name, e) in &extra_projections {
4926            let bound = bind_keys(strip_outer_alias(e));
4927            if splitter.needs_post_aggregation(&bound) {
4928                let rest = splitter.hoist(bound)?;
4929                post_agg_projection.push(rest.alias(name.clone()));
4930            } else if !splitter.requires_group_processing(&bound) {
4931                post_agg_projection.push(bound.alias(name.clone()));
4932            } else {
4933                splitter.aggregates.push(e.clone());
4934            }
4935        }
4936
4937        if let Some(grouping) = grouping {
4938            polars_ensure!(
4939                !splitter.aggregates.iter().any(|e| grouping.contains_placeholder(e)),
4940                SQLSyntax: "GROUPING() cannot be used inside an aggregate function"
4941            );
4942        }
4943
4944        let aggregated = match grouping {
4945            None if group_by_keys.is_empty() => {
4946                // Without GROUP BY there is one aggregated row, which HAVING filters.
4947                let having = having.map(|having| splitter.hoist(having)).transpose()?;
4948                let mut aggregates = splitter.aggregates.into_exprs();
4949                // The row also exists when nothing is aggregated, as in `SELECT 1 ... HAVING`.
4950                if aggregates.is_empty() {
4951                    aggregates.push(len());
4952                }
4953                let mut aggregated = lf.select(aggregates);
4954                if !splitter.subqueries_read_after.is_empty() {
4955                    let mut frames = vec![aggregated];
4956                    frames.extend(
4957                        splitter
4958                            .subqueries_read_after
4959                            .iter()
4960                            .map(|name| self.group_scope.subquery_plans[name].clone()),
4961                    );
4962                    aggregated = concat_lf_horizontal(
4963                        frames,
4964                        HConcatOptions {
4965                            broadcast_unit_length: true,
4966                            ..Default::default()
4967                        },
4968                    )?;
4969                }
4970                match having {
4971                    Some(having) => aggregated.filter(having.cast(DataType::Boolean)),
4972                    None => aggregated,
4973                }
4974            },
4975            None => {
4976                // With a column key, every group has a row. A group-by on scalar keys only runs
4977                // as a select over all rows, which may be none.
4978                let has_column_key = group_by_keys
4979                    .iter()
4980                    .any(|key| matches!(strip_outer_alias(key), Expr::Column(_)));
4981                let in_groups = |e: Expr| {
4982                    if has_column_key {
4983                        assume_groups_have_rows(e)
4984                    } else {
4985                        e
4986                    }
4987                };
4988                let group_by = lf.group_by(group_by_keys);
4989                match having {
4990                    Some(having) => group_by.having(in_groups(strip_aggregate_marks(
4991                        splitter.read_keys_per_group(having),
4992                    ))),
4993                    None => group_by,
4994                }
4995                .agg(
4996                    splitter
4997                        .aggregates
4998                        .into_exprs()
4999                        .into_iter()
5000                        .map(in_groups)
5001                        .collect::<Vec<_>>(),
5002                )
5003            },
5004            Some(grouping) => {
5005                // HAVING runs on the combined rows, where it can also see `GROUPING()`.
5006                let having = having
5007                    .map(|having| {
5008                        let having = bind_stored_keys(having, &stored_keys);
5009                        splitter.hoist(having)
5010                    })
5011                    .transpose()?;
5012                let aggregation_projection = splitter.aggregates.into_exprs();
5013                let agg_names = aggregation_projection
5014                    .iter()
5015                    .map(|e| Ok(e.to_field(&schema_before)?.name))
5016                    .collect::<PolarsResult<Vec<_>>>()?;
5017                let aggregated =
5018                    grouping.aggregate(lf, &key_schema, &aggregation_projection, &agg_names)?;
5019                match having {
5020                    Some(having) => aggregated.filter(having.cast(DataType::Boolean)),
5021                    None => aggregated,
5022                }
5023            },
5024        };
5025
5026        let aggregated = if post_agg_projection.is_empty() {
5027            aggregated
5028        } else {
5029            aggregated.with_columns(&post_agg_projection)
5030        };
5031
5032        let projection_schema =
5033            expressions_to_schema(&projections, &schema_before, |duplicate_name: &str| {
5034                format!("group_by aggregations contained duplicate output name '{duplicate_name}'")
5035            })?;
5036
5037        // A final projection to get the proper order and any deferred transforms/aliases
5038        // (will also drop any temporary columns created for the HAVING post-filter)
5039        let final_projection = projection_schema
5040            .iter_names()
5041            .zip(projections.iter().zip(&projection_group_key))
5042            .map(|(name, (projection_expr, group_key))| {
5043                if let Some(expr) = projection_overrides.get(name.as_str()) {
5044                    expr.clone()
5045                } else if let Some(aliased_name) = aliased_aggregations.get(name) {
5046                    col(aliased_name.clone()).alias(name.clone())
5047                } else if let Some(key_name) = group_key {
5048                    // projection is a group key; reference aggregated key col rather than
5049                    // re-evaluating against aggregated frame (incorrect for computed keys)
5050                    if key_name == name {
5051                        col(name.clone())
5052                    } else {
5053                        col(key_name.clone()).alias(name.clone())
5054                    }
5055                } else if group_by_keys_schema.get(name).is_some()
5056                    || projection_aliases.contains(name.as_str())
5057                    || group_key_aliases.contains(name.as_str())
5058                {
5059                    if has_expr(projection_expr, |e| {
5060                        matches!(e, Expr::Agg(_) | Expr::Len | Expr::Over { .. })
5061                    }) {
5062                        col(name.clone())
5063                    } else {
5064                        bind_keys(projection_expr.clone())
5065                    }
5066                } else if is_constant(projection_expr) {
5067                    // Constants are not part of the aggregation; re-evaluate them.
5068                    projection_expr.clone()
5069                } else {
5070                    col(name.clone())
5071                }
5072            })
5073            .collect::<Vec<_>>();
5074
5075        // Include original GROUP BY columns for ORDER BY access (if aliased).
5076        let mut output_projection = final_projection;
5077        let mut output_names: Vec<PlSmallStr> = projection_schema.iter_names_cloned().collect();
5078        let mut push_hidden = |expr: Expr, name: PlSmallStr| {
5079            output_projection.push(expr);
5080            output_names.push(name);
5081        };
5082        for (name, _) in &extra_projections {
5083            push_hidden(col(name.clone()), name.clone());
5084        }
5085        for key_name in group_by_keys_schema.iter_names() {
5086            if !projection_schema.contains(key_name) {
5087                // Original col name not in output - add for ORDER BY access
5088                push_hidden(col(key_name.clone()), key_name.clone());
5089            } else if group_by_keys.iter().any(|k| is_simple_col_ref(k, key_name)) {
5090                // Original col name in output - check if cross-aliased
5091                let is_cross_aliased = projection_schema
5092                    .iter_names()
5093                    .zip(projections.iter())
5094                    .any(|(name, p)| name == key_name && !is_simple_col_ref(p, key_name));
5095                if is_cross_aliased {
5096                    // Add original name under a prefixed alias for subsequent ORDER BY resolution
5097                    let internal_name = format_pl_smallstr!("__POLARS_ORIG_{}", key_name);
5098                    push_hidden(
5099                        col(key_name.clone()).alias(internal_name.clone()),
5100                        internal_name,
5101                    );
5102                }
5103            }
5104        }
5105        // A literal-only projection (`SELECT 1 ... GROUP BY ()` or `SELECT 1 ... HAVING`)
5106        // must keep the height of the aggregated rows, so the columns are added before the
5107        // frame is narrowed.
5108        let projected = if grouping.is_some() || group_by_keys.is_empty() {
5109            aggregated
5110                .with_columns(&output_projection)
5111                .select(output_names.into_iter().map(col).collect::<Vec<_>>())
5112        } else {
5113            aggregated.select(&output_projection)
5114        };
5115        Ok((
5116            projected,
5117            projection_schema.iter_names_cloned().collect(),
5118            order_by,
5119        ))
5120    }

5122    fn process_limit_offset(
5123        &self,
5124        lf: LazyFrame,
5125        limit_clause: &Option<LimitClause>,
5126        fetch: &Option<Fetch>,
5127    ) -> PolarsResult<LazyFrame> {
5128        // Extract limit and offset from LimitClause
5129        let (limit, offset) = match limit_clause {
5130            Some(LimitClause::LimitOffset {
5131                limit,
5132                offset,
5133                limit_by,
5134            }) => {
5135                if !limit_by.is_empty() {
5136                    // TODO: might be able to support as an aggregate `top_k_by` operation?
5137                    //  (https://clickhouse.com/docs/sql-reference/statements/select/limit-by)
5138                    polars_bail!(SQLSyntax: "`LIMIT <n> BY <exprs>` clause is not supported");
5139                }
5140                (limit.as_ref(), offset.as_ref().map(|o| &o.value))
5141            },
5142            Some(LimitClause::OffsetCommaLimit { offset, limit }) => (Some(limit), Some(offset)),
5143            None => (None, None),
5144        };
5145
5146        // Handle FETCH clause (alternative to LIMIT, mutually exclusive)
5147        let limit = match (fetch, limit) {
5148            (Some(fetch), None) => fetch.quantity.as_ref(),
5149            (Some(_), Some(_)) => {
5150                polars_bail!(SQLSyntax: "cannot use both `LIMIT` and `FETCH` in the same query")
5151            },
5152            (None, limit) => limit,
5153        };

5205        &mut self,
5206        ObjectName(idents): &ObjectName,
5207        options: &WildcardAdditionalOptions,
5208        schema: &Schema,
5209    ) -> PolarsResult<Vec<Expr>> {
5210        let mut idents_with_wildcard: Vec<Ident> = idents
5211            .iter()
5212            .filter_map(|p| p.as_ident().cloned())
5213            .collect();
5214        idents_with_wildcard.push(Ident::new("*"));

5220    /// Apply the options of a wildcard to the columns it selects, in this order: `EXCLUDE`,
5221    /// `EXCEPT` and `ILIKE` drop columns, then `REPLACE` changes them, then `RENAME` renames
5222    /// them.
5223    fn process_wildcard_additional_options(
5224        &mut self,
5225        exprs: Vec<Expr>,
5226        options: &WildcardAdditionalOptions,
5227        schema: &Schema,
5228    ) -> PolarsResult<Vec<Expr>> {
5229        if options.opt_except.is_some() && options.opt_exclude.is_some() {
5230            polars_bail!(SQLInterface: "EXCLUDE and EXCEPT wildcard options cannot be used together (prefer EXCLUDE)")
5231        } else if options.opt_exclude.is_some() && options.opt_ilike.is_some() {
5232            polars_bail!(SQLInterface: "EXCLUDE and ILIKE wildcard options cannot be used together")
5233        }
5234
5235        // SELECT * EXCLUDE
5236        let mut exclude = PlHashSet::new();
5237        if let Some(items) = &options.opt_exclude {
5238            match items {
5239                ExcludeSelectItem::Single(name) => {
5240                    exclude.insert(object_name_to_string(name));
5241                },
5242                ExcludeSelectItem::Multiple(names) => {
5243                    exclude.extend(names.iter().map(object_name_to_string));
5244                },
5245            };
5246        }
5247
5248        // SELECT * EXCEPT
5249        if let Some(items) = &options.opt_except {
5250            exclude.insert(items.first_element.value.clone());
5251            exclude.extend(items.additional_elements.iter().map(|i| i.value.clone()));
5252        }
5253
5254        // SELECT * ILIKE
5255        let ilike = options.opt_ilike.as_ref().map(|item| {
5256            let rx = regex::escape(item.pattern.as_str())
5257                .replace('%', ".*")
5258                .replace('_', ".");
5259            polars_utils::regex_cache::compile_regex(format!("^(?is){rx}$").as_str()).unwrap()
5260        });
5261
5262        // Each column keeps its name in the wildcard, also after a rename, so
5263        // `RENAME (a AS b, b AS c)` renames the input `b` and not the renamed `a`.
5264        let mut named_exprs = Vec::with_capacity(exprs.len());
5265        for expr in exprs {
5266            let name = expr.to_field(schema)?.name;
5267            if !exclude.contains(name.as_str())
5268                && ilike.as_ref().is_none_or(|rx| rx.is_match(&name))
5269            {
5270                named_exprs.push((name, expr));
5271            }
5272        }
5273
5274        // SELECT * REPLACE
5275        if let Some(replacements) = &options.opt_replace {
5276            for rp in &replacements.items {
5277                let name = rp.column_name.value.as_str();
5278                let Some((_, expr)) = named_exprs.iter_mut().find(|(n, _)| n == name) else {
5279                    polars_bail!(SQLSyntax: "REPLACE column '{}' is not selected by the wildcard", name)
5280                };
5281                *expr = parse_sql_expr(&rp.expr, self, Some(schema))?.alias(name);
5282            }
5283        }
5284
5285        // SELECT * RENAME
5286        if let Some(items) = &options.opt_rename {
5287            let renames = match items {
5288                RenameSelectItem::Single(rename) => std::slice::from_ref(rename),
5289                RenameSelectItem::Multiple(renames) => renames.as_slice(),
5290            };
5291            for rn in renames {
5292                let before = rn.ident.value.as_str();
5293                if let Some((_, expr)) = named_exprs.iter_mut().find(|(n, _)| n == before) {
5294                    *expr = strip_outer_alias(expr).alias(rn.alias.value.as_str());
5295                }
5296            }
5297        }
5298        Ok(named_exprs.into_iter().map(|(_, expr)| expr).collect())
5299    }

5491/// Casts an `ASOF JOIN` key pair to one dtype. Temporal keys of different dtypes become Int128
5492/// counts of the finest time unit: a common temporal dtype would lose either precision (with the
5493/// coarser unit) or range (with the finer unit).
5494#[cfg(feature = "asof_join")]
5495fn asof_key_pair(
5496    l: Expr,
5497    l_dtype: &DataType,
5498    r: Expr,
5499    r_dtype: &DataType,
5500) -> PolarsResult<(Expr, Expr)> {
5501    if l_dtype == r_dtype {
5502        return Ok((l, r));
5503    }
5504    let supertype = try_get_supertype(l_dtype, r_dtype)?;
5505    if !matches!(supertype, DataType::Datetime(..) | DataType::Duration(_)) {
5506        return Ok((l.cast(supertype.clone()), r.cast(supertype)));
5507    }
5508    let ticks_per_second = |tu: TimeUnit| -> i64 {
5509        match tu {
5510            TimeUnit::Nanoseconds => 1_000_000_000,
5511            TimeUnit::Microseconds => 1_000_000,
5512            TimeUnit::Milliseconds => 1_000,
5513        }
5514    };
5515    let unit = [l_dtype, r_dtype]
5516        .into_iter()
5517        .filter_map(|dtype| match dtype {
5518            DataType::Datetime(tu, _) | DataType::Duration(tu) => Some(*tu),
5519            _ => None,
5520        })
5521        .max_by_key(|tu| ticks_per_second(*tu))
5522        .unwrap_or(TimeUnit::Milliseconds);
5523    let to_ticks = |e: Expr, dtype: &DataType| -> Expr {
5524        let (e, ticks_per_value) = match (dtype, &supertype) {
5525            (DataType::Datetime(tu, _) | DataType::Duration(tu), _) => {
5526                (e, ticks_per_second(unit) / ticks_per_second(*tu))
5527            },
5528            // Milliseconds hold every date without overflow.
5529            (DataType::Date, DataType::Datetime(_, tz)) => (
5530                e.cast(DataType::Datetime(TimeUnit::Milliseconds, tz.clone())),
5531                ticks_per_second(unit) / ticks_per_second(TimeUnit::Milliseconds),
5532            ),
5533            (_, DataType::Datetime(_, tz)) => (e.cast(DataType::Datetime(unit, tz.clone())), 1),
5534            _ => (e.cast(DataType::Duration(unit)), 1),
5535        };
5536        e.to_physical().cast(DataType::Int128) * lit(ticks_per_value)
5537    };
5538    Ok((to_ticks(l, l_dtype), to_ticks(r, r_dtype)))
5539}

5578/// The key of `map` that `name` refers to: an exact match, else the only key equal to it
5579/// ignoring case. Several keys equal to it ignoring case are an error.
5580fn key_ignoring_case<'a, V>(
5581    map: &'a PlHashMap<String, V>,
5582    name: &str,
5583) -> PolarsResult<Option<&'a String>> {
5584    if let Some((key, _)) = map.get_key_value(name) {
5585        return Ok(Some(key));
5586    }
5587    let mut matches = map.keys().filter(|key| key.eq_ignore_ascii_case(name));
5588    let first = matches.next();
5589    if matches.next().is_some() {
5590        let mut names: Vec<String> = map
5591            .keys()
5592            .filter(|key| key.eq_ignore_ascii_case(name))
5593            .map(|key| format!("'{key}'"))
5594            .collect();
5595        names.sort();

5597            SQLInterface: "relation '{}' matches more than one table ignoring case ({}); write the name with its exact case",
5598            name, names.join(", ")
5599        );
5600    }
5601    Ok(first)

5770/// Read the resolved scalar subqueries in the inputs of the windows of `expr`, and those in
5771/// `read_per_row` in its aggregates, as their column, which holds the value on every row:
5772/// these functions read one value per row.
5773fn broadcast_subqueries_in_inputs(
5774    expr: Expr,
5775    subquery_names: &PlHashSet<PlSmallStr>,
5776    read_per_row: &PlHashSet<PlSmallStr>,
5777) -> Expr {
5778    fn broadcast(
5779        expr: Expr,
5780        in_window: bool,
5781        in_aggregate: bool,
5782        subquery_names: &PlHashSet<PlSmallStr>,
5783        read_per_row: &PlHashSet<PlSmallStr>,
5784    ) -> Expr {
5785        match expr {
5786            Expr::Agg(AggExpr::First(inner))
5787                if matches!(
5788                    inner.as_ref(),
5789                    Expr::Column(name) if subquery_names.contains(name)
5790                        && (in_window || (in_aggregate && read_per_row.contains(name)))
5791                ) =>
5792            {
5793                Arc::unwrap_or_clone(inner)
5794            },
5795            e => {
5796                let in_window = in_window || matches!(e, Expr::Over { .. });
5797                let in_aggregate = in_aggregate || is_marked_aggregate(&e);

5801                            c,
5802                            in_window,
5803                            in_aggregate,
5804                            subquery_names,
5805                            read_per_row,
5806                        ))
5807                    },
5808                    &mut (),
5809                )
5810                .unwrap()
5811            },
5812        }
5813    }
5814    if subquery_names.is_empty() {
5815        return expr;
5816    }
5817    broadcast(expr, false, false, subquery_names, read_per_row)

5879/// Determine which parsed join expressions actually belong in `left_on` and which in `right_on`.
5880///
5881/// This needs to be handled carefully because in SQL joins you can write "join on" constraints
5882/// either way round, and in joins with more than two tables you can also join against an earlier
5883/// table (e.g.: you could be joining `df1` to `df2` to `df3`, but the final join condition where
5884/// we join `df2` to `df3` could refer to `df1.a = df3.b`); this takes a little more work to
5885/// resolve as our native `join` function operates on only two tables at a time.
5886///
5887/// An operand can also be constant (`ON df1.x = df2.x AND df2.flag = 'y'`), in which case it has
5888/// no table of its own and takes whichever side the other operand does not.
5889fn determine_left_right_join_on(
5890    ctx: &mut SQLContext,
5891    expr_left: &SQLExpr,
5892    expr_right: &SQLExpr,
5893    tbl_left: &TableInfo,
5894    tbl_right: &TableInfo,
5895    join_schema: &Schema,
5896) -> PolarsResult<(Vec<Expr>, Vec<Expr>)> {
5897    // parse, removing any aliases that may have been added by `resolve_column`
5898    // (called inside `parse_sql_expr`) as we need the actual/underlying col
5899    let refers_to = |expr: &SQLExpr, tbl: &TableInfo| {
5900        tbl.relations
5901            .iter()
5902            .any(|name| expr_refers_to_table(expr, name, &ctx.active_relations))
5903    };
5904    let left_refs = (
5905        refers_to(expr_left, tbl_left),
5906        refers_to(expr_left, tbl_right),
5907    );
5908    let right_refs = (
5909        refers_to(expr_right, tbl_left),
5910        refers_to(expr_right, tbl_right),
5911    );
5912    // an operand's dtypes come from the table it names; the merged schema keeps the left
5913    // dtype for a column name that exists in both tables
5914    let operand_schema = |refs: (bool, bool)| -> &Schema {
5915        match refs {
5916            (true, false) => &tbl_left.schema,
5917            (false, true) => &tbl_right.schema,
5918            _ => join_schema,
5919        }
5920    };
5921    let (left_schema, right_schema) = (operand_schema(left_refs), operand_schema(right_refs));
5922    let left_on = strip_join_aliases(parse_sql_expr(expr_left, ctx, Some(left_schema))?);
5923    let right_on = strip_join_aliases(parse_sql_expr(expr_right, ctx, Some(right_schema))?);
5924    let (left_on, right_on) = convert_int_literal_for_string(
5925        left_on,
5926        right_on,
5927        |e| is_string_expr(e, left_schema),
5928        |e| is_string_expr(e, right_schema),
5929    );
5930
5931    // a constant operand is a literal, or any other expression referencing no column (such as
5932    // `UPPER('it')`); it can be evaluated against either input, so it has no table affinity
5933    let (left_is_const, right_is_const) = (
5934        !expr_references_any_column(expr_left),
5935        !expr_references_any_column(expr_right),
5936    );
5937
5938    // ------------------------------------------------------------------
5939    // simple/typical case: can fully resolve SQL-level table references
5940    // ------------------------------------------------------------------
5941    // if the SQL-level references unambiguously indicate table ownership, we're done
5942    match (left_refs, right_refs) {
5943        // standard: left expr → left table, right expr → right table
5944        ((true, false), (false, true)) => return Ok((vec![left_on], vec![right_on])),
5945        // reversed: left expr → right table, right expr → left table
5946        ((false, true), (true, false)) => return Ok((vec![right_on], vec![left_on])),
5947        // qualified column vs constant: the qualifier alone settles it (note that this also
5948        // covers a column name that is present in *both* schemas, which schema-based
5949        // resolution below cannot disambiguate)
5950        ((true, false), _) if right_is_const => return Ok((vec![left_on], vec![right_on])),
5951        ((false, true), _) if right_is_const => return Ok((vec![right_on], vec![left_on])),
5952        (_, (false, true)) if left_is_const => return Ok((vec![left_on], vec![right_on])),
5953        (_, (true, false)) if left_is_const => return Ok((vec![right_on], vec![left_on])),
5954        // unsupported: one side references *both* tables
5955        ((true, true), _) | (_, (true, true)) if tbl_left.name != tbl_right.name => {
5956            polars_bail!(
5957               SQLInterface: "unsupported join condition: {} side references both '{}' and '{}'",
5958               if left_refs.0 && left_refs.1 {
5959                    "left"
5960                } else {
5961                    "right"
5962                }, tbl_left.name, tbl_right.name
5963            )
5964        },
5965        // fall through to the more involved col/ref resolution
5966        _ => {},
5967    }
5968
5969    // ------------------------------------------------------------------
5970    // more involved: additionally employ schema-based column resolution
5971    // (applies to unqualified columns and/or chained joins)
5972    // ------------------------------------------------------------------
5973    // a constant resolves against either schema, so we report it as belonging to both and let
5974    // the arms below take the side that the other (schema-unique) operand leaves free
5975    let cols_in_schemas = |expr: &Expr, is_const: bool| {
5976        if is_const {
5977            (true, true)
5978        } else {
5979            (
5980                expr_cols_all_in_schema(expr, &tbl_left.schema),
5981                expr_cols_all_in_schema(expr, &tbl_right.schema),
5982            )
5983        }
5984    };
5985    let left_on_cols_in = cols_in_schemas(&left_on, left_is_const);
5986    let right_on_cols_in = cols_in_schemas(&right_on, right_is_const);
5987
5988    match (left_on_cols_in, right_on_cols_in) {
5989        // each expression's columns exist in exactly one schema
5990        ((true, false), (false, true)) => Ok((vec![left_on], vec![right_on])),
5991        ((false, true), (true, false)) => Ok((vec![right_on], vec![left_on])),
5992        // one expression is in both schemas (or is constant) and the other in only one;
5993        // the unique one decides, and the ambiguous one takes the remaining side
5994        ((true, true), (true, false)) => Ok((vec![right_on], vec![left_on])),
5995        ((true, true), (false, true)) => Ok((vec![left_on], vec![right_on])),
5996        ((true, false), (true, true)) => Ok((vec![left_on], vec![right_on])),
5997        ((false, true), (true, true)) => Ok((vec![right_on], vec![left_on])),
5998        // pass through as-is
5999        _ => Ok((vec![left_on], vec![right_on])),
6000    }
6001}

6003/// Returns `(left_on, right_on, join_where_predicates)`.
6004///
6005/// - Equi-conditions (`=`) are returned as paired `left_on`/`right_on` entries.
6006/// - Any other condition is returned as a `join_where` predicate that references columns
6007///   using their merged-schema names (right columns that conflict with the left schema are
6008///   suffixed).
6009fn process_join_on(
6010    ctx: &mut SQLContext,
6011    sql_expr: &SQLExpr,
6012    tbl_left: &TableInfo,
6013    tbl_right: &TableInfo,
6014) -> PolarsResult<(Vec<Expr>, Vec<Expr>, Vec<Expr>)> {
6015    match sql_expr {
6016        SQLExpr::BinaryOp { left, op, right } => match op {
6017            SQLBinaryOperator::And => {
6018                let (mut left_i, mut right_i, mut preds_i) =
6019                    process_join_on(ctx, left, tbl_left, tbl_right)?;
6020                let (mut left_j, mut right_j, mut preds_j) =
6021                    process_join_on(ctx, right, tbl_left, tbl_right)?;
6022                left_i.append(&mut left_j);
6023                right_i.append(&mut right_j);
6024                preds_i.append(&mut preds_j);
6025                Ok((left_i, right_i, preds_i))
6026            },
6027            SQLBinaryOperator::Eq => {
6028                // Both operands read the same input: not a key pair but a condition on
6029                // that input.
6030                let left_input = join_operand_input(ctx, left, tbl_left, tbl_right);
6031                if left_input.is_some()
6032                    && left_input == join_operand_input(ctx, right, tbl_left, tbl_right)
6033                {
6034                    return process_join_predicate(ctx, sql_expr, tbl_left, tbl_right);
6035                }
6036                let join_schema = build_join_schema(tbl_left, tbl_right)?;
6037                let (l, r) = determine_left_right_join_on(
6038                    ctx,
6039                    left,
6040                    right,
6041                    tbl_left,
6042                    tbl_right,
6043                    &join_schema,
6044                )?;
6045                Ok((l, r, vec![]))
6046            },
6047            _ => process_join_predicate(ctx, sql_expr, tbl_left, tbl_right),
6048        },
6049        SQLExpr::Nested(expr) => process_join_on(ctx, expr, tbl_left, tbl_right),
6050        _ => process_join_predicate(ctx, sql_expr, tbl_left, tbl_right),
6051    }
6052}

6081/// Parse a join condition other than a plain equality into a `join_where` predicate over
6082/// the joined frame, where right-table columns that also exist in the left table carry a suffix.
6083fn process_join_predicate(
6084    ctx: &mut SQLContext,
6085    sql_expr: &SQLExpr,
6086    tbl_left: &TableInfo,
6087    tbl_right: &TableInfo,
6088) -> PolarsResult<(Vec<Expr>, Vec<Expr>, Vec<Expr>)> {
6089    let suffix = format!(":{}", tbl_right.name);
6090    let conflicts = |name: &str| tbl_left.schema.contains(name) && tbl_right.schema.contains(name);
6091
6092    let mut joined_schema = build_join_schema(tbl_left, tbl_right)?;
6093    for (name, dtype) in tbl_right.schema.iter().filter(|(name, _)| conflicts(name)) {
6094        joined_schema.insert(
6095            PlSmallStr::from_string(format!("{name}{suffix}")),
6096            dtype.clone(),
6097        );
6098    }
6099
6100    // `right_table.col` -> `col:right_table` when the name is also a left column
6101    let mut sql_expr = sql_expr.clone();
6102    let _ = visit_expressions_mut(&mut sql_expr, |e| {
6103        if let SQLExpr::CompoundIdentifier(idents) = e
6104            && let Ok(Some((name, len))) = ctx.resolve_qualifier(idents)
6105            && name == tbl_right.name.as_str()
6106            && conflicts(&idents[len].value)
6107        {
6108            let suffixed = Ident::new(format!("{}{suffix}", idents[len].value));
6109            idents.splice(0..=len, [suffixed]);
6110            if idents.len() == 1 {
6111                *e = SQLExpr::Identifier(idents.pop().unwrap());
6112            }
6113        }
6114        ControlFlow::<()>::Continue(())
6115    });
6116    let predicate = strip_join_aliases(parse_sql_expr(&sql_expr, ctx, Some(&joined_schema))?);
6117    Ok((vec![], vec![], vec![predicate]))
6118}

6177fn process_join_constraint(
6178    constraint: &JoinConstraint,
6179    tbl_left: &TableInfo,
6180    tbl_right: &TableInfo,
6181    ctx: &mut SQLContext,
6182) -> PolarsResult<(Vec<Expr>, Vec<Expr>, Vec<Expr>)> {
6183    match constraint {
6184        JoinConstraint::On(expr) => {
6185            let (left_on, right_on, predicates) = process_join_on(ctx, expr, tbl_left, tbl_right)?;
6186            // `a = b AND b = a` gives the same key pair twice.
6187            let mut keys: Vec<(Expr, Expr)> = Vec::with_capacity(left_on.len());
6188            for key in left_on.into_iter().zip(right_on) {
6189                if !keys.contains(&key) {
6190                    keys.push(key);
6191                }
6192            }
6193            let (left_on, right_on) = keys.into_iter().unzip();
6194            Ok((left_on, right_on, predicates))
6195        },
6196        JoinConstraint::Using(idents) if !idents.is_empty() => {
6197            let using: Vec<Expr> = idents
6198                .iter()
6199                .map(|ObjectName(parts)| {
6200                    if parts.len() != 1 {
6201                        polars_bail!(SQLSyntax: "JOIN \"USING\" clause expects simple column names, not qualified names");
6202                    }
6203                    match parts[0].as_ident() {
6204                        Some(ident) => Ok(col(ident.value.as_str())),
6205                        None => polars_bail!(SQLSyntax: "JOIN \"USING\" clause expects identifiers, not functions"),
6206                    }
6207                })
6208                .collect::<PolarsResult<Vec<_>>>()?;
6209            Ok((using.clone(), using, vec![]))
6210        },
6211        JoinConstraint::Natural => {
6212            let left_names = tbl_left.schema.iter_names().collect::<PlHashSet<_>>();
6213            let right_names = tbl_right.schema.iter_names().collect::<PlHashSet<_>>();
6214            let on: Vec<Expr> = left_names
6215                .intersection(&right_names)
6216                .map(|&name| col(name.clone()))
6217                .collect();
6218            if on.is_empty() {
6219                polars_bail!(SQLInterface: "no common columns found for NATURAL JOIN")
6220            }
6221            Ok((on.clone(), on, vec![]))
6222        },
6223        _ => polars_bail!(SQLInterface: "unsupported SQL join constraint:\n{:?}", constraint),
6224    }

6270    right_schema: &Schema,
6271) -> bool {
6272    let refers_to = |expr: &SQLExpr, table: &str| expr_refers_to_table(expr, table, relations);
6273    // A condition naming a relation joined in a later round stays in the
6274    // residual WHERE until that relation is present.
6275    if deferred_tables.iter().any(|table| refers_to(expr, table)) {
6276        return false;
6277    }
6278    // A subquery is not a join key; its correlated columns name outer relations.
6279    if expr_contains_subquery(expr) {
6280        return false;
6281    }
6282    if let SQLExpr::BinaryOp {
6283        left,
6284        op:
6285            SQLBinaryOperator::Eq
6286            | SQLBinaryOperator::Lt
6287            | SQLBinaryOperator::LtEq
6288            | SQLBinaryOperator::Gt
6289            | SQLBinaryOperator::GtEq
6290            | SQLBinaryOperator::NotEq,
6291        right,
6292    } = expr
6293    {
6294        let operand_side = |operand: &SQLExpr| -> (bool, bool) {
6295            let refs_right = refers_to(operand, right_table);
6296            let refs_left = left_tables.iter().any(|t| refers_to(operand, t));
6297            if refs_right || refs_left {
6298                (refs_right, refs_left)
6299            } else {
6300                let in_right = sql_expr_cols_all_in_schema(operand, right_schema);
6301                let in_left = sql_expr_cols_all_in_schema(operand, left_schema);
6302                (in_right && !in_left, in_left && !in_right)
6303            }
6304        };
6305
6306        let (left_is_right, left_is_left) = operand_side(left);
6307        let (right_is_right, right_is_left) = operand_side(right);
6308
6309        // One operand must name only a left table and the other only the right one;
6310        // an operand spanning both stays a filter, as it belongs to neither side.
6311        (left_is_right && !left_is_left && right_is_left && !right_is_right)
6312            || (right_is_right && !right_is_left && left_is_left && !left_is_right)
6313    } else {
6314        false
6315    }
6316}

6318/// Partition a WHERE clause into join predicates (bridging left tables and the
6319/// right table) and residual filter conditions.
6320fn extract_join_predicates(
6321    where_expr: &Option<SQLExpr>,
6322    left_tables: &[String],
6323    right_table: &str,
6324    deferred_tables: &[String],
6325    relations: &[RelationName],
6326    left_schema: &Schema,
6327    right_schema: &Schema,
6328) -> (Option<SQLExpr>, Option<SQLExpr>) {
6329    let Some(expr) = where_expr else {
6330        return (None, None);
6331    };
6332    let conditions = flatten_and_conditions(expr);
6333    let mut join_conds = Vec::new();
6334    let mut filter_conds = Vec::new();
6335    for cond in conditions {
6336        if is_join_comparison(
6337            cond,
6338            left_tables,
6339            right_table,
6340            deferred_tables,
6341            relations,
6342            left_schema,
6343            right_schema,
6344        ) {
6345            join_conds.push(cond.clone());
6346        } else {
6347            filter_conds.push(cond.clone());
6348        }
6349    }
6350    (
6351        combine_conditions(join_conds, SQLBinaryOperator::And),
6352        combine_conditions(filter_conds, SQLBinaryOperator::And),
6353    )
6354}

6380/// The order to register the CTEs of a `WITH RECURSIVE` in, where a CTE may read any other,
6381/// and whether each CTE reads itself. Each CTE comes after the other CTEs it reads; CTEs that
6382/// read each other are not supported.
6383fn recursive_cte_order(with: &With) -> PolarsResult<(Vec<usize>, Vec<bool>)> {
6384    let names: Vec<String> = with
6385        .cte_tables
6386        .iter()
6387        .map(|cte| cte.alias.name.value.clone())
6388        .collect();
6389    let mut seen = PlHashSet::with_capacity(names.len());
6390    if let Some(name) = names
6391        .iter()
6392        .find(|name| !seen.insert(name.to_ascii_lowercase()))
6393    {
6394        polars_bail!(SQLInterface: "CTE name '{}' is used more than once", name)
6395    }
6396    let mut deps: Vec<PlHashSet<usize>> = with
6397        .cte_tables
6398        .iter()
6399        .map(|cte| referenced_ctes(&cte.query, &names))
6400        .collect();
6401    let reads_itself: Vec<bool> = deps
6402        .iter_mut()
6403        .enumerate()
6404        .map(|(idx, deps)| deps.remove(&idx))
6405        .collect();
6406    let mut order = Vec::with_capacity(names.len());
6407    let mut done = vec![false; names.len()];
6408    while order.len() < names.len() {
6409        // Keep the written order among the CTEs that are ready.
6410        let Some(next) =
6411            (0..names.len()).find(|idx| !done[*idx] && deps[*idx].iter().all(|dep| done[*dep]))
6412        else {
6413            let cycle: Vec<&str> = (0..names.len())
6414                .filter(|idx| !done[*idx])
6415                .map(|idx| names[idx].as_str())
6416                .collect();
6417            polars_bail!(
6418                SQLInterface: "mutually recursive CTEs are not supported ({})", cycle.join(", ")
6419            )
6420        };
6421        done[next] = true;
6422        order.push(next);
6423    }
6424    Ok((order, reads_itself))
6425}

6504bitflags::bitflags! {
6505    /// Bitfield indicating whether there exists a projection with the specified height behavior.
6506    ///
6507    /// Used to help determine whether to execute projections in `select()` or `with_columns()`
6508    /// context.
6509    #[derive(PartialEq)]
6510    struct ExprSqlProjectionHeightBehavior: u8 {
6511        /// Maintains the height of input column(s)
6512        const MaintainsColumn = 1 << 0;
6513        /// Height is independent of input, e.g.:
6514        /// * expressions that change length: e.g. slice, explode, filter, gather etc.
6515        /// * aggregations: count(*), first(), sum() etc.
6516        const Independent = 1 << 1;
6517        /// "Inherits" the height of the context, e.g.:
6518        /// * Scalar literals
6519        const InheritsContext = 1 << 2;
6520    }
6521}
6522
6523impl ExprSqlProjectionHeightBehavior {
6524    fn identify_from_expr(expr: &Expr) -> Self {
6525        let mut has_column = false;
6526        let mut has_independent = false;
6527
6528        for e in expr.into_iter() {
6529            use Expr::*;
6530            has_column |= matches!(e, Column(_) | Selector(_));
6531            has_independent |= match e {
6532                // @TODO: This is broken now with functions.
6533                AnonymousFunction { options, .. } => {
6534                    options.returns_scalar() || !options.is_length_preserving()
6535                },
6536                Literal(v) => !v.is_scalar(),
6537                Explode { .. } | Filter { .. } | Gather { .. } | Slice { .. } => true,
6538                Agg { .. } | Len => true,
6539                _ => false,
6540            }
6541        }
6542        if has_independent {
6543            Self::Independent
6544        } else if has_column {
6545            Self::MaintainsColumn
6546        } else {
6547            Self::InheritsContext
6548        }
6549    }
