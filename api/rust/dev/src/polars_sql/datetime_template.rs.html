6/// What a template pattern stands for.
7#[derive(Clone, Copy)]
8enum Item {
9    /// A number written with the strftime spec `%c`, zero padded unless after `FM`.
10    Number(char),
11    /// A strftime spec that `FM` does not change.
12    Fixed(&'static str),
13    /// A fraction of a second, with its strftime spec for writing.
14    Fraction(&'static str),
15    /// A day or month name; `true` if it is padded to 9 characters unless after `FM`.
16    Name(&'static str, bool),
17    Meridian,
18    Unsupported,
19}

81#[derive(Clone, Copy, PartialEq)]
82enum Case {
83    Upper,
84    Title,
85    Lower,
86}

88/// A piece of the text that `TO_CHAR` writes.
89enum Part {
90    Strftime(String),
91    /// A name that strftime cannot write as asked: not in title case, or padded.
92    Name {
93        spec: &'static str,
94        case: Case,
95        pad: bool,
96    },
97}

99/// The pattern at the start of `rest`, with its length and case. Patterns match in upper or
100/// lower case, and names also in title case (`Month`), as in PostgreSQL.
101fn match_pattern(rest: &str) -> Option<(usize, Item, Case)> {
102    PATTERNS.iter().find_map(|&(pattern, item)| {
103        let head = rest.get(..pattern.len())?;
104        if !head.eq_ignore_ascii_case(pattern) {
105            return None;
106        }
107        let is_lower = |s: &str| s.bytes().all(|b| !b.is_ascii_uppercase());
108        let case = if head == pattern {
109            Case::Upper
110        } else if is_lower(head) {
111            Case::Lower
112        } else if matches!(item, Item::Name(..))
113            && head.as_bytes()[0].is_ascii_uppercase()
114            && is_lower(&head[1..])
115        {
116            Case::Title
117        } else {
118            return None;
119        };
120        Some((pattern.len(), item, case))
121    })
122}

124/// A template as strftime parts, with the fields that parsing needs to know about.
125#[derive(Default)]
126struct Template {
127    parts: Vec<Part>,
128    year: bool,
129    month: bool,
130    day: bool,
131    day_of_year: bool,
132    hour: bool,
133    hour12: bool,
134    minute: bool,
135    second: bool,
136    fraction: bool,
137    meridian: bool,
138    iso_year: bool,
139    iso_week: bool,
140    iso_weekday: bool,
141}

143/// The parts of `template`; `output` is true for `TO_CHAR` and false for parsing.
144fn template_parts(template: &str, function: &str, output: bool) -> PolarsResult<Template> {
145    fn push_str(parts: &mut Vec<Part>, s: &str) {
146        match parts.last_mut() {
147            Some(Part::Strftime(fmt)) => fmt.push_str(s),
148            _ => parts.push(Part::Strftime(s.to_string())),
149        }
150    }
151    fn push_literal(parts: &mut Vec<Part>, c: char) {
152        match c {
153            '%' => push_str(parts, "%%"),
154            c => push_str(parts, c.encode_utf8(&mut [0; 4])),
155        }
156    }
157
158    let mut out = Template::default();
159    let parts = &mut out.parts;
160    let mut fractions = 0;
161    let mut fill = false;
162    let mut rest = template;
163    while let Some(c) = rest.chars().next() {
164        if c == '"' {
165            // Quoted text is copied; a backslash takes the next character as is.
166            let mut chars = rest[1..].char_indices();
167            let mut end = rest.len();
168            while let Some((i, c)) = chars.next() {
169                match c {
170                    '"' => {
171                        end = i + 2;
172                        break;
173                    },
174                    '\\' => {
175                        if let Some((_, c)) = chars.next() {
176                            push_literal(parts, c);
177                        }
178                    },
179                    c => push_literal(parts, c),
180                }
181            }
182            rest = &rest[end..];
183            fill = false;
184            continue;
185        }
186        if rest.starts_with("\\\"") {
187            push_literal(parts, '"');
188            rest = &rest[2..];
189            fill = false;
190            continue;
191        }
192        if rest.starts_with("FM") || rest.starts_with("fm") {
193            fill = true;
194            rest = &rest[2..];
195            continue;
196        }
197        let Some((len, item, case)) = match_pattern(rest) else {
198            push_literal(parts, c);
199            rest = &rest[c.len_utf8()..];
200            fill = false;
201            continue;
202        };
203        match item {
204            Item::Number(spec) => {
205                match spec {
206                    'Y' | 'y' => out.year = true,
207                    'm' => out.month = true,
208                    'd' => out.day = true,
209                    'j' => out.day_of_year = true,
210                    'H' => out.hour = true,
211                    'I' => (out.hour, out.hour12) = (true, true),
212                    'M' => out.minute = true,
213                    'S' => out.second = true,
214                    'G' | 'g' => out.iso_year = true,
215                    'V' => out.iso_week = true,
216                    _ => {},
217                }
218                if fill {
219                    push_str(parts, &format!("%-{spec}"))
220                } else {
221                    push_str(parts, &format!("%{spec}"))
222                }
223            },
224            Item::Fixed(spec) => {
225                out.iso_weekday |= spec == "%u";
226                push_str(parts, spec)
227            },
228            Item::Fraction(spec) if output => push_str(parts, spec),
229            Item::Fraction(spec) => {
230                out.fraction = true;
231                fractions += 1;
232                polars_ensure!(
233                    fractions == 1,
234                    SQLInterface: "{} does not support more than one fraction of a second", function
235                );
236                // After a dot, any number of digits is a fraction, as in PostgreSQL.
237                match parts.last_mut() {
238                    Some(Part::Strftime(fmt)) if fmt.ends_with('.') => {
239                        fmt.pop();
240                        fmt.push_str("%.f");
241                    },
242                    _ => push_str(parts, spec),
243                }
244            },
245            Item::Meridian => {
246                out.meridian = true;
247                if output && case == Case::Lower {
248                    push_str(parts, "%P")
249                } else {
250                    push_str(parts, "%p")
251                }
252            },
253            Item::Name(spec, padded) => {
254                out.iso_weekday |= matches!(spec, "%A" | "%a");
255                out.month |= matches!(spec, "%B" | "%b");
256                let pad = padded && !fill;
257                if !output {
258                    push_str(parts, spec);
259                    // A space reads the padding, if any.
260                    if pad {
261                        push_str(parts, " ");
262                    }
263                } else if case == Case::Title && !pad {
264                    push_str(parts, spec);
265                } else {
266                    parts.push(Part::Name { spec, case, pad });
267                }
268            },
269            Item::Unsupported => {
270                polars_bail!(SQLInterface: "{} does not support the template pattern '{}'", function, &rest[..len])
271            },
272        }
273        rest = &rest[len..];
274        fill = false;
275    }
276    Ok(out)
277}

279/// How to parse text written with a PostgreSQL template.
280pub(crate) struct ParseFormat {
281    pub(crate) format: String,
282    /// Text to add to each value for the fields that PostgreSQL fills in when the template has
283    /// none: AM for a 12-hour clock, Monday for an ISO week, 1 for the month or day of a date,
284    /// and 0 for the hour, minute or second of a time.
285    pub(crate) defaults: String,
286}
287
288/// The strftime format that parses text written with a PostgreSQL `template`. A template with
289/// `%` in it is already a strftime format.
290pub(crate) fn parse_format(template: &str, function: &str) -> PolarsResult<ParseFormat> {
291    let mut out = ParseFormat {
292        format: template.to_string(),
293        defaults: String::new(),
294    };
295    if template.contains('%') {
296        return Ok(out);
297    }
298    let template = template_parts(template, function, false)?;
299    polars_ensure!(
300        !template.iso_year || template.iso_week,
301        SQLInterface: "{} needs IW to parse IYYY or IY", function
302    );
303    out.format = template
304        .parts
305        .into_iter()
306        .map(|part| match part {
307            Part::Strftime(fmt) => fmt,
308            Part::Name { spec, .. } => spec.to_string(),
309        })
310        .collect();
311    if template.hour12 && !template.meridian {
312        out.format.push_str(" %p");
313        out.defaults.push_str(" AM");
314    }
315    if template.iso_week && !template.iso_weekday {
316        out.format.push_str(" %u");
317        out.defaults.push_str(" 1");
318    }
319    let date = template.year || template.month || template.day;
320    if date && !template.iso_week && !template.day_of_year {
321        if !template.month {
322            out.format.push_str(" %m");
323            out.defaults.push_str(" 01");
324        }
325        if !template.day {
326            out.format.push_str(" %d");
327            out.defaults.push_str(" 01");
328        }
329    }
330    // Parsing a time needs its hour and minute, and a fraction needs its second.
331    let time = template.hour || template.minute || template.second || template.fraction;
332    if template.fraction && !template.second {
333        out.format.push_str(" %S");
334        out.defaults.push_str(" 00");
335    }
336    if time && !template.hour {
337        out.format.push_str(" %H");
338        out.defaults.push_str(" 00");
339    }
340    if time && !template.minute {
341        out.format.push_str(" %M");
342        out.defaults.push_str(" 00");
343    }
344    Ok(out)
345}

347/// `expr` (a date, datetime or time) written as text with a PostgreSQL `template`. A template
348/// with `%` in it is a strftime format.
349pub(crate) fn format_with_template(expr: Expr, template: &str) -> PolarsResult<Expr> {
350    if template.contains('%') {
351        return Ok(expr.dt().strftime(template));
352    }
353    let mut pieces = template_parts(template, "TO_CHAR", true)?
354        .parts
355        .into_iter()
356        .map(|part| match part {
357            Part::Strftime(fmt) => expr.clone().dt().strftime(&fmt),
358            Part::Name { spec, case, pad } => {
359                let name = expr.clone().dt().strftime(spec);
360                let name = match case {
361                    Case::Upper => name.str().to_uppercase(),
362                    Case::Lower => name.str().to_lowercase(),
363                    Case::Title => name,
364                };
365                if pad {
366                    name.str().pad_end(lit(9), ' ')
367                } else {
368                    name
369                }
370            },
371        })
372        .collect::<Vec<_>>();
373    Ok(match pieces.len() {
374        0 => expr.dt().strftime(""),
375        1 => pieces.pop().unwrap(),
376        _ => concat_str(pieces, "", false),
377    })
