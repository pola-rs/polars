16/// The expressions computed in the group context, without aggregate marks.
17pub(crate) struct AggregateOutputs {
18    exprs: Vec<Expr>,
19}
20
21impl AggregateOutputs {
22    pub(crate) fn with_capacity(capacity: usize) -> Self {
23        Self {
24            exprs: Vec::with_capacity(capacity),
25        }
26    }
27
28    pub(crate) fn push(&mut self, expr: Expr) {
29        self.exprs.push(strip_aggregate_marks(expr));
30    }
31
32    pub(crate) fn iter(&self) -> impl Iterator<Item = &Expr> {
33        self.exprs.iter()
34    }
35
36    pub(crate) fn into_exprs(self) -> Vec<Expr> {
37        self.exprs
38    }
39
40    /// Reference to the output computing `expr`, reusing an existing aggregate with
41    /// the same expression (from SELECT, or the same aggregate repeated in a window).
42    fn get_or_insert_hoisted(&mut self, expr: Expr) -> Expr {
43        let expr = strip_aggregate_marks(expr);
44        let existing = self.exprs.iter().find_map(|agg| match agg {
45            Expr::Alias(inner, name) if **inner == expr => Some(name.clone()),
46            _ => None,
47        });
48        let name = existing.unwrap_or_else(|| {
49            let name = format_pl_smallstr!("__POLARS_HOISTED_AGG_{}", unique_column_name());
50            self.exprs.push(expr.alias(name.clone()));
51            name
52        });
53        col(name)
54    }
55}

57/// Splits post-aggregation expressions from the aggregates they contain.
58pub(crate) struct GroupContextSplitter<'a> {
59    pub(crate) keys: &'a Schema,
60    /// Each group key without its alias, and the column holding it after aggregation.
61    pub(crate) key_exprs: Vec<(Expr, PlSmallStr)>,
62    pub(crate) whole_frame_partition: Option<&'a PlSmallStr>,
63    /// Columns holding the result of a scalar subquery.
64    pub(crate) subquery_names: &'a PlHashSet<PlSmallStr>,
65    /// Whether a scalar subquery outside the aggregates is read from the aggregated rows.
66    /// Without GROUP BY there is one aggregated row even when there is no input row, so
67    /// only that row can hold its value.
68    pub(crate) subqueries_after_aggregation: bool,
69    /// The subqueries that are read from the aggregated rows.
70    pub(crate) subqueries_read_after: PlIndexSet<PlSmallStr>,
71    pub(crate) aggregates: AggregateOutputs,
72}
73
74impl GroupContextSplitter<'_> {
75    /// Whether `expr` has one value per group without being an aggregate call: a scalar
76    /// subquery, or a correlated column.
77    fn is_group_value(&self, expr: &Expr) -> bool {
78        matches!(expr, Expr::Agg(AggExpr::First(inner)) if matches!(inner.as_ref(), Expr::Column(name)
79            if self.subquery_names.contains(name) || is_correlated_result_col(name)))
80    }
81
82    /// The scalar subquery that `expr` reads from the aggregated rows.
83    fn subquery_read_after<'e>(&self, expr: &'e Expr) -> Option<&'e PlSmallStr> {
84        match expr {
85            Expr::Agg(AggExpr::First(inner)) if self.subqueries_after_aggregation => {
86                match inner.as_ref() {
87                    Expr::Column(name) if self.subquery_names.contains(name) => Some(name),
88                    _ => None,
89                }
90            },
91            _ => None,
92        }
93    }
94
95    fn read_after_aggregation(&mut self, name: PlSmallStr) -> Expr {
96        self.subqueries_read_after.insert(name.clone());
97        Expr::Column(name)
98    }
99
100    /// Whether a SELECT projection must be processed in the group context rather
101    /// than passed through as a group key: it contains an aggregate, a window, or a
102    /// function over a non-key column.
103    pub(crate) fn requires_group_processing(&self, expr: &Expr) -> bool {
104        has_expr(expr, |e| match e {
105            _ if is_marked_aggregate(e) => true,
106            Expr::Agg(_) | Expr::Len | Expr::Over { .. } => true,
107            #[cfg(feature = "dynamic_group_by")]
108            Expr::Rolling { .. } => true,
109            Expr::AnonymousFunction { options, .. } => options.returns_scalar(),
110            Expr::Function { function: func, .. }
111                if !matches!(func, FunctionExpr::StructExpr(_)) =>
112            {
113                has_expr(
114                    e,
115                    |e| matches!(e, Expr::Column(name) if !self.keys.contains(name)),
116                )
117            },
118            _ => false,
119        })
120    }
121
122    /// Whether `expr` must run after aggregation: it holds a window or a subquery read from
123    /// the aggregated rows, or combines an aggregate with a grouped key.
124    pub(crate) fn needs_post_aggregation(&self, expr: &Expr) -> bool {
125        struct Finder<'a, 'b> {
126            splitter: &'a GroupContextSplitter<'b>,
127            has_window: bool,
128            has_subquery_read_after: bool,
129            has_group_key: bool,
130            has_aggregate: bool,
131        }
132        impl Visitor for Finder<'_, '_> {
133            type Node = Expr;
134            type Arena = ();
135
136            fn pre_visit(&mut self, node: &Expr, _: &()) -> PolarsResult<VisitRecursion> {
137                // Siblings are still visited so that no aggregate goes unnoticed.
138                Ok(match node {
139                    Expr::Over { .. } => {
140                        self.has_window = true;
141                        VisitRecursion::Skip
142                    },
143                    Expr::Column(name) => {
144                        self.has_group_key |= self.splitter.keys.contains(name);
145                        VisitRecursion::Skip
146                    },
147                    _ if self.splitter.subquery_read_after(node).is_some() => {
148                        self.has_subquery_read_after = true;
149                        VisitRecursion::Skip
150                    },
151                    _ if is_marked_aggregate(node) || self.splitter.is_group_value(node) => {
152                        self.has_aggregate = true;
153                        VisitRecursion::Skip
154                    },
155                    _ => VisitRecursion::Continue,
156                })
157            }
158        }
159        let mut finder = Finder {
160            splitter: self,
161            has_window: false,
162            has_subquery_read_after: false,
163            has_group_key: false,
164            has_aggregate: false,
165        };
166        let _ = expr.visit(&mut finder, &());
167        finder.has_window
168            || finder.has_subquery_read_after
169            || (finder.has_group_key && finder.has_aggregate)
170    }
171
172    /// Replace every aggregate in `expr` with a reference to a hoisted aggregation output,
173    /// recorded in `self.aggregates`. A window runs on the aggregated rows; its inputs are
174    /// bound by `bind_window_input`.
175    pub(crate) fn hoist(&mut self, expr: Expr) -> PolarsResult<Expr> {
176        if let Some(name) = self.subquery_read_after(&expr) {
177            return Ok(self.read_after_aggregation(name.clone()));
178        }
179        match expr {
180            Expr::Over {
181                function,
182                partition_by,
183                order_by,
184                mapping,
185            } => Ok(Expr::Over {
186                function: Arc::new(self.bind_window_input(Arc::unwrap_or_clone(function))?),
187                partition_by: partition_by
188                    .into_iter()
189                    .map(|e| match e {
190                        Expr::Column(name) if Some(&name) == self.whole_frame_partition => {
191                            Ok(lit(1))
192                        },
193                        e => self.bind_window_input(e),
194                    })
195                    .collect::<PolarsResult<_>>()?,
196                order_by: order_by
197                    .map(|(e, options)| {
198                        let e = self.bind_window_input(Arc::unwrap_or_clone(e))?;
199                        PolarsResult::Ok((Arc::new(e), options))
200                    })
201                    .transpose()?,
202                mapping,
203            }),
204            e if is_marked_aggregate(&e) || self.is_group_value(&e) => {
205                Ok(self.aggregates.get_or_insert_hoisted(e))
206            },
207            e => e.map_children(&mut |c, _| self.hoist(c), &mut ()),
208        }
209    }

211    /// Check that `expr`, which runs in the group context, reads the input columns only in
212    /// group keys and aggregates: `x` has no single value per group in `HAVING x > 1`.
213    pub(crate) fn check_columns_per_group(&self, expr: &Expr) -> PolarsResult<()> {
214        struct Finder<'a, 'b> {
215            splitter: &'a GroupContextSplitter<'b>,
216            column: Option<PlSmallStr>,
217        }
218        impl Visitor for Finder<'_, '_> {
219            type Node = Expr;
220            type Arena = ();
221
222            fn pre_visit(&mut self, node: &Expr, _: &()) -> PolarsResult<VisitRecursion> {
223                let splitter = self.splitter;
224                Ok(match node {
225                    // The inputs of a window are checked when they are bound.
226                    Expr::Over { .. } => VisitRecursion::Skip,
227                    _ if is_marked_aggregate(node)
228                        || splitter.is_group_value(node)
229                        || splitter.key_exprs.iter().any(|(key, _)| key == node) =>
230                    {
231                        VisitRecursion::Skip
232                    },
233                    Expr::Column(name)
234                        if !splitter.keys.contains(name)
235                            && !splitter.subquery_names.contains(name)
236                            && !is_correlated_result_col(name) =>
237                    {
238                        self.column = Some(name.clone());
239                        VisitRecursion::Stop
240                    },
241                    _ => VisitRecursion::Continue,
242                })
243            }
244        }
245        let mut finder = Finder {
246            splitter: self,
247            column: None,
248        };
249        expr.visit(&mut finder, &())?;
250        if let Some(name) = finder.column {
251            polars_bail!(SQLSyntax: "'{}' should participate in the GROUP BY clause or an aggregate function", name);
252        }
253        Ok(())
254    }
255
256    /// Read the group keys in `expr`, which runs in the group context, as one value per
257    /// group, as HAVING reads them.
258    pub(crate) fn read_keys_per_group(&self, expr: Expr) -> Expr {
259        if self.key_exprs.iter().any(|(key, _)| *key == expr) {
260            return expr.first();
261        }
262        match expr {
263            e if is_marked_aggregate(&e) || self.is_group_value(&e) => e,
264            e => e
265                .map_children(&mut |c, _| Ok(self.read_keys_per_group(c)), &mut ())
266                .unwrap(),
267        }
268    }
269
270    /// Bind an input of a window function to the aggregated rows: an aggregate reads its
271    /// hoisted output and a group key reads its key column.
272    fn bind_window_input(&mut self, expr: Expr) -> PolarsResult<Expr> {
273        if let Some((_, name)) = self.key_exprs.iter().find(|(key, _)| *key == expr) {
274            return Ok(col(name.clone()));
275        }
276        if let Some(name) = self.subquery_read_after(&expr) {
277            return Ok(self.read_after_aggregation(name.clone()));
278        }
279        match expr {
280            e if is_marked_aggregate(&e) || self.is_group_value(&e) => {
281                Ok(self.aggregates.get_or_insert_hoisted(e))
282            },
283            Expr::Column(name) if self.subquery_names.contains(&name) => {
284                if self.subqueries_after_aggregation {
285                    Ok(self.read_after_aggregation(name))
286                } else {
287                    Ok(self.aggregates.get_or_insert_hoisted(col(name).first()))
288                }
289            },
290            Expr::Column(name) => {
291                polars_ensure!(
292                    self.keys.contains(&name),
293                    SQLSyntax: "'{}' should participate in the GROUP BY clause or an aggregate function", name
294                );
295                Ok(Expr::Column(name))
296            },
297            e => e.map_children(&mut |c, _| self.bind_window_input(c), &mut ()),
298        }
299    }
300}

302/// The names that QUALIFY and ORDER BY can read besides the input columns: the SELECT
303/// aliases. Each stands for an expression over the input columns, of a type that is known
304/// if it could be inferred.
305pub(crate) struct OutputNames {
306    names: PlHashMap<PlSmallStr, (Expr, Option<DataType>)>,
307}
308
309impl OutputNames {
310    /// The output names of `projections` over the input `schema`. An input
311    /// column comes before an output name of the same name. `typed_schema` is `schema` with
312    /// the columns that the projections read before the block resolves them.
313    pub(crate) fn new(projections: &[Expr], schema: &Schema, typed_schema: &Schema) -> Self {
314        let mut names = PlHashMap::new();
315        for projection in projections {
316            if let Expr::Alias(inner, name) = projection
317                && !schema.contains(name)
318                && !names.contains_key(name)
319            {
320                let dtype = projection.to_field(typed_schema).ok().map(|f| f.dtype);
321                names.insert(name.clone(), (inner.as_ref().clone(), dtype));
322            }
323        }
324        Self { names }
325    }
326
327    /// `schema` with the output names of a known type, to parse an expression that can
328    /// read them.
329    pub(crate) fn extend_schema(&self, schema: &Schema) -> Schema {
330        let mut schema = schema.clone();
331        for (name, (_, dtype)) in &self.names {
332            if let Some(dtype) = dtype {
333                schema.with_column(name.clone(), dtype.clone());
334            }
335        }
336        schema
337    }
338
339    /// Replace the output names in `expr` by their expressions.
340    pub(crate) fn resolve(&self, expr: Expr) -> Expr {
341        expr.map_expr(|e| match e {
342            Expr::Column(name) => match self.names.get(&name) {
343                Some((expr, _)) => expr.clone(),
344                None => Expr::Column(name),
345            },
346            e => e,
347        })
348    }
349}

351/// Whether a block without GROUP BY has both windows and aggregates, as in
352/// `SELECT SUM(x), COUNT(*) OVER () FROM t`.
353pub(crate) fn has_windows_over_aggregates(projections: &[Expr]) -> bool {
354    projections
355        .iter()
356        .any(|e| has_expr(e, |e| matches!(e, Expr::Over { .. })))
357        && projections.iter().any(has_marked_aggregate)
358}

360/// Whether a block without GROUP BY reads a scalar subquery outside its aggregates, as in
361/// `SELECT COUNT(*), (SELECT 1) FROM t`. Without input rows, the subquery only has a value
362/// in the aggregated row.
363pub(crate) fn has_subquery_outside_aggregates(
364    projections: &[Expr],
365    subquery_names: &PlHashSet<PlSmallStr>,
366) -> bool {
367    struct Finder<'a> {
368        subquery_names: &'a PlHashSet<PlSmallStr>,
369        found: bool,
370    }
371    impl Visitor for Finder<'_> {
372        type Node = Expr;
373        type Arena = ();
374
375        fn pre_visit(&mut self, node: &Expr, _: &()) -> PolarsResult<VisitRecursion> {
376            Ok(match node {
377                _ if is_marked_aggregate(node) => VisitRecursion::Skip,
378                Expr::Column(name) if self.subquery_names.contains(name) => {
379                    self.found = true;
380                    VisitRecursion::Stop
381                },
382                _ => VisitRecursion::Continue,
383            })
384        }
385    }
386    if subquery_names.is_empty() || !projections.iter().any(has_marked_aggregate) {
387        return false;
388    }
389    projections.iter().any(|expr| {
390        let mut finder = Finder {
391            subquery_names,
392            found: false,
393        };
394        let _ = expr.visit(&mut finder, &());
395        finder.found
396    })
397}

399/// Check that a block that aggregates without GROUP BY reads the input columns only in its
400/// aggregates: `x` has no single value in `SELECT SUM(x), x FROM t`.
401pub(crate) fn check_columns_in_aggregates(
402    projections: &[Expr],
403    subquery_names: &PlHashSet<PlSmallStr>,
404) -> PolarsResult<()> {
405    struct Finder<'a> {
406        subquery_names: &'a PlHashSet<PlSmallStr>,
407        column: Option<PlSmallStr>,
408    }
409    impl Visitor for Finder<'_> {
410        type Node = Expr;
411        type Arena = ();
412
413        fn pre_visit(&mut self, node: &Expr, _: &()) -> PolarsResult<VisitRecursion> {
414            Ok(match node {
415                _ if is_marked_aggregate(node) => VisitRecursion::Skip,
416                Expr::Column(name)
417                    if !self.subquery_names.contains(name) && !is_correlated_result_col(name) =>
418                {
419                    self.column = Some(name.clone());
420                    VisitRecursion::Stop
421                },
422                _ => VisitRecursion::Continue,
423            })
424        }
425    }
426    if !projections.iter().any(has_marked_aggregate) {
427        return Ok(());
428    }
429    for expr in projections {
430        let mut finder = Finder {
431            subquery_names,
432            column: None,
433        };
434        expr.visit(&mut finder, &())?;
435        if let Some(name) = finder.column {
436            polars_bail!(SQLSyntax: "'{}' should participate in the GROUP BY clause or an aggregate function", name);
437        }
438    }
439    Ok(())
440}

442/// Marks the call of an aggregate function: a SQL aggregate, or a user-defined function that
443/// returns one value. The parser sets it in the clauses that run in the group context of a
444/// block (the SELECT list, QUALIFY, HAVING and the aggregates of ORDER BY), and in the inputs
445/// of window functions, as on `COUNT(*)` in `SUM(COUNT(*)) OVER ()`. A window function is not
446/// marked, nor are the reductions that its lowering adds.
447///
448/// The mark is a rename that keeps the name, by a callback that only this module holds, so no
449/// alias written in the query is taken for it. `GroupContextSplitter` computes each marked
450/// aggregate per group, and a block without GROUP BY or windows removes the marks, so none
451/// reach the plan.
452static AGGREGATE_MARK: LazyLock<PlanCallback<PlSmallStr, PlSmallStr>> =
453    LazyLock::new(|| PlanCallback::new(Ok));
454
455pub(crate) fn mark_aggregate(expr: Expr) -> Expr {
456    Expr::RenameAlias {
457        function: RenameAliasFn::Map(AGGREGATE_MARK.clone()),
458        expr: Arc::new(expr),
459    }
460}
461
462pub(crate) fn is_marked_aggregate(expr: &Expr) -> bool {
463    matches!(expr, Expr::RenameAlias { function: RenameAliasFn::Map(f), .. } if *f == *AGGREGATE_MARK)
464}
465
466/// Whether `expr` holds a marked aggregate.
467pub(crate) fn has_marked_aggregate(expr: &Expr) -> bool {
468    has_expr(expr, is_marked_aggregate)
469}
470
471/// `expr` without aggregate marks.
472pub(crate) fn strip_aggregate_marks(expr: Expr) -> Expr {
473    expr.map_expr(|e| match e {
474        Expr::RenameAlias { expr, .. } if is_marked_aggregate(&e) => Arc::unwrap_or_clone(expr),
475        e => e,
476    })
477}

479/// `expr`, evaluated in the groups of a GROUP BY on a column, which have at least one row.
480/// There, the check `len() > 0` that the aggregate of a constant makes (see
481/// `SQLFunctionVisitor::rows_read`) is true, and the optimizer drops it.
482pub(crate) fn assume_groups_have_rows(expr: Expr) -> Expr {
483    expr.map_expr(|e| match e {
484        Expr::BinaryExpr {
485            ref left,
486            op: Operator::Gt,
487            ref right,
488        } if matches!(**left, Expr::Len)
489            && matches!(
490                **right,
491                Expr::Literal(LiteralValue::Dyn(DynLiteralValue::Int(0)))
492            ) =>
493        {
494            lit(true)
495        },
496        e => e,
497    })
