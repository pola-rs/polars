1//! Exact numeric literals and exact folding of arithmetic between them.
2//!
3//! An `<exact numeric literal>` (digits with an optional `.`, no exponent) is a decimal
4//! with the spelled scale, so `1.0000` is `Decimal(5, 4)`. Arithmetic between literals
5//! containing a decimal point is folded with the same kernels and result scales as
6//! decimal column arithmetic, and raises where that would. Integer-only arithmetic is
7//! left to the engine.
8
9use polars_compute::decimal::{
10    DEC128_MAX_PREC, dec128_add_scaled, dec128_mul_scaled, dec128_sub_scaled,
11};
12use polars_core::prelude::*;
13use polars_lazy::prelude::*;
14use polars_plan::prelude::{Expr, Literal};
15use sqlparser::ast::{
16    BinaryOperator as SQLBinaryOperator, Expr as SQLExpr, UnaryOperator as SQLUnaryOperator,
17    Value as SQLValue, ValueWithSpan,
18};

20/// A decimal as `(mantissa, precision, scale)`: `mantissa / 10^scale`.
21pub(crate) type ExactLiteral = (i128, usize, usize);
22
23/// The smallest precision holding `mantissa` at `scale`.
24fn precision_of(mantissa: i128, scale: usize) -> usize {
25    let digits = mantissa
26        .unsigned_abs()
27        .checked_ilog10()
28        .map_or(1, |d| d as usize + 1);
29    digits.max(scale).max(1)
30}

32/// Parse an `<exact numeric literal>` spelling without rounding; `None` if it isn't one,
33/// or it needs more than 38 digits (the caller then raises).
34pub(crate) fn parse_exact_literal(s: &str) -> Option<ExactLiteral> {
35    let (int_part, frac_part) = s.split_once('.').unwrap_or((s, ""));
36    if int_part.is_empty() && frac_part.is_empty()
37        || !int_part
38            .bytes()
39            .chain(frac_part.bytes())
40            .all(|b| b.is_ascii_digit())
41    {
42        return None;
43    }
44    let scale = frac_part.len();
45    if int_part.trim_start_matches('0').len() + scale > DEC128_MAX_PREC {
46        return None;
47    }
48    let mantissa = format!("{int_part}{frac_part}").parse().ok()?;
49    Some((mantissa, precision_of(mantissa, scale), scale))
50}
51
52pub(crate) fn decimal_lit((mantissa, precision, scale): ExactLiteral) -> Expr {
53    Scalar::new_decimal(mantissa, precision, scale).lit()
54}

56/// Whether `expr` is literal arithmetic containing a literal with a decimal point.
57fn has_dotted_literal(expr: &SQLExpr) -> bool {
58    match expr {
59        SQLExpr::Value(ValueWithSpan {
60            value: SQLValue::Number(s, _),
61            ..
62        }) => s.contains('.'),
63        SQLExpr::Nested(e) | SQLExpr::UnaryOp { expr: e, .. } => has_dotted_literal(e),
64        SQLExpr::BinaryOp { left, right, .. } => {
65            has_dotted_literal(left) || has_dotted_literal(right)
66        },
67        _ => false,
68    }
69}

71/// `(l, sl, r, sr, s_out)`: the result of `l` (scale `sl`) with `r` (scale `sr`) at `s_out`.
72type ScaledKernel = fn(i128, usize, i128, usize, usize) -> Option<i128>;
73
74/// `Ok(None)` if `expr` isn't foldable literal arithmetic.
75fn eval(expr: &SQLExpr) -> PolarsResult<Option<(i128, usize)>> {
76    Ok(match expr {
77        SQLExpr::Value(ValueWithSpan {
78            value: SQLValue::Number(s, _),
79            ..
80        }) => parse_exact_literal(s).map(|(m, _, s)| (m, s)),
81        SQLExpr::Nested(e) => eval(e)?,
82        SQLExpr::UnaryOp { op, expr } => match op {
83            SQLUnaryOperator::Plus => eval(expr)?,
84            // Negating a Decimal128 mantissa can't overflow.
85            SQLUnaryOperator::Minus => eval(expr)?.map(|(m, s)| (-m, s)),
86            _ => None,
87        },
88        SQLExpr::BinaryOp { left, op, right } => combine(left, op, right)?,
89        _ => None,
90    })
91}
92
93fn combine(
94    left: &SQLExpr,
95    op: &SQLBinaryOperator,
96    right: &SQLExpr,
97) -> PolarsResult<Option<(i128, usize)>> {
98    let (f, name): (ScaledKernel, _) = match op {
99        SQLBinaryOperator::Plus => (dec128_add_scaled, "addition"),
100        SQLBinaryOperator::Minus => (dec128_sub_scaled, "subtraction"),
101        SQLBinaryOperator::Multiply => (dec128_mul_scaled, "multiplication"),
102        _ => return Ok(None),
103    };
104    let (Some((l, sl)), Some((r, sr))) = (eval(left)?, eval(right)?) else {
105        return Ok(None);
106    };
107    let scale = match op {
108        SQLBinaryOperator::Multiply => {
109            let scale = sl + sr;
110            polars_ensure!(
111                scale <= DEC128_MAX_PREC,
112                SQLInterface: "numeric value out of range: multiplication result scale {} exceeds {}",
113                scale, DEC128_MAX_PREC
114            );
115            scale
116        },
117        _ => sl.max(sr),
118    };
119    let value = f(l, sl, r, sr, scale).ok_or_else(|| {
120        polars_err!(
121            SQLInterface: "numeric value out of range: {} result doesn't fit Decimal({}, {})",
122            name, DEC128_MAX_PREC, scale
123        )
124    })?;
125    Ok(Some((value, scale)))
126}

128/// Fold `left <op> right` when both sides are numeric literal arithmetic and at least one
129/// literal has a decimal point; `Ok(None)` leaves the expression to ordinary translation.
130pub(crate) fn try_fold_decimal_arithmetic(
131    left: &SQLExpr,
132    op: &SQLBinaryOperator,
133    right: &SQLExpr,
134) -> PolarsResult<Option<Expr>> {
135    if !(has_dotted_literal(left) || has_dotted_literal(right)) {
136        return Ok(None);
137    }
138    Ok(combine(left, op, right)?.map(|(m, s)| decimal_lit((m, precision_of(m, s), s))))
139}

141/// Evaluate `expr`, built from literals only, into one literal.
142pub(crate) fn fold_scalar(expr: Expr) -> PolarsResult<Expr> {
143    let df = DataFrame::empty().lazy().select([expr]).collect()?;
144    let value = df.columns()[0].get(0)?.into_static();
145    Ok(lit(Scalar::new(df.columns()[0].dtype().clone(), value)))
146}
