1//! Exact folding of arithmetic between numeric SQL literals.
2//!
3//! `.06 + 0.01` is computed on the literal spellings as fixed-point values (exact within
4//! `i128`) and converted once to the ordinary `Float64` literal. Integer-only arithmetic
5//! is left to the engine.
6
7use polars_compute::decimal::exact;
8use polars_plan::prelude::{Expr, lit};
9use sqlparser::ast::{
10    BinaryOperator as SQLBinaryOperator, Expr as SQLExpr, UnaryOperator as SQLUnaryOperator,
11    Value as SQLValue, ValueWithSpan,
12};
13
14/// A literal as `(mantissa, scale)`: `mantissa / 10^scale`.
15type Fixed = (i128, usize);
16
17/// Read a numeric literal spelling (digits with an optional `.`) without rounding.
18fn parse_spelling(s: &str) -> Option<Fixed> {
19    let (int_part, frac_part) = s.split_once('.').unwrap_or((s, ""));
20    if !int_part
21        .bytes()
22        .chain(frac_part.bytes())
23        .all(|b| b.is_ascii_digit())
24    {
25        return None;
26    }
27    let mantissa = format!("{int_part}{frac_part}").parse().ok()?;
28    Some((mantissa, frac_part.len()))
29}
30
31fn eval(expr: &SQLExpr) -> Option<Fixed> {
32    match expr {
33        SQLExpr::Value(ValueWithSpan {
34            value: SQLValue::Number(s, _),
35            ..
36        }) => parse_spelling(s),
37        SQLExpr::Nested(e) => eval(e),
38        SQLExpr::UnaryOp { op, expr } => match op {
39            SQLUnaryOperator::Plus => eval(expr),
40            SQLUnaryOperator::Minus => exact::neg(eval(expr)?),
41            _ => None,
42        },
43        SQLExpr::BinaryOp { left, op, right } => combine(left, op, right),
44        _ => None,
45    }
46}
47
48fn combine(left: &SQLExpr, op: &SQLBinaryOperator, right: &SQLExpr) -> Option<Fixed> {
49    let f = match op {
50        SQLBinaryOperator::Plus => exact::add,
51        SQLBinaryOperator::Minus => exact::sub,
52        SQLBinaryOperator::Multiply => exact::mul,
53        _ => return None,
54    };
55    f(eval(left)?, eval(right)?)
56}
57
58/// One correctly rounded conversion of the exact result.
59fn to_f64((mantissa, scale): Fixed) -> f64 {
60    format!("{mantissa}e-{scale}").parse().unwrap()
61}
62
63/// Fold `left <op> right` when both sides are numeric literal arithmetic with a
64/// decimal point; `None` leaves the expression to ordinary translation.
65pub(crate) fn try_fold_decimal_arithmetic(
66    left: &SQLExpr,
67    op: &SQLBinaryOperator,
68    right: &SQLExpr,
69) -> Option<Expr> {
70    let value = combine(left, op, right)?;
71    (value.1 > 0).then(|| lit(to_f64(value)))
72}
