29impl SQLContext {
30    /// `<table> PIVOT (<aggregates> FOR <column> IN (<values>))`: a row per group of the input
31    /// columns the PIVOT doesn't read, and per value and aggregate a column with the aggregate
32    /// over the rows of the group whose `<column>` is that value.
33    pub(crate) fn execute_pivot(
34        &mut self,
35        table: &TableFactor,
36        aggregates: &[ExprWithAlias],
37        value_column: &[SQLExpr],
38        value_source: &PivotValueSource,
39    ) -> PolarsResult<LazyFrame> {
40        let [column] = value_column else {
41            polars_bail!(SQLInterface: "PIVOT supports a single column after FOR")
42        };
43        let PivotValueSource::List(values) = value_source else {
44            polars_bail!(SQLInterface: "PIVOT requires a list of values: `IN (<value>, ...)`")
45        };
46        polars_ensure!(
47            !has_aggregate(self, column),
48            SQLInterface: "PIVOT does not support an aggregate after FOR, found `{}`", column
49        );
50        if let Some(value) = values.iter().find(|value| !is_constant(&value.expr)) {
51            polars_bail!(SQLInterface: "PIVOT values must be constants, found `{}`", value.expr)
52        }
53        let relations = factor_relations(std::iter::once(table), self.catalog_spellings());
54        let mut input = InputColumns {
55            relations: &relations,
56            read: PlHashSet::new(),
57            window: false,
58            subquery: false,
59            query_depth: 0,
60        };
61        for aggregate in aggregates {
62            let _ = dyn_visit::visit(&aggregate.expr, &mut input);
63        }
64        polars_ensure!(!input.subquery, SQLInterface: "PIVOT aggregates cannot contain subqueries");
65        let _ = dyn_visit::visit(column, &mut input);
66        polars_ensure!(!input.window, SQLInterface: "PIVOT does not support window functions");
67        let read = input.read;
68
69        self.execute_isolated(|ctx| {
70            let (name, mut lf) = ctx.get_table(table)?;
71            let schema = ctx.get_frame_schema(&mut lf)?;
72            let group_by: Vec<SQLExpr> = schema
73                .iter_names()
74                .filter(|name| !read.contains(name.as_str()))
75                .map(|name| SQLExpr::Identifier(Ident::with_quote('"', name.as_str())))
76                .collect();
77
78            let mut projection: Vec<SelectItem> = group_by
79                .iter()
80                .cloned()
81                .map(SelectItem::UnnamedExpr)
82                .collect();
83            let mut filters = BooleanFilters {
84                ctx: ctx.isolated(),
85                schema: &schema,
86                query_depth: 0,
87            };
88            let exprs: Vec<SQLExpr> = aggregates
89                .iter()
90                .map(|aggregate| {
91                    let mut expr = aggregate.expr.clone();
92                    let _ = dyn_visit::visit_mut(&mut expr, &mut filters);
93                    expr
94                })
95                .collect();
96            for value in values {
97                let value_name = match &value.alias {
98                    Some(alias) => alias.value.clone(),
99                    None => pivot_value_name(&value.expr),
100                };
101                let matches =
102                    SQLExpr::IsNotDistinctFrom(Box::new(column.clone()), Box::new(value.expr.clone()));
103                for (aggregate, expr) in aggregates.iter().zip(&exprs) {
104                    let mut expr = expr.clone();
105                    let mut filter = FilterAggregates {
106                        ctx,
107                        matches: &matches,
108                        found: false,
109                        query_depth: 0,
110                    };
111                    let _ = dyn_visit::visit_mut(&mut expr, &mut filter);
112                    polars_ensure!(
113                        filter.found,
114                        SQLInterface: "PIVOT expects aggregate functions, found `{}`", aggregate.expr
115                    );
116                    let name = match (&aggregate.alias, aggregates.len()) {
117                        (None, 1) => value_name.clone(),
118                        (Some(alias), _) => format!("{value_name}_{}", alias.value),
119                        (None, _) => format!("{value_name}_{}", aggregate.expr),
120                    };
121                    projection.push(SelectItem::ExprWithAlias {
122                        expr,
123                        alias: Ident::with_quote('"', name),
124                    });
125                }
126            }
127
128            ctx.planned_from_item = Some((table.clone(), (name, lf)));
129            ctx.execute_query(&aggregate_query(table, projection, group_by))
130        })
131    }

133    /// `<table> UNPIVOT [INCLUDE | EXCLUDE NULLS] (<value> FOR <name> IN (<columns>))`: per input
134    /// row and listed column, a row with the other columns, the column's name (or alias) as
135    /// `<name>` and its value as `<value>`. Rows with a NULL value are left out unless
136    /// `INCLUDE NULLS` is given.
137    #[cfg(feature = "pivot")]
138    pub(crate) fn execute_unpivot(
139        &mut self,
140        table: &TableFactor,
141        value: &SQLExpr,
142        name: &Ident,
143        columns: &[ExprWithAlias],
144        null_inclusion: Option<&NullInclusion>,
145    ) -> PolarsResult<LazyFrame> {
146        let SQLExpr::Identifier(value_name) = value else {
147            polars_bail!(SQLInterface: "UNPIVOT supports a single value column, found `{}`", value)
148        };
149        let mut lf = self.execute_isolated(|ctx| Ok(ctx.get_table(table)?.1))?;
150        let schema = self.get_frame_schema(&mut lf)?;
151        let relations = factor_relations(std::iter::once(table), self.catalog_spellings());
152
153        let mut sources: Vec<PlSmallStr> = Vec::with_capacity(columns.len());
154        let mut labels: Vec<PlSmallStr> = Vec::with_capacity(columns.len());
155        for column in columns {
156            let source = match &column.expr {
157                SQLExpr::Identifier(ident) => Some(ident),
158                SQLExpr::CompoundIdentifier(parts) => {
159                    match &parts[qualifier_len(parts, &relations)..] {
160                        [ident] => Some(ident),
161                        _ => None,
162                    }
163                },
164                expr => {
165                    polars_bail!(SQLInterface: "UNPIVOT expects column names, found `{}`", expr)
166                },
167            };
168            let source = source
169                .map(|ident| ident.value.as_str())
170                .filter(|name| schema.contains(name))
171                .ok_or_else(
172                    || polars_err!(ColumnNotFound: "UNPIVOT column '{}' was not found", column.expr),
173                )?;
174            sources.push(source.into());
175            labels.push(column.alias.as_ref().map_or(source, |a| &a.value).into());
176        }
177        // A column can be listed more than once and a label can be any name, so the listed
178        // columns are unpivoted as copies with names of their own.
179        let copies: Vec<PlSmallStr> = (0..sources.len())
180            .map(|i| format_pl_smallstr!("__POLARS_UNPIVOT_{i}"))
181            .collect();
182        let index: Vec<PlSmallStr> = schema
183            .iter_names()
184            .filter(|name| !sources.contains(name))
185            .cloned()
186            .collect();
187        let select = index.iter().map(|name| col(name.clone())).chain(
188            sources
189                .iter()
190                .zip(&copies)
191                .map(|(source, copy)| col(source.clone()).alias(copy.clone())),
192        );
193        let strings = |names: &[PlSmallStr]| {
194            let names: Vec<&str> = names.iter().map(PlSmallStr::as_str).collect();
195            lit(Series::new(PlSmallStr::EMPTY, names))
196        };
197        let variable = col(name.value.as_str()).replace(strings(&copies), strings(&labels));
198        let lf = lf
199            .select(select.collect::<Vec<_>>())
200            .unpivot(UnpivotArgsDSL {
201                on: Some(Selector::ByName {
202                    names: copies.into(),
203                    strict: true,
204                }),
205                index: Selector::ByName {
206                    names: index.into(),
207                    strict: true,
208                },
209                variable_name: Some(name.value.as_str().into()),
210                value_name: Some(value_name.value.as_str().into()),
211            })
212            .with_column(variable);
213        Ok(match null_inclusion {
214            Some(NullInclusion::IncludeNulls) => lf,
215            _ => lf.filter(col(value_name.value.as_str()).is_not_null()),
216        })
217    }
218}

220/// Adds the condition `matches` to the FILTER of each aggregate call outside of subqueries. It
221/// does so after visiting the call, so the added condition itself is not visited.
222struct FilterAggregates<'a> {
223    ctx: &'a SQLContext,
224    matches: &'a SQLExpr,
225    found: bool,
226    query_depth: usize,
227}
228
229impl VisitorMut for FilterAggregates<'_> {
230    type Break = ();
231
232    fn pre_visit_query(&mut self, _query: &mut Query) -> ControlFlow<()> {
233        self.query_depth += 1;
234        ControlFlow::Continue(())
235    }
236
237    fn post_visit_query(&mut self, _query: &mut Query) -> ControlFlow<()> {
238        self.query_depth -= 1;
239        ControlFlow::Continue(())
240    }
241
242    fn post_visit_expr(&mut self, expr: &mut SQLExpr) -> ControlFlow<()> {
243        if self.query_depth == 0
244            && is_aggregate_call(self.ctx, expr)
245            && let SQLExpr::Function(function) = expr
246        {
247            self.found = true;
248            function.filter = Some(Box::new(match function.filter.take() {
249                Some(filter) => SQLExpr::BinaryOp {
250                    left: filter,
251                    op: SQLBinaryOperator::And,
252                    right: Box::new(self.matches.clone()),
253                },
254                None => self.matches.clone(),
255            }));
256        }
257        ControlFlow::Continue(())
258    }
259}

261/// Casts the FILTER of each aggregate call outside of subqueries to boolean when it reads no
262/// input, as FILTER does on its own; the PIVOT adds a condition to it.
263struct BooleanFilters<'a> {
264    ctx: SQLContext,
265    schema: &'a Schema,
266    query_depth: usize,
267}
268
269impl VisitorMut for BooleanFilters<'_> {
270    type Break = ();
271
272    fn pre_visit_query(&mut self, _query: &mut Query) -> ControlFlow<()> {
273        self.query_depth += 1;
274        ControlFlow::Continue(())
275    }
276
277    fn post_visit_query(&mut self, _query: &mut Query) -> ControlFlow<()> {
278        self.query_depth -= 1;
279        ControlFlow::Continue(())
280    }
281
282    fn pre_visit_expr(&mut self, expr: &mut SQLExpr) -> ControlFlow<()> {
283        if self.query_depth == 0
284            && is_aggregate_call(&self.ctx, expr)
285            && let SQLExpr::Function(function) = expr
286            && let Some(filter) = &mut function.filter
287            && parse_sql_expr(filter, &mut self.ctx, Some(self.schema))
288                .and_then(|pred| self.ctx.with_typed_subqueries(&pred))
289                .is_ok_and(|pred| reads_no_input(&pred))
290        {
291            let filter = filter.as_mut();
292            *filter = SQLExpr::Cast {
293                kind: CastKind::Cast,
294                expr: Box::new(std::mem::replace(filter, SQLExpr::value(SQLValue::Null))),
295                data_type: SQLDataType::Boolean,
296                format: None,
297            };
298        }
299        ControlFlow::Continue(())
300    }
301}

303/// Collects the input columns read outside of subqueries, and whether a window function or a
304/// subquery is used.
305struct InputColumns<'a> {
306    relations: &'a [RelationName],
307    read: PlHashSet<String>,
308    window: bool,
309    subquery: bool,
310    query_depth: usize,
311}
312
313impl Visitor for InputColumns<'_> {
314    type Break = ();
315
316    fn pre_visit_query(&mut self, _query: &Query) -> ControlFlow<()> {
317        self.query_depth += 1;
318        self.subquery = true;
319        ControlFlow::Continue(())
320    }
321
322    fn post_visit_query(&mut self, _query: &Query) -> ControlFlow<()> {
323        self.query_depth -= 1;
324        ControlFlow::Continue(())
325    }
326
327    fn pre_visit_expr(&mut self, expr: &SQLExpr) -> ControlFlow<()> {
328        if self.query_depth > 0 {
329            return ControlFlow::Continue(());
330        }
331        match expr {
332            SQLExpr::Identifier(ident) => {
333                self.read.insert(ident.value.clone());
334            },
335            SQLExpr::CompoundIdentifier(parts) => {
336                let column = &parts[qualifier_len(parts, self.relations)];
337                self.read.insert(column.value.clone());
338            },
339            SQLExpr::Function(function) if function.over.is_some() => self.window = true,
340            _ => {},
341        }
342        ControlFlow::Continue(())
343    }
344}

346/// The number of leading `parts` that name one of `relations`.
347fn qualifier_len(parts: &[Ident], relations: &[RelationName]) -> usize {
348    match_qualifier(parts, relations.iter()).map_or(0, |(_, len)| len)
349}

351/// Whether `expr` is a literal, possibly negated or cast.
352fn is_constant(expr: &SQLExpr) -> bool {
353    match expr {
354        SQLExpr::Value(_) | SQLExpr::TypedString(_) => true,
355        SQLExpr::Interval(interval) => is_constant(&interval.value),
356        SQLExpr::UnaryOp { expr, .. } | SQLExpr::Nested(expr) | SQLExpr::Cast { expr, .. } => {
357            is_constant(expr)
358        },
359        _ => false,
360    }
361}

363/// The column name a PIVOT value gives: a string as is, else the value as written.
364fn pivot_value_name(expr: &SQLExpr) -> String {
365    match strip_nested(expr) {
366        SQLExpr::Value(value) => value
367            .value
368            .clone()
369            .into_string()
370            .unwrap_or_else(|| value.to_string()),
371        SQLExpr::TypedString(typed) => typed
372            .value
373            .value
374            .clone()
375            .into_string()
376            .unwrap_or_else(|| expr.to_string()),
377        expr => expr.to_string(),
378    }
379}

381/// `SELECT <projection> FROM <table> GROUP BY <group_by>`.
382fn aggregate_query(
383    table: &TableFactor,
384    projection: Vec<SelectItem>,
385    group_by: Vec<SQLExpr>,
386) -> Query {
387    let Statement::Query(mut query) = Parser::parse_sql(&GenericDialect, "SELECT * FROM t")
388        .unwrap()
389        .remove(0)
390    else {
391        unreachable!()
392    };
393    let SetExpr::Select(select) = query.body.as_mut() else {
394        unreachable!()
395    };
396    select.from[0].relation = table.clone();
397    select.projection = projection;
398    select.group_by = GroupByExpr::Expressions(group_by, vec![]);
399    *query
