18/// The values of the parameters of a query.
19#[derive(Clone, Debug)]
20pub enum QueryParams {
21    /// The values of `$1`, `$2`, ... (also written `?1` or `:1`), or of each `?` in order.
22    Positional(Vec<Scalar>),
23    /// The values of `$name` and `:name`, by name.
24    Named(PlHashMap<PlSmallStr, Scalar>),
25}
26
27impl Default for QueryParams {
28    fn default() -> Self {
29        Self::Positional(Vec::new())
30    }
31}
32
33/// The parameter a placeholder refers to.
34#[derive(Clone, Copy)]
35enum ParamRef<'a> {
36    /// `?`: the next parameter.
37    Next,
38    /// `$1`, `?1` or `:1`.
39    Number(usize),
40    /// `$name` or `:name`.
41    Name(&'a str),
42}
43
44fn param_ref(placeholder: &str) -> Option<ParamRef<'_>> {
45    let rest = placeholder.strip_prefix(['$', '?', ':'])?;
46    let question = placeholder.starts_with('?');
47    if rest.is_empty() {
48        return question.then_some(ParamRef::Next);
49    }
50    if rest.bytes().all(|b| b.is_ascii_digit()) {
51        return Some(ParamRef::Number(rest.parse().unwrap_or(usize::MAX)));
52    }
53    (!question).then_some(ParamRef::Name(rest))
54}

56fn placeholder_ref(expr: &SQLExpr) -> Option<&str> {
57    match expr {
58        SQLExpr::Value(ValueWithSpan {
59            value: SQLValue::Placeholder(p),
60            ..
61        }) if param_ref(p).is_some() => Some(p),
62        _ => None,
63    }
64}

66/// The first query parameter in `node`, if any.
67pub(crate) fn first_param(node: &(impl Visit + ?Sized)) -> Option<String> {
68    match dyn_visit::visit_expressions(node, |expr| match placeholder_ref(expr) {
69        Some(p) => ControlFlow::Break(p.to_string()),
70        None => ControlFlow::Continue(()),
71    }) {
72        ControlFlow::Break(p) => Some(p),
73        ControlFlow::Continue(()) => None,
74    }
75}

77/// `literal` with the position of the placeholder it replaces.
78fn at_placeholder(mut literal: SQLExpr, span: Span) -> SQLExpr {
79    let value = match &mut literal {
80        SQLExpr::UnaryOp { expr, .. } => expr.as_mut(),
81        literal => literal,
82    };
83    if let SQLExpr::Value(value) = value {
84        value.span = span;
85    }
86    literal
87}

89/// The SQL literal of `value`, if it has one: as the parser reads it, so a negative number
90/// is a negation and a list or struct is built from literals.
91fn sql_literal(value: &AnyValue) -> Option<SQLExpr> {
92    let number = |s: String| SQLExpr::value(SQLValue::Number(s, false));
93    let negate = |expr: SQLExpr| SQLExpr::UnaryOp {
94        op: UnaryOperator::Minus,
95        expr: Box::new(expr),
96    };
97    Some(match value {
98        AnyValue::Null => SQLExpr::value(SQLValue::Null),
99        AnyValue::Boolean(b) => SQLExpr::value(SQLValue::Boolean(*b)),
100        AnyValue::String(_) | AnyValue::StringOwned(_) => {
101            SQLExpr::value(SQLValue::SingleQuotedString(value.get_str()?.to_string()))
102        },
103        // the magnitude of i64::MIN is not an Int64 literal
104        AnyValue::Int64(v) if *v >= 0 => number(v.to_string()),
105        AnyValue::Int64(v) if *v > i64::MIN => negate(number((-v).to_string())),
106        AnyValue::Float64(v) if v.is_finite() => match v.is_sign_negative() {
107            false => number(format!("{v:e}")),
108            true => negate(number(format!("{:e}", -v))),
109        },
110        AnyValue::List(s) => SQLExpr::Array(Array {
111            elem: s.iter().map(|v| sql_literal(&v)).collect::<Option<_>>()?,
112            named: false,
113        }),
114        AnyValue::StructOwned(payload) => {
115            let (values, fields) = payload.as_ref();
116            SQLExpr::Dictionary(
117                values
118                    .iter()
119                    .zip(fields)
120                    .map(|(v, field)| {
121                        Some(DictionaryField {
122                            key: Ident::with_quote('\'', field.name.as_str()),
123                            value: Box::new(sql_literal(v)?),
124                        })
125                    })
126                    .collect::<Option<_>>()?,
127            )
128        },
129        _ => return None,
130    })
131}

133/// The bare parameter in a GROUP BY item, which would be read as a column position.
134fn grouping_param(expr: &SQLExpr) -> Option<&str> {
135    match expr {
136        SQLExpr::Rollup(sets) | SQLExpr::Cube(sets) | SQLExpr::GroupingSets(sets) => {
137            sets.iter().flatten().find_map(placeholder_ref)
138        },
139        _ => placeholder_ref(expr),
140    }
141}

143/// The placeholders of a statement.
144#[derive(Default)]
145struct Placeholders {
146    /// Where each `?` is.
147    next: Vec<Location>,
148    numbered: bool,
149    named: bool,
150}
151
152impl Visitor for Placeholders {
153    type Break = PolarsError;
154
155    fn pre_visit_query(&mut self, query: &Query) -> ControlFlow<PolarsError> {
156        if let Some(order_by) = &query.order_by
157            && let OrderByKind::Expressions(items) = &order_by.kind
158            && let Some(p) = items.iter().find_map(|item| placeholder_ref(&item.expr))
159        {
160            return ControlFlow::Break(
161                polars_err!(SQLSyntax: "query parameter {} cannot be an ORDER BY item", p),
162            );
163        }
164        ControlFlow::Continue(())
165    }
166
167    fn pre_visit_select(&mut self, select: &Select) -> ControlFlow<PolarsError> {
168        if let GroupByExpr::Expressions(items, _) = &select.group_by
169            && let Some(p) = items.iter().find_map(grouping_param)
170        {
171            return ControlFlow::Break(
172                polars_err!(SQLSyntax: "query parameter {} cannot be a GROUP BY item", p),
173            );
174        }
175        ControlFlow::Continue(())
176    }
177
178    fn pre_visit_expr(&mut self, expr: &SQLExpr) -> ControlFlow<PolarsError> {
179        if let SQLExpr::Value(ValueWithSpan {
180            value: SQLValue::Placeholder(p),
181            span,
182        }) = expr
183        {
184            match param_ref(p) {
185                Some(ParamRef::Next) => self.next.push(span.start),
186                Some(ParamRef::Number(_)) => self.numbered = true,
187                Some(ParamRef::Name(_)) => self.named = true,
188                None => {},
189            }
190        }
191        ControlFlow::Continue(())
192    }
193}

195/// Bind the parameters of `stmt` to `params`. A value with an SQL literal form replaces its
196/// placeholder. The other placeholders become `$<number>` or `$<name>`, and are returned with
197/// their values. Only placeholders in expressions are parameters. Returns `None` if `stmt`
198/// has no parameters.
199pub(crate) fn bind_params(
200    stmt: &mut Statement,
201    params: &QueryParams,
202) -> PolarsResult<Option<PlHashMap<PlSmallStr, Scalar>>> {
203    let mut found = Placeholders::default();
204    if let ControlFlow::Break(err) = dyn_visit::visit(stmt, &mut found) {
205        return Err(err);
206    }
207    if found.next.is_empty() && !found.numbered && !found.named {
208        check_no_params(params)?;
209        return Ok(None);
210    }
211    polars_ensure!(
212        !matches!(stmt, Statement::CreateView(_)),
213        SQLInterface: "query parameters are not supported in CREATE VIEW"
214    );
215    polars_ensure!(
216        found.next.is_empty() || !found.numbered,
217        SQLSyntax: "cannot mix '?' with numbered query parameters"
218    );
219    found.next.sort_unstable();
220    found.next.dedup();
221    if let QueryParams::Positional(values) = params
222        && !found.next.is_empty()
223    {
224        polars_ensure!(
225            found.next.len() == values.len(),
226            SQLInterface: "the query has {} '?' parameters, but {} values were given",
227            found.next.len(), values.len()
228        );
229    }
230
231    let n_numbers = match params {
232        QueryParams::Positional(values) => values.len(),
233        QueryParams::Named(_) => 0,
234    };
235    let mut binder = Binder {
236        params,
237        next: &found.next,
238        used_numbers: vec![false; n_numbers],
239        used_names: PlHashSet::new(),
240        bound: PlHashMap::new(),
241    };
242    if let ControlFlow::Break(err) =
243        dyn_visit::visit_expressions_mut(stmt, |expr| match binder.bind(expr) {
244            Ok(()) => ControlFlow::Continue(()),
245            Err(err) => ControlFlow::Break(err),
246        })
247    {
248        return Err(err);
249    }
250    check_all_used(params, &binder.used_numbers, &binder.used_names)?;
251    Ok(Some(binder.bound))
252}
253
254/// An error if `params` has a value, for a statement that has no parameters.
255pub(crate) fn check_no_params(params: &QueryParams) -> PolarsResult<()> {
256    check_all_used(params, &[], &PlHashSet::new())
257}
258
259fn check_all_used(
260    params: &QueryParams,
261    used_numbers: &[bool],
262    used_names: &PlHashSet<&str>,
263) -> PolarsResult<()> {
264    match params {
265        QueryParams::Positional(values) => {
266            if let Some(i) =
267                (0..values.len()).find(|&i| !used_numbers.get(i).copied().unwrap_or(false))
268            {
269                polars_bail!(SQLInterface: "query parameter ${} is not used in the query", i + 1);
270            }
271        },
272        QueryParams::Named(values) => {
273            let mut unused: Vec<&str> = values
274                .keys()
275                .map(|k| k.as_str())
276                .filter(|k| !used_names.contains(k))
277                .collect();
278            unused.sort_unstable();
279            if let Some(name) = unused.first() {
280                polars_bail!(SQLInterface: "query parameter ${} is not used in the query", name);
281            }
282        },
283    }
284    Ok(())
285}

287struct Binder<'a> {
288    params: &'a QueryParams,
289    next: &'a [Location],
290    used_numbers: Vec<bool>,
291    used_names: PlHashSet<&'a str>,
292    bound: PlHashMap<PlSmallStr, Scalar>,
293}
294
295impl<'a> Binder<'a> {
296    /// Replace the placeholder `expr` by the literal of its value, or by `$<number>` or
297    /// `$<name>` if the value has no literal.
298    fn bind(&mut self, expr: &mut SQLExpr) -> PolarsResult<()> {
299        let SQLExpr::Value(ValueWithSpan {
300            value: SQLValue::Placeholder(p),
301            span,
302        }) = expr
303        else {
304            return Ok(());
305        };
306        let Some(param) = param_ref(p) else {
307            return Ok(());
308        };
309        let (name, value) = self
310            .resolve(p, param, span.start)
311            .map_err(|e| locate(e, SqlLocationKind::Exact, *span))?;
312        *expr = match sql_literal(value.value()) {
313            Some(literal) => at_placeholder(literal, *span),
314            None => {
315                let placeholder = SQLValue::Placeholder(name.to_string()).with_span(*span);
316                self.bound.insert(name, value.clone());
317                SQLExpr::Value(placeholder)
318            },
319        };
320        Ok(())
321    }
322
323    /// The name the placeholder `p` at `location` gets, and its value.
324    fn resolve(
325        &mut self,
326        p: &str,
327        param: ParamRef,
328        location: Location,
329    ) -> PolarsResult<(PlSmallStr, &'a Scalar)> {
330        match (self.params, param) {
331            (QueryParams::Positional(values), ParamRef::Next) => {
332                let i = self.next.binary_search(&location).unwrap();
333                self.used_numbers[i] = true;
334                Ok((format_pl_smallstr!("${}", i + 1), &values[i]))
335            },
336            (QueryParams::Positional(values), ParamRef::Number(n)) => {
337                polars_ensure!(n >= 1, SQLSyntax: "query parameter {} is invalid; numbering starts at 1", p);
338                let value = values.get(n - 1).ok_or_else(
339                    || polars_err!(SQLInterface: "no value given for query parameter {}", p),
340                )?;
341                self.used_numbers[n - 1] = true;
342                Ok((format_pl_smallstr!("${}", n), value))
343            },
344            (QueryParams::Named(values), ParamRef::Name(name)) => {
345                let (key, value) = values.get_key_value(name).ok_or_else(
346                    || polars_err!(SQLInterface: "no value given for query parameter {}", p),
347                )?;
348                self.used_names.insert(key.as_str());
349                Ok((format_pl_smallstr!("${}", name), value))
350            },
351            (QueryParams::Positional(_), ParamRef::Name(_)) => {
352                polars_bail!(SQLInterface: "query parameter {} is named, but positional values were given", p)
353            },
354            (QueryParams::Named(_), _) => {
355                polars_bail!(SQLInterface: "query parameter {} is positional, but named values were given", p)
356            },
357        }
358    }
