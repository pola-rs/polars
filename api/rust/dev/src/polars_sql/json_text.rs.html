1//! The JSONPath that the `->`, `->>`, `#>` and `#>>` operators read from JSON text.
2
3use std::fmt::Write;
4
5use polars_core::prelude::*;
6use polars_lazy::dsl::{Expr, lit};
7use polars_plan::plans::{DynLiteralValue, LiteralValue};
8
9/// The value that `->` or `->>` (or with `path_array`, `#>` or `#>>`) reads with `rhs` from
10/// `lhs`, JSON text; `None` if `rhs` is not a literal path. A key that starts with `$` is a
11/// JSONPath, as in DuckDB, and an integer is an array index.
12pub(crate) fn json_text_access(
13    lhs: Expr,
14    rhs: &Expr,
15    path_array: bool,
16) -> PolarsResult<Option<Expr>> {
17    let Expr::Literal(lv) = rhs else {
18        return Ok(None);
19    };
20    // A NULL path reads nothing, as in PostgreSQL; `lhs` is kept for its number of rows.
21    let null_path = || Expr::Literal(LiteralValue::untyped_null()).cast(DataType::String);
22    if lv.is_null() {
23        return Ok(Some(lhs.str().json_path_match(null_path())));
24    }
25    let mut path = String::from("$");
26    if let LiteralValue::Dyn(DynLiteralValue::Int(idx)) = lv
27        && !path_array
28    {
29        push_index(&mut path, *idx);
30    } else {
31        let Some(key) = lv.extract_str() else {
32            return Ok(None);
33        };
34        if path_array {
35            for elem in text_array_elements(key) {
36                let Some(elem) = elem else {
37                    return Ok(Some(lhs.str().json_path_match(null_path())));
38                };
39                match elem.parse::<i64>() {
40                    Ok(idx) => push_index(&mut path, idx.into()),
41                    Err(_) => push_key(&mut path, &elem)?,
42                }
43            }
44        } else if let Some(rest) = key.strip_prefix('$') {
45            push_duckdb_path(&mut path, rest, key)?;
46        } else {
47            push_key(&mut path, key)?;
48        }
49    }
50    Ok(Some(lhs.str().json_path_match(lit(path))))
51}

53/// A negative index counts from the end. It is written as a slice, which, unlike an index, reads
54/// nothing when it is out of range.
55fn push_index(path: &mut String, idx: i128) {
56    match idx {
57        -1 => path.push_str("[-1:]"),
58        i128::MIN..=-2 => write!(path, "[{idx}:{}]", idx + 1).unwrap(),
59        _ => write!(path, "[{idx}]").unwrap(),
60    }
61}

63fn push_key(path: &mut String, key: &str) -> PolarsResult<()> {
64    // The JSONPath backend reads at most one escaped quote in a key, so the key is quoted with
65    // the quote it has the fewest of.
66    let quote = if key.matches('"').count() < key.matches('\'').count() {
67        '"'
68    } else {
69        '\''
70    };
71    polars_ensure!(
72        !key.contains('\\') && key.matches(quote).count() <= 1,
73        SQLInterface: "unsupported JSON key '{}'; keys with a backslash or with both quotes more than once are not supported", key
74    );
75    let escaped = key.replace(quote, &format!("\\{quote}"));
76    write!(path, "[{quote}{escaped}{quote}]").unwrap();
77    Ok(())
78}

80/// Add the steps of a DuckDB path such as `$.a[0]."b c"[#-1]`, given without its `$`, to `path`.
81fn push_duckdb_path(path: &mut String, mut rest: &str, full: &str) -> PolarsResult<()> {
82    let unsupported = || polars_err!(SQLInterface: "unsupported JSON path '{}'; only keys and array indexes are supported", full);
83    while !rest.is_empty() {
84        if let Some(step) = rest.strip_prefix('.') {
85            let (key, next) = match step.strip_prefix('"') {
86                Some(quoted) => {
87                    // A backslash takes the next character as is.
88                    let mut key = String::new();
89                    let mut chars = quoted.char_indices();
90                    let end = loop {
91                        match chars.next() {
92                            Some((i, '"')) => break i,
93                            Some((_, '\\')) => key.extend(chars.next().map(|(_, c)| c)),
94                            Some((_, c)) => key.push(c),
95                            None => return Err(unsupported()),
96                        }
97                    };
98                    (key, &quoted[end + 1..])
99                },
100                None => {
101                    let end = step.find(['.', '[']).unwrap_or(step.len());
102                    (step[..end].to_string(), &step[end..])
103                },
104            };
105            if key.is_empty() || key == "*" {
106                return Err(unsupported());
107            }
108            push_key(path, &key)?;
109            rest = next;
110        } else if let Some(step) = rest.strip_prefix('[') {
111            let end = step.find(']').ok_or_else(unsupported)?;
112            let idx = match step[..end].strip_prefix('#') {
113                // `#-n` is the n-th element from the end.
114                Some(from_end) => from_end.parse::<i64>().ok().filter(|idx| *idx < 0),
115                None => step[..end].parse::<i64>().ok().filter(|idx| *idx >= 0),
116            };
117            push_index(path, idx.ok_or_else(unsupported)?.into());
118            rest = &step[end + 1..];
119        } else {
120            return Err(unsupported());
121        }
122    }
123    Ok(())
124}

126/// The elements of a PostgreSQL text array literal such as `{a,"b,c",NULL}`, where an unquoted
127/// NULL is `None`; the braces are optional.
128fn text_array_elements(s: &str) -> Vec<Option<String>> {
129    fn finish(mut elem: String, kept: usize, literal: bool) -> Option<String> {
130        let len = elem.trim_end().len().max(kept);
131        elem.truncate(len);
132        (literal || !elem.eq_ignore_ascii_case("NULL")).then_some(elem)
133    }
134
135    let s = s.trim();
136    let inner = s
137        .strip_prefix('{')
138        .and_then(|s| s.strip_suffix('}'))
139        .unwrap_or(s);
140    if inner.trim().is_empty() {
141        return vec![];
142    }
143    let mut out = vec![];
144    let mut elem = String::new();
145    // The length of `elem` that ends in quoted or escaped text, which is not trimmed.
146    let mut kept = 0;
147    let mut literal = false;
148    let mut in_quotes = false;
149    let mut chars = inner.chars();
150    loop {
151        match chars.next() {
152            Some('"') => {
153                in_quotes = !in_quotes;
154                literal = true;
155                kept = elem.len();
156            },
157            Some('\\') => {
158                elem.extend(chars.next());
159                literal = true;
160                kept = elem.len();
161            },
162            Some(c) if c.is_whitespace() && !in_quotes && elem.is_empty() => {},
163            Some(',') if !in_quotes => {
164                out.push(finish(std::mem::take(&mut elem), kept, literal));
165                kept = 0;
166                literal = false;
167            },
168            Some(c) => {
169                elem.push(c);
170                if in_quotes {
171                    kept = elem.len();
172                }
173            },
174            None => {
175                out.push(finish(elem, kept, literal));
176                return out;
177            },
178        }
179    }
