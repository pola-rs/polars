20/// Visitor that checks if an expression tree contains a reference to a specific table.
21pub(crate) struct FindTableIdentifier<'a> {
22    table_name: &'a str,
23    found: bool,
24}
25
26impl<'a> FindTableIdentifier<'a> {
27    fn new(table_name: &'a str) -> Self {
28        Self {
29            table_name,
30            found: false,
31        }
32    }
33}
34
35impl<'a> SQLVisitor for FindTableIdentifier<'a> {
36    type Break = ();
37
38    fn pre_visit_expr(&mut self, expr: &SQLExpr) -> ControlFlow<Self::Break> {
39        if let SQLExpr::CompoundIdentifier(idents) = expr {
40            if idents.len() >= 2 && idents[0].value.as_str() == self.table_name {
41                self.found = true; // return immediately on first match
42                return ControlFlow::Break(());
43            }
44        }
45        ControlFlow::Continue(())
46    }
47}
48
49/// Check if a SQL expression contains a reference to a specific table.
50pub(crate) fn expr_refers_to_table(expr: &SQLExpr, table_name: &str) -> bool {
51    let mut table_finder = FindTableIdentifier::new(table_name);
52    let _ = expr.visit(&mut table_finder);
53    table_finder.found
54}

60/// Visitor that collects unqualified column identifiers (`SQLExpr::Identifier`, as opposed
61/// to `SQLExpr::CompoundIdentifier`) referenced in an expression tree, checking each one
62/// against a given `Schema` as it goes.
63struct UnqualifiedColumnsInSchema<'a> {
64    schema: &'a Schema,
65    found_any: bool,
66    all_in_schema: bool,
67}
68
69impl<'a> UnqualifiedColumnsInSchema<'a> {
70    fn new(schema: &'a Schema) -> Self {
71        Self {
72            schema,
73            found_any: false,
74            all_in_schema: true,
75        }
76    }
77}
78
79impl<'a> SQLVisitor for UnqualifiedColumnsInSchema<'a> {
80    type Break = ();
81
82    fn pre_visit_expr(&mut self, expr: &SQLExpr) -> ControlFlow<Self::Break> {
83        if let SQLExpr::Identifier(ident) = expr {
84            self.found_any = true;
85            if !self.schema.contains(ident.value.as_str()) {
86                self.all_in_schema = false;
87                return ControlFlow::Break(()); // short-circuit on first non-match
88            }
89        }
90        ControlFlow::Continue(())
91    }
92}
93
94/// Check whether a raw SQL expression contains at least one *unqualified* column
95/// identifier, and that every unqualified identifier it contains exists in `schema`.
96pub(crate) fn sql_expr_cols_all_in_schema(expr: &SQLExpr, schema: &Schema) -> bool {
97    let mut visitor = UnqualifiedColumnsInSchema::new(schema);
98    let _ = expr.visit(&mut visitor);
99    visitor.found_any && visitor.all_in_schema
100}

106/// Visitor used to check a SQL expression used in a QUALIFY clause.
107/// (Confirms window functions are present and collects column refs in one pass).
108pub(crate) struct QualifyExpression {
109    has_window_functions: bool,
110    column_refs: PlHashSet<String>,
111}
112
113impl QualifyExpression {
114    fn new() -> Self {
115        Self {
116            has_window_functions: false,
117            column_refs: PlHashSet::new(),
118        }
119    }
120
121    pub(crate) fn analyze(expr: &SQLExpr) -> (bool, PlHashSet<String>) {
122        let mut analyzer = Self::new();
123        let _ = expr.visit(&mut analyzer);
124        (analyzer.has_window_functions, analyzer.column_refs)
125    }
126}
127
128impl SQLVisitor for QualifyExpression {
129    type Break = ();
130
131    fn pre_visit_expr(&mut self, expr: &SQLExpr) -> ControlFlow<Self::Break> {
132        match expr {
133            SQLExpr::Function(func) if func.over.is_some() => {
134                self.has_window_functions = true;
135            },
136            SQLExpr::Identifier(ident) => {
137                self.column_refs.insert(ident.value.clone());
138            },
139            SQLExpr::CompoundIdentifier(idents) if !idents.is_empty() => {
140                self.column_refs
141                    .insert(idents.last().unwrap().value.clone());
142            },
143            _ => {},
144        }
145        ControlFlow::Continue(())
146    }
147}

153/// Format an identifier, quoting only if necessary (or `force` is true).
154fn maybe_quote(s: &str, force: bool) -> String {
155    let needs_quoting = force
156        || s.is_empty()
157        || s.starts_with(|c: char| c.is_ascii_digit())
158        || !s.chars().all(|c| c.is_ascii_alphanumeric() || c == '_')
159        || ALL_KEYWORDS.contains(&s.to_ascii_uppercase().as_str());
160    if needs_quoting {
161        format!("\"{s}\"")
162    } else {
163        s.to_string()
164    }
165}

167/// Visitor that checks for unqualified references to columns that exist in
168/// multiple tables (columns appearing in a USING clause are excluded from
169/// the check as they are implicitly coalesced).
170struct AmbiguousColumnVisitor<'a> {
171    joined_aliases: &'a PlHashMap<String, PlHashMap<String, String>>,
172    base_table_name: &'a str,
173    using_cols: &'a PlHashSet<String>,
174}
175
176impl SQLVisitor for AmbiguousColumnVisitor<'_> {
177    type Break = PolarsError;
178
179    fn pre_visit_expr(&mut self, expr: &SQLExpr) -> ControlFlow<Self::Break> {
180        if let SQLExpr::Identifier(ident) = expr {
181            let col = &ident.value;
182            if self.using_cols.contains(col) {
183                return ControlFlow::Continue(());
184            }
185            let mut tables: Vec<_> = self
186                .joined_aliases
187                .iter()
188                .filter_map(|(t, cols)| cols.contains_key(col).then_some(t.as_str()))
189                .collect();
190
191            if !tables.is_empty() {
192                tables.push(self.base_table_name);
193                tables.sort();
194                let col_hint = maybe_quote(col, false);
195                let hints = tables
196                    .iter()
197                    .map(|t| format!("{}.{}", maybe_quote(t, false), col_hint));
198                return ControlFlow::Break(polars_err!(
199                    SQLInterface: "ambiguous reference to column {} (use one of: {})",
200                    maybe_quote(col, true), hints.collect::<Vec<_>>().join(", ")
201                ));
202            }
203        }
204        ControlFlow::Continue(())
205    }
206}
207
208/// Check a SQL expression for unqualified references to columns that
209/// exist in multiple tables (columns appearing in a USING clause are
210/// excluded from the check as they are implicitly coalesced).
211pub(crate) fn check_for_ambiguous_column_refs(
212    expr: &SQLExpr,
213    joined_aliases: &PlHashMap<String, PlHashMap<String, String>>,
214    base_table_name: &str,
215    using_cols: &PlHashSet<String>,
216) -> PolarsResult<()> {
217    match expr.visit(&mut AmbiguousColumnVisitor {
218        joined_aliases,
219        base_table_name,
220        using_cols,
221    }) {
222        ControlFlow::Break(err) => Err(err),
223        ControlFlow::Continue(()) => Ok(()),
224    }
225}

228// TableIdentifierCollector

230
231/// Visitor that collects all table identifiers referenced in a SQL query.
232#[derive(Default)]
233pub(crate) struct TableIdentifierCollector {
234    pub(crate) tables: Vec<String>,
235    pub(crate) include_schema: bool,
236}
237
238impl TableIdentifierCollector {
239    pub(crate) fn collect_from_set_expr(&mut self, set_expr: &SetExpr) {
240        // Recursively collect table identifiers from SetExpr nodes
241        match set_expr {
242            SetExpr::Table(tbl) => {
243                self.tables.extend(if self.include_schema {
244                    match (&tbl.schema_name, &tbl.table_name) {
245                        (Some(schema), Some(table)) => Some(format!("{schema}.{table}")),
246                        (None, Some(table)) => Some(table.clone()),
247                        _ => None,
248                    }
249                } else {
250                    tbl.table_name.clone()
251                });
252            },
253            SetExpr::SetOperation { left, right, .. } => {
254                self.collect_from_set_expr(left);
255                self.collect_from_set_expr(right);
256            },
257            SetExpr::Query(query) => self.collect_from_set_expr(&query.body),
258            _ => {},
259        }
260    }
261}
262
263impl SQLVisitor for TableIdentifierCollector {
264    type Break = ();
265
266    fn pre_visit_query(&mut self, query: &Query) -> ControlFlow<Self::Break> {
267        // Collect from SetExpr nodes in the query body
268        self.collect_from_set_expr(&query.body);
269        ControlFlow::Continue(())
270    }
271
272    fn pre_visit_relation(&mut self, relation: &ObjectName) -> ControlFlow<Self::Break> {
273        // Table relation (eg: appearing in FROM clause)
274        self.tables.extend(if self.include_schema {
275            let parts: Vec<_> = relation
276                .0
277                .iter()
278                .filter_map(|p| p.as_ident().map(|i| i.value.as_str()))
279                .collect();
280            (!parts.is_empty()).then(|| parts.join("."))
281        } else {
282            relation
283                .0
284                .last()
285                .and_then(|p| p.as_ident())
286                .map(|i| i.value.clone())
287        });
288        ControlFlow::Continue(())
289    }
290}

296/// Visitor that checks if a SQL expression contains explicit window functions.
297/// Uses early-exit for efficiency when only the boolean result is needed.
298struct WindowFunctionFinder;
299
300impl SQLVisitor for WindowFunctionFinder {
301    type Break = ();
302
303    fn pre_visit_expr(&mut self, expr: &SQLExpr) -> ControlFlow<()> {
304        if matches!(expr, SQLExpr::Function(f) if f.over.is_some()) {
305            ControlFlow::Break(())
306        } else {
307            ControlFlow::Continue(())
308        }
309    }
310}
311
312/// Check if a SQL expression contains explicit window functions.
313pub(crate) fn expr_has_window_functions(expr: &SQLExpr) -> bool {
314    expr.visit(&mut WindowFunctionFinder).is_break()
315}

320
321/// Visitor that checks if a SQL expression contains any column reference at all.
322struct ColumnRefFinder;
323
324impl SQLVisitor for ColumnRefFinder {
325    type Break = ();
326
327    fn pre_visit_expr(&mut self, expr: &SQLExpr) -> ControlFlow<()> {
328        let is_column_ref = match expr {
329            SQLExpr::Identifier(_)
330            | SQLExpr::CompoundIdentifier(_)
331            | SQLExpr::Wildcard(_)
332            | SQLExpr::QualifiedWildcard(..) => true,
333            // wildcard arguments, eg: COUNT(*) / COLUMNS(*)
334            SQLExpr::Function(func) => match &func.args {
335                FunctionArguments::List(args) => args.args.iter().any(|arg| {
336                    matches!(
337                        arg,
338                        FunctionArg::Unnamed(
339                            FunctionArgExpr::Wildcard | FunctionArgExpr::QualifiedWildcard(_)
340                        )
341                    )
342                }),
343                _ => false,
344            },
345            _ => false,
346        };
347        if is_column_ref {
348            ControlFlow::Break(())
349        } else {
350            ControlFlow::Continue(())
351        }
352    }
353}

355/// Check if a SQL expression references any column (as opposed to being a
356/// constant expression composed only of literals/operators/functions).
357pub(crate) fn expr_references_any_column(expr: &SQLExpr) -> bool {
358    expr.visit(&mut ColumnRefFinder).is_break()
359}

361/// Whether this expression node is itself a subquery, in any of its forms.
362pub(crate) fn is_subquery_expr(expr: &SQLExpr) -> bool {
363    matches!(
364        expr,
365        SQLExpr::Subquery(_) | SQLExpr::Exists { .. } | SQLExpr::InSubquery { .. }
366    )
367}
368
369/// Check if a SQL expression contains a subquery, in any of its forms.
370pub(crate) fn expr_contains_subquery(expr: &SQLExpr) -> bool {
371    visit_expressions(expr, |e| {
372        if is_subquery_expr(e) {
373            ControlFlow::Break(())
374        } else {
375            ControlFlow::Continue(())
376        }
377    })
378    .is_break()
379}

381/// The argument expressions of a `GROUPING()` / `GROUPING_ID()` call; `None` for
382/// any other function or an argument list that is not plain expressions.
383pub(crate) fn grouping_call_args(func: &SQLFunction) -> Option<Vec<SQLExpr>> {
384    let is_grouping = func
385        .name
386        .0
387        .last()
388        .and_then(|p| p.as_ident())
389        .is_some_and(|i| {
390            i.value.eq_ignore_ascii_case("grouping") || i.value.eq_ignore_ascii_case("grouping_id")
391        });
392    if !is_grouping {
393        return None;
394    }
395    let FunctionArguments::List(FunctionArgumentList { args, .. }) = &func.args else {
396        return None;
397    };
398    args.iter()
399        .map(|arg| match arg {
400            FunctionArg::Unnamed(FunctionArgExpr::Expr(e)) => Some(e.clone()),
401            _ => None,
402        })
403        .collect()
404}
405
406/// Collect the argument lists of the `GROUPING()` calls in a SQL expression,
407/// leaving out those inside subqueries, which belong to their own query block.
408pub(crate) fn collect_grouping_calls(expr: &SQLExpr) -> Vec<Vec<SQLExpr>> {
409    struct Collector {
410        subquery_depth: usize,
411        calls: Vec<Vec<SQLExpr>>,
412    }
413    impl SQLVisitor for Collector {
414        type Break = ();
415
416        fn pre_visit_query(&mut self, _query: &Query) -> ControlFlow<()> {
417            self.subquery_depth += 1;
418            ControlFlow::Continue(())
419        }
420
421        fn post_visit_query(&mut self, _query: &Query) -> ControlFlow<()> {
422            self.subquery_depth -= 1;
423            ControlFlow::Continue(())
424        }
425
426        fn pre_visit_expr(&mut self, expr: &SQLExpr) -> ControlFlow<()> {
427            if self.subquery_depth == 0 {
428                if let SQLExpr::Function(f) = expr {
429                    if let Some(args) = grouping_call_args(f) {
430                        self.calls.push(args);
431                    }
432                }
433            }
434            ControlFlow::Continue(())
435        }
436    }
437    let mut collector = Collector {
438        subquery_depth: 0,
439        calls: Vec::new(),
440    };
441    let _ = expr.visit(&mut collector);
442    collector.calls
443}

449/// Visitor that detects a relation whose execution inserts into the context's table map:
450/// an aliased derived table, an aliased `UNNEST`, or a table function.
451struct TableRegisteringFinder;
452
453impl SQLVisitor for TableRegisteringFinder {
454    type Break = ();
455
456    fn pre_visit_table_factor(&mut self, table_factor: &TableFactor) -> ControlFlow<Self::Break> {
457        let registers = match table_factor {
458            TableFactor::Derived { alias, .. } | TableFactor::UNNEST { alias, .. } => {
459                alias.is_some()
460            },
461            TableFactor::Table { args, .. } => args.is_some(),
462            _ => false,
463        };
464        if registers {
465            ControlFlow::Break(())
466        } else {
467            ControlFlow::Continue(())
468        }
469    }
470}
471
472/// Check whether executing `stmt` registers a relation into the context's table map.
473pub(crate) fn statement_registers_table(stmt: &Statement) -> bool {
474    stmt.visit(&mut TableRegisteringFinder).is_break()
