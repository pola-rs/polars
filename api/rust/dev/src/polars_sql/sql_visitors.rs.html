23/// Visitor that checks if an expression tree contains a reference to a specific table.
24pub(crate) struct FindTableIdentifier<'a> {
25    table_name: &'a str,
26    relations: &'a [RelationName],
27    found: bool,
28}
29
30impl<'a> FindTableIdentifier<'a> {
31    fn new(table_name: &'a str, relations: &'a [RelationName]) -> Self {
32        Self {
33            table_name,
34            relations,
35            found: false,
36        }
37    }
38}
39
40impl<'a> SQLVisitor for FindTableIdentifier<'a> {
41    type Break = ();
42
43    fn pre_visit_expr(&mut self, expr: &SQLExpr) -> ControlFlow<Self::Break> {
44        if let SQLExpr::CompoundIdentifier(idents) = expr {
45            // the longest leading run naming a relation decides (`a.b.k` names `a.b`, not `a`)
46            if match_qualifier(idents, self.relations.iter())
47                .is_some_and(|(matches, _)| matches.iter().any(|r| r.key() == self.table_name))
48            {
49                self.found = true; // return immediately on first match
50                return ControlFlow::Break(());
51            }
52        }
53        ControlFlow::Continue(())
54    }
55}
56
57/// Check if a SQL expression contains a reference to a specific table, among `relations`
58/// in scope.
59pub(crate) fn expr_refers_to_table(
60    expr: &SQLExpr,
61    table_name: &str,
62    relations: &[RelationName],
63) -> bool {
64    let mut table_finder = FindTableIdentifier::new(table_name, relations);
65    let _ = dyn_visit::visit(expr, &mut table_finder);
66    table_finder.found
67}

73/// Visitor that collects unqualified column identifiers (`SQLExpr::Identifier`, as opposed
74/// to `SQLExpr::CompoundIdentifier`) referenced in an expression tree, checking each one
75/// against a given `Schema` as it goes.
76struct UnqualifiedColumnsInSchema<'a> {
77    schema: &'a Schema,
78    found_any: bool,
79    all_in_schema: bool,
80}
81
82impl<'a> UnqualifiedColumnsInSchema<'a> {
83    fn new(schema: &'a Schema) -> Self {
84        Self {
85            schema,
86            found_any: false,
87            all_in_schema: true,
88        }
89    }
90}
91
92impl<'a> SQLVisitor for UnqualifiedColumnsInSchema<'a> {
93    type Break = ();
94
95    fn pre_visit_expr(&mut self, expr: &SQLExpr) -> ControlFlow<Self::Break> {
96        if let SQLExpr::Identifier(ident) = expr {
97            self.found_any = true;
98            if !self.schema.contains(ident.value.as_str()) {
99                self.all_in_schema = false;
100                return ControlFlow::Break(()); // short-circuit on first non-match
101            }
102        }
103        ControlFlow::Continue(())
104    }
105}

107/// Check whether a raw SQL expression contains at least one *unqualified* column
108/// identifier, and that every unqualified identifier it contains exists in `schema`.
109pub(crate) fn sql_expr_cols_all_in_schema(expr: &SQLExpr, schema: &Schema) -> bool {
110    let mut visitor = UnqualifiedColumnsInSchema::new(schema);
111    let _ = dyn_visit::visit(expr, &mut visitor);
112    visitor.found_any && visitor.all_in_schema
113}

119/// Visitor used to check a SQL expression used in a QUALIFY clause.
120/// (Confirms window functions are present and collects column refs in one pass).
121pub(crate) struct QualifyExpression {
122    has_window_functions: bool,
123    column_refs: PlHashSet<String>,
124}
125
126impl QualifyExpression {
127    fn new() -> Self {
128        Self {
129            has_window_functions: false,
130            column_refs: PlHashSet::new(),
131        }
132    }
133
134    pub(crate) fn analyze(expr: &SQLExpr) -> (bool, PlHashSet<String>) {
135        let mut analyzer = Self::new();
136        let _ = dyn_visit::visit(expr, &mut analyzer);
137        (analyzer.has_window_functions, analyzer.column_refs)
138    }
139}
140
141impl SQLVisitor for QualifyExpression {
142    type Break = ();
143
144    fn pre_visit_expr(&mut self, expr: &SQLExpr) -> ControlFlow<Self::Break> {
145        match expr {
146            SQLExpr::Function(func) if func.over.is_some() => {
147                self.has_window_functions = true;
148            },
149            SQLExpr::Identifier(ident) => {
150                self.column_refs.insert(ident.value.clone());
151            },
152            SQLExpr::CompoundIdentifier(idents) => {
153                // The first name can be a SELECT alias holding a struct.
154                self.column_refs
155                    .extend(idents.iter().map(|ident| ident.value.clone()));
156            },
157            _ => {},
158        }
159        ControlFlow::Continue(())
160    }
161}

167/// Format an identifier, quoting only if necessary (or `force` is true).
168fn maybe_quote(s: &str, force: bool) -> String {
169    let needs_quoting = force
170        || s.is_empty()
171        || s.starts_with(|c: char| c.is_ascii_digit())
172        || !s.chars().all(|c| c.is_ascii_alphanumeric() || c == '_')
173        || ALL_KEYWORDS.contains(&s.to_ascii_uppercase().as_str());
174    if needs_quoting {
175        format!("\"{s}\"")
176    } else {
177        s.to_string()
178    }
179}

181/// Visitor that checks for unqualified references to columns that exist in
182/// multiple tables (see `check_ambiguous_column`). Subqueries are skipped, as
183/// their columns resolve against their own FROM first.
184struct AmbiguousColumnVisitor<'a> {
185    joined_aliases: &'a PlHashMap<String, PlHashMap<String, String>>,
186    base_table_name: &'a str,
187    using_cols: &'a [(String, String)],
188    relations: &'a [RelationName],
189    schema: &'a Schema,
190    subquery_depth: usize,
191}
192
193impl AmbiguousColumnVisitor<'_> {
194    /// Check a struct path whose first part `name` names no relation: it is ambiguous if
195    /// the joined frame has more than one column `name`, one of them a struct.
196    fn check_struct_path(&self, name: &str) -> PolarsResult<()> {
197        let columns: PlHashSet<&str> = self
198            .joined_aliases
199            .values()
200            .filter_map(|cols| cols.get(name).map(String::as_str))
201            .chain(self.schema.contains(name).then_some(name))
202            .collect();
203        let has_struct = columns
204            .iter()
205            .any(|c| matches!(self.schema.get(c), Some(DataType::Struct(_))));
206        if columns.len() > 1 && has_struct {
207            check_ambiguous_column(
208                name,
209                self.joined_aliases,
210                self.base_table_name,
211                self.using_cols,
212                self.relations,
213            )?;
214        }
215        Ok(())
216    }
217}
218
219impl SQLVisitor for AmbiguousColumnVisitor<'_> {
220    type Break = PolarsError;
221
222    fn pre_visit_query(&mut self, _query: &Query) -> ControlFlow<Self::Break> {
223        self.subquery_depth += 1;
224        ControlFlow::Continue(())
225    }
226
227    fn post_visit_query(&mut self, _query: &Query) -> ControlFlow<Self::Break> {
228        self.subquery_depth -= 1;
229        ControlFlow::Continue(())
230    }
231
232    fn pre_visit_expr(&mut self, expr: &SQLExpr) -> ControlFlow<Self::Break> {
233        if self.subquery_depth > 0 {
234            return ControlFlow::Continue(());
235        }
236        let col = match expr {
237            SQLExpr::Identifier(ident) => Some(ident),
238            // A struct path (`s.f`) whose first part names no relation reads that column.
239            SQLExpr::CompoundIdentifier(parts)
240                if match_qualifier(parts, self.relations.iter()).is_none() =>
241            {
242                if let Err(err) = self.check_struct_path(&parts[0].value) {
243                    return ControlFlow::Break(err);
244                }
245                None
246            },
247            _ => None,
248        };
249        if let Some(ident) = col {
250            if let Err(err) = check_ambiguous_column(
251                &ident.value,
252                self.joined_aliases,
253                self.base_table_name,
254                self.using_cols,
255                self.relations,
256            ) {
257                return ControlFlow::Break(err);
258            }
259        }
260        ControlFlow::Continue(())
261    }
262}

264/// Check that an unqualified column name exists in at most one of the joined
265/// tables. `using_cols` holds the (joined table, column) pairs of USING clauses,
266/// which are coalesced with the same column on the left of that join.
267pub(crate) fn check_ambiguous_column(
268    col: &str,
269    joined_aliases: &PlHashMap<String, PlHashMap<String, String>>,
270    base_table_name: &str,
271    using_cols: &[(String, String)],
272    relations: &[RelationName],
273) -> PolarsResult<()> {
274    let mut tables: Vec<_> = joined_aliases
275        .iter()
276        .filter(|(t, cols)| {
277            cols.contains_key(col) && !using_cols.iter().any(|(ut, uc)| ut == *t && uc == col)
278        })
279        .map(|(t, _)| t.as_str())
280        .collect();
281
282    if !tables.is_empty() {
283        tables.push(base_table_name);
284        tables.sort();
285        if tables.windows(2).any(|w| w[0] == w[1]) {
286            polars_bail!(
287                SQLInterface: "ambiguous reference to column {} (a relation appears more than once in FROM; give each an alias)",
288                maybe_quote(col, true)
289            );
290        }
291        let col_hint = maybe_quote(col, false);
292        // Quote each part of a relation's name, so `"a.b"` keeps its quotes and `s.t` has none.
293        let hints = tables.iter().map(|t| {
294            let qualifier = match relations.iter().find(|r| r.key() == *t) {
295                Some(relation) => {
296                    let parts: Vec<String> = relation
297                        .parts()
298                        .iter()
299                        .map(|p| maybe_quote(p, false))
300                        .collect();
301                    parts.join(".")
302                },
303                None => maybe_quote(t, false),
304            };
305            format!("{qualifier}.{col_hint}")
306        });
307        polars_bail!(
308            SQLInterface: "ambiguous reference to column {} (use one of: {})",
309            maybe_quote(col, true), hints.collect::<Vec<_>>().join(", ")
310        );
311    }
312    Ok(())
313}

315/// Check a SQL expression for unqualified references to columns that
316/// exist in multiple tables (see `check_ambiguous_column`).
317pub(crate) fn check_for_ambiguous_column_refs(
318    expr: &SQLExpr,
319    joined_aliases: &PlHashMap<String, PlHashMap<String, String>>,
320    base_table_name: &str,
321    using_cols: &[(String, String)],
322    relations: &[RelationName],
323    schema: &Schema,
324) -> PolarsResult<()> {
325    match dyn_visit::visit(
326        expr,
327        &mut AmbiguousColumnVisitor {
328            joined_aliases,
329            base_table_name,
330            using_cols,
331            relations,
332            schema,
333            subquery_depth: 0,
334        },
335    ) {
336        ControlFlow::Break(err) => Err(err),
337        ControlFlow::Continue(()) => Ok(()),
338    }
339}

345/// Visitor that collects all table identifiers referenced in a SQL query.
346#[derive(Default)]
347pub(crate) struct TableIdentifierCollector {
348    pub(crate) tables: Vec<String>,
349    pub(crate) include_schema: bool,
350}
351
352impl TableIdentifierCollector {
353    pub(crate) fn collect_from_set_expr(&mut self, set_expr: &SetExpr) {
354        // Recursively collect table identifiers from SetExpr nodes
355        match set_expr {
356            SetExpr::Table(tbl) => {
357                self.tables.extend(if self.include_schema {
358                    match (&tbl.schema_name, &tbl.table_name) {
359                        (Some(schema), Some(table)) => Some(format!("{schema}.{table}")),
360                        (None, Some(table)) => Some(table.clone()),
361                        _ => None,
362                    }
363                } else {
364                    tbl.table_name.clone()
365                });
366            },
367            SetExpr::SetOperation { left, right, .. } => {
368                self.collect_from_set_expr(left);
369                self.collect_from_set_expr(right);
370            },
371            SetExpr::Query(query) => self.collect_from_set_expr(&query.body),
372            _ => {},
373        }
374    }
375}
376
377impl SQLVisitor for TableIdentifierCollector {
378    type Break = ();
379
380    fn pre_visit_query(&mut self, query: &Query) -> ControlFlow<Self::Break> {
381        // Collect from SetExpr nodes in the query body
382        self.collect_from_set_expr(&query.body);
383        ControlFlow::Continue(())
384    }
385
386    fn pre_visit_relation(&mut self, relation: &ObjectName) -> ControlFlow<Self::Break> {
387        // Table relation (eg: appearing in FROM clause)
388        self.tables.extend(if self.include_schema {
389            let parts: Vec<_> = relation
390                .0
391                .iter()
392                .filter_map(|p| p.as_ident().map(|i| i.value.as_str()))
393                .collect();
394            (!parts.is_empty()).then(|| parts.join("."))
395        } else {
396            relation
397                .0
398                .last()
399                .and_then(|p| p.as_ident())
400                .map(|i| i.value.clone())
401        });
402        ControlFlow::Continue(())
403    }
404}

410/// Visitor that collects which of the CTEs `ctes` a query reads, as `FROM`/`TABLE` resolve
411/// names: the whole name, ignoring case, unless a nested `WITH` defines the same name.
412struct CteReferenceCollector<'a> {
413    ctes: &'a [String],
414    // The CTE names that each nested `WITH` being visited makes visible.
415    scopes: Vec<Vec<String>>,
416    with_stack: Vec<Option<With>>,
417    found: PlHashSet<usize>,
418}
419
420impl CteReferenceCollector<'_> {
421    fn reference(&mut self, name: &str) {
422        let shadowed = self
423            .scopes
424            .iter()
425            .flatten()
426            .any(|n| n.eq_ignore_ascii_case(name));
427        if !shadowed {
428            if let Some(idx) = self.ctes.iter().position(|c| c.eq_ignore_ascii_case(name)) {
429                self.found.insert(idx);
430            }
431        }
432    }
433
434    fn reference_set_expr_tables(&mut self, body: &SetExpr) {
435        match body {
436            SetExpr::SetOperation { left, right, .. } => {
437                self.reference_set_expr_tables(left);
438                self.reference_set_expr_tables(right);
439            },
440            SetExpr::Table(tbl) => {
441                if let Some(table_name) = &tbl.table_name {
442                    match &tbl.schema_name {
443                        Some(schema_name) => self.reference(&format!("{schema_name}.{table_name}")),
444                        None => self.reference(table_name),
445                    }
446                }
447            },
448            _ => {},
449        }
450    }
451}
452
453impl VisitorMut for CteReferenceCollector<'_> {
454    type Break = ();
455
456    fn pre_visit_query(&mut self, query: &mut Query) -> ControlFlow<()> {
457        // Visit the CTE bodies here, each with the names it can see, and hide the `WITH` from
458        // the generic traversal until `post_visit_query`.
459        let mut with = query.with.take();
460        let mut names = Vec::new();
461        if let Some(with) = with.as_mut() {
462            let all: Vec<String> = with
463                .cte_tables
464                .iter()
465                .map(|cte| cte.alias.name.value.clone())
466                .collect();
467            for (idx, cte) in with.cte_tables.iter_mut().enumerate() {
468                let visible = if with.recursive {
469                    all.clone()
470                } else {
471                    all[..idx].to_vec()
472                };
473                self.scopes.push(visible);
474                let _ = dyn_visit::visit_mut(&mut *cte.query, self);
475                self.scopes.pop();
476            }
477            names = all;
478        }
479        self.scopes.push(names);
480        self.with_stack.push(with);
481        self.reference_set_expr_tables(&query.body);
482
483        // The CTEs of a parenthesized query stay visible to the clauses wrapping it.
484        let mut inner_names = Vec::new();
485        let mut body = &*query.body;
486        while let SetExpr::Query(inner) = body {
487            if let Some(inner_with) = &inner.with {
488                inner_names.extend(
489                    inner_with
490                        .cte_tables
491                        .iter()
492                        .map(|cte| cte.alias.name.value.clone()),
493                );
494            }
495            body = &inner.body;
496        }
497        if !inner_names.is_empty() {
498            self.scopes.push(inner_names);
499            let _ = dyn_visit::visit_mut(&mut query.order_by, self);
500            let _ = dyn_visit::visit_mut(&mut query.limit_clause, self);
501            let _ = dyn_visit::visit_mut(&mut query.fetch, self);
502            self.scopes.pop();
503            // Already visited; the visitor works on a copy.
504            query.order_by = None;
505            query.limit_clause = None;
506            query.fetch = None;
507        }
508        ControlFlow::Continue(())
509    }
510
511    fn post_visit_query(&mut self, query: &mut Query) -> ControlFlow<()> {
512        self.scopes.pop();
513        query.with = self.with_stack.pop().flatten();
514        ControlFlow::Continue(())
515    }
516
517    fn pre_visit_table_factor(&mut self, factor: &mut TableFactor) -> ControlFlow<()> {
518        // A name with arguments is a table function.
519        if let TableFactor::Table {
520            name, args: None, ..
521        } = factor
522        {
523            self.reference(&object_name_to_string(name));
524        }
525        ControlFlow::Continue(())
526    }
527}

529/// The indices of the CTEs in `ctes` that `query` reads.
530pub(crate) fn referenced_ctes(query: &Query, ctes: &[String]) -> PlHashSet<usize> {
531    let mut collector = CteReferenceCollector {
532        ctes,
533        scopes: Vec::new(),
534        with_stack: Vec::new(),
535        found: PlHashSet::new(),
536    };
537    // The visitor takes each `WITH` out while visiting, so it works on a copy.
538    let _ = dyn_visit::visit_mut(&mut query.clone(), &mut collector);
539    collector.found
540}

571/// The aliases an expression reads outside its subqueries.
572#[derive(Default)]
573pub(crate) struct AliasReads {
574    /// The first alias read in a window function, including its window.
575    pub(crate) in_window: Option<String>,
576    pub(crate) any: bool,
577    pub(crate) has_window_functions: bool,
578}
579
580struct AliasReadsVisitor<'a> {
581    /// Whether a name, given by its parts, reads an alias.
582    is_alias: &'a dyn Fn(&[Ident]) -> bool,
583    resolve_window: &'a dyn Fn(&WindowType) -> Option<WindowSpec>,
584    query_depth: usize,
585    window_depth: usize,
586    reads: AliasReads,
587}
588
589impl SQLVisitor for AliasReadsVisitor<'_> {
590    type Break = ();
591
592    fn pre_visit_query(&mut self, _query: &Query) -> ControlFlow<()> {
593        self.query_depth += 1;
594        ControlFlow::Continue(())
595    }
596
597    fn post_visit_query(&mut self, _query: &Query) -> ControlFlow<()> {
598        self.query_depth -= 1;
599        ControlFlow::Continue(())
600    }
601
602    fn pre_visit_expr(&mut self, expr: &SQLExpr) -> ControlFlow<()> {
603        if self.query_depth > 0 {
604            return ControlFlow::Continue(());
605        }
606        let idents = match expr {
607            SQLExpr::Identifier(ident) => std::slice::from_ref(ident),
608            SQLExpr::CompoundIdentifier(idents) if !idents.is_empty() => idents.as_slice(),
609            SQLExpr::Function(SQLFunction {
610                over: Some(over), ..
611            }) => {
612                self.reads.has_window_functions = true;
613                self.window_depth += 1;
614                // The window can come from the WINDOW clause.
615                if let Some(spec) = (self.resolve_window)(over) {
616                    let _ = dyn_visit::visit(&spec, self);
617                }
618                return ControlFlow::Continue(());
619            },
620            _ => return ControlFlow::Continue(()),
621        };
622        if (self.is_alias)(idents) {
623            self.reads.any = true;
624            if self.window_depth > 0 && self.reads.in_window.is_none() {
625                self.reads.in_window = Some(idents[0].value.clone());
626            }
627        }
628        ControlFlow::Continue(())
629    }
630
631    fn post_visit_expr(&mut self, expr: &SQLExpr) -> ControlFlow<()> {
632        if self.query_depth == 0 && matches!(expr, SQLExpr::Function(f) if f.over.is_some()) {
633            self.window_depth -= 1;
634        }
635        ControlFlow::Continue(())
636    }
637}

639/// The aliases `expr` reads, where `is_alias` tells whether a name reads an alias and
640/// `resolve_window` gives the window of an `OVER` clause.
641pub(crate) fn alias_reads(
642    expr: &SQLExpr,
643    is_alias: &dyn Fn(&[Ident]) -> bool,
644    resolve_window: &dyn Fn(&WindowType) -> Option<WindowSpec>,
645) -> AliasReads {
646    let mut visitor = AliasReadsVisitor {
647        is_alias,
648        resolve_window,
649        query_depth: 0,
650        window_depth: 0,
651        reads: AliasReads::default(),
652    };
653    let _ = dyn_visit::visit(expr, &mut visitor);
654    visitor.reads
655}

657/// Check if a SQL expression selects columns by the schema it is parsed against: a
658/// wildcard, as an expression or a function argument, or a `COLUMNS(...)` call.
659pub(crate) fn has_column_selector(expr: &SQLExpr) -> bool {
660    visit_expressions(expr, |e| {
661        if is_column_selector(e) {
662            ControlFlow::Break(())
663        } else {
664            ControlFlow::Continue(())
665        }
666    })
667    .is_break()
668}
669
670/// Whether `expr` itself, not counting its children, is a column selector (see
671/// [`has_column_selector`]).
672pub(crate) fn is_column_selector(expr: &SQLExpr) -> bool {
673    match expr {
674        SQLExpr::Wildcard(_) | SQLExpr::QualifiedWildcard(..) => true,
675        SQLExpr::Function(func) => {
676            let is_columns = func.name.0[0]
677                .as_ident()
678                .is_some_and(|ident| ident.value.eq_ignore_ascii_case("columns"));
679            let has_wildcard_arg = match &func.args {
680                FunctionArguments::List(args) => args.args.iter().any(|arg| {
681                    matches!(
682                        arg,
683                        FunctionArg::Unnamed(
684                            FunctionArgExpr::Wildcard | FunctionArgExpr::QualifiedWildcard(_)
685                        )
686                    )
687                }),
688                _ => false,
689            };
690            is_columns || has_wildcard_arg
691        },
692        _ => false,
693    }
694}

700/// The identifiers that do not name a column or parameter: the field names in field accesses
701/// (`(expr).name`) and the form of `NORMALIZE(s, NFKC)`.
702#[derive(Default)]
703pub(crate) struct FieldNames(PlHashSet<*const SQLExpr>);
704
705impl FieldNames {
706    /// Records the names in `expr`, a parent of the expressions `take` is called on.
707    pub(crate) fn record(&mut self, expr: &SQLExpr) {
708        match expr {
709            SQLExpr::CompoundFieldAccess { access_chain, .. } => {
710                for access in access_chain {
711                    if let AccessExpr::Dot(
712                        name @ (SQLExpr::Identifier(_) | SQLExpr::CompoundIdentifier(_)),
713                    ) = access
714                    {
715                        self.0.insert(name);
716                    }
717                }
718            },
719            SQLExpr::Function(SQLFunction {
720                name,
721                args: FunctionArguments::List(list),
722                ..
723            }) if name.0.len() == 1
724                && name.0[0]
725                    .as_ident()
726                    .is_some_and(|ident| ident.value.eq_ignore_ascii_case("normalize")) =>
727            {
728                if let [
729                    _,
730                    FunctionArg::Unnamed(FunctionArgExpr::Expr(form @ SQLExpr::Identifier(_))),
731                ] = list.args.as_slice()
732                {
733                    self.0.insert(form);
734                }
735            },
736            _ => {},
737        }
738    }
739
740    /// Whether `expr` is a recorded field name.
741    pub(crate) fn contains(&self, expr: &SQLExpr) -> bool {
742        !self.0.is_empty() && self.0.contains(&(expr as *const SQLExpr))
743    }
744
745    /// Whether `expr` is a recorded field name, which is then no longer recorded.
746    pub(crate) fn take(&mut self, expr: &SQLExpr) -> bool {
747        !self.0.is_empty() && self.0.remove(&(expr as *const SQLExpr))
748    }
749}

751/// Visitor that checks if a SQL expression contains any column reference at all.
752#[derive(Default)]
753struct ColumnRefFinder;
754
755impl SQLVisitor for ColumnRefFinder {
756    type Break = ();
757
758    fn pre_visit_expr(&mut self, expr: &SQLExpr) -> ControlFlow<()> {
759        let is_column_ref = match expr {
760            SQLExpr::Identifier(_)
761            | SQLExpr::CompoundIdentifier(_)
762            | SQLExpr::Wildcard(_)
763            | SQLExpr::QualifiedWildcard(..) => true,
764            // a column selector, eg: COLUMNS('^x$')
765            SQLExpr::Function(func)
766                if func.name.0.len() == 1
767                    && func.name.0[0]
768                        .as_ident()
769                        .is_some_and(|ident| ident.value.eq_ignore_ascii_case("columns")) =>
770            {
771                true
772            },
773            // wildcard arguments, eg: COUNT(*) / COLUMNS(*)
774            SQLExpr::Function(func) => match &func.args {
775                FunctionArguments::List(args) => args.args.iter().any(|arg| {
776                    matches!(
777                        arg,
778                        FunctionArg::Unnamed(
779                            FunctionArgExpr::Wildcard | FunctionArgExpr::QualifiedWildcard(_)
780                        )
781                    )
782                }),
783                _ => false,
784            },
785            _ => false,
786        };
787        if is_column_ref {
788            ControlFlow::Break(())
789        } else {
790            ControlFlow::Continue(())
791        }
792    }
793}

795/// Check if a SQL expression references any column (as opposed to being a
796/// constant expression composed only of literals/operators/functions).
797pub(crate) fn expr_references_any_column(expr: &SQLExpr) -> bool {
798    dyn_visit::visit(expr, &mut ColumnRefFinder).is_break()
799}

801/// Whether this expression node is itself a subquery, in any of its forms.
802pub(crate) fn is_subquery_expr(expr: &SQLExpr) -> bool {
803    matches!(
804        expr,
805        SQLExpr::Subquery(_) | SQLExpr::Exists { .. } | SQLExpr::InSubquery { .. }
806    )
807}

809/// Check if a SQL expression contains a subquery, in any of its forms.
810pub(crate) fn expr_contains_subquery(expr: &SQLExpr) -> bool {
811    expr_contains(expr, is_subquery_expr)
812}

814/// Check if a SQL expression contains a scalar subquery (`(SELECT ...)` used as
815/// a value), as opposed to a predicate subquery (`EXISTS`, `IN`).
816pub(crate) fn expr_contains_scalar_subquery(expr: &SQLExpr) -> bool {

818}

820/// Finds a `RANDOM()` call, outside the subqueries unless `in_subqueries`.
821struct RandomFinder {
822    in_subqueries: bool,
823    query_depth: usize,
824}
825
826impl SQLVisitor for RandomFinder {
827    type Break = ();
828
829    fn pre_visit_query(&mut self, _query: &Query) -> ControlFlow<()> {
830        self.query_depth += 1;
831        ControlFlow::Continue(())
832    }
833
834    fn post_visit_query(&mut self, _query: &Query) -> ControlFlow<()> {
835        self.query_depth -= 1;
836        ControlFlow::Continue(())
837    }
838
839    fn pre_visit_expr(&mut self, expr: &SQLExpr) -> ControlFlow<()> {
840        let is_random = matches!(expr, SQLExpr::Function(func) if func
841            .name
842            .0
843            .last()
844            .and_then(|p| p.as_ident())
845            .is_some_and(|i| i.value.eq_ignore_ascii_case("random")));
846        if (self.in_subqueries || self.query_depth == 0) && is_random {
847            ControlFlow::Break(())
848        } else {
849            ControlFlow::Continue(())
850        }
851    }
852}
853
854/// Check if `node` calls `RANDOM()`, which yields a new value on every call. Calls in the
855/// subqueries of `node` count only with `in_subqueries`.
856pub(crate) fn contains_random(node: &(impl Visit + ?Sized), in_subqueries: bool) -> bool {
857    let mut finder = RandomFinder {
858        in_subqueries,
859        query_depth: 0,
860    };
861    dyn_visit::visit(node, &mut finder).is_break()
862}
863
864fn expr_contains(expr: &SQLExpr, is_match: impl Fn(&SQLExpr) -> bool) -> bool {
865    visit_expressions(expr, |e| {
866        if is_match(e) {
867            ControlFlow::Break(())
868        } else {
869            ControlFlow::Continue(())
870        }
871    })
872    .is_break()
873}

875/// The argument expressions of a `GROUPING()` / `GROUPING_ID()` call; `None` for
876/// any other function or an argument list that is not plain expressions.
877pub(crate) fn grouping_call_args(func: &SQLFunction) -> Option<Vec<SQLExpr>> {
878    let is_grouping = func
879        .name
880        .0
881        .last()
882        .and_then(|p| p.as_ident())
883        .is_some_and(|i| {
884            i.value.eq_ignore_ascii_case("grouping") || i.value.eq_ignore_ascii_case("grouping_id")
885        });
886    if !is_grouping {
887        return None;
888    }
889    let FunctionArguments::List(FunctionArgumentList { args, .. }) = &func.args else {
890        return None;
891    };
892    args.iter()
893        .map(|arg| match arg {
894            FunctionArg::Unnamed(FunctionArgExpr::Expr(e)) => Some(e.clone()),
895            _ => None,
896        })
897        .collect()
898}

900/// Collect the argument lists of the `GROUPING()` calls in a SQL expression,
901/// leaving out those inside subqueries, which belong to their own query block.
902pub(crate) fn collect_grouping_calls(expr: &SQLExpr) -> Vec<Vec<SQLExpr>> {
903    struct Collector {
904        subquery_depth: usize,
905        calls: Vec<Vec<SQLExpr>>,
906    }
907    impl SQLVisitor for Collector {
908        type Break = ();
909
910        fn pre_visit_query(&mut self, _query: &Query) -> ControlFlow<()> {
911            self.subquery_depth += 1;
912            ControlFlow::Continue(())
913        }
914
915        fn post_visit_query(&mut self, _query: &Query) -> ControlFlow<()> {
916            self.subquery_depth -= 1;
917            ControlFlow::Continue(())
918        }
919
920        fn pre_visit_expr(&mut self, expr: &SQLExpr) -> ControlFlow<()> {
921            if self.subquery_depth == 0 {
922                if let SQLExpr::Function(f) = expr {
923                    if let Some(args) = grouping_call_args(f) {
924                        self.calls.push(args);
925                    }
926                }
927            }
928            ControlFlow::Continue(())
929        }
930    }
931    let mut collector = Collector {
932        subquery_depth: 0,
933        calls: Vec::new(),
934    };
935    let _ = dyn_visit::visit(expr, &mut collector);
936    collector.calls
937}
