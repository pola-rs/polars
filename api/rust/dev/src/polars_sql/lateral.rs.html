1//! A derived table in a join that reads columns of the relations joined before it (a LATERAL
2//! join, written or not) is planned once, over the distinct values of those columns:
3//!
4//! ```sql
5//! l JOIN (SELECT p FROM r WHERE r.k = l.k) d ON c
6//! -- l' JOIN (SELECT p, dom.id AS rid FROM dom CROSS JOIN r WHERE r.k = dom.v) d
7//! --   ON c AND l'.lid = d.rid
8//! ```
9//!
10//! `dom` holds each distinct value `v` of `l.k` with an `id`, and `l'` is `l` with the `id` of
11//! its value as `lid`. A GROUP BY of the derived table also groups by `dom.id`. An aggregate
12//! without GROUP BY gives a row for every outer row, also one without inner rows, so its
13//! aggregates are computed per `dom.id` and joined onto `dom` first:
14//!
15//! ```sql
16//! l JOIN (SELECT MAX(p) + 1 AS m FROM r WHERE r.k = l.k) d ON c
17//! -- dom' = dom LEFT JOIN (SELECT MAX(p) AS a0, dom.id FROM dom CROSS JOIN r
18//! --   WHERE r.k = dom.v GROUP BY dom.id) USING (id)
19//! -- l' JOIN (SELECT dom'.a0 + 1 AS m, dom'.id AS rid FROM dom') d ON c AND l'.lid = d.rid
20//! ```
21//!
22//! A count over no rows is 0, so a NULL count after that join becomes 0.

45/// The inputs of a LATERAL join: the left input with a key column, the derived table with a
46/// key column, and the join on those keys. `keys` names the key columns, which the join drops.
47pub(crate) struct LateralJoin {
48    pub(crate) left: LazyFrame,
49    pub(crate) name: String,

51    pub(crate) join_operator: JoinOperator,
52    pub(crate) keys: [PlSmallStr; 2],
53}

55/// The columns of the relations of the derived table, whose names bind to them first.
56struct Inner<'a> {
57    names: &'a [RelationName],
58    schema: &'a Schema,
59    aliases: &'a [Ident],
60}
61
62impl Inner<'_> {
63    fn binds(&self, expr: &SQLExpr) -> bool {
64        match expr {
65            SQLExpr::Identifier(ident) => {
66                self.schema.contains(&ident.value)
67                    || self.aliases.iter().any(|a| a.value == ident.value)
68            },
69            SQLExpr::CompoundIdentifier(parts) => {
70                match_qualifier(parts, self.names.iter()).is_some()
71                    || self.schema.contains(&parts[0].value)
72            },
73            _ => false,
74        }
75    }
76}

78/// The columns the derived table reads outside its own relations, or `Break` if it reads the
79/// query in a way the rewrite can't keep: a window or ranking function, a column selector, or
80/// an aggregate of outer columns only, which aggregates the outer query.
81struct OuterColumns<'a, 'b> {
82    ctx: &'a SQLContext,
83    inner: &'a Inner<'b>,
84    query_depth: usize,
85    found: Vec<SQLExpr>,
86}
87
88impl Visitor for OuterColumns<'_, '_> {
89    type Break = ();
90
91    fn pre_visit_query(&mut self, _query: &Query) -> ControlFlow<()> {
92        self.query_depth += 1;
93        ControlFlow::Continue(())
94    }
95
96    fn post_visit_query(&mut self, _query: &Query) -> ControlFlow<()> {
97        self.query_depth -= 1;
98        ControlFlow::Continue(())
99    }
100
101    fn pre_visit_expr(&mut self, expr: &SQLExpr) -> ControlFlow<()> {
102        if self.query_depth > 1 {
103            return ControlFlow::Continue(());
104        }
105        match expr {
106            SQLExpr::Function(function)
107                if function.over.is_some() || calls_query_function(self.ctx, expr) =>
108            {
109                return ControlFlow::Break(());
110            },
111            // The columns a nested query reads may be its own.
112            _ if is_aggregate_call(self.ctx, expr) && expr_contains_subquery(expr) => {
113                return ControlFlow::Break(());
114            },
115            _ if is_aggregate_call(self.ctx, expr) => {
116                let (mut reads_inner, mut reads_outer) = (false, false);
117                let _ = dyn_visit::visit_expressions(expr, |e| {
118                    if matches!(e, SQLExpr::Identifier(_) | SQLExpr::CompoundIdentifier(_)) {
119                        if self.inner.binds(e) {
120                            reads_inner = true;
121                        } else {
122                            reads_outer = true;
123                        }
124                    }
125                    ControlFlow::<()>::Continue(())
126                });
127                if reads_outer && !reads_inner {
128                    return ControlFlow::Break(());
129                }
130            },
131            SQLExpr::Identifier(_) | SQLExpr::CompoundIdentifier(_)
132                if !self.inner.binds(expr) && !self.found.contains(expr) =>
133            {
134                self.found.push(expr.clone());
135            },
136            _ if is_column_selector(expr) && !is_count_star(expr) => return ControlFlow::Break(()),
137            _ => {},
138        }
139        ControlFlow::Continue(())
140    }
141}

143/// Replaces the outer columns of the derived table by the columns of the domain table.
144struct ReadDomain<'a> {
145    columns: &'a [(SQLExpr, SQLExpr)],
146    query_depth: usize,
147}
148
149impl VisitorMut for ReadDomain<'_> {
150    type Break = ();
151
152    fn pre_visit_query(&mut self, _query: &mut Query) -> ControlFlow<()> {
153        self.query_depth += 1;
154        ControlFlow::Continue(())
155    }
156
157    fn post_visit_query(&mut self, _query: &mut Query) -> ControlFlow<()> {
158        self.query_depth -= 1;
159        ControlFlow::Continue(())
160    }
161
162    fn pre_visit_expr(&mut self, expr: &mut SQLExpr) -> ControlFlow<()> {
163        if self.query_depth == 1
164            && let Some((_, column)) = self.columns.iter().find(|(outer, _)| outer == expr)
165        {
166            *expr = column.clone();
167        }
168        ControlFlow::Continue(())
169    }
170}

172/// Replaces each aggregate by a column of the domain table, and collects the aggregates,
173/// named after those columns. A name in `aliases` is replaced by the SELECT item it names.
174struct AggregateColumns<'a> {
175    ctx: &'a SQLContext,
176    domain_name: &'a Ident,
177    prefix: &'a str,
178    aggregates: Vec<SelectItem>,
179    aliases: Vec<(String, SQLExpr)>,
180    query_depth: usize,
181}
182
183impl VisitorMut for AggregateColumns<'_> {
184    type Break = ();
185
186    fn pre_visit_query(&mut self, _query: &mut Query) -> ControlFlow<()> {
187        self.query_depth += 1;
188        ControlFlow::Continue(())
189    }
190
191    fn post_visit_query(&mut self, _query: &mut Query) -> ControlFlow<()> {
192        self.query_depth -= 1;
193        ControlFlow::Continue(())
194    }
195
196    fn post_visit_expr(&mut self, expr: &mut SQLExpr) -> ControlFlow<()> {
197        if self.query_depth > 0 {
198            return ControlFlow::Continue(());
199        }
200        if let SQLExpr::Identifier(ident) = expr
201            && let Some((_, item)) = self.aliases.iter().find(|(name, _)| *name == ident.value)
202        {
203            *expr = SQLExpr::Nested(Box::new(item.clone()));
204        } else if is_aggregate_call(self.ctx, expr) {
205            let name = Ident::new(format!("{}_A{}", self.prefix, self.aggregates.len()));
206            let column = SQLExpr::CompoundIdentifier(vec![self.domain_name.clone(), name.clone()]);
207            self.aggregates.push(SelectItem::ExprWithAlias {
208                expr: std::mem::replace(expr, column),
209                alias: name,
210            });
211        }
212        ControlFlow::Continue(())
213    }
214}

216/// Whether an aggregate is 0 over no rows rather than NULL.
217fn zero_over_no_rows(ctx: &SQLContext, aggregate: &SQLExpr) -> bool {
218    use PolarsSQLFunctions::*;
219    let SQLExpr::Function(function) = aggregate else {
220        return false;
221    };
222    match PolarsSQLFunctions::try_from_sql(function, ctx) {
223        Ok(Count | CountIf | Total) => true,
224        #[cfg(feature = "approx_unique")]
225        Ok(ApproxCountDistinct) => true,
226        _ => false,
227    }
228}

230/// Puts the domain table in front of the first FROM item.
231fn read_domain(from: &mut Vec<TableWithJoins>, domain_name: &Ident) {
232    match from.first_mut() {
233        None => from.push(TableWithJoins {
234            relation: table(domain_name),
235            joins: vec![],
236        }),
237        Some(first) => {
238            let relation = std::mem::replace(&mut first.relation, table(domain_name));
239            first.joins.insert(
240                0,
241                Join {
242                    relation,
243                    global: false,
244                    join_operator: JoinOperator::CrossJoin(JoinConstraint::None),
245                },
246            );
247        },
248    }
249}

266/// The operator of a join on `constraint`, of the same kind as `op`, or `None` for a kind
267/// the rewrite doesn't support. A cross join becomes an inner join.
268fn with_constraint(op: &JoinOperator, constraint: JoinConstraint) -> Option<JoinOperator> {
269    Some(match op {
270        JoinOperator::Join(_) => JoinOperator::Join(constraint),
271        JoinOperator::Inner(_) | JoinOperator::CrossJoin(JoinConstraint::None) => {
272            JoinOperator::Inner(constraint)
273        },
274        JoinOperator::Left(_) => JoinOperator::Left(constraint),
275        JoinOperator::LeftOuter(_) => JoinOperator::LeftOuter(constraint),
276        _ => return None,
277    })
278}

280impl SQLContext {
281    /// Computes the aggregates of `query`, an aggregate query without GROUP BY, for each row
282    /// of the domain table as columns of it, and rewrites `query` to read those columns. A row
283    /// without inner rows gets each aggregate's value over no rows. Returns the domain table
284    /// with the columns, or `None` if the query has an ORDER BY or an unnamed aggregate item.
285    fn aggregate_per_domain_row(
286        &mut self,
287        query: &mut Query,
288        domain: LazyFrame,
289        domain_name: &Ident,
290        id: &PlSmallStr,
291        prefix: &str,
292        inner_schema: &Schema,
293    ) -> Option<LazyFrame> {
294        if query.order_by.is_some() {
295            return None;
296        }
297        let SetExpr::Select(select) = query.body.as_mut() else {
298            unreachable!()
299        };
300        let mut columns = AggregateColumns {
301            ctx: self,
302            domain_name,
303            prefix,
304            aggregates: Vec::new(),
305            aliases: Vec::new(),
306            query_depth: 0,
307        };
308        for item in &mut select.projection {
309            match item {
310                SelectItem::UnnamedExpr(expr) if has_aggregate(self, expr) => return None,
311                SelectItem::UnnamedExpr(expr) | SelectItem::ExprWithAlias { expr, .. } => {
312                    let _ = dyn_visit::visit_mut(expr, &mut columns);
313                },
314                _ => {},
315            }
316        }
317        // HAVING may name a SELECT item, if no input column has that name.
318        columns.aliases = select
319            .projection
320            .iter()
321            .filter_map(|item| match item {
322                SelectItem::ExprWithAlias { expr, alias }
323                    if !inner_schema.contains(&alias.value) =>
324                {
325                    Some((alias.value.clone(), expr.clone()))
326                },
327                _ => None,
328            })
329            .collect();
330        let mut having = select.having.take();
331        let _ = dyn_visit::visit_mut(&mut having, &mut columns);
332        let aggregates = columns.aggregates;
333        let fills: Vec<Expr> = aggregates
334            .iter()
335            .filter_map(|item| match item {
336                SelectItem::ExprWithAlias { expr, alias } if zero_over_no_rows(self, expr) => {
337                    Some(col(alias.value.as_str()).fill_null(lit(0)))
338                },
339                _ => None,
340            })
341            .collect();
342
343        let id_column =
344            SQLExpr::CompoundIdentifier(vec![domain_name.clone(), Ident::new(id.as_str())]);
345        let mut grouped = query.clone();
346        let SetExpr::Select(grouped_select) = grouped.body.as_mut() else {
347            unreachable!()
348        };
349        grouped_select.distinct = None;
350        grouped_select.projection = aggregates;
351        grouped_select
352            .projection
353            .push(SelectItem::UnnamedExpr(id_column.clone()));
354        grouped_select.group_by = GroupByExpr::Expressions(vec![id_column], vec![]);
355        read_domain(&mut grouped_select.from, domain_name);
356        let grouped = self
357            .execute_isolated(|ctx| {
358                ctx.table_map
359                    .write()
360                    .unwrap()
361                    .insert(domain_name.value.clone(), domain.clone());
362                ctx.execute_query(&grouped)
363            })
364            .ok()?;
365
366        let SetExpr::Select(select) = query.body.as_mut() else {
367            unreachable!()
368        };
369        select.from.clear();
370        select.selection = having;
371        domain
372            .join_builder()
373            .with(grouped)
374            .left_on([col(id.clone())])
375            .right_on([col(id.clone())])
376            .how(JoinType::Left)
377            .coalesce(JoinCoalesce::CoalesceColumns)
378            .finish()
379            .ok()
380            .map(|lf| lf.with_columns(fills))
381    }

383    /// The inputs of the join of `lf` with `relation`, a derived table that reads columns of
384    /// `lf` (see the module docs), or `None` if it has another shape.
385    pub(crate) fn lateral_join(
386        &mut self,
387        relation: &TableFactor,
388        join_operator: &JoinOperator,
389        lf: &LazyFrame,
390    ) -> Option<LateralJoin> {
391        let TableFactor::Derived {
392            subquery,
393            alias: Some(alias),
394            sample: None,
395            ..
396        } = relation
397        else {
398            return None;
399        };
400        let on = match join_operator {
401            JoinOperator::Join(constraint)
402            | JoinOperator::Inner(constraint)
403            | JoinOperator::Left(constraint)
404            | JoinOperator::LeftOuter(constraint)
405            | JoinOperator::CrossJoin(constraint) => match constraint {
406                JoinConstraint::On(on) => Some(on),
407                JoinConstraint::None => None,
408                _ => return None,
409            },
410            _ => return None,
411        };
412        // A CTE of the derived table may hide a relation whose columns would be read instead.
413        if !alias.columns.is_empty()
414            || subquery.with.is_some()
415            || subquery.limit_clause.is_some()
416            || subquery.fetch.is_some()
417            || !subquery.locks.is_empty()
418            || subquery.for_clause.is_some()
419            || subquery.settings.is_some()
420            || subquery.format_clause.is_some()
421            || !subquery.pipe_operators.is_empty()
422        {
423            return None;
424        }
425        let SetExpr::Select(select) = subquery.body.as_ref() else {
426            return None;
427        };
428        let GroupByExpr::Expressions(group_by, modifiers) = &select.group_by else {
429            return None;
430        };
431        // An empty group yields a row for an outer row without inner rows.
432        let empty_group = group_by.is_empty()
433            && (select.having.is_some()
434                || has_aggregate(self, &select.projection)
435                || has_aggregate(self, &subquery.order_by));
436        if !modifiers.is_empty()
437            || group_by.iter().any(|key| {
438                matches!(
439                    strip_nested(key),
440                    SQLExpr::Rollup(_) | SQLExpr::Cube(_) | SQLExpr::GroupingSets(_)
441                ) || matches!(key, SQLExpr::Tuple(keys) if keys.is_empty())
442            })
443            || matches!(select.distinct, Some(Distinct::On(_)))
444            || select.top.is_some()
445            || select.qualify.is_some()
446            || !select.named_window.is_empty()
447            || !select.lateral_views.is_empty()
448            || select.value_table_mode.is_some()
449            || select.select_modifiers.is_some()
450        {
451            return None;
452        }
453        // The domain table joins before the first FROM item, so a RIGHT or FULL join would
454        // add rows without an outer value.
455        if select.from.first().is_some_and(|tbl| {
456            tbl.joins.iter().any(|join| {
457                matches!(
458                    join.join_operator,
459                    JoinOperator::Right(_)
460                        | JoinOperator::RightOuter(_)
461                        | JoinOperator::FullOuter(_)
462                        | JoinOperator::RightSemi(_)
463                        | JoinOperator::RightAnti(_)
464                )
465            })
466        }) {
467            return None;
468        }
469        // An ordinal past the SELECT items would read the added id column.
470        let order_by_items = match &subquery.order_by {
471            Some(OrderBy {
472                kind: OrderByKind::Expressions(items),
473                ..
474            }) => items.iter().map(|item| &item.expr).collect(),
475            Some(_) => return None,
476            None => vec![],
477        };
478        let has_qualified_wildcard = select
479            .projection
480            .iter()
481            .any(|item| matches!(item, SelectItem::QualifiedWildcard(..)));
482        let invalid_ordinal = |e: &SQLExpr| match e {
483            SQLExpr::Value(ValueWithSpan {
484                value: SQLValue::Number(n, _),
485                ..
486            }) => {
487                has_qualified_wildcard
488                    || n.parse::<usize>()
489                        .map_or(true, |n| n > select.projection.len())
490            },
491            _ => false,
492        };
493        if group_by.iter().chain(order_by_items).any(invalid_ordinal) {
494            return None;
495        }
496
497        // The columns of the derived table's own relations.
498        let factors: Vec<&TableFactor> = select
499            .from
500            .iter()
501            .flat_map(|tbl| {
502                std::iter::once(&tbl.relation).chain(tbl.joins.iter().map(|j| &j.relation))
503            })
504            .collect();
505        let names: Vec<RelationName> = factors
506            .iter()
507            .filter_map(|factor| {
508                Some(relation_name(factor)?.spelled(factor, self.catalog_spellings()))
509            })
510            .collect();
511        let mut schema = Schema::default();
512        let mut ctx = self.isolated();
513        for factor in &factors {
514            let (_, mut frame) = ctx.get_table(factor).ok()?;
515            schema.merge((*ctx.get_frame_schema(&mut frame).ok()?).clone());
516        }
517        let aliases: Vec<Ident> = select
518            .projection
519            .iter()
520            .filter_map(|item| match item {
521                SelectItem::ExprWithAlias { alias, .. } => Some(alias.clone()),
522                _ => None,
523            })
524            .collect();
525        let inner = Inner {
526            names: &names,
527            schema: &schema,
528            aliases: &aliases,
529        };
530        // A wildcard would also select the columns of the domain table, and one that renames
531        // or replaces columns names columns that are not inner.
532        for item in &select.projection {
533            match item {
534                SelectItem::Wildcard(_) => return None,
535                SelectItem::QualifiedWildcard(_, options)
536                    if options.opt_rename.is_some() || options.opt_replace.is_some() =>
537                {
538                    return None;
539                },
540                SelectItem::QualifiedWildcard(
541                    SelectItemQualifiedWildcardKind::ObjectName(name),
542                    _,
543                ) => {
544                    let parts: Vec<Ident> = name
545                        .0
546                        .iter()
547                        .filter_map(|p| p.as_ident().cloned())
548                        .collect();
549                    if !names.iter().any(|n| n.named_by(&parts, false)) {
550                        return None;
551                    }
552                },
553                SelectItem::QualifiedWildcard(..) => return None,
554                _ => {},
555            }
556        }
557
558        let mut outer = OuterColumns {
559            ctx: self,
560            inner: &inner,
561            query_depth: 0,
562            found: Vec::new(),
563        };
564        if dyn_visit::visit(subquery.as_ref(), &mut outer).is_break() {
565            return None;
566        }
567        let found = outer.found;
568
569        // The domain and the left input must see the same rows.
570        let mut lf = lf.clone().cache();
571        let left_schema = self.get_frame_schema(&mut lf).ok()?;
572        let prefix = format!("__POLARS_LATERAL_{}", unique_column_name());
573        let domain_name = Ident::new(&prefix);
574        let id = format_pl_smallstr!("{prefix}_ID");
575        let left_key = format_pl_smallstr!("{prefix}_L");
576        let right_key = format_pl_smallstr!("{prefix}_R");
577        let mut outer_exprs: Vec<Expr> = Vec::new();
578        // The name of each outer value, as the derived table would name a column of it.
579        let mut value_names: Vec<PlSmallStr> = Vec::new();
580        let mut columns: Vec<(SQLExpr, SQLExpr)> = Vec::new();
581        for sql_expr in found {
582            let name = match &sql_expr {
583                SQLExpr::Identifier(ident) => {
584                    // A name of several relations joined before is ambiguous.
585                    let joined_name = format!("{}:", ident.value);
586                    if left_schema
587                        .iter_names()
588                        .any(|name| name.starts_with(&joined_name))
589                    {
590                        return None;
591                    }
592                    &ident.value
593                },
594                SQLExpr::CompoundIdentifier(parts) => &parts.last()?.value,
595                _ => unreachable!(),
596            };
597            let Ok(Some(expr)) = self.try_parse_outer_only_expr(&sql_expr, &left_schema) else {
598                continue;
599            };
600            let index = match outer_exprs.iter().position(|e| *e == expr) {
601                Some(index) => index,
602                None => {
603                    outer_exprs.push(expr);
604                    value_names.push(name.into());
605                    outer_exprs.len() - 1
606                },
607            };
608            let column = SQLExpr::CompoundIdentifier(vec![
609                domain_name.clone(),
610                Ident::new(format!("{prefix}_V{index}")),
611            ]);
612            columns.push((sql_expr, column));
613        }
614        if outer_exprs.is_empty() {
615            return None;
616        }
617
618        let values: Vec<PlSmallStr> = (0..outer_exprs.len())
619            .map(|i| format_pl_smallstr!("{prefix}_V{i}"))
620            .collect();
621        let domain = row_indexed_once(
622            lf.clone()
623                .select(
624                    outer_exprs
625                        .iter()
626                        .zip(&values)
627                        .map(|(e, name)| e.clone().alias(name.clone()))
628                        .collect::<Vec<_>>(),
629                )
630                .unique(None, UniqueKeepStrategy::Any),
631            id.clone(),
632        );
633        let left = lf
634            .join_builder()
635            .with(
636                domain.clone().select(
637                    values
638                        .iter()
639                        .map(|name| col(name.clone()))
640                        .chain(std::iter::once(col(id.clone()).alias(left_key.clone())))
641                        .collect::<Vec<_>>(),
642                ),
643            )
644            .left_on(outer_exprs)
645            .right_on(
646                values
647                    .iter()
648                    .map(|name| col(name.clone()))
649                    .collect::<Vec<_>>(),
650            )
651            .how(JoinType::Left)
652            .join_nulls(true)
653            .coalesce(JoinCoalesce::KeepColumns)
654            .maintain_order(MaintainOrderJoin::Left)
655            .finish()
656            .ok()?
657            .drop(Selector::ByName {
658                names: values.iter().cloned().collect(),
659                strict: true,
660            });
661
662        // The derived table reads the domain table, and keeps the id of the outer values.
663        let mut query = subquery.as_ref().clone();
664        let _ = dyn_visit::visit_mut(
665            &mut query,
666            &mut ReadDomain {
667                columns: &columns,
668                query_depth: 0,
669            },
670        );
671        let domain = if empty_group {
672            self.aggregate_per_domain_row(&mut query, domain, &domain_name, &id, &prefix, &schema)?
673        } else {
674            domain
675        };
676        let SetExpr::Select(select) = query.body.as_mut() else {
677            unreachable!()
678        };
679        let domain_column =
680            |name: &str| SQLExpr::CompoundIdentifier(vec![domain_name.clone(), Ident::new(name)]);
681        read_domain(&mut select.from, &domain_name);
682        select.projection.push(SelectItem::ExprWithAlias {
683            expr: domain_column(&id),
684            alias: Ident::new(right_key.as_str()),
685        });
686        if let GroupByExpr::Expressions(group_by, _) = &mut select.group_by
687            && !group_by.is_empty()
688        {
689            group_by.extend(
690                std::iter::once(id.as_str())
691                    .chain(values.iter().map(|v| v.as_str()))
692                    .map(domain_column),
693            );
694        }
695        let mut right = self
696            .execute_isolated(|ctx| {
697                ctx.table_map
698                    .write()
699                    .unwrap()
700                    .insert(domain_name.value.clone(), domain);
701                ctx.execute_query(&query)
702            })
703            .ok()?;
704        // An unnamed SELECT item named after an outer column takes that column's name.
705        let right_schema = self.get_frame_schema(&mut right).ok()?;
706        let (renamed, names): (Vec<_>, Vec<_>) = values
707            .iter()
708            .zip(&value_names)
709            .filter(|(value, _)| right_schema.contains(value))
710            .map(|(value, name)| (value.clone(), name.clone()))
711            .unzip();
712        let right = right.rename(renamed, names, true);
713
714        let key_condition = SQLExpr::BinaryOp {
715            left: Box::new(SQLExpr::Identifier(Ident::new(left_key.as_str()))),
716            op: SQLBinaryOperator::Eq,
717            right: Box::new(SQLExpr::Identifier(Ident::new(right_key.as_str()))),
718        };
719        let condition = match on {
720            None
721            | Some(SQLExpr::Value(ValueWithSpan {
722                value: SQLValue::Boolean(true),
723                ..
724            })) => key_condition,
725            Some(on) => SQLExpr::BinaryOp {
726                left: Box::new(on.clone()),
727                op: SQLBinaryOperator::And,
728                right: Box::new(key_condition),
729            },
730        };
731        let name = alias.name.value.clone();
732        self.from_item_frames.insert(
733            name.clone(),
734            right.clone().drop(Selector::ByName {
735                names: [right_key.clone()].into(),
736                strict: true,
737            }),
738        );
739        Some(LateralJoin {
740            left,
741            name,
742            right,
743            join_operator: with_constraint(join_operator, JoinConstraint::On(condition))?,
744            keys: [left_key, right_key],
745        })
746    }
747}
