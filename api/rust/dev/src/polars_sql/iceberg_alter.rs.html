1//! The `ALTER TABLE` operations of Iceberg tables that sqlparser does not parse yet:
2//! `SET | RESET PARTITIONED BY`, `SET | RESET SORTED BY`, and the tag and branch statements
3//! (`CREATE [OR REPLACE] TAG | BRANCH`, `REPLACE TAG | BRANCH`, `DROP TAG | BRANCH`).
4//!
5//! Temporary: remove once sqlparser parses these statements.

17/// A statement that sqlparser does not parse.
18pub(crate) enum OtherStatement {
19    Alter(IcebergAlter),
20    Macro(MacroStatement),
21}
22
23/// A parsed `ALTER TABLE [IF EXISTS] <name> <operation>`, for one of the operations above.
24#[derive(Clone, Debug)]
25pub(crate) struct IcebergAlter {
26    pub(crate) name: ObjectName,
27    pub(crate) if_exists: bool,
28    pub(crate) operation: IcebergAlterOp,
29}
30
31/// The operation, as written.
32#[derive(Clone, Debug)]
33pub(crate) enum IcebergAlterOp {
34    /// The new partition fields; none for `RESET`.
35    Partitioning(Vec<Expr>),
36    /// The new sort fields; none for `RESET`.
37    SortOrder(Vec<OrderByExpr>),
38    /// A [`CatalogAlter::SetRef`] or [`CatalogAlter::DropRef`].
39    Ref(CatalogAlter),
40}
41
42/// What follows the table name, up to the words that make this one of the operations above.
43enum Head {
44    Layout {
45        reset: bool,
46        sort: bool,
47    },
48    SetRef {
49        kind: CatalogRefKind,
50        mode: CatalogRefMode,
51    },
52    DropRef {
53        kind: CatalogRefKind,
54    },
55}

57/// The statements of `query`, or the statement that sqlparser does not parse.
58pub(crate) fn parse_statements_or_alter(
59    query: &str,
60) -> PolarsResult<Result<Vec<Statement>, OtherStatement>> {
61    let (mut parser, taken) = sql_parser(query)?;
62    if let Some(alter) = parse_iceberg_alter(&mut parser)? {
63        // a time travel clause has no table to read here
64        taken.refuse()?;
65        return Ok(Err(OtherStatement::Alter(alter)));
66    }
67    match parse_macro_statement(&mut parser) {
68        Ok(None) => {},
69        parsed => {
70            let parse = |parser: &mut Parser<'static>| {
71                Ok(parse_macro_statement(parser)?.into_iter().collect())
72            };
73            let parsed = parsed.map(|stmt| stmt.into_iter().collect());
74            let stmt = taken.attach_to(query, parsed, parse)?.pop();
75            let stmt = stmt.ok_or_else(|| polars_err!(SQLInterface: "invalid macro statement"))?;
76            return Ok(Err(OtherStatement::Macro(stmt)));
77        },
78    }
79    Ok(Ok(taken.attach(query, parser.parse_statements())?))
80}

82/// The single statement of `query`, or the statement that sqlparser does not parse.
83pub(crate) fn parse_single_statement_or_alter(
84    query: &str,
85) -> PolarsResult<Result<Statement, OtherStatement>> {
86    Ok(match parse_statements_or_alter(query)? {
87        Err(other) => Err(other),
88        Ok(statements) => {
89            let [stmt]: [Statement; 1] = statements.try_into().map_err(|_| {
90                polars_err!(SQLInterface: "one (and only one) statement can be parsed at a time")
91            })?;
92            Ok(stmt)
93        },
94    })
95}

97/// The operation at the start of `parser`; if there is none, `parser` stays at the start.
98fn parse_iceberg_alter(parser: &mut Parser) -> PolarsResult<Option<IcebergAlter>> {
99    // Up to the words that make this an operation of this module. Anything else is parsed as usual.
100    let not_ours = || ParserError::ParserError(String::new());
101    let head = parser
102        .maybe_parse(|p| {
103            // as sqlparser, which skips empty statements
104            while p.consume_token(&Token::SemiColon) {}
105            p.expect_keywords(&[Keyword::ALTER, Keyword::TABLE])?;
106            let if_exists = p.parse_keywords(&[Keyword::IF, Keyword::EXISTS]);
107            let only = p.parse_keyword(Keyword::ONLY);
108            let name = p.parse_object_name(false)?;
109            let head =
110                if let Some(keyword) = p.parse_one_of_keywords(&[Keyword::SET, Keyword::RESET]) {
111                    let sort = if p.parse_keywords(&[Keyword::PARTITIONED, Keyword::BY]) {
112                        false
113                    } else if p.parse_keywords(&[Keyword::SORTED, Keyword::BY]) {
114                        true
115                    } else {
116                        return Err(not_ours());
117                    };
118                    Head::Layout {
119                        reset: keyword == Keyword::RESET,
120                        sort,
121                    }
122                } else if p.parse_keyword(Keyword::CREATE) {
123                    let replace = p.parse_keywords(&[Keyword::OR, Keyword::REPLACE]);
124                    let kind = parse_ref_kind(p).ok_or_else(not_ours)?;
125                    let mode = if replace {
126                        CatalogRefMode::CreateOrReplace
127                    } else if p.parse_keywords(&[Keyword::IF, Keyword::NOT, Keyword::EXISTS]) {
128                        CatalogRefMode::CreateIfNotExists
129                    } else {
130                        CatalogRefMode::Create
131                    };
132                    Head::SetRef { kind, mode }
133                } else if p.parse_keyword(Keyword::REPLACE) {
134                    let kind = parse_ref_kind(p).ok_or_else(not_ours)?;
135                    Head::SetRef {
136                        kind,
137                        mode: CatalogRefMode::Replace,
138                    }
139                } else if p.parse_keyword(Keyword::DROP) {
140                    let kind = parse_ref_kind(p).ok_or_else(not_ours)?;
141                    // `DROP tag` alone drops a column of that name
142                    if !starts_ref_name(p) {
143                        return Err(not_ours());
144                    }
145                    Head::DropRef { kind }
146                } else {
147                    return Err(not_ours());
148                };
149            Ok((name, if_exists, only, head))
150        })
151        .map_err(to_sql_interface_err)?;
152    let Some((name, if_exists, only, head)) = head else {
153        return Ok(None);
154    };
155    polars_ensure!(!only, SQLInterface: "`ALTER TABLE ONLY` is not supported");
156    let operation = match head {
157        Head::Layout { reset: true, sort } => match sort {
158            false => IcebergAlterOp::Partitioning(Vec::new()),
159            true => IcebergAlterOp::SortOrder(Vec::new()),
160        },
161        Head::Layout {
162            reset: false,
163            sort: false,
164        } => IcebergAlterOp::Partitioning(parse_fields(
165            parser,
166            "`SET PARTITIONED BY ()` needs a partition field; `RESET PARTITIONED BY` removes the partitioning",
167            Parser::parse_expr,
168        )?),
169        Head::Layout {
170            reset: false,
171            sort: true,
172        } => IcebergAlterOp::SortOrder(parse_fields(
173            parser,
174            "`SET SORTED BY ()` needs a sort field; `RESET SORTED BY` removes the sort order",
175            Parser::parse_order_by_expr,
176        )?),
177        Head::SetRef { kind, mode } => IcebergAlterOp::Ref(parse_set_ref(parser, kind, mode)?),
178        Head::DropRef { kind } => {
179            let if_exists = parser.parse_keywords(&[Keyword::IF, Keyword::EXISTS]);
180            let name = parse_ref_name(parser, kind)?;
181            IcebergAlterOp::Ref(CatalogAlter::DropRef {
182                kind,
183                name,
184                if_exists,
185            })
186        },
187    };
188    polars_ensure!(
189        !parser.consume_token(&Token::Comma),
190        SQLInterface: "`ALTER TABLE` takes one operation per statement"
191    );
192    let ended = parser.consume_token(&Token::SemiColon);
193    while parser.consume_token(&Token::SemiColon) {}
194    let next = parser.peek_token();
195    if next.token != Token::EOF {
196        polars_ensure!(
197            !ended,
198            SQLInterface: "one (and only one) statement can be parsed at a time"
199        );
200        return parser
201            .expected("end of statement", next)
202            .map_err(to_sql_interface_err);
203    }
204    Ok(Some(IcebergAlter {
205        name,
206        if_exists,
207        operation,
208    }))
209}

211/// The parenthesized, non-empty fields at `parser`; `empty` is the message for `()`.
212fn parse_fields<'a, T>(
213    parser: &mut Parser<'a>,
214    empty: &str,
215    field: impl FnMut(&mut Parser<'a>) -> Result<T, ParserError>,
216) -> PolarsResult<Vec<T>> {
217    parser
218        .expect_token(&Token::LParen)
219        .map_err(to_sql_interface_err)?;
220    polars_ensure!(!parser.consume_token(&Token::RParen), SQLInterface: "{}", empty);
221    let fields = parser
222        .parse_comma_separated(field)
223        .map_err(to_sql_interface_err)?;
224    parser
225        .expect_token(&Token::RParen)
226        .map_err(to_sql_interface_err)?;
227    Ok(fields)
228}

230/// Whether the next token is the unquoted `word`, any case; if so, it is taken.
231fn parse_word(parser: &mut Parser, word: &str) -> bool {
232    let next = parser.peek_token();
233    let found = matches!(
234        &next.token,
235        Token::Word(w) if w.quote_style.is_none() && w.value.eq_ignore_ascii_case(word)
236    );
237    if found {
238        parser.next_token();
239    }
240    found
241}

243fn parse_ref_kind(parser: &mut Parser) -> Option<CatalogRefKind> {
244    if parse_word(parser, "TAG") {
245        Some(CatalogRefKind::Tag)
246    } else if parse_word(parser, "BRANCH") {
247        Some(CatalogRefKind::Branch)
248    } else {
249        None
250    }
251}
252
253fn ref_kind_name(kind: CatalogRefKind) -> &'static str {
254    match kind {
255        CatalogRefKind::Tag => "tag",
256        CatalogRefKind::Branch => "branch",
257    }
258}

260/// Whether a ref name (or `IF EXISTS`) follows, rather than the end of a column drop.
261fn starts_ref_name(parser: &Parser) -> bool {
262    match parser.peek_token().token {
263        Token::Word(w) => !matches!(w.keyword, Keyword::CASCADE | Keyword::RESTRICT),
264        Token::SingleQuotedString(_) => true,
265        _ => false,
266    }
267}

269/// The name of a tag or branch: an identifier, bare or quoted, with its case.
270fn parse_ref_name(parser: &mut Parser, kind: CatalogRefKind) -> PolarsResult<String> {
271    let kind = ref_kind_name(kind);
272    polars_ensure!(
273        !matches!(parser.peek_token().token, Token::SingleQuotedString(_)),
274        SQLInterface: "a {} name is an identifier: write it bare or in double quotes, not in single quotes", kind
275    );
276    let name = parser
277        .parse_identifier()
278        .map_err(to_sql_interface_err)?
279        .value;
280    polars_ensure!(!name.is_empty(), SQLInterface: "a {} name cannot be empty", kind);
281    Ok(name)
282}

284/// The name and options of a statement that points a tag or branch at a snapshot:
285/// `<name> [AS OF VERSION <id>] [RETAIN <n> <unit>] [WITH SNAPSHOT RETENTION ...]`.
286fn parse_set_ref(
287    parser: &mut Parser,
288    kind: CatalogRefKind,
289    mode: CatalogRefMode,
290) -> PolarsResult<CatalogAlter> {
291    let name = parse_ref_name(parser, kind)?;
292    // the forms of the read clause (`FOR VERSION AS OF`, `VERSION AS OF`, `AT (...)`)
293    polars_ensure!(
294        !matches!(
295            &parser.peek_token().token,
296            Token::Word(w) if w.quote_style.is_none() && (
297                matches!(w.keyword, Keyword::FOR | Keyword::AT)
298                    || w.value.eq_ignore_ascii_case("VERSION")
299            )
300        ),
301        SQLInterface: "the snapshot of a {} is written `AS OF VERSION <snapshot id>`", ref_kind_name(kind)
302    );
303    let snapshot_id = if parser.parse_keywords(&[Keyword::AS, Keyword::OF]) {
304        if !parse_word(parser, "VERSION") {
305            return parser
306                .expected("VERSION", parser.peek_token())
307                .map_err(to_sql_interface_err);
308        }
309        Some(parse_snapshot_id(parser)?)
310    } else {
311        None
312    };
313    let max_ref_age_ms = match parse_word(parser, "RETAIN") {
314        true => Some(parse_duration_ms(parser, "RETAIN")?),
315        false => None,
316    };
317    let (mut min_snapshots_to_keep, mut max_snapshot_age_ms) = (None, None);
318    if parser.parse_keyword(Keyword::WITH) {
319        if !(parse_word(parser, "SNAPSHOT") && parse_word(parser, "RETENTION")) {
320            return parser
321                .expected("SNAPSHOT RETENTION", parser.peek_token())
322                .map_err(to_sql_interface_err);
323        }
324        polars_ensure!(
325            kind == CatalogRefKind::Branch,
326            SQLInterface: "`WITH SNAPSHOT RETENTION` is for branches: a tag points at one snapshot"
327        );
328        let what = "WITH SNAPSHOT RETENTION";
329        let n = parse_positive(parser, what)?;
330        if parse_word(parser, "SNAPSHOTS") {
331            min_snapshots_to_keep = Some(i32::try_from(n).map_err(
332                |_| polars_err!(SQLInterface: "`{}` takes at most {} snapshots", what, i32::MAX),
333            )?);
334            if matches!(parser.peek_token().token, Token::Number(..)) {
335                max_snapshot_age_ms = Some(parse_duration_ms(parser, what)?);
336            }
337        } else {
338            max_snapshot_age_ms = Some(times_unit(parser, n, what)?);
339        }
340    }
341    Ok(CatalogAlter::SetRef {
342        kind,
343        name,
344        mode,
345        snapshot_id,
346        max_ref_age_ms,
347        min_snapshots_to_keep,
348        max_snapshot_age_ms,
349    })
350}

352/// The snapshot id of `AS OF VERSION`.
353fn parse_snapshot_id(parser: &mut Parser) -> PolarsResult<i64> {
354    let negative = parser.consume_token(&Token::Minus);
355    let next = parser.next_token();
356    match &next.token {
357        Token::Number(digits, _) => {
358            let digits = if negative {
359                format!("-{digits}")
360            } else {
361                digits.clone()
362            };
363            digits.parse().map_err(
364                |_| polars_err!(SQLInterface: "`AS OF VERSION` takes a snapshot id; found {}", digits),
365            )
366        },
367        Token::SingleQuotedString(_) if !negative => polars_bail!(
368            SQLInterface: "`AS OF VERSION` takes a snapshot id, not a tag or branch: the `$refs` table (e.g. `SELECT * FROM t$refs`) lists the snapshot of each"
369        ),
370        _ => parser
371            .expected("a snapshot id", next)
372            .map_err(to_sql_interface_err),
373    }
374}

376/// A whole number greater than zero, for the option `what`.
377fn parse_positive(parser: &mut Parser, what: &str) -> PolarsResult<i64> {
378    let next = parser.next_token();
379    match &next.token {
380        Token::Number(digits, _) => match digits.parse::<i64>() {
381            Ok(n) if n > 0 => Ok(n),
382            _ => {
383                polars_bail!(SQLInterface: "`{}` takes a whole number greater than 0; found {}", what, digits)
384            },
385        },
386        _ => parser
387            .expected("a whole number", next)
388            .map_err(to_sql_interface_err),
389    }
390}

392/// `<n> DAYS | HOURS | MINUTES`, in milliseconds.
393fn parse_duration_ms(parser: &mut Parser, what: &str) -> PolarsResult<i64> {
394    let n = parse_positive(parser, what)?;
395    times_unit(parser, n, what)
396}

398/// `n` times the unit that follows, in milliseconds.
399fn times_unit(parser: &mut Parser, n: i64, what: &str) -> PolarsResult<i64> {
400    let unit_ms: i64 = if parse_word(parser, "DAYS") {
401        86_400_000
402    } else if parse_word(parser, "HOURS") {
403        3_600_000
404    } else if parse_word(parser, "MINUTES") {
405        60_000
406    } else {
407        return parser
408            .expected("DAYS, HOURS or MINUTES", parser.peek_token())
409            .map_err(to_sql_interface_err);
410    };
411    n.checked_mul(unit_ms)
412        .ok_or_else(|| polars_err!(SQLInterface: "`{}` is too long", what))
