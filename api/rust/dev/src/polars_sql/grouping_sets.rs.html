14/// Cap on the number of grouping sets a single `GROUP BY` may expand to.
15const MAX_GROUPING_SETS: usize = 4096;
16
17/// Widest `GROUPING()` call whose bits fit an `Int64`.
18pub(crate) const MAX_GROUPING_ARGS: usize = 63;

20const PLACEHOLDER_PREFIX: &str = "__POLARS_GROUPING_";

22const GLOBAL_COUNT_COL: &str = "__POLARS_GSET_COUNT";

24/// A `GROUPING(...)` call parsed in the current query block.
25///
26/// Its arguments are bound to the grouping keys once the `GROUP BY` clause is
27/// known; until then the call is a reference to `placeholder`.
28#[derive(Clone)]
29pub(crate) struct GroupingCall {
30    pub args: Vec<SQLExpr>,
31    pub placeholder: PlSmallStr,
32}
33
34pub(crate) fn new_placeholder() -> PlSmallStr {
35    format_pl_smallstr!("{}{}", PLACEHOLDER_PREFIX, unique_column_name())
36}

38/// Whether `expr` refers to any of the given `GROUPING()` placeholders.
39pub(crate) fn contains_grouping_placeholder<'a>(
40    expr: &Expr,
41    placeholders: impl Iterator<Item = &'a PlSmallStr>,
42) -> bool {
43    let placeholders: Vec<&PlSmallStr> = placeholders.collect();
44    has_expr(
45        expr,
46        |e| matches!(e, Expr::Column(name) if placeholders.contains(&name)),
47    )
48}

50/// The `GROUP BY` items of a query with grouping sets, expanded into the distinct
51/// key expressions they reference and the sets, given as indices into those keys.
52pub(crate) struct ExpandedGroupBy {
53    pub keys: Vec<SQLExpr>,
54    pub sets: Vec<Vec<usize>>,
55}

57/// The alternative key lists a single `GROUP BY` item contributes, or `None` for a
58/// plain expression (which every grouping set then includes).
59fn grouping_item_alternatives(item: &SQLExpr) -> PolarsResult<Option<Vec<Vec<SQLExpr>>>> {
60    let flatten =
61        |groups: &[Vec<SQLExpr>]| -> Vec<SQLExpr> { groups.iter().flatten().cloned().collect() };
62    Ok(Some(match item {
63        // ROLLUP(a, b, c) => (a,b,c), (a,b), (a), ()
64        SQLExpr::Rollup(groups) => (0..=groups.len())
65            .rev()
66            .map(|n| flatten(&groups[..n]))
67            .collect(),
68        // CUBE(a, b, c) => every subset of (a, b, c)
69        SQLExpr::Cube(groups) => {
70            let n = groups.len();
71            polars_ensure!(
72                n < usize::BITS as usize && (1usize << n) <= MAX_GROUPING_SETS,
73                SQLInterface: "CUBE with {} grouping expressions exceeds the {} grouping set limit", n, MAX_GROUPING_SETS
74            );
75            (0..(1usize << n))
76                .rev()
77                .map(|mask| {
78                    (0..n)
79                        .filter(|i| (mask >> i) & 1 == 1)
80                        .flat_map(|i| groups[i].iter().cloned())
81                        .collect()
82                })
83                .collect()
84        },
85        SQLExpr::GroupingSets(sets) => sets.clone(),
86        // `GROUP BY ()`
87        SQLExpr::Tuple(items) if items.is_empty() => vec![vec![]],
88        _ => return Ok(None),
89    }))
90}

92/// Expand a `GROUP BY` item list; `None` when it holds no grouping-set construct
93/// and can be grouped directly. Multiple items combine as a Cartesian product.
94pub(crate) fn expand_grouping_sets(items: &[SQLExpr]) -> PolarsResult<Option<ExpandedGroupBy>> {
95    let alternatives = items
96        .iter()
97        .map(grouping_item_alternatives)
98        .collect::<PolarsResult<Vec<_>>>()?;
99    if alternatives.iter().all(Option::is_none) {
100        return Ok(None);
101    }
102
103    let mut keys: PlIndexSet<SQLExpr> = PlIndexSet::default();
104    let mut sets: Vec<Vec<usize>> = vec![Vec::new()];
105    for (item, alts) in items.iter().zip(alternatives) {
106        let alts = alts.unwrap_or_else(|| vec![vec![item.clone()]]);
107        let n_expanded = sets
108            .len()
109            .checked_mul(alts.len())
110            .filter(|n| *n <= MAX_GROUPING_SETS);
111        let Some(n_expanded) = n_expanded else {
112            polars_bail!(SQLInterface: "GROUP BY expands to more than {} grouping sets", MAX_GROUPING_SETS);
113        };
114        let mut expanded = Vec::with_capacity(n_expanded);
115        for base in &sets {
116            for alt in &alts {
117                let mut set = base.clone();
118                for e in alt {
119                    let (idx, _) = keys.insert_full(e.clone());
120                    if !set.contains(&idx) {
121                        set.push(idx);
122                    }
123                }
124                expanded.push(set);
125            }
126        }
127        sets = expanded;
128    }
129    Ok(Some(ExpandedGroupBy {
130        keys: keys.into_iter().collect(),
131        sets,
132    }))
133}

135/// Merge keys that resolved to the same expression, remapping the sets onto the
136/// canonical key list.
137pub(crate) fn canonicalize_keys(
138    resolved: Vec<Expr>,
139    sets: Vec<Vec<usize>>,
140) -> (Vec<Expr>, Vec<Vec<usize>>) {
141    let mut keys: Vec<Expr> = Vec::with_capacity(resolved.len());
142    let mut stripped: Vec<Expr> = Vec::with_capacity(resolved.len());
143    let mut remap = Vec::with_capacity(resolved.len());
144    for key in resolved {
145        let s = strip_outer_alias(&key);
146        let idx = stripped.iter().position(|k| *k == s).unwrap_or_else(|| {
147            keys.push(key);
148            stripped.push(s);
149            keys.len() - 1
150        });
151        remap.push(idx);
152    }
153    let sets = sets
154        .into_iter()
155        .map(|set| {
156            let mut out: Vec<usize> = Vec::with_capacity(set.len());
157            for i in set {
158                if !out.contains(&remap[i]) {
159                    out.push(remap[i]);
160                }
161            }
162            out
163        })
164        .collect();
165    (keys, sets)
166}

168/// Everything the aggregation stage needs to know about a grouping-sets query.
169pub(crate) struct GroupingSets {
170    /// Canonical key expressions, as handed to `group_by`.
171    keys: Vec<Expr>,
172    /// The keys without their output alias, for matching by expression.
173    stripped_keys: Vec<Expr>,
174    /// One entry per grouping-set occurrence, holding the active canonical keys.
175    sets: Vec<Vec<usize>>,
176    /// Each `GROUPING()` placeholder with the canonical keys of its arguments.
177    calls: Vec<(PlSmallStr, Vec<usize>)>,
178}
179
180impl GroupingSets {
181    /// Bind every parsed `GROUPING()` call to the canonical keys.
182    pub(crate) fn new(
183        keys: Vec<Expr>,
184        sets: Vec<Vec<usize>>,
185        calls: &[GroupingCall],
186        resolved_args: Vec<Vec<Expr>>,
187    ) -> PolarsResult<Self> {
188        let stripped_keys: Vec<Expr> = keys.iter().map(strip_outer_alias).collect();
189        let calls = calls
190            .iter()
191            .zip(resolved_args)
192            .map(|(call, args)| {
193                let indices = args
194                    .iter()
195                    .zip(&call.args)
196                    .map(|(arg, sql_arg)| {
197                        let arg = strip_outer_alias(arg);
198                        stripped_keys.iter().position(|k| *k == arg).ok_or_else(|| {
199                            polars_err!(SQLSyntax: "GROUPING() argument '{}' does not appear in the GROUP BY clause", sql_arg)
200                        })
201                    })
202                    .collect::<PolarsResult<Vec<_>>>()?;
203                Ok((call.placeholder.clone(), indices))
204            })
205            .collect::<PolarsResult<Vec<_>>>()?;
206        Ok(Self {
207            keys,
208            stripped_keys,
209            sets,
210            calls,
211        })
212    }
213
214    pub(crate) fn placeholders(&self) -> impl Iterator<Item = &PlSmallStr> {
215        self.calls.iter().map(|(name, _)| name)
216    }
217
218    pub(crate) fn contains_placeholder(&self, expr: &Expr) -> bool {
219        contains_grouping_placeholder(expr, self.placeholders())
220    }

222    /// Point computed grouping expressions in a post-aggregation expression at the
223    /// columns holding their grouped values. A key is matched as a whole before its
224    /// children are visited, and aggregate arguments keep reading the original input.
225    pub(crate) fn bind_stored_keys(&self, expr: Expr, key_schema: &Schema) -> Expr {
226        struct Binder<'a> {
227            keys: Vec<(Expr, &'a PlSmallStr)>,
228        }
229        impl RewritingVisitor for Binder<'_> {
230            type Node = Expr;
231            type Arena = ();
232
233            fn pre_visit(&mut self, node: &Expr, _: &mut ()) -> PolarsResult<RewriteRecursion> {
234                Ok(match node {
235                    Expr::Agg(_) | Expr::Len => RewriteRecursion::Stop,
236                    _ if self.keys.iter().any(|(k, _)| k == node) => {
237                        RewriteRecursion::MutateAndStop
238                    },
239                    _ => RewriteRecursion::NoMutateAndContinue,
240                })
241            }
242
243            fn mutate(&mut self, node: Expr, _: &mut ()) -> PolarsResult<Expr> {
244                let (_, name) = self.keys.iter().find(|(k, _)| *k == node).unwrap();
245                Ok(col((*name).clone()))
246            }
247        }
248        let keys = self
249            .stripped_keys
250            .iter()
251            .zip(key_schema.iter_names())
252            .map(|(key, name)| (key.clone(), name))
253            .filter(|(key, _)| !matches!(key, Expr::Column(_)))
254            .collect::<Vec<_>>();
255        if keys.is_empty() {
256            return expr;
257        }
258        expr.rewrite(&mut Binder { keys }, &mut ()).unwrap()
259    }

261    /// The `GROUPING()` bits of a call within one set: an argument the set omits
262    /// contributes a set bit, the last argument being the least significant.
263    fn call_value(set: &[usize], args: &[usize]) -> i64 {
264        args.iter()
265            .fold(0i64, |acc, i| (acc << 1) | i64::from(!set.contains(i)))
266    }
267
268    /// Run the ordinary aggregation once per grouping set over one shared input and
269    /// concatenate the branches onto a common schema of canonical keys (NULL where a
270    /// set omits them), aggregate outputs and `GROUPING()` values.
271    ///
272    /// `group_aggs` run in `group_by().agg()`; `global_aggs` are the same aggregates
273    /// as they run in `select()` for the empty set.
274    pub(crate) fn aggregate(
275        &self,
276        lf: LazyFrame,
277        key_schema: &Schema,
278        group_aggs: &[Expr],
279        global_aggs: &[Expr],
280        agg_names: &[PlSmallStr],
281    ) -> PolarsResult<LazyFrame> {
282        // Compute non-column keys once, before the input is shared.
283        let mut prepared = Vec::new();
284        let group_keys: Vec<Expr> = self
285            .keys
286            .iter()
287            .zip(&self.stripped_keys)
288            .zip(key_schema.iter_names())
289            .map(|((key, inner), name)| {
290                if matches!(inner, Expr::Column(_)) {
291                    key.clone()
292                } else {
293                    let hidden = unique_column_name();
294                    prepared.push(inner.clone().alias(hidden.clone()));
295                    col(hidden).alias(name.clone())
296                }
297            })
298            .collect();
299        let lf = if prepared.is_empty() {
300            lf
301        } else {
302            lf.with_columns(prepared)
303        };
304        let shared = if self.sets.len() > 1 { lf.cache() } else { lf };
305
306        // With neither keys nor aggregates only a row count can carry the height.
307        let need_count = key_schema.is_empty() && agg_names.is_empty();
308
309        let branches = self
310            .sets
311            .iter()
312            .map(|set| {
313                let branch = if set.is_empty() {
314                    let mut aggs = global_aggs.to_vec();
315                    if aggs.is_empty() {
316                        aggs.push(len().alias(GLOBAL_COUNT_COL));
317                    }
318                    shared.clone().select(aggs)
319                } else {
320                    let keys: Vec<Expr> = set.iter().map(|&i| group_keys[i].clone()).collect();
321                    shared.clone().group_by(keys).agg(group_aggs)
322                };
323
324                let mut out: Vec<Expr> =
325                    Vec::with_capacity(key_schema.len() + agg_names.len() + self.calls.len() + 1);
326                for (i, (name, dtype)) in key_schema.iter().enumerate() {
327                    out.push(if set.contains(&i) {
328                        col(name.clone())
329                    } else {
330                        Scalar::null(dtype.clone()).lit().alias(name.clone())
331                    });
332                }
333                out.extend(agg_names.iter().map(|n| col(n.clone())));
334                out.extend(self.calls.iter().map(|(placeholder, args)| {
335                    typed_lit(Self::call_value(set, args)).alias(placeholder.clone())
336                }));
337                if need_count {
338                    out.push(col(GLOBAL_COUNT_COL));
339                }
340                branch.select(out)
341            })
342            .collect::<Vec<_>>();
343
344        if branches.len() == 1 {
345            Ok(branches.into_iter().next().unwrap())
346        } else {
347            concat(
348                branches,
349                UnionArgs {
350                    maintain_order: false,
351                    ..Default::default()
352                },
353            )
354        }
355    }
