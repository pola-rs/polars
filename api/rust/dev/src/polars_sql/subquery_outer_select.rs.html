1//! A subquery whose only SELECT item reads columns of the queries around it, and none
2//! of its own relations, has that value on every row. The item is moved out of the
3//! subquery into the query around it, which can read those columns:
4//!
5//! ```sql
6//! lhs IN (SELECT f FROM r WHERE c)
7//! -- CASE WHEN EXISTS (SELECT 1 FROM r WHERE c) THEN lhs = f ELSE FALSE END
8//! (SELECT f FROM r WHERE c)
9//! -- CASE WHEN (SELECT 1 FROM r WHERE c) IS NOT NULL THEN f END
10//! ```
11//!
12//! An `IN` over no rows is false; over rows that all hold `f` it is `lhs = f`. The
13//! scalar `(SELECT 1 ...)` has no row, one row or more rows exactly when the subquery
14//! has, so it is NULL, `f` or raises in the same cases.

35impl SQLContext {
36    /// `select` with the subqueries of its clauses whose SELECT item reads only columns of
37    /// queries around them rewritten to read those columns here, or `None` if there are none.
38    /// The subqueries of its derived tables are left to those tables.
39    pub(crate) fn hoist_outer_select_items(&self, select: &Select) -> Option<Select> {
40        let has_candidate = visit_expressions(select, |e| match e {
41            SQLExpr::Subquery(_) | SQLExpr::InSubquery { .. } => ControlFlow::Break(()),
42            _ => ControlFlow::Continue(()),
43        });
44        if has_candidate.is_continue() {
45            return None;
46        }
47        struct Hoist<'a> {
48            ctx: &'a SQLContext,
49            query_depth: usize,
50            // Per visited expression: whether it hid its quantified subquery.
51            skips: Vec<bool>,
52            changed: bool,
53        }
54
55        impl VisitorMut for Hoist<'_> {
56            type Break = ();
57
58            fn pre_visit_query(&mut self, _query: &mut Query) -> ControlFlow<()> {
59                self.query_depth += 1;
60                ControlFlow::Continue(())
61            }
62
63            fn post_visit_query(&mut self, _query: &mut Query) -> ControlFlow<()> {
64                self.query_depth -= 1;
65                ControlFlow::Continue(())
66            }
67
68            fn pre_visit_expr(&mut self, expr: &mut SQLExpr) -> ControlFlow<()> {
69                // The subquery of `x > ANY (subquery)` is a set of values, not a scalar.
70                let skip = self.query_depth == 0
71                    && matches!(
72                        expr,
73                        SQLExpr::AnyOp { right, .. } | SQLExpr::AllOp { right, .. }
74                            if matches!(strip_nested(right), SQLExpr::Subquery(_))
75                    );
76                if skip && let SQLExpr::AnyOp { left, .. } | SQLExpr::AllOp { left, .. } = expr {
77                    dyn_visit::visit_mut(left.as_mut(), self)?;
78                    self.query_depth += 1;
79                }
80                self.skips.push(skip);
81                if self.query_depth == 0
82                    && let Some(hoisted) = self.ctx.hoist_outer_select_item(expr)
83                {
84                    *expr = hoisted;
85                    self.changed = true;
86                }
87                ControlFlow::Continue(())
88            }
89
90            fn post_visit_expr(&mut self, _expr: &mut SQLExpr) -> ControlFlow<()> {
91                if self.skips.pop() == Some(true) {
92                    self.query_depth -= 1;
93                }
94                ControlFlow::Continue(())
95            }
96        }
97
98        let mut select = select.clone();
99        let mut hoist = Hoist {
100            ctx: self,
101            query_depth: 0,
102            skips: Vec::new(),
103            changed: false,
104        };
105        let _ = dyn_visit::visit_mut(&mut select, &mut hoist);
106        hoist.changed.then_some(select)
107    }
108
109    /// `expr` rewritten as in the module docs when it is `[NOT] IN` or a scalar subquery
110    /// whose SELECT item reads only columns of queries around it.
111    fn hoist_outer_select_item(&self, expr: &SQLExpr) -> Option<SQLExpr> {
112        match expr {
113            SQLExpr::InSubquery {
114                expr: lhs,
115                subquery,
116                negated,
117            } => {
118                let (item, exists_query) = self.outer_select_item(subquery)?;
119                // The item is only evaluated where the subquery has rows.
120                let in_set = when_then(
121                    SQLExpr::Exists {
122                        subquery: Box::new(exists_query),
123                        negated: false,
124                    },
125                    SQLExpr::BinaryOp {
126                        left: lhs.clone(),
127                        op: SQLBinaryOperator::Eq,
128                        right: Box::new(SQLExpr::Nested(Box::new(item))),
129                    },
130                    Some(SQLExpr::value(SQLValue::Boolean(false))),
131                );
132                Some(match negated {
133                    true => SQLExpr::UnaryOp {
134                        op: SQLUnaryOperator::Not,
135                        expr: Box::new(in_set),
136                    },
137                    false => in_set,
138                })
139            },
140            SQLExpr::Subquery(subquery) => {
141                let (item, row_query) = self.outer_select_item(subquery)?;
142                Some(when_then(
143                    SQLExpr::IsNotNull(Box::new(SQLExpr::Subquery(Box::new(row_query)))),
144                    SQLExpr::Nested(Box::new(item)),
145                    None,
146                ))
147            },
148            _ => None,
149        }
150    }

152    /// The SELECT item of `subquery` if it reads only columns of queries around it, with
153    /// the subquery that selects `1` in its place.
154    fn outer_select_item(&self, subquery: &Query) -> Option<(SQLExpr, Query)> {
155        if subquery.with.is_some() {
156            return None;
157        }
158        let SetExpr::Select(select) = subquery.body.as_ref() else {
159            return None;
160        };
161        let [SelectItem::UnnamedExpr(item) | SelectItem::ExprWithAlias { expr: item, .. }] =
162            select.projection.as_slice()
163        else {
164            return None;
165        };
166        if !expr_references_any_column(item)
167            || expr_contains_subquery(item)
168            || expr_has_window_functions(item)
169            || has_aggregate(self, item)
170            || has_column_selector(item)
171            || calls_query_function(self, item)
172        {
173            return None;
174        }
175        // A grouping set null-extends the item on some rows.
176        let has_grouping_sets = match &select.group_by {
177            GroupByExpr::All(modifiers) => !modifiers.is_empty(),
178            GroupByExpr::Expressions(exprs, modifiers) => {
179                !modifiers.is_empty()
180                    || exprs.iter().any(|e| {
181                        matches!(
182                            strip_nested(e),
183                            SQLExpr::Rollup(_) | SQLExpr::Cube(_) | SQLExpr::GroupingSets(_)
184                        )
185                    })
186            },
187        };
188        if has_grouping_sets {
189            return None;
190        }
191
192        let (inner_names, inner_schema) = if select.from.is_empty() {
193            // Without FROM only a plain `SELECT f` is moved: it has exactly one row.
194            let filters_rows = select.selection.is_some()
195                || select.having.is_some()
196                || select.qualify.is_some()
197                || subquery.limit_clause.is_some()
198                || subquery.fetch.is_some();
199            if filters_rows {
200                return None;
201            }
202            (Vec::new(), SchemaRef::default())
203        } else {
204            // An error leaves the subquery to fail as it did.
205            let (names, _, schema) = self.isolated().resolve_subquery_from(select).ok()??;
206            (names, schema)
207        };
208        if reads_inner_column(item, &inner_names, &inner_schema) {
209            return None;
210        }
211
212        let mut row_query = subquery.clone();
213        let SetExpr::Select(row_select) = row_query.body.as_mut() else {
214            unreachable!()
215        };
216        let (SelectItem::UnnamedExpr(slot) | SelectItem::ExprWithAlias { expr: slot, .. }) =
217            &mut row_select.projection[0]
218        else {
219            unreachable!()
220        };
221        *slot = SQLExpr::value(SQLValue::Number("1".into(), false));
222        // Another clause reading the alias would read `1` instead.
223        if let SelectItem::ExprWithAlias { alias, .. } = &select.projection[0]
224            && visit_expressions(&row_query, |e| match e {
225                SQLExpr::Identifier(ident) if ident.value.eq_ignore_ascii_case(&alias.value) => {
226                    ControlFlow::Break(())
227                },
228                _ => ControlFlow::Continue(()),
229            })
230            .is_break()
231        {
232            return None;
233        }
234        Some((item.clone(), row_query))
235    }

238pub(crate) fn when_then(
239    condition: SQLExpr,
240    result: SQLExpr,
241    else_result: Option<SQLExpr>,
242) -> SQLExpr {
243    SQLExpr::Case {
244        case_token: AttachedToken::empty(),
245        end_token: AttachedToken::empty(),
246        operand: None,
247        conditions: vec![CaseWhen { condition, result }],
248        else_result: else_result.map(Box::new),
249    }
250}

252/// Whether `expr` calls a function whose value depends on the query it is in: `UNNEST`,
253/// which can return several rows for one, `GROUPING` / `GROUPING_ID`, or a ranking or
254/// offset function, which may be called without OVER.
255pub(crate) fn calls_query_function(ctx: &SQLContext, expr: &SQLExpr) -> bool {
256    visit_expressions(expr, |e| match e {
257        SQLExpr::Function(function)
258            if matches!(
259                PolarsSQLFunctions::try_from_sql(function, ctx),
260                Ok(PolarsSQLFunctions::Explode
261                    | PolarsSQLFunctions::Grouping
262                    | PolarsSQLFunctions::GroupingId
263                    | PolarsSQLFunctions::RowNumber
264                    | PolarsSQLFunctions::Rank
265                    | PolarsSQLFunctions::DenseRank
266                    | PolarsSQLFunctions::PercentRank
267                    | PolarsSQLFunctions::CumeDist
268                    | PolarsSQLFunctions::Ntile
269                    | PolarsSQLFunctions::Lag
270                    | PolarsSQLFunctions::Lead)
271            ) =>
272        {
273            ControlFlow::Break(())
274        },
275        _ => ControlFlow::Continue(()),
276    })
277    .is_break()
278}

280/// Whether `expr` reads a column of the relations `inner_names`, whose columns are
281/// `inner_schema`. An unqualified name binds to them when they have it; a qualified one
282/// when its qualifier names one of them or its first part is one of their (struct)
283/// columns.
284pub(crate) fn reads_inner_column(
285    expr: &SQLExpr,
286    inner_names: &[RelationName],
287    inner_schema: &Schema,
288) -> bool {
289    visit_expressions(expr, |e| {
290        let inner = match e {
291            SQLExpr::Identifier(ident) => inner_schema.contains(&ident.value),
292            SQLExpr::CompoundIdentifier(parts) => {
293                match_qualifier(parts, inner_names.iter()).is_some()
294                    || inner_schema.contains(&parts[0].value)
295            },
296            _ => false,
297        };
298        if inner {
299            ControlFlow::Break(())
300        } else {
301            ControlFlow::Continue(())
302        }
303    })
304    .is_break()
