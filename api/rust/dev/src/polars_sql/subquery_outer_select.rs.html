1//! A subquery whose only SELECT item reads columns of the queries around it, and none
2//! of its own relations, has that value on every row. The item is moved out of the
3//! subquery into the query around it, which can read those columns:
4//!
5//! ```sql
6//! lhs IN (SELECT f FROM r WHERE c)
7//! -- CASE WHEN EXISTS (SELECT 1 FROM r WHERE c) THEN lhs = f ELSE FALSE END
8//! (SELECT f FROM r WHERE c)
9//! -- CASE WHEN (SELECT 1 FROM r WHERE c) IS NOT NULL THEN f END
10//! ```
11//!
12//! An `IN` over no rows is false; over rows that all hold `f` it is `lhs = f`. The
13//! scalar `(SELECT 1 ...)` has no row, one row or more rows exactly when the subquery
14//! has, so it is NULL, `f` or raises in the same cases.
15//!
16//! A scalar subquery whose only FROM item is a derived table that reads columns of the
17//! queries around it is first merged with that table. It reads the table's FROM and
18//! WHERE itself, with the table's SELECT items in place of its columns:
19//!
20//! ```sql
21//! (SELECT MAX(d.v) FROM (SELECT x AS v FROM r WHERE r.k = o.k) AS d WHERE c)
22//! -- (SELECT MAX(x) FROM r WHERE r.k = o.k AND c')
23//! ```

46impl SQLContext {
47    /// `select` with the subqueries of its clauses whose SELECT item reads only columns of
48    /// queries around them rewritten to read those columns here, and its scalar subqueries
49    /// merged with a derived table as in the module docs, or `None` if there are none. The
50    /// subqueries of its derived tables are left to those tables.
51    pub(crate) fn hoist_outer_select_items(&self, select: &Select) -> Option<Select> {
52        let has_candidate = visit_expressions(select, |e| match e {
53            SQLExpr::Subquery(_) | SQLExpr::InSubquery { .. } => ControlFlow::Break(()),
54            _ => ControlFlow::Continue(()),
55        });
56        if has_candidate.is_continue() {
57            return None;
58        }
59        struct Hoist<'a> {
60            ctx: &'a SQLContext,
61            query_depth: usize,
62            // Per visited expression: whether it hid its quantified subquery.
63            skips: Vec<bool>,
64            changed: bool,
65        }
66
67        impl VisitorMut for Hoist<'_> {
68            type Break = ();
69
70            fn pre_visit_query(&mut self, _query: &mut Query) -> ControlFlow<()> {
71                self.query_depth += 1;
72                ControlFlow::Continue(())
73            }
74
75            fn post_visit_query(&mut self, _query: &mut Query) -> ControlFlow<()> {
76                self.query_depth -= 1;
77                ControlFlow::Continue(())
78            }
79
80            fn pre_visit_expr(&mut self, expr: &mut SQLExpr) -> ControlFlow<()> {
81                // The subquery of `x > ANY (subquery)` is a set of values, not a scalar.
82                let skip = self.query_depth == 0
83                    && matches!(
84                        expr,
85                        SQLExpr::AnyOp { right, .. } | SQLExpr::AllOp { right, .. }
86                            if matches!(strip_nested(right), SQLExpr::Subquery(_))
87                    );
88                if skip && let SQLExpr::AnyOp { left, .. } | SQLExpr::AllOp { left, .. } = expr {
89                    dyn_visit::visit_mut(left.as_mut(), self)?;
90                    self.query_depth += 1;
91                }
92                self.skips.push(skip);
93                if self.query_depth == 0
94                    && let SQLExpr::Subquery(subquery) = expr
95                    && let Some(merged) = self.ctx.merge_outer_derived_table(subquery)
96                {
97                    **subquery = merged;
98                    self.changed = true;
99                }
100                if self.query_depth == 0
101                    && let Some(hoisted) = self.ctx.hoist_outer_select_item(expr)
102                {
103                    *expr = hoisted;
104                    self.changed = true;
105                }
106                ControlFlow::Continue(())
107            }
108
109            fn post_visit_expr(&mut self, _expr: &mut SQLExpr) -> ControlFlow<()> {
110                if self.skips.pop() == Some(true) {
111                    self.query_depth -= 1;
112                }
113                ControlFlow::Continue(())
114            }
115
116            // The expressions of a PIVOT read its input, not the relations of this select.
117            fn pre_visit_table_factor(&mut self, factor: &mut TableFactor) -> ControlFlow<()> {
118                if matches!(factor, TableFactor::Pivot { .. }) {
119                    self.query_depth += 1;
120                }
121                ControlFlow::Continue(())
122            }
123
124            fn post_visit_table_factor(&mut self, factor: &mut TableFactor) -> ControlFlow<()> {
125                if matches!(factor, TableFactor::Pivot { .. }) {
126                    self.query_depth -= 1;
127                }
128                ControlFlow::Continue(())
129            }
130        }
131
132        let mut select = select.clone();
133        let mut hoist = Hoist {
134            ctx: self,
135            query_depth: 0,
136            skips: Vec::new(),
137            changed: false,
138        };
139        let _ = dyn_visit::visit_mut(&mut select, &mut hoist);
140        hoist.changed.then_some(select)
141    }

143    /// `expr` rewritten as in the module docs when it is `[NOT] IN` or a scalar subquery
144    /// whose SELECT item reads only columns of queries around it.
145    fn hoist_outer_select_item(&self, expr: &SQLExpr) -> Option<SQLExpr> {
146        match expr {
147            SQLExpr::InSubquery {
148                expr: lhs,
149                subquery,
150                negated,
151            } => {
152                let (item, exists_query) = self.outer_select_item(subquery)?;
153                // The item is only evaluated where the subquery has rows.
154                let in_set = when_then(
155                    SQLExpr::Exists {
156                        subquery: Box::new(exists_query),
157                        negated: false,
158                    },
159                    SQLExpr::BinaryOp {
160                        left: lhs.clone(),
161                        op: SQLBinaryOperator::Eq,
162                        right: Box::new(SQLExpr::Nested(Box::new(item))),
163                    },
164                    Some(SQLExpr::value(SQLValue::Boolean(false))),
165                );
166                Some(match negated {
167                    true => SQLExpr::UnaryOp {
168                        op: SQLUnaryOperator::Not,
169                        expr: Box::new(in_set),
170                    },
171                    false => in_set,
172                })
173            },
174            SQLExpr::Subquery(subquery) => {
175                let (item, row_query) = self.outer_select_item(subquery)?;
176                Some(when_then(
177                    SQLExpr::IsNotNull(Box::new(SQLExpr::Subquery(Box::new(row_query)))),
178                    SQLExpr::Nested(Box::new(item)),
179                    None,
180                ))
181            },
182            _ => None,
183        }
184    }

186    /// The SELECT item of `subquery` if it reads only columns of queries around it, with
187    /// the subquery that selects `1` in its place.
188    fn outer_select_item(&self, subquery: &Query) -> Option<(SQLExpr, Query)> {
189        if subquery.with.is_some() {
190            return None;
191        }
192        let SetExpr::Select(select) = subquery.body.as_ref() else {
193            return None;
194        };
195        let [SelectItem::UnnamedExpr(item) | SelectItem::ExprWithAlias { expr: item, .. }] =
196            select.projection.as_slice()
197        else {
198            return None;
199        };
200        if !expr_references_any_column(item)
201            || expr_contains_subquery(item)
202            || expr_has_window_functions(item)
203            || has_aggregate(self, item)
204            || has_column_selector(item)
205            || calls_query_function(self, item)
206        {
207            return None;
208        }
209        // A grouping set null-extends the item on some rows.
210        let has_grouping_sets = match &select.group_by {
211            GroupByExpr::All(modifiers) => !modifiers.is_empty(),
212            GroupByExpr::Expressions(exprs, modifiers) => {
213                !modifiers.is_empty()
214                    || exprs.iter().any(|e| {
215                        matches!(
216                            strip_nested(e),
217                            SQLExpr::Rollup(_) | SQLExpr::Cube(_) | SQLExpr::GroupingSets(_)
218                        )
219                    })
220            },
221        };
222        if has_grouping_sets {
223            return None;
224        }
225
226        let (inner_names, inner_schema) = if select.from.is_empty() {
227            // Without FROM only a plain `SELECT f` is moved: it has exactly one row.
228            let filters_rows = select.selection.is_some()
229                || select.having.is_some()
230                || select.qualify.is_some()
231                || subquery.limit_clause.is_some()
232                || subquery.fetch.is_some();
233            if filters_rows {
234                return None;
235            }
236            (Vec::new(), SchemaRef::default())
237        } else {
238            // An error leaves the subquery to fail as it did.
239            let (names, _, schema) = self.isolated().resolve_subquery_from(select).ok()??;
240            (names, schema)
241        };
242        if reads_inner_column(item, &inner_names, &inner_schema) {
243            return None;
244        }
245
246        let mut row_query = subquery.clone();
247        let SetExpr::Select(row_select) = row_query.body.as_mut() else {
248            unreachable!()
249        };
250        let (SelectItem::UnnamedExpr(slot) | SelectItem::ExprWithAlias { expr: slot, .. }) =
251            &mut row_select.projection[0]
252        else {
253            unreachable!()
254        };
255        *slot = SQLExpr::value(SQLValue::Number("1".into(), false));
256        // Another clause reading the alias would read `1` instead.
257        if let SelectItem::ExprWithAlias { alias, .. } = &select.projection[0]
258            && visit_expressions(&row_query, |e| match e {
259                SQLExpr::Identifier(ident) if ident.value.eq_ignore_ascii_case(&alias.value) => {
260                    ControlFlow::Break(())
261                },
262                _ => ControlFlow::Continue(()),
263            })
264            .is_break()
265        {
266            return None;
267        }
268        Some((item.clone(), row_query))
269    }

271    /// `subquery` merged with its only FROM item as in the module docs, or `None` if that
272    /// item is not a derived table that reads a relation of a query further out, or either
273    /// of them has another shape. The subquery may only read the derived table's columns,
274    /// and each of those must be a plain expression of a column.
275    fn merge_outer_derived_table(&self, subquery: &Query) -> Option<Query> {
276        let select = eligible_query_block(subquery)?;
277        let [
278            TableWithJoins {
279                relation:
280                    TableFactor::Derived {
281                        subquery: derived,
282                        alias,
283                        sample: None,
284                        ..
285                    },
286                joins,
287            },
288        ] = select.from.as_slice()
289        else {
290            return None;
291        };
292        // ORDER BY and DISTINCT ON may read a SELECT item rather than a column.
293        if !joins.is_empty()
294            || subquery.order_by.is_some()
295            || matches!(select.distinct, Some(Distinct::On(_)))
296            || select.exclude.is_some()
297            || !select.named_window.is_empty()
298            || alias.as_ref().is_some_and(|a| !a.columns.is_empty())
299            || contains_random(subquery, true)
300            || has_unchecked_clauses(subquery, select)
301            || !select.projection.iter().all(|item| {
302                matches!(
303                    item,
304                    SelectItem::UnnamedExpr(_) | SelectItem::ExprWithAlias { .. }
305                )
306            })
307        {
308            return None;
309        }
310        let derived_select = eligible_query_block(derived)?;
311        if derived.order_by.is_some()
312            || derived.limit_clause.is_some()
313            || derived.fetch.is_some()
314            || derived_select.distinct.is_some()
315            || derived_select.exclude.is_some()
316            || !derived_select.named_window.is_empty()
317            || has_unchecked_clauses(derived, derived_select)
318        {
319            return None;
320        }
321        let mut items: Vec<(&str, &SQLExpr)> = Vec::new();
322        let mut aliases: Vec<&str> = Vec::new();
323        for item in &derived_select.projection {
324            let (name, expr) = match item {
325                SelectItem::ExprWithAlias { expr, alias } => {
326                    aliases.push(alias.value.as_str());
327                    (alias.value.as_str(), expr)
328                },
329                SelectItem::UnnamedExpr(expr @ SQLExpr::Identifier(ident)) => {
330                    (ident.value.as_str(), expr)
331                },
332                SelectItem::UnnamedExpr(expr @ SQLExpr::CompoundIdentifier(parts)) => {
333                    (parts.last()?.value.as_str(), expr)
334                },
335                _ => return None,
336            };
337            // A column's type, not a literal's, decides the type of the item.
338            if items.iter().any(|(n, _)| *n == name)
339                || !is_plain_column_expr(expr)
340                || !expr_references_any_column(expr)
341            {
342                return None;
343            }
344            items.push((name, expr));
345        }
346        // An item may read the alias of another one, which is not kept.
347        let reads_alias = items.iter().any(|(own_name, expr)| {
348            visit_expressions(*expr, |e| {
349                let name = match e {
350                    SQLExpr::Identifier(ident) => ident,
351                    SQLExpr::CompoundIdentifier(parts) => &parts[0],
352                    _ => return ControlFlow::Continue(()),
353                };
354                if name.value != *own_name && aliases.contains(&name.value.as_str()) {
355                    ControlFlow::Break(())
356                } else {
357                    ControlFlow::Continue(())
358                }
359            })
360            .is_break()
361        });
362        if reads_alias {
363            return None;
364        }
365
366        // Only a derived table that reads a relation it does not declare, rather than a
367        // field of one of its struct columns, could not be planned on its own.
368        if derived_select
369            .selection
370            .as_ref()
371            .is_some_and(expr_contains_subquery)
372        {
373            return None;
374        }
375        let (names, schema) = if derived_select.from.is_empty() {
376            (Vec::new(), SchemaRef::default())
377        } else {
378            let mut from_only = derived_select.clone();
379            from_only.projection.clear();
380            from_only.selection = None;
381            let (names, _, schema) = self.isolated().resolve_subquery_from(&from_only).ok()??;
382            (names, schema)
383        };
384        let reads_outer = |expr: &SQLExpr| {
385            visit_expressions(expr, |e| match e {
386                SQLExpr::CompoundIdentifier(parts)
387                    if match_qualifier(parts, names.iter()).is_none()
388                        && !schema.contains(&parts[0].value) =>
389                {
390                    ControlFlow::Break(())
391                },
392                _ => ControlFlow::Continue(()),
393            })
394            .is_break()
395        };
396        if !items.iter().any(|(_, expr)| reads_outer(expr))
397            && !derived_select.selection.as_ref().is_some_and(reads_outer)
398        {
399            return None;
400        }
401
402        // The clauses of the query block read the derived table, and LIMIT is left as is.
403        let mut merged = subquery.clone();
404        let SetExpr::Select(merged_select) = merged.body.as_mut() else {
405            unreachable!()
406        };
407        merged_select.from.clear();
408        let alias = alias
409            .as_ref()
410            .map(|a| RelationName::new(vec![a.name.value.clone()]));
411        let mut read_items = ReadItems {
412            items: &items,
413            alias: alias.as_ref(),
414            read: vec![false; items.len()],
415        };
416        // An item that is not read would not be checked.
417        if dyn_visit::visit_mut(merged_select.as_mut(), &mut read_items).is_break()
418            || read_items.read.contains(&false)
419        {
420            return None;
421        }
422        // An aggregate of an item that reads no column of the derived table's relations
423        // would aggregate a query further out.
424        if has_aggregate(self, merged_select.as_ref())
425            && items
426                .iter()
427                .any(|(_, expr)| !reads_inner_column(expr, &names, &schema))
428        {
429            return None;
430        }
431        merged_select.from = derived_select.from.clone();
432        merged_select.selection = combine_conditions(
433            derived_select
434                .selection
435                .iter()
436                .cloned()
437                .chain(merged_select.selection.take())
438                .collect(),
439            SQLBinaryOperator::And,
440        );
441        Some(merged)
442    }
443}

445pub(crate) fn when_then(
446    condition: SQLExpr,
447    result: SQLExpr,
448    else_result: Option<SQLExpr>,
449) -> SQLExpr {
450    SQLExpr::Case {
451        case_token: AttachedToken::empty(),
452        end_token: AttachedToken::empty(),
453        operand: None,
454        conditions: vec![CaseWhen { condition, result }],
455        else_result: else_result.map(Box::new),
456    }
457}

459/// Whether `expr` calls a function whose value depends on the query it is in: `UNNEST`,
460/// which can return several rows for one, `GROUPING` / `GROUPING_ID`, or a ranking or
461/// offset function, which may be called without OVER.
462pub(crate) fn calls_query_function(ctx: &SQLContext, expr: &SQLExpr) -> bool {
463    visit_expressions(expr, |e| match e {
464        SQLExpr::Function(function)
465            if matches!(
466                PolarsSQLFunctions::try_from_sql(function, ctx),
467                Ok(PolarsSQLFunctions::Explode
468                    | PolarsSQLFunctions::Grouping
469                    | PolarsSQLFunctions::GroupingId
470                    | PolarsSQLFunctions::RowNumber
471                    | PolarsSQLFunctions::Rank
472                    | PolarsSQLFunctions::DenseRank
473                    | PolarsSQLFunctions::PercentRank
474                    | PolarsSQLFunctions::CumeDist
475                    | PolarsSQLFunctions::Ntile
476                    | PolarsSQLFunctions::Lag
477                    | PolarsSQLFunctions::Lead)
478            ) =>
479        {
480            ControlFlow::Break(())
481        },
482        _ => ControlFlow::Continue(()),
483    })
484    .is_break()
485}

487/// Whether `expr` is built from columns and literals by operators and casts only.
488fn is_plain_column_expr(expr: &SQLExpr) -> bool {
489    match expr {
490        SQLExpr::Identifier(_) | SQLExpr::CompoundIdentifier(_) | SQLExpr::Value(_) => true,
491        SQLExpr::Nested(expr)
492        | SQLExpr::UnaryOp { expr, .. }
493        | SQLExpr::Cast { expr, .. }
494        | SQLExpr::IsNull(expr)
495        | SQLExpr::IsNotNull(expr) => is_plain_column_expr(expr),
496        SQLExpr::BinaryOp { left, right, .. } => {
497            is_plain_column_expr(left) && is_plain_column_expr(right)
498        },
499        _ => false,
500    }
501}

503/// Whether `query` has a clause that is not checked once `select` has been decorrelated.
504fn has_unchecked_clauses(query: &Query, select: &Select) -> bool {
505    !query.locks.is_empty()
506        || query.format_clause.is_some()
507        || !select.cluster_by.is_empty()
508        || !select.distribute_by.is_empty()
509        || !select.sort_by.is_empty()
510}

512/// Replaces the columns of a derived table in the query that reads it by the items that
513/// define them, and marks in `read` which items. Breaks on any other name, a column
514/// selector, a window function, a nested query or a lambda.
515struct ReadItems<'a> {
516    items: &'a [(&'a str, &'a SQLExpr)],
517    alias: Option<&'a RelationName>,
518    read: Vec<bool>,
519}
520
521impl VisitorMut for ReadItems<'_> {
522    type Break = ();
523
524    fn pre_visit_query(&mut self, _query: &mut Query) -> ControlFlow<()> {
525        ControlFlow::Break(())
526    }
527
528    fn pre_visit_expr(&mut self, expr: &mut SQLExpr) -> ControlFlow<()> {
529        let window = matches!(expr, SQLExpr::Function(f) if f.over.is_some());
530        if window
531            || matches!(expr, SQLExpr::Lambda(_))
532            || (is_column_selector(expr) && !is_count_star(expr))
533        {
534            return ControlFlow::Break(());
535        }
536        ControlFlow::Continue(())
537    }
538
539    fn post_visit_expr(&mut self, expr: &mut SQLExpr) -> ControlFlow<()> {
540        let column = match expr {
541            SQLExpr::Identifier(column) => column,
542            SQLExpr::CompoundIdentifier(parts) => match parts.as_slice() {
543                [relation, column]
544                    if self
545                        .alias
546                        .is_some_and(|a| a.named_by(std::slice::from_ref(relation), true)) =>
547                {
548                    column
549                },
550                _ => return ControlFlow::Break(()),
551            },
552            _ => return ControlFlow::Continue(()),
553        };
554        let Some(index) = self.items.iter().position(|(n, _)| *n == column.value) else {
555            return ControlFlow::Break(());
556        };
557        let item = self.items[index].1;
558        *expr = match item {
559            SQLExpr::Identifier(_) | SQLExpr::CompoundIdentifier(_) => item.clone(),
560            _ => SQLExpr::Nested(Box::new(item.clone())),
561        };
562        self.read[index] = true;
563        ControlFlow::Continue(())
564    }
565}

567/// Whether `expr` reads a column of the relations `inner_names`, whose columns are
568/// `inner_schema`. An unqualified name binds to them when they have it; a qualified one
569/// when its qualifier names one of them or its first part is one of their (struct)
570/// columns.
571pub(crate) fn reads_inner_column(
572    expr: &SQLExpr,
573    inner_names: &[RelationName],
574    inner_schema: &Schema,
575) -> bool {
576    visit_expressions(expr, |e| {
577        let inner = match e {
578            SQLExpr::Identifier(ident) => inner_schema.contains(&ident.value),
579            SQLExpr::CompoundIdentifier(parts) => {
580                match_qualifier(parts, inner_names.iter()).is_some()
581                    || inner_schema.contains(&parts[0].value)
582            },
583            _ => false,
584        };
585        if inner {
586            ControlFlow::Break(())
587        } else {
588            ControlFlow::Continue(())
589        }
590    })
591    .is_break()
