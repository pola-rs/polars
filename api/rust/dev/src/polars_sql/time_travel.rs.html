1//! Time travel on catalog tables: `[FOR] VERSION AS OF x`, `[FOR] TIMESTAMP AS OF ts`,
2//! `FOR SYSTEM_TIME AS OF ts` and `AT (VERSION|TIMESTAMP => x)`.
3//!
4//! sqlparser's generic dialect parses none of them, so [`extract_table_versions`] takes the
5//! clause out of the tokens and [`TableVersions::attach`] puts it on the table factor it
6//! follows. A clause that no table factor claims is an error, so it is never lost.
7//! Temporary: replace this pass once a sqlparser release parses these forms.

24/// The time travel clauses taken out of a query, by the position (line, column) of the table
25/// name each follows. Each must be attached to a statement, or refused.
26#[must_use]
27pub(crate) struct TableVersions(PlHashMap<(u64, u64), TableVersion>);
28
29impl TableVersions {
30    /// Put each clause on the table factor whose name starts at its position.
31    pub(crate) fn attach(mut self, stmts: &mut [Statement]) -> PolarsResult<()> {
32        if self.0.is_empty() {
33            return Ok(());
34        }
35        let mut attach = Attach(&mut self.0);
36        for stmt in stmts {
37            if let ControlFlow::Break(err) = dyn_visit::visit_mut(stmt, &mut attach) {
38                return Err(err);
39            }
40        }
41        self.refuse()
42    }
43
44    /// An error if a clause was taken out where no table factor can take it.
45    pub(crate) fn refuse(self) -> PolarsResult<()> {
46        match self.0.iter().min_by_key(|(at, _)| **at) {
47            None => Ok(()),
48            Some((_, version)) => polars_bail!(
49                SQLInterface: "`{}` must follow the name of a table", version
50            ),
51        }
52    }
53}
54
55struct Attach<'a>(&'a mut PlHashMap<(u64, u64), TableVersion>);
56
57impl VisitorMut for Attach<'_> {
58    type Break = PolarsError;
59
60    fn pre_visit_table_factor(&mut self, factor: &mut TableFactor) -> ControlFlow<PolarsError> {
61        let TableFactor::Table { name, version, .. } = factor else {
62            return ControlFlow::Continue(());
63        };
64        if let Some(found) = name_start(name).and_then(|at| self.0.remove(&at)) {
65            if version.is_some() {
66                return ControlFlow::Break(polars_err!(
67                    SQLInterface: "table '{}' takes one time travel clause", name
68                ));
69            }
70            *version = Some(found);
71        }
72        ControlFlow::Continue(())
73    }
74}

76fn name_start(name: &ObjectName) -> Option<(u64, u64)> {
77    let start = name.0.first()?.as_ident()?.span.start;
78    Some((start.line, start.column))
79}

81/// Take the time travel clauses out of `tokens` (see the module docs), in one pass.
82pub(crate) fn extract_table_versions(
83    tokens: &mut Vec<TokenWithSpan>,
84) -> PolarsResult<TableVersions> {
85    let input = std::mem::take(tokens);
86    let mut found = PlHashMap::new();
87    // Per output token: whether a comma or `(` starts a FROM item, and for a `)` the index of
88    // its `(`.
89    let mut starts_item = Vec::with_capacity(input.len());
90    let mut opened_at = Vec::with_capacity(input.len());
91    let mut open = Vec::new();
92    // Whether the tokens are in a FROM list, per parenthesis depth.
93    let mut in_from = vec![false];
94    let mut i = 0;
95    while i < input.len() {
96        if let Some((kind, at_form, value_start)) = clause_head(&input, i)
97            && let Some(name) = table_name_before(tokens, &starts_item, &opened_at)
98        {
99            let (value, end) = clause_value(&input, value_start, at_form)?;
100            let version = match kind {
101                Kind::Version => TableVersion::VersionAsOf(value),
102                Kind::Timestamp => TableVersion::TimestampAsOf(value),
103            };
104            let at = tokens[name].span.start;
105            found.insert((at.line, at.column), version);
106            i = end;
107            continue;
108        }
109        let token = &input[i].token;
110        let from = in_from.last().copied().unwrap_or(false);
111        starts_item.push(from && matches!(token, Token::Comma | Token::LParen));
112        opened_at.push(usize::MAX);
113        match token {
114            Token::LParen => {
115                open.push(tokens.len());
116                in_from.push(from);
117            },
118            Token::RParen => {
119                if let Some(at) = open.pop() {
120                    *opened_at.last_mut().unwrap() = at;
121                }
122                if in_from.len() > 1 {
123                    in_from.pop();
124                }
125            },
126            // a keyword after `AS` is an alias
127            Token::Word(w)
128                if w.quote_style.is_none()
129                    && !prev(tokens, tokens.len())
130                        .is_some_and(|j| is_keyword(&tokens[j].token, Keyword::AS)) =>
131            {
132                use Keyword::*;
133                // clauses that end a FROM list, and can't appear inside an expression
134                let ends = [
135                    SELECT, WHERE, GROUP, HAVING, ORDER, LIMIT, OFFSET, UNION, EXCEPT, INTERSECT,
136                    QUALIFY, WINDOW, RETURNING,
137                ];
138                if let Some(state) = in_from.last_mut() {
139                    if matches!(w.keyword, FROM | JOIN | USING) {
140                        *state = true;
141                    } else if ends.contains(&w.keyword) {
142                        *state = false;
143                    }
144                }
145            },
146            _ => {},
147        }
148        tokens.push(input[i].clone());
149        i += 1;
150    }
151    Ok(TableVersions(found))
152}

154/// The kind of value a time travel clause selects by.
155#[derive(Clone, Copy)]
156enum Kind {
157    Version,
158    Timestamp,
159}
160
161fn is_keyword(token: &Token, keyword: Keyword) -> bool {
162    matches!(token, Token::Word(w) if w.keyword == keyword && w.quote_style.is_none())
163}
164
165fn is_significant(token: &Token) -> bool {
166    !matches!(token, Token::Whitespace(_))
167}
168
169/// The index of the significant token after `i`.
170fn next(tokens: &[TokenWithSpan], i: usize) -> Option<usize> {
171    (i + 1..tokens.len()).find(|&j| is_significant(&tokens[j].token))
172}
173
174/// The index of the significant token before `i`.
175fn prev(tokens: &[TokenWithSpan], i: usize) -> Option<usize> {
176    (0..i).rev().find(|&j| is_significant(&tokens[j].token))
177}
178
179/// Whether the significant tokens from `i` are `keywords`; the index after the last.
180fn keywords_at(tokens: &[TokenWithSpan], i: usize, keywords: &[Keyword]) -> Option<usize> {
181    let mut at = Some(i);
182    let mut last = i;
183    for keyword in keywords {
184        let j = at?;
185        if !is_keyword(&tokens[j].token, *keyword) {
186            return None;
187        }
188        last = j;
189        at = next(tokens, j);
190    }
191    Some(last + 1)
192}

194/// A clause head at `i`: its kind, whether it is the `AT (... => ...)` form, and where its
195/// value starts.
196fn clause_head(tokens: &[TokenWithSpan], i: usize) -> Option<(Kind, bool, usize)> {
197    use Keyword::{AS, AT, FOR, OF, SYSTEM_TIME, TIMESTAMP, VERSION};
198    for (kind, keyword) in [(Kind::Version, VERSION), (Kind::Timestamp, TIMESTAMP)] {
199        if let Some(end) = keywords_at(tokens, i, &[FOR, keyword, AS, OF])
200            .or_else(|| keywords_at(tokens, i, &[keyword, AS, OF]))
201        {
202            return Some((kind, false, end));
203        }
204    }
205    if let Some(end) = keywords_at(tokens, i, &[FOR, SYSTEM_TIME, AS, OF]) {
206        return Some((Kind::Timestamp, false, end));
207    }
208    if is_keyword(&tokens[i].token, AT) {
209        let open = next(tokens, i)?;
210        let word = next(tokens, open)?;
211        let arrow = next(tokens, word)?;
212        if tokens[open].token == Token::LParen && tokens[arrow].token == Token::RArrow {
213            let kind = match &tokens[word].token {
214                t if is_keyword(t, VERSION) => Kind::Version,
215                t if is_keyword(t, TIMESTAMP) => Kind::Timestamp,
216                _ => return None,
217            };
218            return Some((kind, true, arrow + 1));
219        }
220    }
221    None
222}

224/// A word that can be part of a table name or be its alias without `AS`.
225fn is_name_word(token: &Token) -> bool {
226    match token {
227        Token::Word(w) if w.quote_style.is_some() => true,
228        Token::Word(w) => {
229            !RESERVED_FOR_TABLE_ALIAS.contains(&w.keyword)
230                && !matches!(w.keyword, Keyword::FROM | Keyword::AS)
231        },
232        _ => false,
233    }
234}

236/// The first token of the dotted name ending at `end`, if a FROM item starts there: after
237/// `FROM`, `JOIN` or `USING`, or a comma or the parenthesis of a grouped relation in a FROM
238/// list (`starts_item`). A part next to a `.` may be any word.
239fn table_name_ending_at(
240    tokens: &[TokenWithSpan],
241    starts_item: &[bool],
242    end: usize,
243) -> Option<usize> {
244    let is_dot = |j: Option<usize>| j.is_some_and(|j| tokens[j].token == Token::Period);
245    let mut start = end;
246    let qualified = is_dot(prev(tokens, end));
247    let is_part = |j: usize, qualified: bool| match &tokens[j].token {
248        Token::Word(_) if qualified => true,
249        token => is_name_word(token),
250    };
251    if !is_part(end, qualified) {
252        return None;
253    }
254    while let Some(dot) = prev(tokens, start)
255        && tokens[dot].token == Token::Period
256    {
257        let part = prev(tokens, dot)?;
258        if !is_part(part, true) {
259            return None;
260        }
261        start = part;
262    }
263    let before = prev(tokens, start)?;
264    let starts = starts_item[before]
265        || [Keyword::FROM, Keyword::JOIN, Keyword::USING]
266            .iter()
267            .any(|k| is_keyword(&tokens[before].token, *k));
268    starts.then_some(start)
269}

271/// The first token of the table name that a clause after the last of `tokens` follows: right
272/// after the name, or after its `[AS] alias [(columns)]`.
273fn table_name_before(
274    tokens: &[TokenWithSpan],
275    starts_item: &[bool],
276    opened_at: &[usize],
277) -> Option<usize> {
278    let last = prev(tokens, tokens.len())?;
279    if let Some(name) = table_name_ending_at(tokens, starts_item, last) {
280        return Some(name);
281    }
282    let mut alias = last;
283    if tokens[alias].token == Token::RParen {
284        let open = *opened_at.get(alias).filter(|&&at| at != usize::MAX)?;
285        alias = prev(tokens, open)?;
286    }
287    let mut end = prev(tokens, alias)?;
288    // after an explicit `AS`, any word is an alias
289    if is_keyword(&tokens[end].token, Keyword::AS) {
290        if !matches!(tokens[alias].token, Token::Word(_)) {
291            return None;
292        }
293        end = prev(tokens, end)?;
294    } else if !is_name_word(&tokens[alias].token) {
295        return None;
296    }
297    table_name_ending_at(tokens, starts_item, end)
298}

300/// The literal value of a clause starting at `start`, and the index after the clause.
301fn clause_value(
302    tokens: &[TokenWithSpan],
303    start: usize,
304    at_form: bool,
305) -> PolarsResult<(SQLExpr, usize)> {
306    // a literal is a few tokens: parse from a slice of 16 significant ones, so the work stays
307    // linear, and only a prefix expression, so a following alias is not taken as an operator
308    let stop = (start..tokens.len())
309        .filter(|&j| is_significant(&tokens[j].token))
310        .nth(16)
311        .unwrap_or(tokens.len());
312    let mut parser =
313        Parser::new(&GenericDialect).with_tokens_with_locations(tokens[start..stop].to_vec());
314    let value = parser
315        .parse_prefix()
316        .map_err(|e| polars_err!(SQLInterface: "invalid time travel value: {}", e))?;
317    let mut end = start + parser.index();
318    if at_form {
319        let close = next(tokens, end - 1)
320            .filter(|&j| tokens[j].token == Token::RParen)
321            .ok_or_else(|| polars_err!(SQLInterface: "expected `)` after the time travel value `{}`", value))?;
322        end = close + 1;
323    }
324    polars_ensure!(
325        is_literal(&value),
326        SQLInterface: "time travel takes a literal value, not `{}`", value
327    );
328    Ok((value, end))
329}

331fn is_literal(expr: &SQLExpr) -> bool {
332    match expr {
333        SQLExpr::Value(v) => matches!(
334            v.value,
335            SQLValue::Number(..) | SQLValue::SingleQuotedString(_)
336        ),
337        SQLExpr::UnaryOp {
338            op: UnaryOperator::Minus,
339            expr,
340        } => matches!(&**expr, SQLExpr::Value(v) if matches!(v.value, SQLValue::Number(..))),
341        SQLExpr::TypedString(_) => true,
342        _ => false,
343    }
344}

346/// What a time travel clause selects.
347#[derive(Clone, Debug, PartialEq, Eq)]
348pub enum VersionSpec {
349    /// A snapshot by its id.
350    Snapshot(i64),
351    /// A branch or tag by its name.
352    Ref(String),
353    /// The snapshot current on the main branch at this time, in microseconds since the epoch.
354    Timestamp(i64),
355}
356
357/// Marks the name part that holds the version of a versioned catalog table. No table name
358/// starts with it.
359const MARKER: char = '\0';
360
361impl VersionSpec {
362    /// The version a clause selects. A timestamp without time zone is UTC.
363    pub(crate) fn from_table_version(version: &TableVersion) -> PolarsResult<Self> {
364        let number = |expr: &SQLExpr| match expr {
365            SQLExpr::Value(v) => match &v.value {
366                SQLValue::Number(n, _) => Some(n.to_string()),
367                _ => None,
368            },
369            SQLExpr::UnaryOp {
370                op: UnaryOperator::Minus,
371                expr,
372            } => match &**expr {
373                SQLExpr::Value(v) => match &v.value {
374                    SQLValue::Number(n, _) => Some(format!("-{n}")),
375                    _ => None,
376                },
377                _ => None,
378            },
379            _ => None,
380        };
381        match version {
382            TableVersion::VersionAsOf(expr) => {
383                if let Some(n) = number(expr) {
384                    let id = n.parse::<i64>().map_err(
385                        |_| polars_err!(SQLInterface: "invalid snapshot id {}: expected a 64-bit integer", n),
386                    )?;
387                    return Ok(Self::Snapshot(id));
388                }
389                match expr {
390                    SQLExpr::Value(v) => match &v.value {
391                        SQLValue::SingleQuotedString(name) => Ok(Self::Ref(name.clone())),
392                        _ => {
393                            polars_bail!(SQLInterface: "VERSION AS OF takes a snapshot id or a branch or tag name, not `{}`", expr)
394                        },
395                    },
396                    _ => {
397                        polars_bail!(SQLInterface: "VERSION AS OF takes a snapshot id or a branch or tag name, not `{}`", expr)
398                    },
399                }
400            },
401            TableVersion::TimestampAsOf(expr) => {
402                polars_ensure!(
403                    number(expr).is_none(),
404                    SQLInterface: "TIMESTAMP AS OF takes a timestamp, not the number {} (for a snapshot id, use VERSION AS OF)", expr
405                );
406                Ok(Self::Timestamp(timestamp_micros(expr)?))
407            },
408            other => polars_bail!(SQLInterface: "unsupported time travel clause `{}`", other),
409        }
410    }
411
412    /// The name part that holds this version in a versioned table name.
413    pub(crate) fn marker(&self) -> String {
414        match self {
415            Self::Snapshot(id) => format!("{MARKER}snapshot:{id}"),
416            Self::Ref(name) => format!("{MARKER}ref:{name}"),
417            Self::Timestamp(us) => format!("{MARKER}timestamp:{us}"),
418        }
419    }
420
421    fn from_marker(part: &str) -> Option<Self> {
422        let (kind, value) = part.strip_prefix(MARKER)?.split_once(':')?;
423        match kind {
424            "snapshot" => value.parse().ok().map(Self::Snapshot),
425            "ref" => Some(Self::Ref(value.to_owned())),
426            "timestamp" => value.parse().ok().map(Self::Timestamp),
427            _ => None,
428        }
429    }
430}

432impl std::fmt::Display for VersionSpec {
433    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
434        match self {
435            Self::Snapshot(id) => write!(f, "VERSION AS OF {id}"),
436            Self::Ref(name) => write!(f, "VERSION AS OF '{}'", name.replace('\'', "''")),
437            Self::Timestamp(us) => match chrono::DateTime::from_timestamp_micros(*us) {
438                Some(ts) => write!(
439                    f,
440                    "TIMESTAMP AS OF '{}'",
441                    ts.format("%Y-%m-%d %H:%M:%S%.f%:z")
442                ),
443                None => write!(f, "TIMESTAMP AS OF {us}"),
444            },
445        }
446    }
447}

449/// The instant of a timestamp literal: ISO text, read as `CAST('<text>' AS TIMESTAMPTZ)`
450/// reads it. A UTC offset in the text is kept (a `TIMESTAMP` literal would drop it), no
451/// offset is UTC. A `DATE` is its midnight.
452fn timestamp_micros(expr: &SQLExpr) -> PolarsResult<i64> {
453    let invalid = || polars_err!(SQLInterface: "TIMESTAMP AS OF takes an ISO timestamp or date, not `{}`", expr);
454    let (text, date_only) = match expr {
455        SQLExpr::Value(v) => match &v.value {
456            SQLValue::SingleQuotedString(text) => (text, false),
457            _ => return Err(invalid()),
458        },
459        SQLExpr::TypedString(typed) => match (&typed.data_type, &typed.value.value) {
460            (SQLDataType::Date, SQLValue::SingleQuotedString(text)) => (text, true),
461            (
462                SQLDataType::Timestamp(_, TimezoneInfo::WithTimeZone | TimezoneInfo::Tz),
463                SQLValue::SingleQuotedString(text),
464            ) => (text, false),
465            (
466                SQLDataType::Timestamp(..) | SQLDataType::Datetime(_),
467                SQLValue::SingleQuotedString(text),
468            ) => {
469                // Polars SQL drops such an offset elsewhere, other engines keep it
470                polars_ensure!(
471                    strip_utc_offset(text) == text,
472                    SQLInterface: "TIMESTAMP AS OF: `{}` has a UTC offset but no time zone type; write TIMESTAMPTZ '{}' or '{}'", expr, text, text
473                );
474                (text, false)
475            },
476            _ => return Err(invalid()),
477        },
478        _ => return Err(invalid()),
479    };
480    let iso = is_iso_date(text) || (!date_only && is_iso_datetime(text));
481    polars_ensure!(iso, SQLInterface: "{}", invalid());
482    let value = SQLValue::SingleQuotedString(text.clone());
483    let df = SQLContext::new()
484        .execute(&format!("SELECT CAST({value} AS TIMESTAMPTZ) AS ts"))
485        .and_then(|lf| lf.collect())
486        .map_err(|_| invalid())?;
487    let ts = df
488        .column("ts")?
489        .cast(&DataType::Datetime(TimeUnit::Microseconds, None))?;
490    ts.cast(&DataType::Int64)?.i64()?.get(0).ok_or_else(invalid)
491}

493/// Split a table key into the table name and its version, if versioned.
494pub(crate) fn split_version(key: &[String]) -> (&[String], Option<VersionSpec>) {
495    match key.split_last() {
496        Some((last, name)) => match VersionSpec::from_marker(last) {
497            Some(spec) => (name, Some(spec)),
498            None => (key, None),
499        },
500        None => (key, None),
501    }
502}

504/// Whether a name part is reserved for versions; no written name may have one.
505pub(crate) fn is_version_part(part: &str) -> bool {
506    part.starts_with(MARKER)
507}

509/// A dotted table key without its version, if versioned.
510pub(crate) fn unversioned_key(key: &str) -> &str {
511    key.find(&format!(".{MARKER}")).map_or(key, |at| &key[..at])
512}
