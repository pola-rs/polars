1//! Join keys and `join_where` predicates over operands of different dtypes, which a SQL
2//! comparison compares in a common type but the engine join only upcasts losslessly.

8use polars_utils::arena::{Arena, Node};

12/// `l` and `r` cast to the type a SQL comparison of them uses, for a key pair that the
13/// engine join doesn't cast itself. `merging` allows casts that map distinct values to one
14/// key, such as integers and floats to a float. `None` when no cast is needed or none is
15/// known.
16pub(crate) fn sql_join_key_pair(
17    l: Expr,
18    l_dtype: &DataType,
19    r: Expr,
20    r_dtype: &DataType,
21    merging: bool,
22) -> Option<(Expr, Expr)> {
23    use DataType::*;
24    // A key of untyped literals, such as `CASE .. THEN 0 ELSE 1 END`, is not typed for the
25    // join: both keys take the type a comparison of them resolves to.
26    if (matches!(l_dtype, Unknown(_)) || matches!(r_dtype, Unknown(_)))
27        && let Ok(dtype) = try_get_supertype(l_dtype, r_dtype)
28        && let Ok(dtype) = dtype.materialize_unknown(true)
29        && (dtype.is_primitive_numeric() || dtype.is_decimal())
30    {
31        return Some((l.cast(dtype.clone()), r.cast(dtype)));
32    }
33    // The join computes a key that isn't a column, which gives an untyped literal its type.
34    let l_dtype = &l_dtype.clone().materialize_unknown(true).ok()?;
35    let r_dtype = &r_dtype.clone().materialize_unknown(true).ok()?;
36    if l_dtype == r_dtype || get_numeric_upcast_supertype_lossless(l_dtype, r_dtype).is_some() {
37        return None;
38    }
39    // A decimal compares exactly with another decimal or an integer at the larger scale.
40    let decimal = |scale: usize| {
41        let dtype = Decimal(polars_compute::decimal::DEC128_MAX_PREC, scale);
42        Some((
43            l.clone().strict_cast(dtype.clone()),
44            r.clone().strict_cast(dtype),
45        ))
46    };
47    let both = |dtype: DataType| Some((l.clone().cast(dtype.clone()), r.clone().cast(dtype)));
48    match (l_dtype, r_dtype) {
49        (Decimal(_, l_scale), Decimal(_, r_scale)) => decimal(*l_scale.max(r_scale)),
50        (Decimal(_, scale), dtype) | (dtype, Decimal(_, scale)) if dtype.is_integer() => {
51            decimal(*scale)
52        },
53        (Null, dtype) => Some((l.cast(dtype.clone()), r)),
54        (dtype, Null) => Some((l, r.cast(dtype.clone()))),
55        _ if !merging => None,
56        // A decimal literal takes the float's type, as in a SQL comparison.
57        (Decimal(..), dtype) if dtype.is_float() && is_decimal_literal_valued(&l) => {
58            Some((l.cast(dtype.clone()), r))
59        },
60        (dtype, Decimal(..)) if dtype.is_float() && is_decimal_literal_valued(&r) => {
61            Some((l, r.cast(dtype.clone())))
62        },
63        (Decimal(..), dtype) | (dtype, Decimal(..)) if dtype.is_float() => both(Float64),
64        _ if l_dtype.is_primitive_numeric() && r_dtype.is_primitive_numeric() => {
65            both(try_get_supertype(l_dtype, r_dtype).ok()?)
66        },
67        _ => None,
68    }
69}

71/// `predicate` with each comparison of a left and a right operand that `join_where`
72/// rejects cast to their supertype: a primitive numeric pair without a lossless upcast,
73/// such as an integer and a float, or a Null and a primitive numeric operand. Columns `left_schema` has are left
74/// ones; `joined_schema` holds both inputs as `join_where` resolves them. The predicate
75/// itself when nothing is cast.
76pub(crate) fn sql_join_where_predicate(
77    predicate: Expr,
78    left_schema: &Schema,
79    joined_schema: &Schema,
80) -> Expr {
81    // `join_where` checks the lowered predicate, where SQL arithmetic has become binary
82    // expressions or functions.
83    let mut arena = Arena::new();
84    let mut ctx = ExprToIRContext::new(&mut arena, joined_schema);
85    let Ok(ir) = to_expr_ir(predicate.clone(), &mut ctx) else {
86        return predicate;
87    };
88    let mut casts = Vec::new();
89    if comparison_casts(ir.node(), left_schema, joined_schema, &arena, &mut casts).is_err()
90        || casts.is_empty()
91    {
92        return predicate;
93    }
94    for (node, dtype) in casts {
95        let input = arena.duplicate(node);
96        arena.replace(
97            node,
98            AExpr::Cast {
99                expr: input,
100                dtype,
101                options: CastOptions::NonStrict,
102            },
103        );
104    }
105    node_to_expr(ir.node(), &arena)
106}

108/// Which join inputs an expression reads.
109#[derive(Clone, Copy, PartialEq, Eq)]
110enum Origin {
111    None,
112    Left,
113    Right,
114    Both,
115}
116
117impl Origin {
118    fn union(self, other: Self) -> Self {
119        match (self, other) {
120            (Self::None, o) | (o, Self::None) => o,
121            (a, b) if a == b => a,
122            _ => Self::Both,
123        }
124    }
125}

127// The walk of the `join_where` dtype check (`build_upcast_node_list` in polars-plan),
128// recording the operands of the comparisons it rejects with the dtype to cast them to.
129fn comparison_casts(
130    node: Node,
131    left_schema: &Schema,
132    joined_schema: &Schema,
133    arena: &Arena<AExpr>,
134    casts: &mut Vec<(Node, DataType)>,
135) -> PolarsResult<Origin> {
136    Ok(match arena.get(node) {
137        AExpr::Column(name) if left_schema.contains(name) => Origin::Left,
138        AExpr::Column(_) => Origin::Right,
139        AExpr::Cast { expr, .. } => {
140            comparison_casts(*expr, left_schema, joined_schema, arena, casts)?
141        },
142        AExpr::BinaryExpr { left, op, right } => {
143            let (left, op, right) = (*left, *op, *right);
144            let left_origin = comparison_casts(left, left_schema, joined_schema, arena, casts)?;
145            let right_origin = comparison_casts(right, left_schema, joined_schema, arena, casts)?;
146            if op.is_comparison()
147                && matches!(
148                    (left_origin, right_origin),
149                    (Origin::Left, Origin::Right) | (Origin::Right, Origin::Left)
150                )
151            {
152                let ctx = ToFieldContext::new(arena, joined_schema);
153                let left_dtype = arena.get(left).to_dtype(&ctx)?;
154                let right_dtype = arena.get(right).to_dtype(&ctx)?;
155                // Only the rejected pairs a SQL comparison compares numerically.
156                let numeric_or_null = |dt: &DataType| dt.is_primitive_numeric() || dt.is_null();
157                let rejected = left_dtype != right_dtype
158                    && numeric_or_null(&left_dtype)
159                    && numeric_or_null(&right_dtype)
160                    && get_numeric_upcast_supertype_lossless(&left_dtype, &right_dtype).is_none();
161                if rejected {
162                    let dtype = try_get_supertype(&left_dtype, &right_dtype)?;
163                    if dtype != left_dtype {
164                        casts.push((left, dtype.clone()));
165                    }
166                    if dtype != right_dtype {
167                        casts.push((right, dtype));
168                    }
169                }
170            }
171            left_origin.union(right_origin)
172        },
173        _ => Origin::None,
174    })
