43impl SQLContext {
44    // Entry point: offer each WHERE conjunct to the rewrites, returning the
45    // (possibly join-extended) frame together with the conjuncts left for the
46    // ordinary filter path. In `KeepTrue` mode each top-level AND-conjunct is
47    // offered independently. In `RemoveTrue` mode conjuncts can't be split
48    // (`NOT (a AND b)` is a disjunction), so only a sole (possibly
49    // parenthesized) subquery predicate rewrites, and anything else is
50    // returned whole as the residual.
51    pub(crate) fn rewrite_subquery_conjuncts<'a>(
52        &mut self,
53        mut lf: LazyFrame,
54        expr: &'a SQLExpr,
55        filter_mode: FilterMode,
56        schema: &Schema,
57        stage: RewriteStage,
58    ) -> PolarsResult<(LazyFrame, Vec<&'a SQLExpr>)> {
59        let residual = match filter_mode {
60            FilterMode::RemoveTrue => {
61                let unwrapped = unwrap_nested(expr);
62                match self.try_rewrite_subquery_conjunct(&lf, unwrapped, filter_mode, schema)? {
63                    Some(new_lf) => {
64                        lf = new_lf;
65                        Vec::new()
66                    },
67                    None => vec![expr],
68                }
69            },
70            FilterMode::KeepTrue => {
71                let (plain, with_subquery): (Vec<&SQLExpr>, Vec<&SQLExpr>) =
72                    MintermIter::new(expr).partition(|c| !expr_contains_subquery(c));
73
74                if with_subquery.is_empty() {
75                    return Ok((lf, plain));
76                }
77
78                // A subquery rewrite can row-index the frame, which blocks predicate
79                // pushdown. Apply the conjuncts holding no subquery before that.
80                let mut residual = Vec::new();
81                let mut early = Vec::new();
82                for conj in plain {
83                    let parsed = parse_sql_expr(conj, self, Some(schema))?;
84                    if stage.offers(&parsed, schema) {
85                        early.push(parsed);
86                    } else {
87                        residual.push(conj);
88                    }
89                }
90                if !early.is_empty() {
91                    lf = lf.filter(all_horizontal(early)?);
92                }
93
94                for conj in with_subquery {
95                    if self.stage_offers_membership(conj, schema, stage)?
96                        && let Some(new_lf) =
97                            self.try_rewrite_subquery_conjunct(&lf, conj, filter_mode, schema)?
98                    {
99                        lf = new_lf;
100                    } else {
101                        residual.push(conj);
102                    }
103                }
104                residual
105            },
106        };
107        Ok((lf, residual))
108    }

110    // Whether the stage offers a subquery conjunct to the rewrites: before scalar
111    // lowering only when its membership key is harmless on every row. A key
112    // holding a subquery of its own can't be parsed before that subquery is
113    // lowered.
114    fn stage_offers_membership(
115        &mut self,
116        conj: &SQLExpr,
117        schema: &Schema,
118        stage: RewriteStage,
119    ) -> PolarsResult<bool> {
120        if stage == RewriteStage::AfterScalarLowering {
121            return Ok(true);
122        }
123        Ok(match membership_predicate(conj).map(|m| m.key) {
124            None => true,
125            Some(key) if expr_contains_subquery(key) => false,
126            Some(key) => harmless_on_every_row(&parse_sql_expr(key, self, Some(schema))?, schema),
127        })
128    }

130    // Dispatch one conjunct to the matching rewrite. `RemoveTrue` mode (DELETE)
131    // flips the join polarity. Removing `IN` rows keeps
132    // rows whose membership is false or NULL, exactly what an anti-join
133    // produces; removing `NOT IN` rows would additionally keep NULL keys, which
134    // a semi join can't express, so it stays on the filter path.
135    fn try_rewrite_subquery_conjunct(
136        &mut self,
137        lf: &LazyFrame,
138        conj: &SQLExpr,
139        filter_mode: FilterMode,
140        schema: &Schema,
141    ) -> PolarsResult<Option<LazyFrame>> {
142        let removing = filter_mode == FilterMode::RemoveTrue;
143        match conj {
144            SQLExpr::Exists { subquery, negated } => {
145                self.try_rewrite_exists_as_join(lf, subquery, *negated != removing, schema)
146            },
147            SQLExpr::InSubquery {
148                expr: lhs,
149                subquery,
150                negated,
151            } if !(*negated && removing) => self.try_rewrite_in_subquery_as_join(
152                lf,
153                lhs,
154                subquery,
155                *negated != removing,
156                filter_mode,
157                schema,
158            ),
159            _ => Ok(None),
160        }
161    }

163    // Lower `lhs [NOT] IN (subquery)` and `lhs = (subquery)` over an uncorrelated
164    // subquery of any shape to a semi / anti join against the subquery evaluated
165    // whole. A scalar subquery is reduced to its single value first, so more than
166    // one row still raises.
167    #[cfg(feature = "semi_anti_join")]
168    pub(crate) fn try_rewrite_uncorrelated_subquery_conjunct(
169        &mut self,
170        lf: &LazyFrame,
171        conj: &SQLExpr,
172        filter_mode: FilterMode,
173        outer_schema: &Schema,
174    ) -> PolarsResult<Option<LazyFrame>> {
175        let removing = filter_mode == FilterMode::RemoveTrue;
176        let Some(Membership {
177            key: lhs,
178            subquery,
179            negated,
180            scalar,
181        }) = membership_predicate(conj)
182        else {
183            return Ok(None);
184        };
185        if negated && removing {
186            return Ok(None);
187        }
188        let anti = negated != removing;
189        if is_correlated_subquery(subquery, self.catalog_spellings()) || expr_contains_subquery(lhs)
190        {
191            return Ok(None);
192        }
193        let Some(left_key) = self.try_parse_outer_only_expr(lhs, outer_schema)? else {
194            return Ok(None);
195        };
196
197        let mut inner_lf = self.execute_isolated(|ctx| ctx.execute_query(subquery))?;
198        let inner_schema = self.get_frame_schema(&mut inner_lf)?;
199        polars_ensure!(inner_schema.len() == 1, SQLSyntax: "SQL subquery returns more than one column");
200        let right_key = col(inner_schema.get_at_index(0).unwrap().0.clone());
201        if scalar {
202            inner_lf = inner_lf.select([first().as_expr().item(true)]);
203        }
204
205        build_semi_anti_join(
206            lf,
207            inner_lf,
208            outer_schema,
209            &inner_schema,
210            vec![left_key],
211            vec![right_key],
212            anti,
213            // Only `KeepTrue` "NOT IN" needs 3VL correction.
214            anti && !scalar && filter_mode == FilterMode::KeepTrue,
215        )
216    }

218    // Lower `[NOT] EXISTS (SELECT ... FROM rel WHERE rel.k = outer.k ...)` to a
219    // semi / anti join by decorrelating the equi-correlation predicate(s) into
220    // join keys. DISTINCT is ignored: existence is invariant under
221    // deduplication. A non-equality correlation predicate falls back to the
222    // count-filter rewrite below.
223    #[cfg(feature = "semi_anti_join")]
224    fn try_rewrite_exists_as_join(
225        &mut self,
226        lf: &LazyFrame,
227        subquery: &Query,
228        negated: bool,
229        outer_schema: &Schema,
230    ) -> PolarsResult<Option<LazyFrame>> {
231        let Some(select) = eligible_subquery_select(subquery) else {
232            return Ok(None);
233        };
234        let Some(selection) = &select.selection else {
235            return Ok(None);
236        };
237        // Both rewrites below split this into conjuncts.
238        let selection = factored_selection(selection);
239        // Resolve and parse the inner relation in an isolated context so its
240        // table/alias registrations don't leak into the outer query's scope.
241        let mut ctx = self.isolated();
242        let Some((inner_names, inner_lf, inner_schema)) = ctx.resolve_subquery_from(select)? else {
243            return Ok(None);
244        };
245        // An aggregating subquery is left to `try_decorrelate_exists_subquery`.
246        if exists_aggregates(&ctx, subquery, select, &inner_names, &inner_schema) != Some(false) {
247            return Ok(None);
248        }
249        if let Some(conjuncts) = ctx.split_subquery_conjuncts(
250            &selection,
251            &inner_names,
252            &inner_schema,
253            outer_schema,
254            self,
255        )? {
256            // An uncorrelated EXISTS (no correlation key found) has no join key
257            // to build from, so leave it to the existing path.
258            return Ok(if conjuncts.left_on.is_empty() {
259                None
260            } else {
261                ctx.finish_decorrelated_join(
262                    lf,
263                    inner_lf,
264                    outer_schema,
265                    &inner_schema,
266                    conjuncts,
267                    negated,
268                )?
269            });
270        }
271        ctx.try_rewrite_exists_as_count_filter(
272            lf,
273            inner_lf,
274            &selection,
275            &inner_names,
276            &inner_schema,
277            outer_schema,
278            self,
279            negated,
280        )
281    }

283    // Lower `[NOT] EXISTS (SELECT ... FROM rel WHERE <comparison-correlated>)`,
284    // whose correlation predicates aren't all plain equalities, to a
285    // `count(*) > 0` filter (`== 0` for NOT EXISTS).
286    #[cfg(feature = "semi_anti_join")]
287    #[expect(clippy::too_many_arguments)]
288    fn try_rewrite_exists_as_count_filter(
289        mut self,
290        lf: &LazyFrame,
291        inner_lf: LazyFrame,
292        selection: &SQLExpr,
293        inner_names: &[RelationName],
294        inner_schema: &Schema,
295        outer_schema: &Schema,
296        outer: &SQLContext,
297        negated: bool,
298    ) -> PolarsResult<Option<LazyFrame>> {
299        let mut corr_preds = Vec::new();
300        let mut local_filters = Vec::new();
301        for conj in MintermIter::new(selection) {
302            if let Some(pred) =
303                scalar_correlation_predicate(conj, inner_schema, outer_schema, &self, outer)?
304            {
305                corr_preds.push(pred);
306            } else if let Some(filter) =
307                self.try_parse_inner_only_expr(conj, inner_names, inner_schema)?
308            {
309                local_filters.push(filter);
310            } else {
311                return Ok(None);
312            }
313        }
314        // No correlation: leave it to the equi-only path.
315        if corr_preds.is_empty() {
316            return Ok(None);
317        }
318
319        let prefix = format_pl_smallstr!("{CORRELATED_COL_PREFIX}{}_", unique_column_name());
320        let idx_name = format_pl_smallstr!("{prefix}idx");
321        let count_name = format_pl_smallstr!("{prefix}cnt");
322
323        // Apply inner-only filters, then rename inner columns to collision-free
324        // names so they can't clash with outer columns of the same name (as in
325        // a self-correlated `t1 AS x` subquery over outer `t1`).
326        let inner_filtered = local_filters.into_iter().fold(inner_lf, LazyFrame::filter);
327        let rename_from: Vec<PlSmallStr> = inner_schema.iter_names().cloned().collect();
328        let rename_to: Vec<PlSmallStr> = rename_from
329            .iter()
330            .map(|name| prefixed_inner(&prefix, name))
331            .collect();
332        let inner_renamed = inner_filtered.rename(&rename_from, &rename_to, true);
333        inner_renamed.set_cached_arena(self.lp_arena, self.expr_arena);
334
335        let join_preds: Vec<Expr> = corr_preds.iter().map(|p| p.to_expr(&prefix)).collect();
336
337        let outer_indexed = row_indexed_once(lf.clone(), idx_name.clone());
338        let matched = outer_indexed
339            .clone()
340            .join_builder()
341            .with(inner_renamed)
342            .how(JoinType::Inner)
343            .join_where(join_preds);
344        let grouped = matched
345            .group_by([col(idx_name.clone())])
346            .agg([len().alias(count_name.clone())]);
347
348        let joined = outer_indexed
349            .join_builder()
350            .with(grouped)
351            .left_on([col(idx_name.clone())])
352            .right_on([col(idx_name.clone())])
353            .how(JoinType::Left)
354            .coalesce(JoinCoalesce::CoalesceColumns)
355            .maintain_order(MaintainOrderJoin::Left)
356            .finish()?
357            .with_columns([col(count_name.clone()).fill_null(lit(0))]);
358
359        let matches = if negated {
360            col(count_name.clone()).eq(lit(0))
361        } else {
362            col(count_name.clone()).gt(lit(0))
363        };
364        Ok(Some(joined.filter(matches).drop(Selector::ByName {
365            names: Arc::from([idx_name, count_name]),
366            strict: true,
367        })))
368    }

370    // Lower `lhs [NOT] IN (SELECT col FROM rel ...)` to a semi / anti join: the
371    // projected column is the membership key and any equi-correlations in the
372    // subquery WHERE become additional join keys.
373    #[cfg(feature = "semi_anti_join")]
374    fn try_rewrite_in_subquery_as_join(
375        &mut self,
376        lf: &LazyFrame,
377        lhs: &SQLExpr,
378        subquery: &Query,
379        anti: bool,
380        filter_mode: FilterMode,
381        outer_schema: &Schema,
382    ) -> PolarsResult<Option<LazyFrame>> {
383        let Some(select) = eligible_subquery_select(subquery) else {
384            return Ok(None);
385        };
386        // DISTINCT is membership-invariant, but DISTINCT ON drops rows per key.
387        if matches!(&select.distinct, Some(Distinct::On(_))) {
388            return Ok(None);
389        }
390        let [SelectItem::UnnamedExpr(proj) | SelectItem::ExprWithAlias { expr: proj, .. }] =
391            select.projection.as_slice()
392        else {
393            return Ok(None);
394        };
395
396        let Some(left_key) = self.try_parse_outer_only_expr(lhs, outer_schema)? else {
397            return Ok(None);
398        };
399
400        let mut ctx = self.isolated();
401        let Some((inner_names, inner_lf, inner_schema)) = ctx.resolve_subquery_from(select)? else {
402            return Ok(None);
403        };
404        // The membership key must be a plain expression over the inner relation;
405        // any alias it carries is cosmetic and not allowed in a join key.
406        let Some(right_key) = ctx.try_parse_inner_only_expr(proj, &inner_names, &inner_schema)?
407        else {
408            return Ok(None);
409        };
410        let right_key = right_key.meta().undo_aliases();
411
412        let SubqueryConjuncts {
413            mut left_on,
414            mut right_on,
415            local_filters,
416        } = match &select.selection {
417            Some(selection) => {
418                let Some(split) = ctx.split_subquery_conjuncts(
419                    selection,
420                    &inner_names,
421                    &inner_schema,
422                    outer_schema,
423                    self,
424                )?
425                else {
426                    return Ok(None);
427                };
428                split
429            },
430            None => SubqueryConjuncts::default(),
431        };
432
433        left_on.insert(0, left_key);
434        right_on.insert(0, right_key);
435
436        let inner_lf = local_filters.into_iter().fold(inner_lf, LazyFrame::filter);
437        inner_lf.set_cached_arena(ctx.lp_arena, ctx.expr_arena);
438        build_semi_anti_join(
439            lf,
440            inner_lf,
441            outer_schema,
442            &inner_schema,
443            left_on,
444            right_on,
445            anti,
446            // Only `KeepTrue` "NOT IN" needs 3VL correction.
447            anti && filter_mode == FilterMode::KeepTrue,
448        )
449    }

484    // Resolve the subquery's FROM (relations, possibly with joins) into the inner
485    // LazyFrame, its schema, and the set of relation names/aliases used to
486    // classify qualified correlation columns. Also checks the SELECT and WHERE
487    // for unqualified columns that are ambiguous between the joined relations.
488    fn resolve_subquery_from(
489        &mut self,
490        select: &Select,
491    ) -> PolarsResult<Option<(Vec<RelationName>, LazyFrame, SchemaRef)>> {
492        let from = select.from.as_slice();
493        let Some(inner_names) = from
494            .iter()
495            .flat_map(|tbl_expr| {
496                std::iter::once(&tbl_expr.relation)
497                    .chain(tbl_expr.joins.iter().map(|j| &j.relation))
498            })
499            .map(|factor| Some(relation_name(factor)?.spelled(factor, self.catalog_spellings())))
500            .collect::<Option<Vec<_>>>()
501        else {
502            return Ok(None);
503        };
504        let Some((first, rest)) = from.split_first() else {
505            return Ok(None);
506        };
507        // Qualifiers in the inner predicates resolve against the inner relations first.
508        self.set_active_relations(from);
509        let mut inner_lf = self.execute_from_statement(first)?;
510        // Comma-separated relations cross join; the equalities linking them are
511        // inner-only filters.
512        for tbl_expr in rest {
513            let Some(r_name) = get_table_name(&tbl_expr.relation) else {
514                return Ok(None);
515            };
516            let mut rf = self.execute_from_statement(tbl_expr)?;
517            let left_schema = self.get_frame_schema(&mut inner_lf)?;
518            let right_schema = self.get_frame_schema(&mut rf)?;
519            inner_lf = inner_lf.cross_join(rf, Some(format_pl_smallstr!(":{r_name}")));
520            let joined_schema = self.get_frame_schema(&mut inner_lf)?;
521            let right_relations: Vec<String> =
522                declared_relations(std::slice::from_ref(tbl_expr), self.catalog_spellings())
523                    .iter()
524                    .map(RelationName::key)
525                    .collect();
526            self.record_joined_aliases(
527                &r_name,
528                &right_relations,
529                &left_schema,
530                &right_schema,
531                &joined_schema,
532            )?;
533        }
534        self.set_join_base(&select.from);
535        self.check_select_column_refs(select)?;
536        let inner_schema = self.get_frame_schema(&mut inner_lf)?;
537        Ok(Some((inner_names, inner_lf, inner_schema)))
538    }

540    // Split a subquery's WHERE conjuncts into a `SubqueryConjuncts`, or `None`
541    // when a conjunct is neither a correlation key pair nor an inner-only
542    // filter (an outer column in a non-equi shape, an unresolvable name).
543    #[cfg(feature = "semi_anti_join")]
544    fn split_subquery_conjuncts(
545        &mut self,
546        selection: &SQLExpr,
547        inner_names: &[RelationName],
548        inner_schema: &Schema,
549        outer_schema: &Schema,
550        outer: &SQLContext,
551    ) -> PolarsResult<Option<SubqueryConjuncts>> {
552        let mut left_on = Vec::new();
553        let mut right_on = Vec::new();
554        let mut local_filters = Vec::new();
555        for conj in MintermIter::new(selection) {
556            if let Some((outer_key, inner_key)) =
557                correlation_key_pair(conj, inner_schema, outer_schema, self, outer)?
558            {
559                left_on.push(col(outer_key));
560                right_on.push(col(inner_key));
561                continue;
562            }
563            let Some(filter) = self.try_parse_inner_only_expr(conj, inner_names, inner_schema)?
564            else {
565                return Ok(None);
566            };
567            local_filters.push(filter);
568        }
569        Ok(Some(SubqueryConjuncts {
570            left_on,
571            right_on,
572            local_filters,
573        }))
574    }

595    // Split a subquery's WHERE into its correlation comparisons and its
596    // inner-only filters, or `None` when a conjunct is neither. A comparison
597    // spelled twice is kept once: it would duplicate a group-by key.
598    fn split_correlation_conjuncts(
599        &mut self,
600        selection: &SQLExpr,
601        inner_names: &[RelationName],
602        inner_schema: &Schema,
603        outer_schema: &Schema,
604        outer: &SQLContext,
605    ) -> PolarsResult<Option<(Vec<CorrPredicate>, Vec<Expr>)>> {
606        let mut corr_preds: Vec<CorrPredicate> = Vec::new();
607        let mut local_filters = Vec::new();
608        let selection = factored_selection(selection);
609        for conj in MintermIter::new(&selection) {
610            if let Some(pred) =
611                scalar_correlation_predicate(conj, inner_schema, outer_schema, self, outer)?
612            {
613                if !corr_preds.iter().any(|p| p.same_comparison(&pred)) {
614                    corr_preds.push(pred);
615                }
616            } else if let Some(filter) =
617                self.try_parse_inner_only_expr(conj, inner_names, inner_schema)?
618            {
619                local_filters.push(filter);
620            } else {
621                return Ok(None);
622            }
623        }
624        Ok(Some((corr_preds, local_filters)))
625    }

662    // `values` sorted by the `ORDER BY` of a `LIMIT 1` subquery, whose first value
663    // is then its result. The sort keys are pushed onto `row_columns`, to be
664    // computed per inner row. `None` for an order that isn't read from the inner
665    // row alone.
666    #[allow(clippy::too_many_arguments)]
667    fn try_first_row_order(
668        &mut self,
669        subquery: &Query,
670        alias: Option<&Ident>,
671        values: Expr,
672        prefix: &str,
673        inner_names: &[RelationName],
674        inner_schema: &Schema,
675        row_columns: &mut Vec<Expr>,
676    ) -> PolarsResult<Option<Expr>> {
677        let Some(order_by) = &subquery.order_by else {
678            return Ok(Some(values));
679        };
680        let OrderByKind::Expressions(exprs) = &order_by.kind else {
681            return Ok(None);
682        };
683        if order_by.interpolate.is_some() || exprs.is_empty() {
684            return Ok(None);
685        }
686        let mut by = Vec::with_capacity(exprs.len());
687        let mut descending = Vec::with_capacity(exprs.len());
688        let mut nulls_last = Vec::with_capacity(exprs.len());
689        for ob in exprs {
690            if ob.with_fill.is_some() || expr_has_window_functions(&ob.expr) {
691                return Ok(None);
692            }
693            let mut expr = &ob.expr;
694            while let SQLExpr::Nested(inner) = expr {
695                expr = inner;
696            }
697            // The output column, by name or by position.
698            let is_output = match expr {
699                SQLExpr::Identifier(ident) => alias.is_some_and(|a| a.value == ident.value),
700                SQLExpr::Value(ValueWithSpan {
701                    value: SQLValue::Number(n, _),
702                    ..
703                }) => {
704                    if n != "1" {
705                        return Ok(None);
706                    }
707                    true
708                },
709                _ => false,
710            };
711            let key = if is_output {
712                values.clone()
713            } else {
714                let Some(key) = self.marking_aggregates(|ctx| {
715                    ctx.try_parse_inner_only_expr(&ob.expr, inner_names, inner_schema)
716                })?
717                else {
718                    return Ok(None);
719                };
720                if has_marked_aggregate(&key)
721                    || has_expr(&key, |e| matches!(e, Expr::SubPlan(_, _)))
722                {
723                    return Ok(None);
724                }
725                let name = format_pl_smallstr!("{prefix}ord{}", row_columns.len());
726                row_columns.push(key.alias(name.clone()));
727                col(name)
728            };
729            let options = order_by_sort_options(&ob.options);
730            by.push(key);
731            descending.push(options.descending);
732            nulls_last.push(options.nulls_last);
733        }
734        Ok(Some(
735            values.sort_by(
736                by,
737                SortMultipleOptions::default()
738                    .with_order_descending_multi(descending)
739                    .with_nulls_last_multi(nulls_last),
740            ),
741        ))
742    }

744    // The join tree of a block whose frame is its FROM relations joined and
745    // filtered by the WHERE conjuncts that hold no subquery, and nothing more.
746    // `residual` is what the early subquery rewrites left of `where_expr`: the
747    // frame is still the tree when that is every subquery conjunct and nothing
748    // else. `None` as well when the FROM isn't a list of distinct plain tables
749    // whose columns all have distinct names.
750    pub(crate) fn outer_join_tree(
751        &mut self,
752        select: &Select,
753        where_expr: &SQLExpr,
754        residual: &[&SQLExpr],
755    ) -> PolarsResult<Option<OuterJoinTree>> {
756        let Some(selection) = &select.selection else {
757            return Ok(None);
758        };
759        let n_with_subquery = MintermIter::new(where_expr)
760            .filter(|c| expr_contains_subquery(c))
761            .count();
762        if select.from.len() < 2
763            || residual.len() != n_with_subquery
764            || !residual.iter().all(|c| expr_contains_subquery(c))
765        {
766            return Ok(None);
767        }
768
769        let mut relations: Vec<TreeRelation> = Vec::with_capacity(select.from.len());
770        for tbl_expr in &select.from {
771            let Some((table, name)) = plain_table(tbl_expr) else {
772                return Ok(None);
773            };
774            if relations.iter().any(|r| r.table == table || r.name == name) {
775                return Ok(None);
776            }
777            let mut rel_lf = self.execute_from_statement(tbl_expr)?;
778            let schema = self.get_frame_schema(&mut rel_lf)?;
779            relations.push(TreeRelation {
780                table: table.to_string(),
781                name: name.to_string(),
782                schema,
783            });
784        }
785        let n_columns: usize = relations.iter().map(|r| r.schema.len()).sum();
786        let n_distinct = relations
787            .iter()
788            .flat_map(|r| r.schema.iter_names())
789            .collect::<PlHashSet<_>>()
790            .len();
791        if n_distinct != n_columns {
792            return Ok(None);
793        }
794
795        let own: Vec<(&str, usize)> = relations
796            .iter()
797            .enumerate()
798            .map(|(idx, r)| (r.name.as_str(), idx))
799            .collect();
800        let selection = factored_selection(selection);
801        let mut conjuncts = Vec::new();
802        for conj in MintermIter::new(&selection).filter(|c| !expr_contains_subquery(c)) {
803            let Some(spelled) = spell_conjunct(conj, &own, &relations, false) else {
804                return Ok(None);
805            };
806            conjuncts.push(spelled);
807        }
808        Ok(Some(OuterJoinTree {
809            relations,
810            conjuncts,
811        }))
812    }

814    // Lower every correlated subquery reachable from `expr` into a decorrelated
815    // join over `lf`, substituting each one for the column its lowering
816    // materialised. Returns the updated frame together with the rewritten
817    // expression, which borrows `expr` unchanged when nothing was lowered.
818    // Subqueries that can't be lowered are left in place.
819    //
820    // See [`SubqueryBindings`] for when a set may be shared across calls. Pass
821    // `join_tree` only while `lf` is still the frame it describes.
822    pub(crate) fn lower_correlated_subqueries<'a>(
823        &mut self,
824        lf: LazyFrame,
825        outer_schema: &Schema,
826        expr: &'a SQLExpr,
827        scope: LowerScope,
828        bindings: &mut SubqueryBindings,
829        join_tree: Option<&OuterJoinTree>,
830    ) -> PolarsResult<(LazyFrame, Cow<'a, SQLExpr>)> {
831        // The mutable visit needs an owned expression; skip the clone when there is
832        // nothing to rewrite.
833        if !expr_contains_subquery(expr) {
834            return Ok((lf, Cow::Borrowed(expr)));
835        }
836        let mut rewritten = expr.clone();
837        let mut lowering = CorrelatedLowering {
838            ctx: self,
839            lf,
840            outer_schema,
841            scope,
842            bindings,
843            join_tree,
844            query_depth: 0,
845            skips: Vec::new(),
846            changed: false,
847        };
848        if let ControlFlow::Break(e) = dyn_visit::visit_mut(&mut rewritten, &mut lowering) {
849            return Err(e);
850        }
851        let rewritten = if lowering.changed {
852            Cow::Owned(rewritten)
853        } else {
854            Cow::Borrowed(expr)
855        };
856        Ok((lowering.lf, rewritten))
857    }

859    // Lower the correlated subqueries of a `JOIN ... ON` condition against the
860    // input they read, one conjunct at a time, so that the condition reads
861    // plain columns of the inputs. A conjunct that `correlated_join_side`
862    // declines is left as it is. Returns the inputs with the lowered columns
863    // added and the rewritten condition, or `None` when nothing was lowered.
864    pub(crate) fn lower_join_on_subqueries(
865        &mut self,
866        tbl_left: &TableInfo,
867        tbl_right: &TableInfo,
868        on: &SQLExpr,
869        join_type: &JoinType,
870    ) -> PolarsResult<Option<(TableInfo, TableInfo, SQLExpr)>> {
871        let mut left = tbl_left.frame.clone();
872        let mut right = tbl_right.frame.clone();
873        let mut left_bindings = SubqueryBindings::new();
874        let mut right_bindings = SubqueryBindings::new();
875        let mut changed = false;
876        let mut conjuncts = Vec::new();
877        for conj in flatten_and_conditions(on) {
878            let lowered;
879            match correlated_join_side(self, conj, tbl_left, tbl_right) {
880                Some(true) => {
881                    (left, lowered) = self.lower_correlated_subqueries(
882                        left,
883                        &tbl_left.schema,
884                        conj,
885                        LowerScope::ScalarAndPredicates,
886                        &mut left_bindings,
887                        None,
888                    )?;
889                },
890                Some(false) => {
891                    (right, lowered) = self.lower_correlated_subqueries(
892                        right,
893                        &tbl_right.schema,
894                        conj,
895                        LowerScope::ScalarAndPredicates,
896                        &mut right_bindings,
897                        None,
898                    )?;
899                },
900                None => {
901                    lowered = match self.try_attach_join_membership(
902                        conj, &mut left, &mut right, tbl_left, tbl_right, join_type,
903                    )? {
904                        Some(rewritten) => Cow::Owned(rewritten),
905                        None => Cow::Borrowed(conj),
906                    }
907                },
908            }
909            changed |= matches!(lowered, Cow::Owned(_));
910            conjuncts.push(lowered.into_owned());
911        }
912        if !changed {
913            return Ok(None);
914        }
915        let left_schema = self.get_frame_schema(&mut left)?;
916        let right_schema = self.get_frame_schema(&mut right)?;
917        let on = combine_conditions(conjuncts, SQLBinaryOperator::And)
918            .expect("a condition has at least one conjunct");
919        Ok(Some((
920            TableInfo {
921                frame: left,
922                name: tbl_left.name.clone(),
923                schema: left_schema,
924                relations: tbl_left.relations.clone(),
925            },
926            TableInfo {
927                frame: right,
928                name: tbl_right.name.clone(),
929                schema: right_schema,
930                relations: tbl_right.relations.clone(),
931            },
932            on,
933        )))
934    }

936    // Join the distinct inner rows that make `lhs IN (subquery)` or `EXISTS (subquery)`
937    // true onto one input of a join, when the subquery correlates with an input that the
938    // rest of the conjunct doesn't read. The conjunct then becomes equalities between the
939    // other input and the joined columns. For a pair of input rows at most one joined row
940    // matches, so only an input whose unmatched rows the join drops can take them.
941    fn try_attach_join_membership(
942        &mut self,
943        conj: &SQLExpr,
944        left: &mut LazyFrame,
945        right: &mut LazyFrame,
946        tbl_left: &TableInfo,
947        tbl_right: &TableInfo,
948        join_type: &JoinType,
949    ) -> PolarsResult<Option<SQLExpr>> {
950        let (attach_left, attach_right) = match join_type {
951            JoinType::Inner => (true, true),
952            JoinType::Left => (false, true),
953            JoinType::Right => (true, false),
954            _ => return Ok(None),
955        };
956        let (lhs, subquery) = match strip_nested(conj) {
957            SQLExpr::InSubquery {
958                expr,
959                subquery,
960                negated: false,
961            } => (Some(expr.as_ref()), subquery.as_ref()),
962            SQLExpr::Exists {
963                subquery,
964                negated: false,
965            } => (None, subquery.as_ref()),
966            _ => return Ok(None),
967        };
968        let Some(select) = eligible_subquery_select(subquery) else {
969            return Ok(None);
970        };
971        let Some(selection) = &select.selection else {
972            return Ok(None);
973        };
974        if matches!(&select.distinct, Some(Distinct::On(_))) {
975            return Ok(None);
976        }
977        // Planning the subquery is only worth it if its qualifiers read the other input.
978        let lhs_is_left = match lhs.map(|lhs| self.join_inputs_read(lhs, tbl_left, tbl_right)) {
979            Some((true, false)) => Some(true),
980            Some((false, true)) => Some(false),
981            Some(_) => return Ok(None),
982            None => None,
983        };
984        let (corr_left, corr_right) = self.foreign_join_inputs(subquery, tbl_left, tbl_right);
985        let reads_both = match lhs_is_left {
986            Some(true) => corr_right,
987            Some(false) => corr_left,
988            None => corr_left && corr_right,
989        };
990        if !reads_both {
991            return Ok(None);
992        }
993        let mut ctx = self.isolated();
994        let Some((inner_names, inner_lf, inner_schema)) = ctx.resolve_subquery_from(select)? else {
995            return Ok(None);
996        };
997        if exists_aggregates(&ctx, subquery, select, &inner_names, &inner_schema) != Some(false) {
998            return Ok(None);
999        }
1000
1001        // Pairs of an input expression and the inner expression it must equal.
1002        let mut keys: Vec<JoinMembershipKey> = Vec::new();
1003        if let Some(lhs) = lhs {
1004            let [SelectItem::UnnamedExpr(proj) | SelectItem::ExprWithAlias { expr: proj, .. }] =
1005                select.projection.as_slice()
1006            else {
1007                return Ok(None);
1008            };
1009            let Some(value) = ctx.try_parse_inner_only_expr(proj, &inner_names, &inner_schema)?
1010            else {
1011                return Ok(None);
1012            };
1013            // The value is taken per inner row, before the rows are grouped by outer row.
1014            if !is_elementwise_over(&value, &inner_schema) {
1015                return Ok(None);
1016            }
1017            let is_left = lhs_is_left.unwrap();
1018            let tbl = if is_left { tbl_left } else { tbl_right };
1019            let Some(outer) = self.try_parse_outer_only_expr(lhs, &tbl.schema)? else {
1020                return Ok(None);
1021            };
1022            keys.push(JoinMembershipKey {
1023                is_left,
1024                outer,
1025                inner: value.meta().undo_aliases(),
1026            });
1027        }
1028        let mut local_filters = Vec::new();
1029        let selection = factored_selection(selection);
1030        for c in MintermIter::new(&selection) {
1031            if let Some(key) =
1032                self.join_correlation_key(c, &ctx, &inner_schema, tbl_left, tbl_right)?
1033            {
1034                keys.push(key);
1035            } else if let Some(filter) =
1036                ctx.try_parse_inner_only_expr(c, &inner_names, &inner_schema)?
1037                && !has_expr(&filter, |e| matches!(e, Expr::SubPlan(_, _)))
1038            {
1039                local_filters.push(filter);
1040            } else {
1041                return Ok(None);
1042            }
1043        }
1044        let reads_left = keys.iter().any(|k| k.is_left);
1045        let reads_right = keys.iter().any(|k| !k.is_left);
1046        if !(reads_left && reads_right) {
1047            return Ok(None);
1048        }
1049        // With a choice, the rows join onto the input the `IN` operand doesn't read.
1050        let to_left = match (attach_left, attach_right) {
1051            (true, true) => lhs.is_none() || !keys[0].is_left,
1052            (to_left, _) => to_left,
1053        };
1054
1055        let prefix = format_pl_smallstr!("{CORRELATED_COL_PREFIX}{}_", unique_column_name());
1056        let names: Vec<PlSmallStr> = (0..keys.len())
1057            .map(|i| format_pl_smallstr!("{prefix}{i}"))
1058            .collect();
1059        let mut rows = local_filters
1060            .into_iter()
1061            .fold(inner_lf, LazyFrame::filter)
1062            .select(
1063                keys.iter()
1064                    .zip(&names)
1065                    .map(|(k, name)| k.inner.clone().alias(name.clone()))
1066                    .collect::<Vec<_>>(),
1067            );
1068        let rows_schema = ctx.get_frame_schema(&mut rows)?;
1069        rows.set_cached_arena(ctx.lp_arena, ctx.expr_arena);
1070
1071        // Each key is compared as a column of its input, both sides cast to a shared type.
1072        let (mut left_frame, mut right_frame) = (left.clone(), right.clone());
1073        let mut row_casts = Vec::with_capacity(keys.len());
1074        let (mut target_on, mut rows_on, mut equalities) = (Vec::new(), Vec::new(), Vec::new());
1075        for (i, (key, name)) in keys.into_iter().zip(names).enumerate() {
1076            let frame = if key.is_left {
1077                &mut left_frame
1078            } else {
1079                &mut right_frame
1080            };
1081            let key_name = format_pl_smallstr!("{prefix}key{i}");
1082            let mut keyed = frame.clone().with_column(key.outer.alias(key_name.clone()));
1083            let keyed_schema = self.get_frame_schema(&mut keyed)?;
1084            let Some((outer, inner)) = try_prepare_join_keys(
1085                &keyed_schema,
1086                &rows_schema,
1087                vec![col(key_name.clone())],
1088                vec![col(name.clone())],
1089            ) else {
1090                return Ok(None);
1091            };
1092            let [outer] = <[Expr; 1]>::try_from(outer).unwrap();
1093            let [inner] = <[Expr; 1]>::try_from(inner).unwrap();
1094            *frame = keyed.with_column(outer.alias(key_name.clone()));
1095            row_casts.push(inner.alias(name.clone()));
1096            if key.is_left == to_left {
1097                target_on.push(col(key_name));
1098                rows_on.push(col(name));
1099            } else {
1100                equalities.push(SQLExpr::BinaryOp {
1101                    left: Box::new(SQLExpr::Identifier(Ident::new(key_name.as_str()))),
1102                    op: SQLBinaryOperator::Eq,
1103                    right: Box::new(SQLExpr::Identifier(Ident::new(name.as_str()))),
1104                });
1105            }
1106        }
1107        let rows = rows
1108            .with_columns(row_casts)
1109            .unique(None, UniqueKeepStrategy::Any);
1110        let (target, other, mut target_frame, other_frame) = if to_left {
1111            (left, right, left_frame, right_frame)
1112        } else {
1113            (right, left, right_frame, left_frame)
1114        };
1115        target_frame = target_frame
1116            .join_builder()
1117            .with(rows)
1118            .left_on(target_on)
1119            .right_on(rows_on)
1120            .how(JoinType::Inner)
1121            .coalesce(JoinCoalesce::KeepColumns)
1122            .finish()?;
1123        *target = target_frame;
1124        *other = other_frame;
1125        Ok(combine_conditions(equalities, SQLBinaryOperator::And))
1126    }

1128    // Which inputs of a join the qualifiers that `subquery` doesn't declare name, as
1129    // `(reads_left, reads_right)`.
1130    fn foreign_join_inputs(
1131        &self,
1132        subquery: &Query,
1133        tbl_left: &TableInfo,
1134        tbl_right: &TableInfo,
1135    ) -> (bool, bool) {
1136        let (mut reads_left, mut reads_right) = (false, false);
1137        visit_foreign_qualifiers(subquery, self.catalog_spellings(), |parts| {
1138            if let Ok(Some((name, _))) = self.resolve_qualifier(parts) {
1139                if self.is_right_relation(&name, tbl_right) {
1140                    reads_right = true;
1141                } else if self.is_joined_relation(&name, tbl_left) {
1142                    reads_left = true;
1143                }
1144            }
1145            ControlFlow::Continue(())
1146        });
1147        (reads_left, reads_right)
1148    }

1150    // An equality between a column of the subquery and a column of one input of the
1151    // join around it, either way round.
1152    fn join_correlation_key(
1153        &self,
1154        conj: &SQLExpr,
1155        inner: &SQLContext,
1156        inner_schema: &Schema,
1157        tbl_left: &TableInfo,
1158        tbl_right: &TableInfo,
1159    ) -> PolarsResult<Option<JoinMembershipKey>> {
1160        let SQLExpr::BinaryOp {
1161            left,
1162            op: SQLBinaryOperator::Eq,
1163            right,
1164        } = strip_nested(conj)
1165        else {
1166            return Ok(None);
1167        };
1168        for (inner_sql, outer_sql) in [(left, right), (right, left)] {
1169            let Some((CorrelationSide::Inner, inner_name)) = classify_correlation_column(
1170                inner_sql,
1171                inner_schema,
1172                &Schema::default(),
1173                inner,
1174                self,
1175            )?
1176            else {
1177                continue;
1178            };
1179            let Some(is_left) = self.column_join_input(outer_sql, tbl_left, tbl_right) else {
1180                continue;
1181            };
1182            let tbl = if is_left { tbl_left } else { tbl_right };
1183            let Some((CorrelationSide::Outer, outer_name)) =
1184                classify_correlation_column(outer_sql, inner_schema, &tbl.schema, inner, self)?
1185            else {
1186                continue;
1187            };
1188            return Ok(Some(JoinMembershipKey {
1189                is_left,
1190                outer: col(outer_name),
1191                inner: col(inner_name),
1192            }));
1193        }
1194        Ok(None)
1195    }

1197    // Attempt the decorrelation of a single scalar subquery:
1198    //   (SELECT AGG(...) FROM inner WHERE <corr-preds> AND <inner-filters>)
1199    // A projection without an aggregate is taken as the `item` of its values,
1200    // which raises when an outer row matches more than one inner row.
1201    // Equality-only correlation: `inner GROUP BY <corr cols>` once, joined onto
1202    // the outer frame on those columns directly. Otherwise: row-index the outer
1203    // frame, inner-join outer × inner on the correlation predicates (`join_where`
1204    // handles inequality), aggregate per outer row, then left-join back on the
1205    // row index. Either way, `COUNT` over no matches is 0, every other aggregate
1206    // is NULL. Returns `None` when uncorrelated or not a supported shape.
1207    // When `join_tree` describes the outer frame and the subquery repeats a part
1208    // of it, the aggregate is taken over the outer frame itself instead.
1209    fn try_decorrelate_scalar_subquery(
1210        &mut self,
1211        mut lf: LazyFrame,
1212        outer_schema: &Schema,
1213        subquery: &Query,
1214        join_tree: Option<&OuterJoinTree>,
1215    ) -> PolarsResult<Option<(LazyFrame, PlSmallStr)>> {
1216        let Some(first_row) = first_row_limit(subquery) else {
1217            return Ok(None);
1218        };
1219        let Some(select) = eligible_subquery_body(subquery) else {
1220            return Ok(None);
1221        };
1222        let Some(selection) = &select.selection else {
1223            return Ok(None);
1224        };
1225        let [SelectItem::UnnamedExpr(proj) | SelectItem::ExprWithAlias { expr: proj, .. }] =
1226            select.projection.as_slice()
1227        else {
1228            return Ok(None);
1229        };
1230
1231        let mut ctx = self.isolated();
1232        let Some((inner_names, inner_lf, inner_schema)) = ctx.resolve_subquery_from(select)? else {
1233            return Ok(None);
1234        };
1235
1236        // A window gives one value per inner row, not one per group.
1237        if expr_has_window_functions(proj) {
1238            return Ok(None);
1239        }
1240        let Some(proj_expr) = ctx.marking_aggregates(|ctx| {
1241            ctx.try_parse_inner_only_expr(proj, &inner_names, &inner_schema)
1242        })?
1243        else {
1244            return Ok(None);
1245        };
1246        let prefix = format_pl_smallstr!("{CORRELATED_COL_PREFIX}{}_", unique_column_name());
1247        let has_aggregate = has_marked_aggregate(&proj_expr);
1248        let proj_expr = strip_aggregate_marks(proj_expr);
1249        // The projection is either a scalar aggregate over the inner relation, or a
1250        // value per inner row. Without `LIMIT 1`, each outer row may match at most
1251        // one inner row. With it, the first row in the subquery's order is taken.
1252        let mut row_columns: Vec<Expr> = Vec::new();
1253        let (agg_expr, single_row_check) =
1254            if has_expr(&proj_expr, |e| matches!(e, Expr::Agg(_) | Expr::Len)) {
1255                // An aggregate within a row-expanding function gives several rows.
1256                if !is_scalar_over(&proj_expr, &inner_schema) {
1257                    return Ok(None);
1258                }
1259                (proj_expr, false)
1260            } else if !has_aggregate
1261                && !matches!(select.distinct, Some(Distinct::On(_)))
1262                && !has_expr(&proj_expr, |e| matches!(e, Expr::SubPlan(_, _)))
1263            {
1264                let value_name = format_pl_smallstr!("{prefix}val");
1265                let values = col(value_name.clone());
1266                row_columns.push(proj_expr.alias(value_name));
1267                if first_row {
1268                    let alias = match &select.projection[0] {
1269                        SelectItem::ExprWithAlias { alias, .. } => Some(alias),
1270                        _ => None,
1271                    };
1272                    let Some(ordered) = ctx.try_first_row_order(
1273                        subquery,
1274                        alias,
1275                        values,
1276                        &prefix,
1277                        &inner_names,
1278                        &inner_schema,
1279                        &mut row_columns,
1280                    )?
1281                    else {
1282                        return Ok(None);
1283                    };
1284                    (ordered.first(), false)
1285                } else {
1286                    let values = match select.distinct {
1287                        Some(Distinct::Distinct) => values.unique(),
1288                        _ => values,
1289                    };
1290                    (values.item(true), true)
1291                }
1292            } else {
1293                return Ok(None);
1294            };
1295
1296        let Some((corr_preds, local_filters)) = ctx.split_correlation_conjuncts(
1297            selection,
1298            &inner_names,
1299            &inner_schema,
1300            outer_schema,
1301            self,
1302        )?
1303        else {
1304            return Ok(None);
1305        };
1306        // No correlation: leave it to the uncorrelated scalar-subquery path.
1307        if corr_preds.is_empty() {
1308            return Ok(None);
1309        }
1310
1311        let result_name = format_pl_smallstr!("{prefix}res");
1312        // Without a semi join to restrict the inner groups, a value per row is taken
1313        // per matched outer row instead.
1314        let equality_only = corr_preds.iter().all(|p| p.op == SQLBinaryOperator::Eq)
1315            && (cfg!(feature = "semi_anti_join") || !single_row_check);
1316
1317        // When the subquery repeats a part of the outer join tree, the outer frame
1318        // holds each outer row's subquery rows, each one repeated once per match
1319        // of the row's other relations on the keys. An aggregate that ignores such
1320        // repeats is then read from the outer frame, grouped by the outer keys.
1321        if equality_only
1322            && ignores_uniform_repeats(&agg_expr)
1323            && let Some(tree) = join_tree
1324            && tree.covers(&select.from, &factored_selection(selection), &inner_schema)
1325        {
1326            let mut keys: Vec<Expr> = Vec::with_capacity(corr_preds.len());
1327            for p in &corr_preds {
1328                let key = col(p.outer.clone());
1329                if !keys.contains(&key) {
1330                    keys.push(key);
1331                }
1332            }
1333            if expr_to_leaf_column_names_iter(&agg_expr)
1334                .chain(corr_preds.iter().map(|p| p.outer.clone()))
1335                .all(|name| outer_schema.contains(&name))
1336            {
1337                let outer = lf.cache();
1338                let aggregate = ScalarAggregate::new(agg_expr, &result_name, &prefix, outer_schema);
1339                let joined = equality_correlated_aggregate(
1340                    outer.clone(),
1341                    outer,
1342                    keys.clone(),
1343                    keys,
1344                    &aggregate,
1345                    false,
1346                )?;
1347                return Ok(Some((joined, result_name)));
1348            }
1349        }
1350
1351        // Apply inner-only filters, then rename inner columns to collision-free
1352        // names so they can't clash with outer columns of the same name (as in a
1353        // self-correlated `t1 AS x` subquery over outer `t1`).
1354        let mut inner_filtered = local_filters.into_iter().fold(inner_lf, LazyFrame::filter);
1355        if !row_columns.is_empty() {
1356            inner_filtered = inner_filtered.with_columns(row_columns);
1357        }
1358        let rename_from: Vec<PlSmallStr> = inner_schema.iter_names().cloned().collect();
1359        let rename_to: Vec<PlSmallStr> = rename_from
1360            .iter()
1361            .map(|name| prefixed_inner(&prefix, name))
1362            .collect();
1363        let mut inner_renamed = inner_filtered.rename(&rename_from, &rename_to, true);
1364        // The single-row check must only see the groups that outer rows ask for.
1365        let restrict = equality_only
1366            && group_local_aggregate(&agg_expr)
1367            && (single_row_check || {
1368                let inner_rows = ctx.estimated_rows(&mut inner_renamed)?;
1369                let outer_rows = self.estimated_rows(&mut lf)?;
1370                restriction_pays_off(outer_rows.as_ref(), inner_rows.as_ref())
1371            });
1372        inner_renamed.set_cached_arena(ctx.lp_arena, ctx.expr_arena);
1373
1374        let aggregate = ScalarAggregate::new(agg_expr, &result_name, &prefix, &inner_schema)
1375            .map_inputs(|e| {
1376                e.map_expr(|e| match e {
1377                    Expr::Column(name) if inner_schema.contains(name.as_str()) => {
1378                        col(prefixed_inner(&prefix, &name))
1379                    },
1380                    other => other,
1381                })
1382            });
1383
1384        if equality_only {
1385            let outer_on: Vec<Expr> = corr_preds.iter().map(|p| col(p.outer.clone())).collect();
1386            let inner_on: Vec<Expr> = corr_preds
1387                .iter()
1388                .map(|p| col(prefixed_inner(&prefix, &p.inner)))
1389                .collect();
1390            let joined = equality_correlated_aggregate(
1391                lf,
1392                inner_renamed,
1393                outer_on,
1394                inner_on,
1395                &aggregate,
1396                restrict,
1397            )?;
1398            return Ok(Some((joined, result_name)));
1399        }
1400
1401        let join_preds: Vec<Expr> = corr_preds.iter().map(|p| p.to_expr(&prefix)).collect();
1402        let idx_name = format_pl_smallstr!("{prefix}idx");
1403
1404        let outer_indexed = row_indexed_once(lf, idx_name.clone());
1405        let matched = outer_indexed
1406            .clone()
1407            .join_builder()
1408            .with(inner_renamed)
1409            .how(JoinType::Inner)
1410            .join_where(join_preds);
1411        let grouped = matched
1412            .group_by([col(idx_name.clone())])
1413            .agg(aggregate.aggs.clone());
1414
1415        let joined = aggregate
1416            .finish(left_join_aggregate(
1417                outer_indexed,
1418                grouped,
1419                vec![col(idx_name.clone())],
1420                vec![col(idx_name.clone())],
1421            )?)
1422            .drop(Selector::ByName {
1423                names: Arc::from([idx_name]),
1424                strict: true,
1425            });
1426        Ok(Some((joined, result_name)))
1427    }

1429    // Attempt the decorrelation of a correlated `IN (subquery)` into a boolean
1430    // column. The correlated candidate values are collected per outer row into a
1431    // list, and membership is then evaluated against that list under SQL's
1432    // three-valued logic. An uncorrelated subquery is left to the generic path.
1433    fn try_decorrelate_in_subquery(
1434        &mut self,
1435        lf: LazyFrame,
1436        outer_schema: &Schema,
1437        lhs: &SQLExpr,
1438        subquery: &Query,
1439    ) -> PolarsResult<Option<(LazyFrame, PlSmallStr)>> {
1440        let Some(select) = eligible_subquery_select(subquery) else {
1441            return Ok(None);
1442        };
1443        if matches!(&select.distinct, Some(Distinct::On(_))) {
1444            return Ok(None);
1445        }
1446        let [SelectItem::UnnamedExpr(proj) | SelectItem::ExprWithAlias { expr: proj, .. }] =
1447            select.projection.as_slice()
1448        else {
1449            return Ok(None);
1450        };
1451        let Some(selection) = &select.selection else {
1452            return Ok(None);
1453        };
1454
1455        let Some(needle) = self.try_parse_outer_only_expr(lhs, outer_schema)? else {
1456            return Ok(None);
1457        };
1458
1459        let mut ctx = self.isolated();
1460        let Some((inner_names, inner_lf, inner_schema)) = ctx.resolve_subquery_from(select)? else {
1461            return Ok(None);
1462        };
1463        let Some(value) = ctx.try_parse_inner_only_expr(proj, &inner_names, &inner_schema)? else {
1464            return Ok(None);
1465        };
1466        let value = value.meta().undo_aliases();
1467
1468        let Some((corr_preds, local_filters)) = ctx.split_correlation_conjuncts(
1469            selection,
1470            &inner_names,
1471            &inner_schema,
1472            outer_schema,
1473            self,
1474        )?
1475        else {
1476            return Ok(None);
1477        };
1478        // No correlation: the generic `IN` path already handles this.
1479        if corr_preds.is_empty() {
1480            return Ok(None);
1481        }
1482
1483        let prefix = format_pl_smallstr!("{CORRELATED_COL_PREFIX}{}_", unique_column_name());
1484        let set_name = format_pl_smallstr!("{prefix}set");
1485
1486        let inner_filtered = local_filters.into_iter().fold(inner_lf, LazyFrame::filter);
1487        let rename_from: Vec<PlSmallStr> = inner_schema.iter_names().cloned().collect();
1488        let rename_to: Vec<PlSmallStr> = rename_from
1489            .iter()
1490            .map(|name| prefixed_inner(&prefix, name))
1491            .collect();
1492        let inner_renamed = inner_filtered.rename(&rename_from, &rename_to, true);
1493        inner_renamed.set_cached_arena(ctx.lp_arena, ctx.expr_arena);
1494
1495        let value = value.map_expr(|e| match e {
1496            Expr::Column(name) if inner_schema.contains(name.as_str()) => {
1497                col(prefixed_inner(&prefix, &name))
1498            },
1499            other => other,
1500        });
1501
1502        // Grouping collects the candidate values of each outer row into a list.
1503        let joined = if corr_preds.iter().all(|p| p.op == SQLBinaryOperator::Eq) {
1504            let outer_on: Vec<Expr> = corr_preds.iter().map(|p| col(p.outer.clone())).collect();
1505            let inner_on: Vec<Expr> = corr_preds
1506                .iter()
1507                .map(|p| col(prefixed_inner(&prefix, &p.inner)))
1508                .collect();
1509            let grouped = inner_renamed
1510                .group_by(inner_on.clone())
1511                .agg([value.alias(set_name.clone())]);
1512            left_join_aggregate(lf, grouped, outer_on, inner_on)?
1513        } else {
1514            let join_preds: Vec<Expr> = corr_preds.iter().map(|p| p.to_expr(&prefix)).collect();
1515            let idx_name = format_pl_smallstr!("{prefix}idx");
1516            let outer_indexed = row_indexed_once(lf, idx_name.clone());
1517            let matched = outer_indexed
1518                .clone()
1519                .join_builder()
1520                .with(inner_renamed)
1521                .how(JoinType::Inner)
1522                .join_where(join_preds);
1523            let grouped = matched
1524                .group_by([col(idx_name.clone())])
1525                .agg([value.alias(set_name.clone())]);
1526            left_join_aggregate(
1527                outer_indexed,
1528                grouped,
1529                vec![col(idx_name.clone())],
1530                vec![col(idx_name.clone())],
1531            )?
1532            .drop(Selector::ByName {
1533                names: Arc::from([idx_name]),
1534                strict: true,
1535            })
1536        };
1537
1538        // An outer row with no matching inner rows joins to null, which is the
1539        // empty candidate set rather than an unknown one.
1540        let set = col(set_name.clone());
1541        let is_in = sql_in_membership(
1542            sql_is_in(needle, set.clone()),
1543            set.clone(),
1544            set.clone().is_null().or(set.list().len().eq(lit(0u32))),
1545        );
1546        // The boolean replaces the candidate list in place.
1547        let joined = joined.with_columns([is_in.alias(set_name.clone())]);
1548        Ok(Some((joined, set_name)))
1549    }

1551    /// Whether the uncorrelated `subquery` aggregates without `GROUP BY`, and so returns
1552    /// exactly one row.
1553    pub(crate) fn exists_over_aggregate(&mut self, subquery: &Query) -> PolarsResult<bool> {
1554        let Some(select) = eligible_subquery_select(subquery) else {
1555            return Ok(false);
1556        };
1557        let mut ctx = self.isolated();
1558        let Some((inner_names, inner_lf, inner_schema)) = ctx.resolve_subquery_from(select)? else {
1559            return Ok(false);
1560        };
1561        match exists_aggregates(&ctx, subquery, select, &inner_names, &inner_schema) {
1562            Some(true) => {
1563                // The rows are not read, but WHERE must still be valid.
1564                if let Some(selection) = &select.selection {
1565                    let mut filtered = ctx.process_where(
1566                        inner_lf,
1567                        Some(selection),
1568                        FilterMode::KeepTrue,
1569                        Some(inner_schema),
1570                    )?;
1571                    ctx.get_frame_schema(&mut filtered)?;
1572                }
1573                Ok(true)
1574            },
1575            Some(false) => Ok(false),
1576            None => polars_bail!(
1577                SQLInterface:
1578                "EXISTS subquery is not currently supported in this position: {:?}", subquery
1579            ),
1580        }
1581    }
1582
1583    // Attempt the decorrelation of a single `EXISTS` subquery into a boolean flag
1584    // column holding `count(*) > 0` over the correlated match set. An uncorrelated
1585    // subquery is one value instead, kept in `exists_values` for `visit_exists`, and
1586    // `None` is returned. Also `None` for shapes that can't be soundly classified.
1587    fn try_decorrelate_exists_subquery(
1588        &mut self,
1589        lf: LazyFrame,
1590        outer_schema: &Schema,
1591        subquery: &Query,
1592    ) -> PolarsResult<Option<(LazyFrame, PlSmallStr)>> {
1593        let Some(select) = eligible_subquery_select(subquery) else {
1594            return Ok(None);
1595        };
1596
1597        let mut ctx = self.isolated();
1598        let Some((inner_names, inner_lf, inner_schema)) = ctx.resolve_subquery_from(select)? else {
1599            return Ok(None);
1600        };
1601
1602        let (corr_preds, local_filters) = match &select.selection {
1603            Some(selection) => {
1604                let Some(split) = ctx.split_correlation_conjuncts(
1605                    selection,
1606                    &inner_names,
1607                    &inner_schema,
1608                    outer_schema,
1609                    self,
1610                )?
1611                else {
1612                    return Ok(None);
1613                };
1614                split
1615            },
1616            None => (Vec::new(), Vec::new()),
1617        };
1618
1619        let prefix = format_pl_smallstr!("{CORRELATED_COL_PREFIX}{}_", unique_column_name());
1620        let flag_name = format_pl_smallstr!("{prefix}exists");
1621        // An uncorrelated subquery is one value, which `visit_exists` takes.
1622        if corr_preds.is_empty() && !is_correlated_subquery(subquery, self.catalog_spellings()) {
1623            let exists =
1624                match exists_aggregates(&ctx, subquery, select, &inner_names, &inner_schema) {
1625                    None => return Ok(None),
1626                    Some(true) => lit(true),
1627                    Some(false) => {
1628                        let inner = local_filters
1629                            .into_iter()
1630                            .fold(inner_lf, LazyFrame::filter)
1631                            .select([len().gt(lit(0))]);
1632                        inner.set_cached_arena(ctx.lp_arena, ctx.expr_arena);
1633                        reduced_subquery(inner, first().as_expr().item(true))
1634                    },
1635                };
1636            self.exists_values.push((subquery.clone(), exists));
1637            return Ok(None);
1638        }
1639        match exists_aggregates(&ctx, subquery, select, &inner_names, &inner_schema) {
1640            None => return Ok(None),
1641            // An aggregate without GROUP BY returns one row, even over no rows.
1642            Some(true) => {
1643                return Ok(Some((
1644                    lf.with_column(lit(true).alias(flag_name.clone())),
1645                    flag_name,
1646                )));
1647            },
1648            Some(false) => {},
1649        }
1650        let inner_filtered = local_filters.into_iter().fold(inner_lf, LazyFrame::filter);
1651
1652        if corr_preds.is_empty() {
1653            // Outer columns are only in the projection, which EXISTS ignores, so the
1654            // existence is a single constant boolean broadcast onto every row.
1655            let inner_flag = inner_filtered.select([len().gt(lit(0)).alias(flag_name.clone())]);
1656            inner_flag.set_cached_arena(ctx.lp_arena, ctx.expr_arena);
1657            let joined = concat_lf_horizontal(
1658                [lf, inner_flag],
1659                HConcatOptions {
1660                    broadcast_unit_length: true,
1661                    ..Default::default()
1662                },
1663            )?;
1664            return Ok(Some((joined, flag_name)));
1665        }
1666
1667        let idx_name = format_pl_smallstr!("{prefix}idx");
1668        let count_name = format_pl_smallstr!("{prefix}cnt");
1669
1670        // Rename inner columns to collision-free names so they can't clash
1671        // with outer columns of the same name (as in a self-correlated
1672        // `t1 AS x` subquery over outer `t1`).
1673        let rename_from: Vec<PlSmallStr> = inner_schema.iter_names().cloned().collect();
1674        let rename_to: Vec<PlSmallStr> = rename_from
1675            .iter()
1676            .map(|name| prefixed_inner(&prefix, name))
1677            .collect();
1678        let inner_renamed = inner_filtered.rename(&rename_from, &rename_to, true);
1679        inner_renamed.set_cached_arena(ctx.lp_arena, ctx.expr_arena);
1680
1681        let join_preds: Vec<Expr> = corr_preds.iter().map(|p| p.to_expr(&prefix)).collect();
1682
1683        let outer_indexed = row_indexed_once(lf, idx_name.clone());
1684        let matched = outer_indexed
1685            .clone()
1686            .join_builder()
1687            .with(inner_renamed)
1688            .how(JoinType::Inner)
1689            .join_where(join_preds);
1690        let grouped = matched
1691            .group_by([col(idx_name.clone())])
1692            .agg([len().alias(count_name.clone())]);
1693
1694        let joined = outer_indexed
1695            .join_builder()
1696            .with(grouped)
1697            .left_on([col(idx_name.clone())])
1698            .right_on([col(idx_name.clone())])
1699            .how(JoinType::Left)
1700            .coalesce(JoinCoalesce::CoalesceColumns)
1701            .maintain_order(MaintainOrderJoin::Left)
1702            .finish()?
1703            .with_columns([col(count_name.clone())
1704                .fill_null(lit(0))
1705                .gt(lit(0))
1706                .alias(flag_name.clone())])
1707            .drop(Selector::ByName {
1708                names: Arc::from([idx_name, count_name]),
1709                strict: true,
1710            });
1711        Ok(Some((joined, flag_name)))
1712    }

1739/// When WHERE conjuncts are offered to the join rewrites.
1740#[derive(Clone, Copy, PartialEq, Eq)]
1741pub(crate) enum RewriteStage {
1742    /// Before the correlated scalar subqueries are lowered: a conjunct moves
1743    /// ahead of them only if it is harmless on every row.
1744    BeforeScalarLowering,
1745    /// After: every conjunct is offered.
1746    AfterScalarLowering,
1747}

1749impl RewriteStage {

1751        self == Self::AfterScalarLowering || harmless_on_every_row(expr, schema)
1752    }

1755// A `[NOT] IN (subquery)` or `= (subquery)` conjunct taken apart.
1756#[cfg_attr(not(feature = "semi_anti_join"), allow(dead_code))]
1757struct Membership<'a> {
1758    /// The outer-side expression compared against the subquery.
1759    key: &'a SQLExpr,
1760    subquery: &'a Query,
1761    negated: bool,
1762    /// `= (subquery)`: the subquery stands for its single value.
1763    scalar: bool,
1764}
1765
1766fn membership_predicate(conj: &SQLExpr) -> Option<Membership<'_>> {
1767    match unwrap_nested(conj) {
1768        SQLExpr::InSubquery {
1769            expr,
1770            subquery,
1771            negated,
1772        } => Some(Membership {
1773            key: expr,
1774            subquery,
1775            negated: *negated,
1776            scalar: false,
1777        }),
1778        SQLExpr::BinaryOp {
1779            left,
1780            op: SQLBinaryOperator::Eq,
1781            right,
1782        } => match (left.as_ref(), right.as_ref()) {
1783            (key, SQLExpr::Subquery(subquery)) | (SQLExpr::Subquery(subquery), key) => {
1784                Some(Membership {
1785                    key,
1786                    subquery,
1787                    negated: false,
1788                    scalar: true,
1789                })
1790            },
1791            _ => None,
1792        },
1793        _ => None,
1794    }
1795}

1797fn unwrap_nested(mut expr: &SQLExpr) -> &SQLExpr {
1798    while let SQLExpr::Nested(inner) = expr {
1799        expr = inner;
1800    }
1801    expr
1802}

1818// Semi/anti join the outer frame against the (filtered, arena-cached) inner, or
1819// `None` for keys the join cannot stand in for the comparison on. The first key
1820// pair is the membership key, the rest correlation keys. With `correct_not_in`
1821// the anti join is refined to `NOT IN`'s three-valued logic.
1822#[cfg(feature = "semi_anti_join")]
1823#[allow(clippy::too_many_arguments)]
1824fn build_semi_anti_join(
1825    lf: &LazyFrame,
1826    inner_lf: LazyFrame,
1827    outer_schema: &Schema,
1828    inner_schema: &Schema,
1829    left_on: Vec<Expr>,
1830    right_on: Vec<Expr>,
1831    anti: bool,
1832    correct_not_in: bool,
1833) -> PolarsResult<Option<LazyFrame>> {
1834    let join_type = if anti { JoinType::Anti } else { JoinType::Semi };
1835    let Some((left_on, right_on)) =
1836        try_prepare_join_keys(outer_schema, inner_schema, left_on, right_on)
1837    else {
1838        return Ok(None);
1839    };
1840    let joined = lf
1841        .clone()
1842        .join_builder()
1843        .with(inner_lf.clone())
1844        .left_on(left_on.clone())
1845        .right_on(right_on.clone())
1846        .how(join_type)
1847        .finish()?;
1848    if !correct_not_in {
1849        return Ok(Some(joined));
1850    }
1851    let (left_key, corr_outer) = left_on.split_first().unwrap();
1852    let (right_key, corr_inner) = right_on.split_first().unwrap();
1853    Ok(Some(refine_not_in_anti_join(
1854        joined, inner_lf, left_key, right_key, corr_outer, corr_inner,
1855    )?))
1856}

1858// The keys of a semi / anti join standing in for a comparison, numeric pairs cast
1859// to their supertype as the comparison would. `None` for a pair the join cannot
1860// stand in for: not elementwise over columns, reading no column, holding an
1861// untyped literal (whose type the comparison would take from the other operand),
1862// of unresolvable types, or of different non-numeric types.
1863fn try_prepare_join_keys(
1864    outer_schema: &Schema,
1865    inner_schema: &Schema,
1866    mut left_on: Vec<Expr>,
1867    mut right_on: Vec<Expr>,
1868) -> Option<(Vec<Expr>, Vec<Expr>)> {
1869    let untyped_literal = |e: &Expr| matches!(e, Expr::Literal(LiteralValue::Dyn(_)));
1870    for (left, right) in left_on.iter_mut().zip(right_on.iter_mut()) {
1871        if !is_elementwise(left)
1872            || !is_elementwise(right)
1873            || expr_to_leaf_column_names_iter(left).next().is_none()
1874            || has_expr(left, untyped_literal)
1875            || has_expr(right, untyped_literal)
1876        {
1877            return None;
1878        }
1879        let l = left.to_field(outer_schema).ok()?.dtype;
1880        let r = right.to_field(inner_schema).ok()?.dtype;
1881        if l == r {
1882            continue;
1883        }
1884        if !(l.is_primitive_numeric() && r.is_primitive_numeric()) {
1885            return None;
1886        }
1887        let st = try_get_supertype(&l, &r).ok()?;
1888        *left = std::mem::take(left).cast(st.clone());
1889        *right = std::mem::take(right).cast(st);
1890    }
1891    Some((left_on, right_on))
1892}

1894// Account for 3VL interaction with NULL values
1895#[cfg(feature = "semi_anti_join")]
1896fn refine_not_in_anti_join(
1897    joined: LazyFrame,
1898    inner_lf: LazyFrame,
1899    left_key: &Expr,
1900    right_key: &Expr,
1901    corr_outer: &[Expr],
1902    corr_inner: &[Expr],
1903) -> PolarsResult<LazyFrame> {
1904    if corr_inner.is_empty() {
1905        // Uncorrelated
1906        let flag_name = unique_column_name();
1907        let flag = when(len().eq(lit(0u32)))
1908            .then(lit(NULL).cast(DataType::Boolean))
1909            .otherwise(right_key.clone().is_null().any(true))
1910            .alias(flag_name.clone());
1911        let keep = when(col(flag_name.clone()).is_null())
1912            .then(lit(true)) // empty set
1913            .when(col(flag_name.clone())) // set has a NULL
1914            .then(lit(false))
1915            .otherwise(left_key.clone().is_not_null());
1916
1917        return Ok(joined
1918            .join_builder()
1919            .with(inner_lf.select([flag]))
1920            .how(JoinType::Cross)
1921            .finish()?
1922            .filter(keep)
1923            .drop(Selector::ByName {
1924                names: [flag_name].into(),
1925                strict: true,
1926            }));
1927    }
1928
1929    // Correlated
1930    let corr_keys = |lf: LazyFrame| lf.select(corr_inner).unique(None, UniqueKeepStrategy::Any);
1931    let exclude_groups = |rows: LazyFrame, groups: LazyFrame| {
1932        rows.join_builder()
1933            .with(groups)
1934            .left_on(corr_outer)
1935            .right_on(corr_inner)
1936            .how(JoinType::Anti)
1937            .finish()
1938    };
1939    let kept_non_null = exclude_groups(
1940        joined.clone().filter(left_key.clone().is_not_null()),
1941        corr_keys(inner_lf.clone().filter(right_key.clone().is_null())),
1942    )?;
1943    let kept_null = exclude_groups(
1944        joined.filter(left_key.clone().is_null()),
1945        corr_keys(inner_lf),
1946    )?;
1947
1948    Ok(concat(
1949        [kept_non_null, kept_null],
1950        UnionArgs {
1951            rechunk: false,
1952            parallel: true,
1953            ..Default::default()
1954        },
1955    )
1956    .expect("'NOT IN' 3VL union has identical schemas"))
1957}

2029/// Rewrite `(A AND X) OR (A AND Y)` as `A AND (X OR Y)`.
2030///
2031/// The SQL-AST analogue of `polars_plan::plans::aexpr::or_factoring`, which runs
2032/// after the decorrelation decision made here.
2033fn factor_common_conjuncts(expr: &SQLExpr) -> Option<SQLExpr> {
2034    let branches: Vec<Vec<&SQLExpr>> = DisjunctIter::new(expr)
2035        .map(|branch| MintermIter::new(branch).collect())
2036        .collect();
2037    let (first, rest) = branches.split_first()?;
2038    if rest.is_empty() {
2039        return None;
2040    }
2041    let common: Vec<&SQLExpr> = first
2042        .iter()
2043        .copied()
2044        .filter(|term| rest.iter().all(|branch| branch.contains(term)))
2045        .collect();
2046    let factored = combine_conditions(
2047        common.iter().copied().cloned().collect(),
2048        SQLBinaryOperator::And,
2049    )?;
2050
2051    // A branch holding nothing but the common terms absorbs the others:
2052    // `A OR (A AND Y)` is just `A`.
2053    let remainders: Option<Vec<SQLExpr>> = branches
2054        .iter()
2055        .map(|terms| {
2056            combine_conditions(
2057                terms
2058                    .iter()
2059                    .copied()
2060                    .filter(|term| !common.contains(term))
2061                    .cloned()
2062                    .collect(),
2063                SQLBinaryOperator::And,
2064            )
2065        })
2066        .collect();
2067    let Some(remainders) = remainders else {
2068        return Some(factored);
2069    };
2070    let disjunction = combine_conditions(remainders, SQLBinaryOperator::Or)?;
2071    Some(SQLExpr::BinaryOp {
2072        left: Box::new(factored),
2073        op: SQLBinaryOperator::And,
2074        right: Box::new(SQLExpr::Nested(Box::new(disjunction))),
2075    })
2076}

2078/// Hoist predicates shared by every branch of an `OR` to the top level.
2079fn factored_selection(selection: &SQLExpr) -> Cow<'_, SQLExpr> {
2080    let factored: Vec<Option<SQLExpr>> = MintermIter::new(selection)
2081        .map(factor_common_conjuncts)
2082        .collect();
2083    if factored.iter().all(Option::is_none) {
2084        return Cow::Borrowed(selection);
2085    }
2086    let conjuncts = MintermIter::new(selection)
2087        .zip(factored)
2088        .map(|(conj, factored)| factored.unwrap_or_else(|| conj.clone()))
2089        .collect();
2090    combine_conditions(conjuncts, SQLBinaryOperator::And)
2091        .map_or(Cow::Borrowed(selection), Cow::Owned)
2092}

2174// Classify a correlation operand as an inner- or outer-query column and return
2175// its column name in that frame. `None` for non-identifiers or names in neither
2176// schema.
2177fn classify_correlation_column(
2178    expr: &SQLExpr,
2179    inner_schema: &Schema,
2180    outer_schema: &Schema,
2181    inner: &SQLContext,
2182    outer: &SQLContext,
2183) -> PolarsResult<Option<(CorrelationSide, PlSmallStr)>> {
2184    let (qualifier, name) = match expr {
2185        SQLExpr::Identifier(ident) => (None, ident.value.as_str()),
2186        SQLExpr::CompoundIdentifier(parts) => {
2187            // A relation of the outer query is outer (so same-named columns like
2188            // `o.id = c.id` resolve). Any other qualifier, or a struct column of the
2189            // subquery, is not an outer column.
2190            let (side, q, len) = match inner.resolve_qualifier(parts)? {
2191                Some((q, len)) => (CorrelationSide::Inner, q, len),
2192                None => match outer.resolve_qualifier(parts)? {
2193                    Some((q, len))
2194                        if !matches!(
2195                            inner_schema.get(&parts[0].value),
2196                            Some(DataType::Struct(_))
2197                        ) =>
2198                    {
2199                        (CorrelationSide::Outer, q, len)
2200                    },
2201                    _ => return Ok(None),
2202                },
2203            };
2204            // A struct field is not a column of either frame.
2205            if len + 1 != parts.len() {
2206                return Ok(None);
2207            }
2208            (Some((side, q)), parts[len].value.as_str())
2209        },
2210        _ => return Ok(None),
2211    };
2212    // A joined relation's column may be stored under another name.
2213    Ok(match qualifier {
2214        Some((CorrelationSide::Inner, q)) => {
2215            let name = inner.resolve_name(&q, name);
2216            inner_schema
2217                .contains(name.as_str())
2218                .then(|| (CorrelationSide::Inner, name.into()))
2219        },
2220        Some((CorrelationSide::Outer, q)) => {
2221            let name = outer.resolve_name(&q, name);
2222            outer_schema
2223                .contains(name.as_str())
2224                .then(|| (CorrelationSide::Outer, name.into()))
2225        },
2226        // An unqualified name binds to the innermost scope that holds it. An
2227        // outer one raises if it is ambiguous between the outer joins.
2228        None => match (inner_schema.contains(name), outer_schema.contains(name)) {
2229            (true, _) => Some((CorrelationSide::Inner, name.into())),
2230            (false, true) => {
2231                outer.check_unqualified_column(name)?;
2232                Some((CorrelationSide::Outer, name.into()))
2233            },
2234            (false, false) => None,
2235        },
2236    })
2237}

2431// `eligible_subquery_select` without the limit check. Exhaustive destructuring
2432// (no `..`) is on purpose: a new sqlparser clause must not compile until it gets
2433// an explicit keep-or-bail decision here.
2434fn eligible_subquery_body(subquery: &Query) -> Option<&Select> {
2435    let Query {
2436        with, // CTEs aren't resolved inside the rewrite: bail
2437        body,
2438        order_by: _,      // row order can't affect existence/membership
2439        limit_clause: _,  // checked by the callers
2440        fetch: _,         // checked by the callers
2441        locks: _,         // row locking doesn't change the rows
2442        for_clause,       // FOR XML/JSON reshape the result: bail
2443        settings,         // ClickHouse SETTINGS can change results: bail
2444        format_clause: _, // output serialization only
2445        pipe_operators,   // `|>` operators transform the rows: bail
2446    } = subquery;
2447    if with.is_some() || for_clause.is_some() || settings.is_some() || !pipe_operators.is_empty() {
2448        return None;
2449    }
2450    let SetExpr::Select(select) = body.as_ref() else {
2451        return None;
2452    };
2453    let Select {
2454        select_token: _,
2455        // Deduplication is existence/membership-invariant; the IN rewrite
2456        // separately bails on `DISTINCT ON`, which is not.
2457        distinct: _,
2458        top,                    // TOP is LIMIT spelled differently: bail
2459        top_before_distinct: _, // only meaningful with `top`
2460        // The projection is validated by the callers: EXISTS ignores it, IN
2461        // requires a single plain expression (which also rules out wildcards
2462        // and the `exclude` modifier).
2463        projection: _,
2464        exclude: _,
2465        into,          // SELECT INTO is not a pure subquery: bail
2466        from,          // at least one relation, comma-separated ones cross join
2467        lateral_views, // row-multiplying: bail
2468        prewhere,      // an extra filter we don't fold in: bail
2469        selection: _,  // split into join keys/filters by callers
2470        group_by,      // aggregation changes the yielded rows: bail
2471        cluster_by: _, // layout/order hints: row-set preserving
2472        distribute_by: _,
2473        sort_by: _,
2474        having,                   // aggregation filter: bail
2475        named_window: _,          // definitions only; uses are parsed later
2476        qualify,                  // post-window filter changes the rows: bail
2477        window_before_qualify: _, // only meaningful with `qualify`
2478        value_table_mode,         // changes what a row is: bail
2479        connect_by,               // hierarchical recursion: bail
2480        optimizer_hints,          // unsupported: bail
2481        select_modifiers,         // unsupported: bail
2482        flavor: _,                // surface syntax only
2483    } = select.as_ref();
2484    let no_group_by = matches!(
2485        group_by,
2486        GroupByExpr::Expressions(e, m) if e.is_empty() && m.is_empty()
2487    );
2488    if from.is_empty()
2489        || !no_group_by
2490        || top.is_some()
2491        || into.is_some()
2492        || having.is_some()
2493        || qualify.is_some()
2494        || prewhere.is_some()
2495        || !connect_by.is_empty()
2496        || value_table_mode.is_some()
2497        || !lateral_views.is_empty()
2498        || !optimizer_hints.is_empty()
2499        || select_modifiers.is_some()
2500    {
2501        return None;
2502    }
2503    Some(select)
2504}

2536/// Which input of a join the correlated subqueries in `conj` read, together with the
2537/// left operand of an `IN`: `Some(true)` for the left input, `Some(false)` for the right
2538/// one. `None` if they read neither or both, or a relation whose columns can't be found
2539/// by name.
2540fn correlated_join_side(
2541    ctx: &SQLContext,
2542    conj: &SQLExpr,
2543    tbl_left: &TableInfo,
2544    tbl_right: &TableInfo,
2545) -> Option<bool> {
2546    struct JoinSides<'a> {
2547        ctx: &'a SQLContext,
2548        tbl_left: &'a TableInfo,
2549        tbl_right: &'a TableInfo,
2550        query_depth: usize,
2551        is_correlated: bool,
2552        unresolved: bool,
2553        reads_left: bool,
2554        reads_right: bool,
2555    }
2556
2557    impl Visitor for JoinSides<'_> {
2558        type Break = ();
2559
2560        fn pre_visit_query(&mut self, _query: &Query) -> ControlFlow<()> {
2561            self.query_depth += 1;
2562            ControlFlow::Continue(())
2563        }
2564
2565        fn post_visit_query(&mut self, _query: &Query) -> ControlFlow<()> {
2566            self.query_depth -= 1;
2567            ControlFlow::Continue(())
2568        }
2569
2570        fn pre_visit_expr(&mut self, expr: &SQLExpr) -> ControlFlow<()> {
2571            // A nested subquery's qualifiers may name relations of the subquery
2572            // around it.
2573            if self.query_depth > 0 {
2574                return ControlFlow::Continue(());
2575            }
2576            let (subquery, lhs) = match expr {
2577                SQLExpr::Subquery(subquery) | SQLExpr::Exists { subquery, .. } => (subquery, None),
2578                SQLExpr::InSubquery {
2579                    expr: lhs,
2580                    subquery,
2581                    ..
2582                } => (subquery, Some(lhs)),
2583                _ => return ControlFlow::Continue(()),
2584            };
2585            visit_foreign_qualifiers(subquery, self.ctx.catalog_spellings(), |parts| {
2586                self.is_correlated = true;
2587                match self.ctx.resolve_qualifier(parts) {
2588                    // A struct field path is not resolved.
2589                    Ok(Some((name, len))) if len + 1 == parts.len() => {
2590                        if self.ctx.is_right_relation(&name, self.tbl_right) {
2591                            self.reads_right = true;
2592                        } else if self.ctx.is_joined_relation(&name, self.tbl_left) {
2593                            self.reads_left = true;
2594                        } else {
2595                            self.unresolved = true;
2596                        }
2597                    },
2598                    _ => self.unresolved = true,
2599                }
2600                ControlFlow::Continue(())
2601            });
2602            // The left operand of a lowered `IN` is read on the same input.
2603            if let Some(lhs) = lhs {
2604                let (reads_left, reads_right) =
2605                    self.ctx
2606                        .join_inputs_read(lhs, self.tbl_left, self.tbl_right);
2607                self.reads_left |= reads_left;
2608                self.reads_right |= reads_right;
2609            }
2610            ControlFlow::Continue(())
2611        }
2612    }
2613
2614    let mut sides = JoinSides {
2615        ctx,
2616        tbl_left,
2617        tbl_right,
2618        query_depth: 0,
2619        is_correlated: false,
2620        unresolved: false,
2621        reads_left: false,
2622        reads_right: false,
2623    };
2624    let _ = dyn_visit::visit(conj, &mut sides);
2625    if !sides.is_correlated || sides.unresolved {
2626        return None;
2627    }
2628    match (sides.reads_left, sides.reads_right) {
2629        (true, false) => Some(true),
2630        (false, true) => Some(false),
2631        _ => None,
2632    }
2633}

2741/// Rewrites correlated subqueries in-place into references to the columns their
2742/// decorrelated lowering materialises, threading the growing `LazyFrame` through the
2743/// traversal. `query_depth` keeps each subquery a leaf rather than recursing into it.
2744/// `skips` holds, per expression being visited, whether it was skipped as a whole.
2745struct CorrelatedLowering<'a> {
2746    ctx: &'a mut SQLContext,
2747    lf: LazyFrame,
2748    outer_schema: &'a Schema,
2749    scope: LowerScope,
2750    bindings: &'a mut SubqueryBindings,
2751    join_tree: Option<&'a OuterJoinTree>,
2752    query_depth: usize,
2753    skips: Vec<bool>,
2754    changed: bool,
2755}
2756
2757impl CorrelatedLowering<'_> {
2758    /// Lower `subquery` (or reuse an earlier identical lowering), returning the
2759    /// materialised column name, or `None` if it can't be soundly lowered.
2760    fn lower(&mut self, subquery: &Query, kind: SubqueryKind) -> PolarsResult<Option<PlSmallStr>> {
2761        if let Some((_, _, name)) = self
2762            .bindings
2763            .iter()
2764            .find(|(q, bound, _)| *bound == kind && q == subquery)
2765        {
2766            return Ok(Some(name.clone()));
2767        }
2768        // The frame is cloned rather than moved so it survives a declined lowering.
2769        let lowered = match &kind {
2770            SubqueryKind::Exists => self.ctx.try_decorrelate_exists_subquery(
2771                self.lf.clone(),
2772                self.outer_schema,
2773                subquery,
2774            )?,
2775            SubqueryKind::Scalar => self.ctx.try_decorrelate_scalar_subquery(
2776                self.lf.clone(),
2777                self.outer_schema,
2778                subquery,
2779                self.join_tree,
2780            )?,
2781            SubqueryKind::In(lhs) => self.ctx.try_decorrelate_in_subquery(
2782                self.lf.clone(),
2783                self.outer_schema,
2784                lhs,
2785                subquery,
2786            )?,
2787        };
2788        let Some((new_lf, name)) = lowered else {
2789            return Ok(None);
2790        };
2791        self.lf = new_lf;
2792        self.bindings.push((subquery.clone(), kind, name.clone()));
2793        Ok(Some(name))
2794    }
2795
2796    /// Replace `expr` by the column of its lowering when it is a lowerable subquery.
2797    fn lower_expr(&mut self, expr: &mut SQLExpr) -> ControlFlow<PolarsError> {
2798        // Inside a nested query: not ours to lower against the outer schema.
2799        if self.query_depth > 0 || !lowerable_kind(expr, self.scope) {
2800            return ControlFlow::Continue(());
2801        }
2802        let (subquery, kind, negated) = match &*expr {
2803            SQLExpr::Subquery(subquery) => (subquery.as_ref(), SubqueryKind::Scalar, false),
2804            SQLExpr::Exists { subquery, negated } => {
2805                (subquery.as_ref(), SubqueryKind::Exists, *negated)
2806            },
2807            SQLExpr::InSubquery {
2808                expr: lhs,
2809                subquery,
2810                negated,
2811            } => (subquery.as_ref(), SubqueryKind::In(lhs.clone()), *negated),
2812            _ => unreachable!("guarded by lowerable_kind"),
2813        };
2814        let name = match self.lower(subquery, kind) {
2815            Ok(Some(name)) => name,
2816            Ok(None) => return ControlFlow::Continue(()),
2817            Err(e) => return ControlFlow::Break(e),
2818        };
2819        // `EXISTS` is never NULL; `IN` may be, and negating a NULL keeps it NULL,
2820        // which is what SQL asks for.
2821        let resolved = SQLExpr::Identifier(Ident::new(name.as_str()));
2822        *expr = if negated {
2823            SQLExpr::UnaryOp {
2824                op: SQLUnaryOperator::Not,
2825                expr: Box::new(resolved),
2826            }
2827        } else {
2828            resolved
2829        };
2830        self.changed = true;
2831        ControlFlow::Continue(())
2832    }
2833}

2835impl VisitorMut for CorrelatedLowering<'_> {
2836    type Break = PolarsError;
2837
2838    fn pre_visit_query(&mut self, _query: &mut Query) -> ControlFlow<PolarsError> {
2839        self.query_depth += 1;
2840        ControlFlow::Continue(())
2841    }
2842
2843    fn post_visit_query(&mut self, _query: &mut Query) -> ControlFlow<PolarsError> {
2844        self.query_depth -= 1;
2845        ControlFlow::Continue(())
2846    }
2847
2848    fn pre_visit_expr(&mut self, expr: &mut SQLExpr) -> ControlFlow<PolarsError> {
2849        // The subquery of `x > ANY (subquery)` is a set of values rather than a
2850        // scalar: only the left operand is lowered, and the rest is skipped like a
2851        // nested query.
2852        let skip = self.query_depth == 0
2853            && matches!(
2854                expr,
2855                SQLExpr::AnyOp { right, .. } | SQLExpr::AllOp { right, .. }
2856                    if matches!(strip_nested(right), SQLExpr::Subquery(_))
2857            );

2859            dyn_visit::visit_mut(left.as_mut(), self)?;
2860            self.query_depth += 1;
2861        }
2862        self.skips.push(skip);
2863        self.lower_expr(expr)
2864    }
2865
2866    fn post_visit_expr(&mut self, _expr: &mut SQLExpr) -> ControlFlow<PolarsError> {
2867        if self.skips.pop() == Some(true) {
2868            self.query_depth -= 1;
2869        }
2870        ControlFlow::Continue(())
2871    }

2874// Whether an expression is elementwise: one value per row, from that row alone.
2875// Join keys must be, so that every key is as long as the frame it is built from.
2876// Anything this does not recognise declines the rewrite, which costs an
2877// optimisation rather than correctness.
2878fn is_elementwise(expr: &Expr) -> bool {
2879    match expr {
2880        Expr::Column(_) | Expr::Literal(_) => true,
2881        Expr::Alias(inner, _) | Expr::Cast { expr: inner, .. } => is_elementwise(inner),
2882        Expr::BinaryExpr { left, op: _, right } => is_elementwise(left) && is_elementwise(right),
2883        // SQL operators that are lowered once the dtypes are known are elementwise.
2884        Expr::Function {
2885            input,
2886            function: FunctionExpr::Sql(_),
2887        } => input.iter().all(is_elementwise),
2888        Expr::Ternary {
2889            predicate,
2890            truthy,
2891            falsy,
2892        } => is_elementwise(predicate) && is_elementwise(truthy) && is_elementwise(falsy),
2893        _ => false,
2894    }
2895}

2923/// The relations and filters of a query block whose frame is its FROM relations
2924/// joined and filtered by its WHERE conjuncts that hold no subquery, and nothing
2925/// more. A correlated scalar subquery over some of these relations can then be
2926/// answered from the frame itself.
2927pub(crate) struct OuterJoinTree {
2928    relations: Vec<TreeRelation>,
2929    conjuncts: Vec<SpelledConjunct>,
2930}
2931
2932struct TreeRelation {
2933    // The name of the table it reads.
2934    table: String,
2935    // The name that qualifies its columns: its alias, else its table name.
2936    name: String,
2937    schema: SchemaRef,
2938}

2940// A conjunct with each column spelled `<relation index>.<column>`, so that one
2941// condition written in two scopes, under different aliases, compares equal.
2942struct SpelledConjunct {
2943    expr: SQLExpr,
2944    // The relations it reads through its own scope's FROM.
2945    own: Vec<usize>,
2946    // The relations it reads as correlated outer columns.
2947    outer: Vec<usize>,
2948}
2949
2950impl OuterJoinTree {
2951    // Whether a subquery over `from` filtered by `selection` reads, for each row
2952    // of the frame, exactly the rows of its relations that the row was joined
2953    // with: its relations are tree relations, its filters are exactly the
2954    // tree's conditions on those relations alone, and its correlation
2955    // predicates are exactly the tree's conditions linking them to the other
2956    // relations. `inner_schema` is the schema the subquery's FROM resolved to.
2957    fn covers(&self, from: &[TableWithJoins], selection: &SQLExpr, inner_schema: &Schema) -> bool {
2958        let mut own: Vec<(&str, usize)> = Vec::with_capacity(from.len());
2959        for tbl_expr in from {
2960            let Some((table, name)) = plain_table(tbl_expr) else {
2961                return false;
2962            };
2963            let Some(idx) = self.relations.iter().position(|r| r.table == table) else {
2964                return false;
2965            };
2966            if own.iter().any(|&(n, i)| n == name || i == idx) {
2967                return false;
2968            }
2969            own.push((name, idx));
2970        }
2971        // An alias spelled like a table name can make the subquery read another table.
2972        let n_columns: usize = own
2973            .iter()
2974            .map(|&(_, idx)| self.relations[idx].schema.len())
2975            .sum();
2976        let same_tables = own.iter().all(|&(_, idx)| {
2977            self.relations[idx]
2978                .schema
2979                .iter()
2980                .all(|(name, dtype)| inner_schema.get(name) == Some(dtype))
2981        });
2982        if n_columns != inner_schema.len() || !same_tables {
2983            return false;
2984        }
2985
2986        let is_own = |idx: &usize| own.iter().any(|&(_, i)| i == *idx);
2987        let mut filters = Vec::new();
2988        let mut links = Vec::new();
2989        for conj in MintermIter::new(selection) {
2990            let Some(spelled) = spell_conjunct(conj, &own, &self.relations, true) else {
2991                return false;
2992            };
2993            if spelled.outer.is_empty() {
2994                filters.push(spelled.expr);
2995            } else if spelled.outer.iter().any(is_own) {
2996                // The outer row's column of a relation the subquery reads itself.
2997                return false;
2998            } else {
2999                links.push(spelled.expr);
3000            }
3001        }
3002        let mut tree_filters = Vec::new();
3003        let mut tree_links = Vec::new();
3004        for conj in &self.conjuncts {
3005            match conj.own.iter().filter(|idx| is_own(idx)).count() {
3006                n if n == conj.own.len() => tree_filters.push(&conj.expr),
3007                0 => {},
3008                _ => tree_links.push(&conj.expr),
3009            }
3010        }
3011        let same = |ours: &[SQLExpr], tree: &[&SQLExpr]| {

3041// Spell a conjunct's columns by relation index. A column resolves against the
3042// relations its scope declares, `own` as (name, relation index), and then, when
3043// `correlated`, against all relations as an outer column. `None` when a column
3044// doesn't resolve to exactly one relation, or the conjunct holds a subquery.
3045fn spell_conjunct(
3046    expr: &SQLExpr,
3047    own: &[(&str, usize)],
3048    relations: &[TreeRelation],
3049    correlated: bool,
3050) -> Option<SpelledConjunct> {
3051    let has = |idx: usize, column: &str| relations[idx].schema.contains(column);
3052    let resolve = |qualifier: Option<&str>, column: &str| -> Option<(usize, bool)> {
3053        let own_matches: Vec<usize> = own
3054            .iter()
3055            .filter(|&&(name, idx)| qualifier.map_or(has(idx, column), |q| q == name))
3056            .map(|&(_, idx)| idx)
3057            .collect();
3058        match own_matches.as_slice() {
3059            [idx] => return has(*idx, column).then_some((*idx, false)),
3060            [] if correlated => {},
3061            _ => return None,
3062        }
3063        let outer_matches: Vec<usize> = (0..relations.len())
3064            .filter(|&idx| qualifier.is_none_or(|q| relations[idx].name == q) && has(idx, column))
3065            .collect();
3066        match outer_matches.as_slice() {
3067            [idx] => Some((*idx, true)),
3068            _ => None,
3069        }
3070    };
3071
3072    let mut spelled = expr.clone();
3073    let mut own_read = Vec::new();
3074    let mut outer_read = Vec::new();
3075    let flow = visit_expressions_mut(&mut spelled, |e| {
3076        if is_subquery_expr(e) {
3077            return ControlFlow::Break(());
3078        }
3079        if let SQLExpr::Nested(inner) = e {
3080            let inner = inner.as_ref().clone();
3081            *e = inner;
3082            return ControlFlow::Continue(());
3083        }
3084        // An (in)equality reads the same either way round.
3085        if let SQLExpr::BinaryOp {
3086            left,
3087            op: SQLBinaryOperator::Eq | SQLBinaryOperator::NotEq,
3088            right,
3089        } = e
3090        {
3091            if left.to_string() > right.to_string() {
3092                std::mem::swap(left, right);
3093            }
3094            return ControlFlow::Continue(());
3095        }
3096        let resolved = match &*e {
3097            SQLExpr::Identifier(ident) => resolve(None, &ident.value).zip(Some(ident.clone())),
3098            SQLExpr::CompoundIdentifier(parts) => match parts.as_slice() {
3099                [qualifier, ident] => {
3100                    resolve(Some(&qualifier.value), &ident.value).zip(Some(ident.clone()))
3101                },
3102                _ => None,
3103            },
3104            _ => return ControlFlow::Continue(()),
3105        };
3106        let Some(((idx, outer), ident)) = resolved else {
3107            return ControlFlow::Break(());
3108        };
3109        if outer {
3110            outer_read.push(idx);
3111        } else {
3112            own_read.push(idx);
3113        }
3114        *e = SQLExpr::CompoundIdentifier(vec![Ident::new(idx.to_string()), ident]);
3115        ControlFlow::Continue(())
3116    });
3117    flow.is_continue().then_some(SpelledConjunct {
3118        expr: spelled,
3119        own: own_read,
3120        outer: outer_read,
3121    })
3122}

3166// Restrict the inner frame to the rows whose keys occur in the outer frame by a
3167// semi join against the outer keys. The outer frame is cached so that both the
3168// restriction and the join back read one evaluation of it. A semi join never
3169// duplicates inner rows.
3170#[cfg(feature = "semi_anti_join")]
3171fn restrict_to_outer_keys(
3172    outer: LazyFrame,
3173    inner: LazyFrame,
3174    outer_on: &[Expr],
3175    inner_on: &[Expr],
3176) -> PolarsResult<(LazyFrame, LazyFrame)> {
3177    let outer = outer.cache();
3178    // One outer column may serve several key pairs; it is projected once.
3179    let mut requested: Vec<Expr> = Vec::with_capacity(outer_on.len());
3180    for key in outer_on {
3181        if !requested.contains(key) {
3182            requested.push(key.clone());
3183        }
3184    }
3185    let requested_keys = outer.clone().select(requested);
3186    let restricted = inner
3187        .join_builder()
3188        .with(requested_keys)
3189        .left_on(inner_on)
3190        .right_on(outer_on)
3191        .how(JoinType::Semi)
3192        .finish()?;
3193    Ok((outer, restricted))
3194}

3235// The projection of a correlated scalar aggregate subquery, as the aggregates
3236// computed per group plus what turns them into the result after the left join.
3237//
3238// An outer row without inner rows must get the projection's value over no
3239// rows. The left join gives NULL there. That is right for most projections,
3240// but not for `COUNT(*)`, `COUNT(*) + 1` or `COALESCE(MAX(x), 0)`. For those,
3241// each aggregate gets its value over no rows, and a projection that combines
3242// several aggregates is evaluated per outer row after the join. It then only
3243// runs on outer rows that exist, so it can't fail on a value no row needs.
3244struct ScalarAggregate {
3245    // Computed per group.
3246    aggs: Vec<Expr>,
3247    // Columns of `aggs` and the value they take for an outer row without inner
3248    // rows, where that is not NULL.
3249    fill: Vec<(PlSmallStr, Expr)>,
3250    // The result over the `aggs` columns, and those columns to drop after.
3251    combine: Option<(Expr, Vec<PlSmallStr>)>,
3252}
3253
3254impl ScalarAggregate {
3255    // `schema` is the frame `agg` reads.
3256    fn new(agg: Expr, result_name: &PlSmallStr, prefix: &str, schema: &Schema) -> Self {
3257        let whole = |fill: Vec<(PlSmallStr, Expr)>| Self {
3258            aggs: vec![agg.clone().alias(result_name.clone())],
3259            fill,
3260            combine: None,
3261        };
3262        match agg_output_root(&agg) {
3263            e if zero_over_no_rows(e) && widening_integer_casts(&agg, schema) => {
3264                return whole(vec![(result_name.clone(), lit(0))]);
3265            },
3266            // A cast of NULL is NULL.
3267            e @ Expr::Agg(_) if !zero_over_no_rows(e) => return whole(vec![]),
3268            _ => {},
3269        }
3270
3271        let mut aggs = Vec::new();
3272        let combine = split_aggregates(agg.clone(), prefix, &mut aggs);
3273        let Ok(fields) = aggs
3274            .iter()
3275            .map(|e| e.to_field(schema))
3276            .collect::<PolarsResult<Vec<_>>>()
3277        else {
3278            return whole(vec![]);
3279        };
3280        let agg_schema = Schema::from_iter(fields);
3281        let fill: Vec<(PlSmallStr, Expr)> = aggs
3282            .iter()
3283            .zip(agg_schema.iter())
3284            .filter(|(e, _)| zero_over_no_rows(agg_output_root(e)))
3285            .map(|(_, (name, dtype))| (name.clone(), lit(0).cast(dtype.clone())))
3286            .collect();
3287        if !is_elementwise_over(&combine, &agg_schema) {
3288            return whole(vec![]);
3289        }
3290        // Only a cheap expression is evaluated here. Any other one is split,
3291        // which is right whatever its value over no rows is.
3292        if is_cheap(&combine) && null_over_no_rows(&combine, &agg_schema, &fill) {
3293            return whole(vec![]);
3294        }
3295        let names = agg_schema.iter_names().cloned().collect();
3296        Self {
3297            aggs,
3298            fill,
3299            combine: Some((combine.alias(result_name.clone()), names)),
3300        }
3301    }

3303    // Apply `f` to every per-group aggregate.
3304    fn map_inputs(mut self, f: impl Fn(Expr) -> Expr) -> Self {
3305        self.aggs = self.aggs.into_iter().map(f).collect();

3307    }

3310    fn finish(&self, joined: LazyFrame) -> LazyFrame {

3312            joined
3313        } else {
3314            joined.with_columns(
3315                self.fill
3316                    .iter()
3317                    .map(|(name, value)| col(name.clone()).fill_null(value.clone()))
3318                    .collect::<Vec<_>>(),
3319            )
3320        };
3321        match &self.combine {
3322            Some((combine, names)) => joined.with_column(combine.clone()).drop(Selector::ByName {
3323                names: Arc::from(names.as_slice()),

3326            None => joined,
3327        }
3328    }

3460// A correlation conjunct `<inner col> <cmp> <outer col>` (either way round),
3461// tracking which side is the inner column so the comparison can be rebuilt in
3462// its original orientation as a join predicate.
3463struct CorrPredicate {
3464    outer: PlSmallStr,
3465    inner: PlSmallStr,
3466    op: SQLBinaryOperator,
3467    inner_on_left: bool,
3468}
3469
3470impl CorrPredicate {
3471    // Whether both conjuncts compare the same columns the same way; an equality
3472    // reads the same from either side.
3473    fn same_comparison(&self, other: &Self) -> bool {
3474        let symmetric = self.op == SQLBinaryOperator::Eq && other.op == SQLBinaryOperator::Eq;
3475        self.outer == other.outer
3476            && self.inner == other.inner
3477            && (symmetric || self.op == other.op && self.inner_on_left == other.inner_on_left)
3478    }
3479
3480    fn to_expr(&self, prefix: &str) -> Expr {
3481        let outer = col(self.outer.clone());
3482        let inner = col(prefixed_inner(prefix, &self.inner));
3483        let (l, r) = if self.inner_on_left {
3484            (inner, outer)
3485        } else {
3486            (outer, inner)
3487        };
3488        match self.op {
3489            SQLBinaryOperator::Eq => l.eq(r),
3490            SQLBinaryOperator::NotEq => l.neq(r),
3491            SQLBinaryOperator::Lt => l.lt(r),
3492            SQLBinaryOperator::LtEq => l.lt_eq(r),
3493            SQLBinaryOperator::Gt => l.gt(r),
3494            SQLBinaryOperator::GtEq => l.gt_eq(r),
3495            // Guarded by `scalar_correlation_predicate`.
3496            _ => unreachable!("non-comparison correlation operator"),
3497        }
3498    }
