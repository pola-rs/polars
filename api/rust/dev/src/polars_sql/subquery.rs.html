2//! anti joins, decorrelating equi-correlation predicates into join keys.
3//! Subquery shapes the rewrites can't soundly express return `None` so the
4//! caller falls back to the generic filter path.

27    // Entry point: offer each WHERE conjunct to the rewrite, returning the
28    // (possibly join-extended) frame together with the conjuncts left for the
29    // ordinary filter path. In `KeepTrue` mode each top-level AND-conjunct is
30    // offered independently. In `RemoveTrue` mode conjuncts can't be split
31    // (`NOT (a AND b)` is a disjunction), so only a sole (possibly
32    // parenthesized) subquery predicate rewrites, and anything else is
33    // returned whole as the residual.
34    pub(crate) fn rewrite_subquery_conjuncts<'a>(
35        &mut self,
36        mut lf: LazyFrame,
37        expr: &'a SQLExpr,
38        filter_mode: FilterMode,
39        schema: &Schema,
40    ) -> PolarsResult<(LazyFrame, Vec<&'a SQLExpr>)> {
41        let residual = match filter_mode {
42            FilterMode::RemoveTrue => {
43                let mut unwrapped = expr;
44                while let SQLExpr::Nested(inner) = unwrapped {
45                    unwrapped = inner;
46                }
47                match self.try_rewrite_subquery_conjunct(&lf, unwrapped, filter_mode, schema)? {
48                    Some(new_lf) => {
49                        lf = new_lf;
50                        Vec::new()
51                    },
52                    None => vec![expr],
53                }
54            },
55            FilterMode::KeepTrue => {
56                let (plain, with_subquery): (Vec<&SQLExpr>, Vec<&SQLExpr>) =
57                    MintermIter::new(expr).partition(|c| !expr_contains_subquery(c));
58
59                if with_subquery.is_empty() {
60                    return Ok((lf, plain));
61                }
62
63                // A subquery rewrite can row-index the frame, which blocks predicate
64                // pushdown. Apply the conjuncts holding no subquery before that.
65                if !plain.is_empty() {
66                    let early = plain
67                        .iter()
68                        .map(|c| parse_sql_expr(c, self, Some(schema)))
69                        .collect::<PolarsResult<Vec<_>>>()?;
70                    lf = lf.filter(all_horizontal(early)?);
71                }
72
73                let mut residual = Vec::new();
74                for conj in with_subquery {
75                    if let Some(new_lf) =
76                        self.try_rewrite_subquery_conjunct(&lf, conj, filter_mode, schema)?
77                    {
78                        lf = new_lf;
79                    } else {
80                        residual.push(conj);
81                    }
82                }
83                residual
84            },
85        };
86        Ok((lf, residual))
87    }

89    // Dispatch one conjunct to the matching rewrite. `RemoveTrue` mode (DELETE)
90    // flips the join polarity. Removing `IN` rows keeps
91    // rows whose membership is false or NULL, exactly what an anti-join
92    // produces; removing `NOT IN` rows would additionally keep NULL keys, which
93    // a semi join can't express, so it stays on the filter path.
94    fn try_rewrite_subquery_conjunct(
95        &mut self,
96        lf: &LazyFrame,
97        conj: &SQLExpr,
98        filter_mode: FilterMode,
99        schema: &Schema,
100    ) -> PolarsResult<Option<LazyFrame>> {
101        let removing = filter_mode == FilterMode::RemoveTrue;
102        match conj {
103            SQLExpr::Exists { subquery, negated } => {
104                self.try_rewrite_exists_as_join(lf, subquery, *negated != removing, schema)
105            },
106            SQLExpr::InSubquery {
107                expr: lhs,
108                subquery,
109                negated,
110            } if !(*negated && removing) => self.try_rewrite_in_subquery_as_join(
111                lf,
112                lhs,
113                subquery,
114                *negated != removing,
115                filter_mode,
116                schema,
117            ),
118            _ => Ok(None),
119        }
120    }

122    // Lower `[NOT] EXISTS (SELECT ... FROM rel WHERE rel.k = outer.k ...)` to a
123    // semi / anti join by decorrelating the equi-correlation predicate(s) into
124    // join keys. DISTINCT is ignored: existence is invariant under
125    // deduplication. A non-equality correlation predicate falls back to the
126    // count-filter rewrite below.
127    #[cfg(feature = "semi_anti_join")]
128    fn try_rewrite_exists_as_join(
129        &mut self,
130        lf: &LazyFrame,
131        subquery: &Query,
132        negated: bool,
133        outer_schema: &Schema,
134    ) -> PolarsResult<Option<LazyFrame>> {
135        let Some(select) = eligible_subquery_select(subquery) else {
136            return Ok(None);
137        };
138        let Some(selection) = &select.selection else {
139            return Ok(None);
140        };
141        // Both rewrites below split this into conjuncts.
142        let selection = factored_selection(selection);
143        // Resolve and parse the inner relation in an isolated context so its
144        // table/alias registrations don't leak into the outer query's scope.
145        let mut ctx = self.isolated();
146        let Some((inner_names, inner_lf, inner_schema)) =
147            ctx.resolve_subquery_from(&select.from)?
148        else {
149            return Ok(None);
150        };
151        if let Some(SubqueryConjuncts {
152            left_on,
153            right_on,
154            local_filters,
155        }) =
156            ctx.split_subquery_conjuncts(&selection, &inner_names, &inner_schema, outer_schema)?
157        {
158            // An uncorrelated EXISTS (no correlation key found) has no join key
159            // to build from, so leave it to the existing path.
160            return Ok(if left_on.is_empty() {
161                None
162            } else {
163                Some(ctx.finish_decorrelated_join(
164                    lf,
165                    inner_lf,
166                    left_on,
167                    right_on,
168                    local_filters,
169                    negated,
170                )?)
171            });
172        }
173        ctx.try_rewrite_exists_as_count_filter(
174            lf,
175            inner_lf,
176            &selection,
177            &inner_names,
178            &inner_schema,
179            outer_schema,
180            negated,
181        )
182    }

184    // Lower `[NOT] EXISTS (SELECT ... FROM rel WHERE <comparison-correlated>)`,
185    // whose correlation predicates aren't all plain equalities, to a
186    // `count(*) > 0` filter (`== 0` for NOT EXISTS).
187    #[cfg(feature = "semi_anti_join")]
188    #[expect(clippy::too_many_arguments)]
189    fn try_rewrite_exists_as_count_filter(
190        mut self,
191        lf: &LazyFrame,
192        inner_lf: LazyFrame,
193        selection: &SQLExpr,
194        inner_names: &PlHashSet<String>,
195        inner_schema: &Schema,
196        outer_schema: &Schema,
197        negated: bool,
198    ) -> PolarsResult<Option<LazyFrame>> {
199        let mut corr_preds = Vec::new();
200        let mut local_filters = Vec::new();
201        for conj in MintermIter::new(selection) {
202            if let Some(pred) =
203                scalar_correlation_predicate(conj, inner_names, inner_schema, outer_schema)
204            {
205                corr_preds.push(pred);
206            } else if let Some(filter) =
207                self.try_parse_inner_only_expr(conj, inner_names, inner_schema)?
208            {
209                local_filters.push(filter);
210            } else {
211                return Ok(None);
212            }
213        }
214        // No correlation: leave it to the equi-only path.
215        if corr_preds.is_empty() {
216            return Ok(None);
217        }
218
219        let prefix = format_pl_smallstr!("{CORRELATED_COL_PREFIX}{}_", unique_column_name());
220        let idx_name = format_pl_smallstr!("{prefix}idx");
221        let count_name = format_pl_smallstr!("{prefix}cnt");
222
223        // Apply inner-only filters, then rename inner columns to collision-free
224        // names so they can't clash with outer columns of the same name (as in
225        // a self-correlated `t1 AS x` subquery over outer `t1`).
226        let inner_filtered = local_filters.into_iter().fold(inner_lf, LazyFrame::filter);
227        let rename_from: Vec<PlSmallStr> = inner_schema.iter_names().cloned().collect();
228        let rename_to: Vec<PlSmallStr> = rename_from
229            .iter()
230            .map(|name| prefixed_inner(&prefix, name))
231            .collect();
232        let inner_renamed = inner_filtered.rename(&rename_from, &rename_to, true);
233        inner_renamed.set_cached_arena(self.lp_arena, self.expr_arena);
234
235        let join_preds: Vec<Expr> = corr_preds.iter().map(|p| p.to_expr(&prefix)).collect();
236
237        let outer_indexed = row_indexed_once(lf.clone(), idx_name.clone());
238        let matched = outer_indexed
239            .clone()
240            .join_builder()
241            .with(inner_renamed)
242            .how(JoinType::Inner)
243            .join_where(join_preds);
244        let grouped = matched
245            .group_by([col(idx_name.clone())])
246            .agg([len().alias(count_name.clone())]);
247
248        let joined = outer_indexed
249            .join_builder()
250            .with(grouped)
251            .left_on([col(idx_name.clone())])
252            .right_on([col(idx_name.clone())])
253            .how(JoinType::Left)
254            .coalesce(JoinCoalesce::CoalesceColumns)
255            .maintain_order(MaintainOrderJoin::Left)
256            .finish()?
257            .with_columns([col(count_name.clone()).fill_null(lit(0))]);
258
259        let matches = if negated {
260            col(count_name.clone()).eq(lit(0))
261        } else {
262            col(count_name.clone()).gt(lit(0))
263        };
264        Ok(Some(joined.filter(matches).drop(Selector::ByName {
265            names: Arc::from([idx_name, count_name]),
266            strict: true,
267        })))
268    }

270    // Lower `lhs [NOT] IN (SELECT col FROM rel ...)` to a semi / anti join: the
271    // projected column is the membership key and any equi-correlations in the
272    // subquery WHERE become additional join keys.
273    #[cfg(feature = "semi_anti_join")]
274    fn try_rewrite_in_subquery_as_join(
275        &mut self,
276        lf: &LazyFrame,
277        lhs: &SQLExpr,
278        subquery: &Query,
279        anti: bool,
280        filter_mode: FilterMode,
281        outer_schema: &Schema,
282    ) -> PolarsResult<Option<LazyFrame>> {
283        let Some(select) = eligible_subquery_select(subquery) else {
284            return Ok(None);
285        };
286        // DISTINCT is membership-invariant, but DISTINCT ON drops rows per key.
287        if matches!(&select.distinct, Some(Distinct::On(_))) {
288            return Ok(None);
289        }
290        let [SelectItem::UnnamedExpr(proj) | SelectItem::ExprWithAlias { expr: proj, .. }] =
291            select.projection.as_slice()
292        else {
293            return Ok(None);
294        };
295
296        let Some(left_key) = self.try_parse_outer_only_expr(lhs, outer_schema)? else {
297            return Ok(None);
298        };
299
300        let mut ctx = self.isolated();
301        let Some((inner_names, inner_lf, inner_schema)) =
302            ctx.resolve_subquery_from(&select.from)?
303        else {
304            return Ok(None);
305        };
306        // The membership key must be a plain expression over the inner relation;
307        // any alias it carries is cosmetic and not allowed in a join key.
308        let Some(right_key) = ctx.try_parse_inner_only_expr(proj, &inner_names, &inner_schema)?
309        else {
310            return Ok(None);
311        };
312        let right_key = right_key.meta().undo_aliases();
313        if !usable_as_join_key(&left_key) || !usable_as_join_key(&right_key) {
314            return Ok(None);
315        }
316
317        let SubqueryConjuncts {
318            mut left_on,
319            mut right_on,
320            local_filters,
321        } = match &select.selection {
322            Some(selection) => {
323                let Some(split) = ctx.split_subquery_conjuncts(
324                    selection,
325                    &inner_names,
326                    &inner_schema,
327                    outer_schema,
328                )?
329                else {
330                    return Ok(None);
331                };
332                split
333            },
334            None => SubqueryConjuncts::default(),
335        };
336
337        // Correlation keys for the "NOT IN" 3VL correction
338        let corr_outer = left_on.clone();
339        let corr_inner = right_on.clone();
340        left_on.insert(0, left_key.clone());
341        right_on.insert(0, right_key.clone());
342
343        // Inline, so filtered inner frame can be reused for the correction
344        let inner_lf = local_filters.into_iter().fold(inner_lf, LazyFrame::filter);
345        inner_lf.set_cached_arena(ctx.lp_arena, ctx.expr_arena);
346        let joined = build_semi_anti_join(lf, inner_lf.clone(), left_on, right_on, anti)?;
347
348        // Only `KeepTrue` "NOT IN" needs 3VL correction.
349        if !(anti && filter_mode == FilterMode::KeepTrue) {
350            return Ok(Some(joined));
351        }
352        Ok(Some(refine_not_in_anti_join(
353            joined,
354            inner_lf,
355            &left_key,
356            &right_key,
357            &corr_outer,
358            &corr_inner,
359        )?))
360    }

362    // Apply the local filters to the inner relation, hand this (isolated, now
363    // finished) context's arenas to it, and build the semi / anti join against
364    // the outer frame. Consumes the context: nothing may be parsed in the
365    // subquery's scope after the join is built.
366    #[cfg(feature = "semi_anti_join")]
367    fn finish_decorrelated_join(
368        self,
369        lf: &LazyFrame,
370        inner_lf: LazyFrame,
371        left_on: Vec<Expr>,
372        right_on: Vec<Expr>,
373        local_filters: Vec<Expr>,
374        anti: bool,
375    ) -> PolarsResult<LazyFrame> {
376        let inner_lf = local_filters.into_iter().fold(inner_lf, LazyFrame::filter);
377        inner_lf.set_cached_arena(self.lp_arena, self.expr_arena);
378        build_semi_anti_join(lf, inner_lf, left_on, right_on, anti)
379    }

381    // Resolve the subquery's FROM (a single relation, possibly with joins) into
382    // the inner LazyFrame, its schema, and the set of relation names/aliases
383    // used to classify qualified correlation columns.
384    fn resolve_subquery_from(
385        &mut self,
386        from: &[TableWithJoins],
387    ) -> PolarsResult<Option<(PlHashSet<String>, LazyFrame, SchemaRef)>> {
388        let Some(inner_names) = from
389            .iter()
390            .flat_map(|tbl_expr| {
391                std::iter::once(&tbl_expr.relation)
392                    .chain(tbl_expr.joins.iter().map(|j| &j.relation))
393            })
394            .map(get_table_name)
395            .collect::<Option<PlHashSet<_>>>()
396        else {
397            return Ok(None);
398        };
399        let Some((first, rest)) = from.split_first() else {
400            return Ok(None);
401        };
402        let mut inner_lf = self.execute_from_statement(first)?;
403        // Comma-separated relations cross join; the equalities linking them are
404        // inner-only filters.
405        for tbl_expr in rest {
406            let rf = self.execute_from_statement(tbl_expr)?;
407            inner_lf = inner_lf.cross_join(rf, None);
408        }
409        let inner_schema = self.get_frame_schema(&mut inner_lf)?;
410        Ok(Some((inner_names, inner_lf, inner_schema)))
411    }

413    // Split a subquery's WHERE conjuncts into a `SubqueryConjuncts`, or `None`
414    // when a conjunct is neither a correlation key pair nor an inner-only
415    // filter (an outer column in a non-equi shape, an unresolvable name).
416    #[cfg(feature = "semi_anti_join")]
417    fn split_subquery_conjuncts(
418        &mut self,
419        selection: &SQLExpr,
420        inner_names: &PlHashSet<String>,
421        inner_schema: &Schema,
422        outer_schema: &Schema,
423    ) -> PolarsResult<Option<SubqueryConjuncts>> {
424        let mut left_on = Vec::new();
425        let mut right_on = Vec::new();
426        let mut local_filters = Vec::new();
427        for conj in MintermIter::new(selection) {
428            if let Some((outer_key, inner_key)) =
429                correlation_key_pair(conj, inner_names, inner_schema, outer_schema)
430            {
431                left_on.push(col(outer_key));
432                right_on.push(col(inner_key));
433                continue;
434            }
435            let Some(filter) = self.try_parse_inner_only_expr(conj, inner_names, inner_schema)?
436            else {
437                return Ok(None);
438            };
439            local_filters.push(filter);
440        }
441        Ok(Some(SubqueryConjuncts {
442            left_on,
443            right_on,
444            local_filters,
445        }))
446    }

448    // Parse an outer-query expression, or `None` if it doesn't stand on the outer
449    // relation alone: an unlowered subquery, or a column the outer frame lacks.
450    // Any alias is cosmetic here and stripped.
451    fn try_parse_outer_only_expr(
452        &mut self,
453        sql_expr: &SQLExpr,
454        outer_schema: &Schema,
455    ) -> PolarsResult<Option<Expr>> {
456        let expr = parse_sql_expr(sql_expr, self, Some(outer_schema))?
457            .meta()
458            .undo_aliases();
459        if has_expr(&expr, |e| matches!(e, Expr::SubPlan(_, _)))
460            || !expr_to_leaf_column_names_iter(&expr)
461                .all(|name| outer_schema.contains(name.as_str()))
462        {
463            return Ok(None);
464        }
465        Ok(Some(expr))
466    }

483    // Lower every correlated subquery reachable from `expr` into a decorrelated
484    // join over `lf`, substituting each one for the column its lowering
485    // materialised. Returns the updated frame together with the rewritten
486    // expression, which borrows `expr` unchanged when nothing was lowered.
487    // Subqueries that can't be lowered are left in place.
488    //
489    // See [`SubqueryBindings`] for when a set may be shared across calls.
490    pub(crate) fn lower_correlated_subqueries<'a>(
491        &mut self,
492        lf: LazyFrame,
493        outer_schema: &Schema,
494        expr: &'a SQLExpr,
495        scope: LowerScope,
496        bindings: &mut SubqueryBindings,
497    ) -> PolarsResult<(LazyFrame, Cow<'a, SQLExpr>)> {
498        // The mutable visit needs an owned expression; skip the clone when there is
499        // nothing to rewrite.
500        if !expr_contains_subquery(expr) {
501            return Ok((lf, Cow::Borrowed(expr)));
502        }
503        let mut rewritten = expr.clone();
504        let mut lowering = CorrelatedLowering {
505            ctx: self,
506            lf,
507            outer_schema,
508            scope,
509            bindings,
510            query_depth: 0,
511            changed: false,
512        };
513        if let ControlFlow::Break(e) = VisitMut::visit(&mut rewritten, &mut lowering) {
514            return Err(e);
515        }
516        let rewritten = if lowering.changed {
517            Cow::Owned(rewritten)
518        } else {
519            Cow::Borrowed(expr)
520        };
521        Ok((lowering.lf, rewritten))
522    }

524    // Attempt the decorrelation of a single scalar aggregate subquery:
525    //   (SELECT AGG(...) FROM inner WHERE <corr-preds> AND <inner-filters>)
526    // Equality-only correlation: `inner GROUP BY <corr cols>` once, joined onto
527    // the outer frame on those columns directly. Otherwise: row-index the outer
528    // frame, inner-join outer × inner on the correlation predicates (`join_where`
529    // handles inequality), aggregate per outer row, then left-join back on the
530    // row index. Either way, `COUNT` over no matches is 0, every other aggregate
531    // is NULL. Returns `None` when uncorrelated or not a scalar-aggregate shape.
532    fn try_decorrelate_scalar_subquery(
533        &mut self,
534        lf: LazyFrame,
535        outer_schema: &Schema,
536        subquery: &Query,
537    ) -> PolarsResult<Option<(LazyFrame, PlSmallStr)>> {
538        let Some(select) = eligible_subquery_select(subquery) else {
539            return Ok(None);
540        };
541        let Some(selection) = &select.selection else {
542            return Ok(None);
543        };
544        let [SelectItem::UnnamedExpr(proj) | SelectItem::ExprWithAlias { expr: proj, .. }] =
545            select.projection.as_slice()
546        else {
547            return Ok(None);
548        };
549
550        let mut ctx = self.isolated();
551        let Some((inner_names, inner_lf, inner_schema)) =
552            ctx.resolve_subquery_from(&select.from)?
553        else {
554            return Ok(None);
555        };
556
557        // The projection must be a scalar aggregate over the inner relation.
558        let Some(agg_expr) = ctx.try_parse_inner_only_expr(proj, &inner_names, &inner_schema)?
559        else {
560            return Ok(None);
561        };
562        if !has_expr(&agg_expr, |e| matches!(e, Expr::Agg(_) | Expr::Len)) {
563            return Ok(None);
564        }
565        let count_like = matches!(
566            agg_output_root(&agg_expr),
567            Expr::Len | Expr::Agg(AggExpr::Count { .. })
568        );
569
570        // Split the WHERE into correlation predicates and inner-only filters.
571        let mut corr_preds = Vec::new();
572        let mut local_filters = Vec::new();
573        let selection = factored_selection(selection);
574        for conj in MintermIter::new(&selection) {
575            if let Some(pred) =
576                scalar_correlation_predicate(conj, &inner_names, &inner_schema, outer_schema)
577            {
578                corr_preds.push(pred);
579            } else if let Some(filter) =
580                ctx.try_parse_inner_only_expr(conj, &inner_names, &inner_schema)?
581            {
582                local_filters.push(filter);
583            } else {
584                return Ok(None);
585            }
586        }
587        // No correlation: leave it to the uncorrelated scalar-subquery path.
588        if corr_preds.is_empty() {
589            return Ok(None);
590        }
591
592        let prefix = format_pl_smallstr!("{CORRELATED_COL_PREFIX}{}_", unique_column_name());
593        let result_name = format_pl_smallstr!("{prefix}res");
594
595        // Apply inner-only filters, then rename inner columns to collision-free
596        // names so they can't clash with outer columns of the same name (as in a
597        // self-correlated `t1 AS x` subquery over outer `t1`).
598        let inner_filtered = local_filters.into_iter().fold(inner_lf, LazyFrame::filter);
599        let rename_from: Vec<PlSmallStr> = inner_schema.iter_names().cloned().collect();
600        let rename_to: Vec<PlSmallStr> = rename_from
601            .iter()
602            .map(|name| prefixed_inner(&prefix, name))
603            .collect();
604        let inner_renamed = inner_filtered.rename(&rename_from, &rename_to, true);
605        inner_renamed.set_cached_arena(ctx.lp_arena, ctx.expr_arena);
606
607        let agg_expr = agg_expr.map_expr(|e| match e {
608            Expr::Column(name) if inner_schema.contains(name.as_str()) => {
609                col(prefixed_inner(&prefix, &name))
610            },
611            other => other,
612        });
613
614        if corr_preds.iter().all(|p| p.op == SQLBinaryOperator::Eq) {
615            let outer_on: Vec<Expr> = corr_preds.iter().map(|p| col(p.outer.clone())).collect();
616            let inner_on: Vec<Expr> = corr_preds
617                .iter()
618                .map(|p| col(prefixed_inner(&prefix, &p.inner)))
619                .collect();
620            let grouped = inner_renamed
621                .group_by(inner_on.clone())
622                .agg([agg_expr.alias(result_name.clone())]);
623            let joined =
624                left_join_aggregate(lf, grouped, outer_on, inner_on, &result_name, count_like)?;
625            return Ok(Some((joined, result_name)));
626        }
627
628        let join_preds: Vec<Expr> = corr_preds.iter().map(|p| p.to_expr(&prefix)).collect();
629        let idx_name = format_pl_smallstr!("{prefix}idx");
630
631        let outer_indexed = row_indexed_once(lf, idx_name.clone());
632        let matched = outer_indexed
633            .clone()
634            .join_builder()
635            .with(inner_renamed)
636            .how(JoinType::Inner)
637            .join_where(join_preds);
638        let grouped = matched
639            .group_by([col(idx_name.clone())])
640            .agg([agg_expr.alias(result_name.clone())]);
641
642        let joined = left_join_aggregate(
643            outer_indexed,
644            grouped,
645            vec![col(idx_name.clone())],
646            vec![col(idx_name.clone())],
647            &result_name,
648            count_like,
649        )?
650        .drop(Selector::ByName {
651            names: Arc::from([idx_name]),
652            strict: true,
653        });
654        Ok(Some((joined, result_name)))
655    }
656
657    // Attempt the decorrelation of a correlated `IN (subquery)` into a boolean
658    // column. The correlated candidate values are collected per outer row into a
659    // list, and membership is then evaluated against that list under SQL's
660    // three-valued logic. An uncorrelated subquery is left to the generic path.
661    fn try_decorrelate_in_subquery(
662        &mut self,
663        lf: LazyFrame,
664        outer_schema: &Schema,
665        lhs: &SQLExpr,
666        subquery: &Query,
667    ) -> PolarsResult<Option<(LazyFrame, PlSmallStr)>> {
668        let Some(select) = eligible_subquery_select(subquery) else {
669            return Ok(None);
670        };
671        if matches!(&select.distinct, Some(Distinct::On(_))) {
672            return Ok(None);
673        }
674        let [SelectItem::UnnamedExpr(proj) | SelectItem::ExprWithAlias { expr: proj, .. }] =
675            select.projection.as_slice()
676        else {
677            return Ok(None);
678        };
679        let Some(selection) = &select.selection else {
680            return Ok(None);
681        };
682
683        let Some(needle) = self.try_parse_outer_only_expr(lhs, outer_schema)? else {
684            return Ok(None);
685        };
686
687        let mut ctx = self.isolated();
688        let Some((inner_names, inner_lf, inner_schema)) =
689            ctx.resolve_subquery_from(&select.from)?
690        else {
691            return Ok(None);
692        };
693        let Some(value) = ctx.try_parse_inner_only_expr(proj, &inner_names, &inner_schema)? else {
694            return Ok(None);
695        };
696        let value = value.meta().undo_aliases();
697
698        let mut corr_preds = Vec::new();
699        let mut local_filters = Vec::new();
700        let selection = factored_selection(selection);
701        for conj in MintermIter::new(&selection) {
702            if let Some(pred) =
703                scalar_correlation_predicate(conj, &inner_names, &inner_schema, outer_schema)
704            {
705                corr_preds.push(pred);
706            } else if let Some(filter) =
707                ctx.try_parse_inner_only_expr(conj, &inner_names, &inner_schema)?
708            {
709                local_filters.push(filter);
710            } else {
711                return Ok(None);
712            }
713        }
714        // No correlation: the generic `IN` path already handles this.
715        if corr_preds.is_empty() {
716            return Ok(None);
717        }
718
719        let prefix = format_pl_smallstr!("{CORRELATED_COL_PREFIX}{}_", unique_column_name());
720        let set_name = format_pl_smallstr!("{prefix}set");
721
722        let inner_filtered = local_filters.into_iter().fold(inner_lf, LazyFrame::filter);
723        let rename_from: Vec<PlSmallStr> = inner_schema.iter_names().cloned().collect();
724        let rename_to: Vec<PlSmallStr> = rename_from
725            .iter()
726            .map(|name| prefixed_inner(&prefix, name))
727            .collect();
728        let inner_renamed = inner_filtered.rename(&rename_from, &rename_to, true);
729        inner_renamed.set_cached_arena(ctx.lp_arena, ctx.expr_arena);
730
731        let value = value.map_expr(|e| match e {
732            Expr::Column(name) if inner_schema.contains(name.as_str()) => {
733                col(prefixed_inner(&prefix, &name))
734            },
735            other => other,
736        });
737
738        // Grouping collects the candidate values of each outer row into a list.
739        let joined = if corr_preds.iter().all(|p| p.op == SQLBinaryOperator::Eq) {
740            let outer_on: Vec<Expr> = corr_preds.iter().map(|p| col(p.outer.clone())).collect();
741            let inner_on: Vec<Expr> = corr_preds
742                .iter()
743                .map(|p| col(prefixed_inner(&prefix, &p.inner)))
744                .collect();
745            let grouped = inner_renamed
746                .group_by(inner_on.clone())
747                .agg([value.alias(set_name.clone())]);
748            left_join_aggregate(lf, grouped, outer_on, inner_on, &set_name, false)?
749        } else {
750            let join_preds: Vec<Expr> = corr_preds.iter().map(|p| p.to_expr(&prefix)).collect();
751            let idx_name = format_pl_smallstr!("{prefix}idx");
752            let outer_indexed = row_indexed_once(lf, idx_name.clone());
753            let matched = outer_indexed
754                .clone()
755                .join_builder()
756                .with(inner_renamed)
757                .how(JoinType::Inner)
758                .join_where(join_preds);
759            let grouped = matched
760                .group_by([col(idx_name.clone())])
761                .agg([value.alias(set_name.clone())]);
762            left_join_aggregate(
763                outer_indexed,
764                grouped,
765                vec![col(idx_name.clone())],
766                vec![col(idx_name.clone())],
767                &set_name,
768                false,
769            )?
770            .drop(Selector::ByName {
771                names: Arc::from([idx_name]),
772                strict: true,
773            })
774        };
775
776        // An outer row with no matching inner rows joins to null, which is the
777        // empty candidate set rather than an unknown one.
778        let set = col(set_name.clone());
779        let is_in = sql_in_membership(
780            needle.is_in(set.clone(), false),
781            set.clone(),
782            set.clone().is_null().or(set.list().len().eq(lit(0u32))),
783        );
784        // The boolean replaces the candidate list in place.
785        let joined = joined.with_columns([is_in.alias(set_name.clone())]);
786        Ok(Some((joined, set_name)))
787    }
788
789    // Attempt the decorrelation of a single `EXISTS` subquery into a boolean flag
790    // column holding `count(*) > 0` over the correlated match set. An uncorrelated
791    // subquery yields a single constant boolean broadcast onto every outer row
792    // instead, no join being needed to know whether a fixed relation is non-empty.
793    // Returns `None` for shapes that can't be soundly classified.
794    fn try_decorrelate_exists_subquery(
795        &mut self,
796        lf: LazyFrame,
797        outer_schema: &Schema,
798        subquery: &Query,
799    ) -> PolarsResult<Option<(LazyFrame, PlSmallStr)>> {
800        let Some(select) = eligible_subquery_select(subquery) else {
801            return Ok(None);
802        };
803
804        let mut ctx = self.isolated();
805        let Some((inner_names, inner_lf, inner_schema)) =
806            ctx.resolve_subquery_from(&select.from)?
807        else {
808            return Ok(None);
809        };
810
811        let mut corr_preds = Vec::new();
812        let mut local_filters = Vec::new();
813        if let Some(selection) = &select.selection {
814            let selection = factored_selection(selection);
815            for conj in MintermIter::new(&selection) {
816                if let Some(pred) =
817                    scalar_correlation_predicate(conj, &inner_names, &inner_schema, outer_schema)
818                {
819                    corr_preds.push(pred);
820                } else if let Some(filter) =
821                    ctx.try_parse_inner_only_expr(conj, &inner_names, &inner_schema)?
822                {
823                    local_filters.push(filter);
824                } else {
825                    return Ok(None);
826                }
827            }
828        }
829
830        let prefix = format_pl_smallstr!("{CORRELATED_COL_PREFIX}{}_", unique_column_name());
831        let flag_name = format_pl_smallstr!("{prefix}exists");
832        let inner_filtered = local_filters.into_iter().fold(inner_lf, LazyFrame::filter);
833
834        if corr_preds.is_empty() {
835            // Uncorrelated: the inner relation doesn't depend on the outer row, so
836            // its existence is a single constant boolean broadcast onto every row.
837            let inner_flag = inner_filtered.select([len().gt(lit(0)).alias(flag_name.clone())]);
838            inner_flag.set_cached_arena(ctx.lp_arena, ctx.expr_arena);
839            let joined = concat_lf_horizontal(
840                [lf, inner_flag],
841                HConcatOptions {
842                    broadcast_unit_length: true,
843                    ..Default::default()
844                },
845            )?;
846            return Ok(Some((joined, flag_name)));
847        }
848
849        let idx_name = format_pl_smallstr!("{prefix}idx");
850        let count_name = format_pl_smallstr!("{prefix}cnt");
851
852        // Rename inner columns to collision-free names so they can't clash
853        // with outer columns of the same name (as in a self-correlated
854        // `t1 AS x` subquery over outer `t1`).
855        let rename_from: Vec<PlSmallStr> = inner_schema.iter_names().cloned().collect();
856        let rename_to: Vec<PlSmallStr> = rename_from
857            .iter()
858            .map(|name| prefixed_inner(&prefix, name))
859            .collect();
860        let inner_renamed = inner_filtered.rename(&rename_from, &rename_to, true);
861        inner_renamed.set_cached_arena(ctx.lp_arena, ctx.expr_arena);
862
863        let join_preds: Vec<Expr> = corr_preds.iter().map(|p| p.to_expr(&prefix)).collect();
864
865        let outer_indexed = row_indexed_once(lf, idx_name.clone());
866        let matched = outer_indexed
867            .clone()
868            .join_builder()
869            .with(inner_renamed)
870            .how(JoinType::Inner)
871            .join_where(join_preds);
872        let grouped = matched
873            .group_by([col(idx_name.clone())])
874            .agg([len().alias(count_name.clone())]);
875
876        let joined = outer_indexed
877            .join_builder()
878            .with(grouped)
879            .left_on([col(idx_name.clone())])
880            .right_on([col(idx_name.clone())])
881            .how(JoinType::Left)
882            .coalesce(JoinCoalesce::CoalesceColumns)
883            .maintain_order(MaintainOrderJoin::Left)
884            .finish()?
885            .with_columns([col(count_name.clone())
886                .fill_null(lit(0))
887                .gt(lit(0))
888                .alias(flag_name.clone())])
889            .drop(Selector::ByName {
890                names: Arc::from([idx_name, count_name]),
891                strict: true,
892            });
893        Ok(Some((joined, flag_name)))
894    }

897    fn try_rewrite_exists_as_join(
898        &mut self,
899        _lf: &LazyFrame,
900        _subquery: &Query,
901        _negated: bool,
902        _outer_schema: &Schema,
903    ) -> PolarsResult<Option<LazyFrame>> {

940// Account for 3VL interaction with NULL values
941#[cfg(feature = "semi_anti_join")]
942fn refine_not_in_anti_join(
943    joined: LazyFrame,
944    inner_lf: LazyFrame,
945    left_key: &Expr,
946    right_key: &Expr,
947    corr_outer: &[Expr],
948    corr_inner: &[Expr],
949) -> PolarsResult<LazyFrame> {
950    if corr_inner.is_empty() {
951        // Uncorrelated
952        let flag_name = unique_column_name();
953        let flag = when(len().eq(lit(0u32)))
954            .then(lit(NULL).cast(DataType::Boolean))
955            .otherwise(right_key.clone().is_null().any(true))
956            .alias(flag_name.clone());
957        let keep = when(col(flag_name.clone()).is_null())
958            .then(lit(true)) // empty set
959            .when(col(flag_name.clone())) // set has a NULL
960            .then(lit(false))
961            .otherwise(left_key.clone().is_not_null());
962
963        return Ok(joined
964            .join_builder()
965            .with(inner_lf.select([flag]))
966            .how(JoinType::Cross)
967            .finish()?
968            .filter(keep)
969            .drop(Selector::ByName {
970                names: [flag_name].into(),
971                strict: true,
972            }));
973    }
974
975    // Correlated
976    let corr_keys = |lf: LazyFrame| lf.select(corr_inner).unique(None, UniqueKeepStrategy::Any);
977    let exclude_groups = |rows: LazyFrame, groups: LazyFrame| {
978        rows.join_builder()
979            .with(groups)
980            .left_on(corr_outer)
981            .right_on(corr_inner)
982            .how(JoinType::Anti)
983            .finish()
984    };
985    let kept_non_null = exclude_groups(
986        joined.clone().filter(left_key.clone().is_not_null()),
987        corr_keys(inner_lf.clone().filter(right_key.clone().is_null())),
988    )?;
989    let kept_null = exclude_groups(
990        joined.filter(left_key.clone().is_null()),
991        corr_keys(inner_lf),
992    )?;
993
994    Ok(concat(
995        [kept_non_null, kept_null],
996        UnionArgs {
997            rechunk: false,
998            parallel: true,
999            ..Default::default()
1000        },
1001    )
1002    .expect("'NOT IN' 3VL union has identical schemas"))
1003}

1005/// An iterator over all the minterms in an SQL boolean expression: the terms
1006/// that `AND` together to form it, descending through parenthesized `Nested`
1007/// expressions. The SQL-AST analogue of the `AExpr`-level
1008/// `polars_plan::plans::aexpr::MintermIter`.
1009struct MintermIter<'a> {
1010    stack: Vec<&'a SQLExpr>,
1011}
1012
1013impl<'a> Iterator for MintermIter<'a> {
1014    type Item = &'a SQLExpr;
1015
1016    fn next(&mut self) -> Option<Self::Item> {
1017        let mut top = self.stack.pop()?;
1018        loop {
1019            match top {
1020                SQLExpr::Nested(inner) => top = inner,
1021                SQLExpr::BinaryOp {
1022                    left,
1023                    op: SQLBinaryOperator::And,
1024                    right,
1025                } => {
1026                    self.stack.push(right);
1027                    top = left;
1028                },
1029                _ => return Some(top),
1030            }
1031        }
1032    }
1033}
1034
1035impl<'a> MintermIter<'a> {
1036    fn new(root: &'a SQLExpr) -> Self {
1037        Self { stack: vec![root] }
1038    }
1039}

1041/// An iterator over the terms that `OR` together to form an SQL boolean
1042/// expression, descending through parenthesized `Nested` expressions.
1043struct DisjunctIter<'a> {
1044    stack: Vec<&'a SQLExpr>,
1045}
1046
1047impl<'a> Iterator for DisjunctIter<'a> {
1048    type Item = &'a SQLExpr;
1049
1050    fn next(&mut self) -> Option<Self::Item> {
1051        let mut top = self.stack.pop()?;
1052        loop {
1053            match top {
1054                SQLExpr::Nested(inner) => top = inner,
1055                SQLExpr::BinaryOp {
1056                    left,
1057                    op: SQLBinaryOperator::Or,
1058                    right,
1059                } => {
1060                    self.stack.push(right);
1061                    top = left;
1062                },
1063                _ => return Some(top),
1064            }
1065        }
1066    }
1067}

1070    fn new(root: &'a SQLExpr) -> Self {

1072    }

1075/// Rewrite `(A AND X) OR (A AND Y)` as `A AND (X OR Y)`.
1076///
1077/// The SQL-AST analogue of `polars_plan::plans::aexpr::or_factoring`, which runs
1078/// after the decorrelation decision made here.
1079fn factor_common_conjuncts(expr: &SQLExpr) -> Option<SQLExpr> {
1080    let branches: Vec<Vec<&SQLExpr>> = DisjunctIter::new(expr)
1081        .map(|branch| MintermIter::new(branch).collect())
1082        .collect();
1083    let (first, rest) = branches.split_first()?;
1084    if rest.is_empty() {
1085        return None;
1086    }
1087    let common: Vec<&SQLExpr> = first
1088        .iter()
1089        .copied()
1090        .filter(|term| rest.iter().all(|branch| branch.contains(term)))
1091        .collect();
1092    let factored = combine_conditions(
1093        common.iter().copied().cloned().collect(),
1094        SQLBinaryOperator::And,
1095    )?;
1096
1097    // A branch holding nothing but the common terms absorbs the others:
1098    // `A OR (A AND Y)` is just `A`.
1099    let remainders: Option<Vec<SQLExpr>> = branches
1100        .iter()
1101        .map(|terms| {
1102            combine_conditions(
1103                terms
1104                    .iter()
1105                    .copied()
1106                    .filter(|term| !common.contains(term))
1107                    .cloned()
1108                    .collect(),
1109                SQLBinaryOperator::And,
1110            )
1111        })
1112        .collect();
1113    let Some(remainders) = remainders else {
1114        return Some(factored);
1115    };
1116    let disjunction = combine_conditions(remainders, SQLBinaryOperator::Or)?;
1117    Some(SQLExpr::BinaryOp {
1118        left: Box::new(factored),
1119        op: SQLBinaryOperator::And,
1120        right: Box::new(SQLExpr::Nested(Box::new(disjunction))),
1121    })
1122}
1123
1124/// Hoist predicates shared by every branch of an `OR` to the top level.
1125fn factored_selection(selection: &SQLExpr) -> Cow<'_, SQLExpr> {
1126    let factored: Vec<Option<SQLExpr>> = MintermIter::new(selection)
1127        .map(factor_common_conjuncts)
1128        .collect();
1129    if factored.iter().all(Option::is_none) {
1130        return Cow::Borrowed(selection);
1131    }
1132    let conjuncts = MintermIter::new(selection)
1133        .zip(factored)
1134        .map(|(conj, factored)| factored.unwrap_or_else(|| conj.clone()))
1135        .collect();
1136    combine_conditions(conjuncts, SQLBinaryOperator::And)
1137        .map_or(Cow::Borrowed(selection), Cow::Owned)
1138}

1185// Whether every column the expression references resolves to the inner relation,
1186// so it can be evaluated against the inner frame alone. A qualified name resolves
1187// through its qualifier: only an inner relation's name or alias counts as inner.
1188// An unqualified name binds to the innermost scope that has it, so the inner
1189// schema wins over the outer one.
1190fn binds_to_inner_relation(
1191    expr: &SQLExpr,
1192    inner_names: &PlHashSet<String>,
1193    inner_schema: &Schema,
1194) -> bool {
1195    visit_expressions(expr, |e| {
1196        let resolves = match e {
1197            // A nested subquery is its own scope, which this cannot resolve against.
1198            _ if is_subquery_expr(e) => false,
1199            SQLExpr::Identifier(_) | SQLExpr::CompoundIdentifier(_) => qualifier_and_name(e)
1200                .is_some_and(|(qualifier, name)| {
1201                    qualifier.is_none_or(|q| inner_names.contains(q)) && inner_schema.contains(name)
1202                }),
1203            _ => true,
1204        };
1205        if resolves {
1206            ControlFlow::Continue(())
1207        } else {
1208            ControlFlow::Break(())
1209        }
1210    })
1211    .is_continue()
1212}

1214// Split an identifier into its optional table qualifier and its bare column name.
1215// Catalog and schema prefixes are dropped, matching how `get_table_name` builds
1216// `inner_names`.
1217fn qualifier_and_name(expr: &SQLExpr) -> Option<(Option<&str>, &str)> {
1218    match expr {
1219        SQLExpr::Identifier(ident) => Some((None, ident.value.as_str())),
1220        SQLExpr::CompoundIdentifier(parts) => {
1221            let (last, init) = parts.split_last()?;
1222            Some((init.last().map(|q| q.value.as_str()), last.value.as_str()))
1223        },
1224        _ => None,
1225    }
1226}

1228// Classify a correlation operand as an inner- or outer-query column and return
1229// its bare name. A qualified identifier (`tbl.col`) resolves by its qualifier:
1230// an inner relation's name/alias means inner, anything else means outer (so
1231// same-named columns like `o.id = c.id` resolve). An unqualified identifier
1232// resolves by schema membership. `None` for non-identifiers or names that can't
1233// be placed (in neither schema, or ambiguous).
1234fn classify_correlation_column(
1235    expr: &SQLExpr,
1236    inner_names: &PlHashSet<String>,
1237    inner_schema: &Schema,
1238    outer_schema: &Schema,
1239) -> Option<(CorrelationSide, PlSmallStr)> {
1240    let (qualifier, name) = qualifier_and_name(expr)?;
1241    let name: PlSmallStr = name.into();
1242    match qualifier {
1243        Some(q) if inner_names.contains(q) => inner_schema
1244            .contains(name.as_str())
1245            .then_some((CorrelationSide::Inner, name)),
1246        // Any other qualifier is taken as outer: the outer query may span
1247        // several relations and their names/aliases aren't visible here, so
1248        // only the column's schema membership can be checked.
1249        Some(_) => outer_schema
1250            .contains(name.as_str())
1251            .then_some((CorrelationSide::Outer, name)),
1252        // An unqualified name binds to the innermost scope that holds it.
1253        None => match (
1254            inner_schema.contains(name.as_str()),
1255            outer_schema.contains(name.as_str()),
1256        ) {
1257            (true, _) => Some((CorrelationSide::Inner, name)),
1258            (false, true) => Some((CorrelationSide::Outer, name)),
1259            (false, false) => None,
1260        },
1261    }
1262}

1264// Shared eligibility gate for the rewrites: bail on any clause that changes
1265// which rows the subquery yields. Exhaustive destructuring (no `..`) is on
1266// purpose: a new sqlparser clause must not compile until it gets an explicit
1267// keep-or-bail decision here.
1268fn eligible_subquery_select(subquery: &Query) -> Option<&Select> {
1269    let Query {
1270        with, // CTEs aren't resolved inside the rewrite: bail
1271        body,
1272        order_by: _,      // row order can't affect existence/membership
1273        limit_clause,     // LIMIT/OFFSET change the yielded rows: bail
1274        fetch,            // FETCH FIRST is LIMIT spelled differently: bail
1275        locks: _,         // row locking doesn't change the rows
1276        for_clause,       // FOR XML/JSON reshape the result: bail
1277        settings,         // ClickHouse SETTINGS can change results: bail
1278        format_clause: _, // output serialization only
1279        pipe_operators,   // `|>` operators transform the rows: bail
1280    } = subquery;
1281    if with.is_some()
1282        || limit_clause.is_some()
1283        || fetch.is_some()
1284        || for_clause.is_some()
1285        || settings.is_some()
1286        || !pipe_operators.is_empty()
1287    {
1288        return None;
1289    }
1290    let SetExpr::Select(select) = body.as_ref() else {
1291        return None;
1292    };
1293    let Select {
1294        select_token: _,
1295        // Deduplication is existence/membership-invariant; the IN rewrite
1296        // separately bails on `DISTINCT ON`, which is not.
1297        distinct: _,
1298        top,                    // TOP is LIMIT spelled differently: bail
1299        top_before_distinct: _, // only meaningful with `top`
1300        // The projection is validated by the callers: EXISTS ignores it, IN
1301        // requires a single plain expression (which also rules out wildcards
1302        // and the `exclude` modifier).
1303        projection: _,
1304        exclude: _,
1305        into,          // SELECT INTO is not a pure subquery: bail
1306        from,          // at least one relation, comma-separated ones cross join
1307        lateral_views, // row-multiplying: bail
1308        prewhere,      // an extra filter we don't fold in: bail
1309        selection: _,  // split into join keys/filters by callers
1310        group_by,      // aggregation changes the yielded rows: bail
1311        cluster_by: _, // layout/order hints: row-set preserving
1312        distribute_by: _,
1313        sort_by: _,
1314        having,                   // aggregation filter: bail
1315        named_window: _,          // definitions only; uses are parsed later
1316        qualify,                  // post-window filter changes the rows: bail
1317        window_before_qualify: _, // only meaningful with `qualify`
1318        value_table_mode,         // changes what a row is: bail
1319        connect_by,               // hierarchical recursion: bail
1320        optimizer_hints,          // unsupported: bail
1321        select_modifiers,         // unsupported: bail
1322        flavor: _,                // surface syntax only
1323    } = select.as_ref();
1324    let no_group_by = matches!(
1325        group_by,
1326        GroupByExpr::Expressions(e, m) if e.is_empty() && m.is_empty()
1327    );
1328    if from.is_empty()
1329        || !no_group_by
1330        || top.is_some()
1331        || into.is_some()
1332        || having.is_some()
1333        || qualify.is_some()
1334        || prewhere.is_some()
1335        || !connect_by.is_empty()
1336        || value_table_mode.is_some()
1337        || !lateral_views.is_empty()
1338        || !optimizer_hints.is_empty()
1339        || select_modifiers.is_some()
1340    {
1341        return None;
1342    }
1343    Some(select)
1344}

1365/// Whether a subquery reads a qualifier it does not declare, which is what
1366/// correlates it with the query around it.
1367pub(crate) fn is_correlated_subquery(query: &Query) -> bool {
1368    #[derive(Default)]
1369    struct Declared(PlHashSet<String>);
1370
1371    impl Visitor for Declared {
1372        type Break = ();
1373
1374        fn pre_visit_table_factor(&mut self, factor: &TableFactor) -> ControlFlow<()> {
1375            if let Some(name) = get_table_name(factor) {
1376                self.0.insert(name);
1377            }
1378            ControlFlow::Continue(())
1379        }
1380    }
1381
1382    let mut declared = Declared::default();
1383    let _ = query.visit(&mut declared);
1384
1385    struct Foreign<'a>(&'a PlHashSet<String>);
1386
1387    impl Visitor for Foreign<'_> {
1388        type Break = ();
1389
1390        fn pre_visit_expr(&mut self, expr: &SQLExpr) -> ControlFlow<()> {
1391            match qualifier_and_name(expr) {
1392                Some((Some(qualifier), _)) if !self.0.contains(qualifier) => ControlFlow::Break(()),
1393                _ => ControlFlow::Continue(()),
1394            }
1395        }
1396    }
1397
1398    query.visit(&mut Foreign(&declared.0)).is_break()
1399}

1401/// Rewrite `x = ANY (subquery)` to `x IN (subquery)` and `x <> ALL (subquery)`
1402/// to `x NOT IN (subquery)`, which they are equivalent to.
1403struct DesugarQuantified;
1404
1405impl VisitorMut for DesugarQuantified {
1406    type Break = ();
1407
1408    fn pre_visit_expr(&mut self, expr: &mut SQLExpr) -> ControlFlow<()> {
1409        let (left, right, negated) = match &*expr {
1410            SQLExpr::AnyOp {
1411                left,
1412                compare_op: SQLBinaryOperator::Eq,
1413                right,
1414                ..
1415            } => (left, right, false),
1416            SQLExpr::AllOp {
1417                left,
1418                compare_op: SQLBinaryOperator::NotEq,
1419                right,
1420            } => (left, right, true),
1421            _ => return ControlFlow::Continue(()),
1422        };
1423        let SQLExpr::Subquery(subquery) = right.as_ref() else {
1424            return ControlFlow::Continue(());
1425        };
1426        *expr = SQLExpr::InSubquery {
1427            expr: left.clone(),
1428            subquery: subquery.clone(),
1429            negated,
1430        };
1431        ControlFlow::Continue(())
1432    }
1433}

1435pub(crate) fn desugar_quantified_subqueries(stmt: &mut Statement) {
1436    let _ = VisitMut::visit(stmt, &mut DesugarQuantified);
1437}

1439/// Which correlated-subquery node kinds a lowering pass should claim.
1440#[derive(Clone, Copy, PartialEq, Eq)]
1441pub(crate) enum LowerScope {
1442    /// Scalar subqueries only, leaving predicate subqueries for the semi/anti join rewrite.
1443    ScalarOnly,
1444    /// Scalar subqueries and the predicate subqueries `[NOT] EXISTS` and `[NOT] IN`.
1445    ScalarAndPredicates,
1446}
1447
1448fn lowerable_kind(expr: &SQLExpr, scope: LowerScope) -> bool {
1449    match expr {
1450        SQLExpr::Subquery(_) => true,
1451        SQLExpr::Exists { .. } | SQLExpr::InSubquery { .. } => {
1452            scope == LowerScope::ScalarAndPredicates
1453        },
1454        _ => false,
1455    }
1456}

1458/// Rewrites correlated subqueries in-place into references to the columns their
1459/// decorrelated lowering materialises, threading the growing `LazyFrame` through the
1460/// traversal. `query_depth` keeps each subquery a leaf rather than recursing into it.
1461struct CorrelatedLowering<'a> {
1462    ctx: &'a mut SQLContext,
1463    lf: LazyFrame,
1464    outer_schema: &'a Schema,
1465    scope: LowerScope,
1466    bindings: &'a mut SubqueryBindings,
1467    query_depth: usize,
1468    changed: bool,
1469}
1470
1471impl CorrelatedLowering<'_> {
1472    /// Lower `subquery` (or reuse an earlier identical lowering), returning the
1473    /// materialised column name, or `None` if it can't be soundly lowered.
1474    fn lower(&mut self, subquery: &Query, kind: SubqueryKind) -> PolarsResult<Option<PlSmallStr>> {
1475        if let Some((_, _, name)) = self
1476            .bindings
1477            .iter()
1478            .find(|(q, bound, _)| *bound == kind && q == subquery)
1479        {
1480            return Ok(Some(name.clone()));
1481        }
1482        // The frame is cloned rather than moved so it survives a declined lowering.
1483        let lowered = match &kind {
1484            SubqueryKind::Exists => self.ctx.try_decorrelate_exists_subquery(
1485                self.lf.clone(),
1486                self.outer_schema,
1487                subquery,
1488            )?,
1489            SubqueryKind::Scalar => self.ctx.try_decorrelate_scalar_subquery(
1490                self.lf.clone(),
1491                self.outer_schema,
1492                subquery,
1493            )?,
1494            SubqueryKind::In(lhs) => self.ctx.try_decorrelate_in_subquery(
1495                self.lf.clone(),
1496                self.outer_schema,
1497                lhs,
1498                subquery,
1499            )?,
1500        };
1501        let Some((new_lf, name)) = lowered else {
1502            return Ok(None);
1503        };
1504        self.lf = new_lf;
1505        self.bindings.push((subquery.clone(), kind, name.clone()));
1506        Ok(Some(name))
1507    }
1508}
1509
1510impl VisitorMut for CorrelatedLowering<'_> {
1511    type Break = PolarsError;
1512
1513    fn pre_visit_query(&mut self, _query: &mut Query) -> ControlFlow<PolarsError> {
1514        self.query_depth += 1;
1515        ControlFlow::Continue(())
1516    }
1517
1518    fn post_visit_query(&mut self, _query: &mut Query) -> ControlFlow<PolarsError> {
1519        self.query_depth -= 1;
1520        ControlFlow::Continue(())
1521    }
1522
1523    fn pre_visit_expr(&mut self, expr: &mut SQLExpr) -> ControlFlow<PolarsError> {
1524        // Inside a nested query: not ours to lower against the outer schema.
1525        if self.query_depth > 0 || !lowerable_kind(expr, self.scope) {
1526            return ControlFlow::Continue(());
1527        }
1528        let (subquery, kind, negated) = match &*expr {
1529            SQLExpr::Subquery(subquery) => (subquery.as_ref(), SubqueryKind::Scalar, false),
1530            SQLExpr::Exists { subquery, negated } => {
1531                (subquery.as_ref(), SubqueryKind::Exists, *negated)
1532            },
1533            SQLExpr::InSubquery {
1534                expr: lhs,
1535                subquery,
1536                negated,
1537            } => (subquery.as_ref(), SubqueryKind::In(lhs.clone()), *negated),
1538            _ => unreachable!("guarded by lowerable_kind"),
1539        };
1540        let name = match self.lower(subquery, kind) {
1541            Ok(Some(name)) => name,
1542            Ok(None) => return ControlFlow::Continue(()),
1543            Err(e) => return ControlFlow::Break(e),
1544        };
1545        // `EXISTS` is never NULL; `IN` may be, and negating a NULL keeps it NULL,
1546        // which is what SQL asks for.
1547        let resolved = SQLExpr::Identifier(Ident::new(name.as_str()));
1548        *expr = if negated {
1549            SQLExpr::UnaryOp {
1550                op: SQLUnaryOperator::Not,
1551                expr: Box::new(resolved),
1552            }
1553        } else {
1554            resolved
1555        };
1556        self.changed = true;
1557        ControlFlow::Continue(())
1558    }
1559}

1561// Whether an expression can serve as a join key. Join keys must be elementwise,
1562// so that every key is as long as the frame it is built from. Anything this does
1563// not recognise declines the rewrite, which costs an optimisation rather than
1564// correctness.
1565fn usable_as_join_key(expr: &Expr) -> bool {
1566    match expr {
1567        Expr::Column(_) | Expr::Literal(_) => true,
1568        Expr::Alias(inner, _) | Expr::Cast { expr: inner, .. } => usable_as_join_key(inner),
1569        Expr::BinaryExpr { left, op: _, right } => {
1570            usable_as_join_key(left) && usable_as_join_key(right)
1571        },
1572        Expr::Ternary {
1573            predicate,
1574            truthy,
1575            falsy,
1576        } => {
1577            usable_as_join_key(predicate) && usable_as_join_key(truthy) && usable_as_join_key(falsy)
1578        },
1579        _ => false,
1580    }
1581}

1594// Left-join a per-group aggregate onto the outer frame, filling unmatched `COUNT`
1595// results with 0 (every other aggregate stays NULL for unmatched rows).
1596fn left_join_aggregate(
1597    outer: LazyFrame,
1598    grouped: LazyFrame,
1599    left_on: Vec<Expr>,
1600    right_on: Vec<Expr>,
1601    result_name: &PlSmallStr,
1602    count_like: bool,
1603) -> PolarsResult<LazyFrame> {
1604    let joined = outer
1605        .join_builder()
1606        .with(grouped)
1607        .left_on(left_on)
1608        .right_on(right_on)
1609        .how(JoinType::Left)
1610        .coalesce(JoinCoalesce::CoalesceColumns)
1611        .maintain_order(MaintainOrderJoin::Left)
1612        .finish()?;
1613    Ok(if count_like {
1614        joined.with_columns([col(result_name.clone()).fill_null(lit(0))])
1615    } else {
1616        joined
1617    })
1618}

1630// A correlation conjunct `<inner col> <cmp> <outer col>` (either way round),
1631// tracking which side is the inner column so the comparison can be rebuilt in
1632// its original orientation as a join predicate.
1633struct CorrPredicate {
1634    outer: PlSmallStr,
1635    inner: PlSmallStr,
1636    op: SQLBinaryOperator,
1637    inner_on_left: bool,
1638}
1639
1640impl CorrPredicate {
1641    fn to_expr(&self, prefix: &str) -> Expr {
1642        let outer = col(self.outer.clone());
1643        let inner = col(prefixed_inner(prefix, &self.inner));
1644        let (l, r) = if self.inner_on_left {
1645            (inner, outer)
1646        } else {
1647            (outer, inner)
1648        };
1649        match self.op {
1650            SQLBinaryOperator::Eq => l.eq(r),
1651            SQLBinaryOperator::NotEq => l.neq(r),
1652            SQLBinaryOperator::Lt => l.lt(r),
1653            SQLBinaryOperator::LtEq => l.lt_eq(r),
1654            SQLBinaryOperator::Gt => l.gt(r),
1655            SQLBinaryOperator::GtEq => l.gt_eq(r),
1656            // Guarded by `scalar_correlation_predicate`.
1657            _ => unreachable!("non-comparison correlation operator"),
1658        }
1659    }
1660}

1662// A comparison conjunct with one bare inner column and one bare outer column,
1663// or `None` for anything else. Both sides must be simple columns; more complex
1664// correlation shapes leave the whole subquery to the generic path.
1665fn scalar_correlation_predicate(
1666    conj: &SQLExpr,
1667    inner_names: &PlHashSet<String>,
1668    inner_schema: &Schema,
1669    outer_schema: &Schema,
1670) -> Option<CorrPredicate> {
1671    let SQLExpr::BinaryOp { left, op, right } = conj else {
1672        return None;
1673    };
1674    if !matches!(
1675        op,
1676        SQLBinaryOperator::Eq
1677            | SQLBinaryOperator::NotEq
1678            | SQLBinaryOperator::Lt
1679            | SQLBinaryOperator::LtEq
1680            | SQLBinaryOperator::Gt
1681            | SQLBinaryOperator::GtEq
1682    ) {
1683        return None;
1684    }
1685    let (lside, lname) =
1686        classify_correlation_column(left, inner_names, inner_schema, outer_schema)?;
1687    let (rside, rname) =
1688        classify_correlation_column(right, inner_names, inner_schema, outer_schema)?;
1689    match (lside, rside) {
1690        (CorrelationSide::Inner, CorrelationSide::Outer) => Some(CorrPredicate {
1691            outer: rname,
1692            inner: lname,
1693            op: op.clone(),
1694            inner_on_left: true,
1695        }),
1696        (CorrelationSide::Outer, CorrelationSide::Inner) => Some(CorrPredicate {
1697            outer: lname,
1698            inner: rname,
1699            op: op.clone(),
1700            inner_on_left: false,
1701        }),
1702        _ => None,
1703    }
