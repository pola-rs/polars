27/// Table functions that are supported by Polars
28#[allow(clippy::enum_variant_names)]
29pub(crate) enum PolarsTableFunctions {
30    /// SQL 'read_csv' function, with optional named options as in Python's `scan_csv`.
31    /// ```sql
32    /// SELECT * FROM read_csv('path/to/file.csv')
33    /// SELECT * FROM read_csv('path/to/file.csv', has_header => false, separator => ';')
34    /// ```
35    #[cfg(feature = "csv")]
36    ReadCsv,
37    /// SQL 'read_parquet' function.
38    /// ```sql
39    /// SELECT * FROM read_parquet('path/to/file.parquet')
40    /// ```
41    #[cfg(feature = "parquet")]
42    ReadParquet,
43    /// SQL 'read_ipc' function.
44    /// ```sql
45    /// SELECT * FROM read_ipc('path/to/file.ipc')
46    /// ```
47    #[cfg(feature = "ipc")]
48    ReadIpc,
49    /// SQL 'read_json' function: reads newline-delimited JSON from a path only; for named options,
50    /// use `read_ndjson`. It stays path-only so that it can later read JSON documents, as Python's
51    /// `read_json` does, without a change of meaning for its options.
52    /// ```sql
53    /// SELECT * FROM read_json('path/to/file.ndjson')
54    /// ```
55    #[cfg(feature = "json")]
56    ReadJson,
57    /// SQL 'read_ndjson' function, with optional named options as in Python's `scan_ndjson`.
58    /// ```sql
59    /// SELECT * FROM read_ndjson('path/to/file.ndjson')
60    /// SELECT * FROM read_ndjson('path/to/file.ndjson', infer_schema_length => NULL)
61    /// ```
62    #[cfg(feature = "json")]
63    ReadNdjson,
64    /// SQL 'generate_series' function (stop included).
65    /// ```sql
66    /// SELECT * FROM generate_series(1, 10, 2)
67    /// SELECT * FROM generate_series(DATE '2026-01-01', DATE '2026-01-31', INTERVAL '1 day')
68    /// ```
69    GenerateSeries,
70    /// SQL 'range' function (stop not included).
71    /// ```sql
72    /// SELECT * FROM range(0, 10, 2)
73    /// ```
74    Range,
75}
76
77impl FromStr for PolarsTableFunctions {
78    type Err = PolarsError;
79
80    #[allow(unreachable_code)]
81    fn from_str(s: &str) -> Result<Self, Self::Err> {
82        Ok(match s.to_lowercase().as_str() {
83            #[cfg(feature = "csv")]
84            "read_csv" => PolarsTableFunctions::ReadCsv,
85            #[cfg(feature = "parquet")]
86            "read_parquet" => PolarsTableFunctions::ReadParquet,
87            #[cfg(feature = "ipc")]
88            "read_ipc" => PolarsTableFunctions::ReadIpc,
89            #[cfg(feature = "json")]
90            "read_json" => PolarsTableFunctions::ReadJson,
91            #[cfg(feature = "json")]
92            "read_ndjson" => PolarsTableFunctions::ReadNdjson,
93            "generate_series" => PolarsTableFunctions::GenerateSeries,
94            "range" => PolarsTableFunctions::Range,
95            _ => polars_bail!(SQLInterface: "'{}' is not a supported table function", s),
96        })
97    }
98}

100impl PolarsTableFunctions {
101    #[allow(unreachable_code)]
102    pub(crate) fn execute(
103        &self,
104        args: &[SQLFunctionArg],
105        ctx: &mut SQLContext,
106    ) -> PolarsResult<(String, LazyFrame)> {
107        let (path, lf): (PlRefPath, LazyFrame) = match self {
108            #[cfg(feature = "csv")]
109            PolarsTableFunctions::ReadCsv => self.read_csv(args)?,
110            #[cfg(feature = "parquet")]
111            PolarsTableFunctions::ReadParquet => self.read_parquet(args)?,
112            #[cfg(feature = "ipc")]
113            PolarsTableFunctions::ReadIpc => self.read_ipc(args)?,
114            #[cfg(feature = "json")]
115            PolarsTableFunctions::ReadJson => self.read_json(args)?,
116            #[cfg(feature = "json")]
117            PolarsTableFunctions::ReadNdjson => self.read_ndjson(args)?,
118            PolarsTableFunctions::GenerateSeries => return self.series(args, ctx, true),
119            PolarsTableFunctions::Range => return self.series(args, ctx, false),
120        };
121        Ok((path.to_string(), lf))
122    }
123
124    fn series(
125        &self,
126        args: &[SQLFunctionArg],
127        ctx: &mut SQLContext,
128        include_end: bool,
129    ) -> PolarsResult<(String, LazyFrame)> {
130        let fname = if include_end {
131            "generate_series"
132        } else {
133            "range"
134        };
135        let args = args
136            .iter()
137            .map(|arg| match arg {
138                SQLFunctionArg::Unnamed(SQLFunctionArgExpr::Expr(e)) => Ok(e),
139                _ => polars_bail!(SQLSyntax: "`{}` expects positional arguments; found {}", fname, arg),
140            })
141            .collect::<PolarsResult<Vec<_>>>()?;
142        let mut expr = series_expr(fname, &args, include_end, |e| {
143            let expr = parse_sql_expr(e, ctx, None)?;
144            polars_ensure!(
145                expr_to_leaf_column_names_iter(&expr).next().is_none(),
146                SQLInterface: "`{}` in FROM takes only constant arguments; use UNNEST({}(...)) in the SELECT list for a series per row", fname, fname
147            );
148            Ok(expr)
149        })?;
150        let (lf, _) = ctx.process_subqueries(
151            DataFrame::empty_with_height(1).lazy(),
152            vec![&mut expr],
153            SubqueryShape::Scalar,
154        )?;
155        let lf = lf.select([expr
156            .explode(ExplodeOptions {
157                empty_as_null: false,
158                keep_nulls: false,
159            })
160            .alias(fname)]);
161        Ok((fname.to_string(), lf))
162    }

164    #[cfg(feature = "csv")]
165    fn read_csv(&self, args: &[SQLFunctionArg]) -> PolarsResult<(PlRefPath, LazyFrame)> {
166        use std::sync::Arc;
167
168        use polars_io::csv::read::{CsvEncoding, NullValues};
169        use polars_lazy::frame::LazyFileListReader;
170        use polars_plan::dsl::{ExtraColumnsPolicy, MissingColumnsPolicy};
171
172        let (path_arg, options) = TableFunctionOptions::parse(
173            "read_csv",
174            args,
175            READ_CSV_OPTIONS,
176            READ_CSV_EXCLUDED_OPTIONS,
177        )?;
178        let path = self.get_file_path_from_arg(path_arg)?;
179
180        let has_header = options.bool("has_header")?.unwrap_or(true);
181        let schema = options.schema("schema")?;
182        let mut reader = LazyCsvReader::new(path.clone())
183            .with_has_header(has_header)
184            .with_try_parse_dates(options.bool("try_parse_dates")?.unwrap_or(true))
185            // As in Python: an empty file only errors when there is no schema to return.
186            .with_raise_if_empty(schema.is_none() || has_header)
187            .with_schema(schema.map(Arc::new));
188
189        if let Some(separator) = options.byte("separator")? {
190            reader = reader.with_separator(separator);
191        }
192        if let Some(quote_char) = options.nullable_byte("quote_char")? {
193            reader = reader.with_quote_char(quote_char);
194        }
195        if let Some(prefix) = options.nullable_string("comment_prefix")? {
196            reader = reader.with_comment_prefix(prefix.map(PlSmallStr::from));
197        }
198        if let Some(n) = options.usize("skip_rows")? {
199            reader = reader.with_skip_rows(n);
200        }
201        if let Some(n) = options.usize("skip_rows_after_header")? {
202            reader = reader.with_skip_rows_after_header(n);
203        }
204        if let Some(n) = options.usize("skip_lines")? {
205            reader = reader.with_skip_lines(n);
206        }
207        if let Some(encoding) = options.one_of(
208            "encoding",
209            &[
210                ("utf8", CsvEncoding::Utf8),
211                ("utf8-lossy", CsvEncoding::LossyUtf8),
212            ],
213        )? {
214            reader = reader.with_encoding(encoding);
215        }
216        if let Some(names) = options.strings("new_columns")? {
217            let names: Vec<PlSmallStr> = names.into_iter().map(PlSmallStr::from).collect();
218            reader = reader.with_column_names_overwrite(names.into());
219        }
220        match options.dtype_overrides("schema_overrides")? {
221            Some(DtypeOverrides::ByName(schema)) => {
222                reader = reader.with_dtype_overwrite(Some(Arc::new(schema)));
223            },
224            Some(DtypeOverrides::ByPosition(dtypes)) => {
225                reader = reader.with_dtype_overwrite_by_position(Some(Arc::new(dtypes)));
226            },
227            None => {},
228        }
229        if let Some(n) = options.nullable_usize("infer_schema_length")? {
230            reader = reader.with_infer_schema_length(n);
231        }
232        if let Some(n) = options.positive_usize("infer_schema_files")? {
233            reader = reader.with_infer_schema_files(n);
234        }
235        // As in Python: `infer_schema => false` reads every column as a string.
236        if options.bool("infer_schema")? == Some(false) {
237            reader = reader.with_infer_schema_length(Some(0));
238        }
239        if let Some(decimal_comma) = options.bool("decimal_comma")? {
240            reader = reader.with_decimal_comma(decimal_comma);
241        }
242        let null_values = options
243            .string_values("null_values")?
244            .map(|values| match values {
245                StringValues::One(value) => NullValues::AllColumnsSingle(value.into()),
246                StringValues::List(values) => {
247                    NullValues::AllColumns(values.into_iter().map(Into::into).collect())
248                },
249                StringValues::Dict(pairs) => NullValues::Named(
250                    pairs
251                        .into_iter()
252                        .map(|(k, v)| (k.into(), v.into()))
253                        .collect(),
254                ),
255            });
256        reader = reader.with_null_values(null_values);
257        if let Some(ignore_errors) = options.bool("ignore_errors")? {
258            reader = reader.with_ignore_errors(ignore_errors);
259        }
260        if let Some(policy) = options.one_of(
261            "missing_columns",
262            &[
263                ("insert", MissingColumnsPolicy::Insert),
264                ("raise", MissingColumnsPolicy::Raise),
265            ],
266        )? {
267            reader = reader.with_missing_columns_policy(Some(policy));
268        }
269        let extra_columns = options.one_of(
270            "extra_columns",
271            &[
272                ("ignore", ExtraColumnsPolicy::Ignore),
273                ("raise", ExtraColumnsPolicy::Raise),
274            ],
275        )?;
276        // As in Python: ignoring extra columns also truncates rows with extra fields, unless
277        // `truncate_ragged_lines` is given. The reader doesn't apply this rule itself, so `scan_csv`
278        // and `read_csv` each do, as for `raise_if_empty` above.
279        let ignore_extra_columns = extra_columns == Some(ExtraColumnsPolicy::Ignore);
280        let truncate = options.bool("truncate_ragged_lines")?;
281        polars_ensure!(
282            !(ignore_extra_columns && truncate == Some(false)),
283            SQLSyntax: "`read_csv` cannot set truncate_ragged_lines => false with extra_columns => 'ignore'"
284        );
285        if let Some(truncate) = truncate.or(ignore_extra_columns.then_some(true)) {
286            reader = reader.with_truncate_ragged_lines(truncate);
287        }
288        if let Some(policy) = extra_columns {
289            reader = reader.with_extra_columns_policy(policy);
290        }
291        if let Some(column) = options.string("include_file_paths")? {
292            reader = reader.with_include_file_paths(Some(column.into()));
293        }
294
295        let lf = reader.finish()?;
296        Ok((path, lf))
297    }

299    #[cfg(feature = "parquet")]
300    fn read_parquet(&self, args: &[SQLFunctionArg]) -> PolarsResult<(PlRefPath, LazyFrame)> {
301        polars_ensure!(args.len() == 1, SQLSyntax: "`read_parquet` expects a single file path; found {:?} arguments", args.len());
302
303        let path = self.get_file_path_from_arg(&args[0])?;
304        let lf = LazyFrame::scan_parquet(path.clone(), Default::default())?;
305        Ok((path, lf))
306    }

316    #[cfg(feature = "json")]
317    fn read_json(&self, args: &[SQLFunctionArg]) -> PolarsResult<(PlRefPath, LazyFrame)> {
318        use polars_lazy::frame::LazyFileListReader;
319        use polars_lazy::prelude::LazyJsonLineReader;
320
321        polars_ensure!(
322            args.len() == 1,
323            SQLSyntax: "`read_json` expects a single file path; found {} arguments{}",
324            args.len(),
325            if args.len() > 1 { " (for options, use `read_ndjson`)" } else { "" }
326        );
327        let path = self.get_file_path_from_arg(&args[0])?;
328        let lf = LazyJsonLineReader::new(path.clone()).finish()?;
329        Ok((path, lf))
330    }
331
332    #[cfg(feature = "json")]
333    fn read_ndjson(&self, args: &[SQLFunctionArg]) -> PolarsResult<(PlRefPath, LazyFrame)> {
334        use std::sync::Arc;
335
336        use polars_lazy::frame::LazyFileListReader;
337        use polars_lazy::prelude::LazyJsonLineReader;
338
339        let (path_arg, options) = TableFunctionOptions::parse(
340            "read_ndjson",
341            args,
342            READ_NDJSON_OPTIONS,
343            READ_NDJSON_EXCLUDED_OPTIONS,
344        )?;
345        let path = self.get_file_path_from_arg(path_arg)?;
346
347        let mut reader = LazyJsonLineReader::new(path.clone())
348            .with_schema(options.schema("schema")?.map(Arc::new))
349            .with_schema_overwrite(options.schema("schema_overrides")?.map(Arc::new));
350        if let Some(n) = options.nullable_positive_usize("infer_schema_length")? {
351            reader = reader.with_infer_schema_length(n);
352        }
353        if let Some(ignore_errors) = options.bool("ignore_errors")? {
354            reader = reader.with_ignore_errors(ignore_errors);
355        }
356        if let Some(column) = options.string("include_file_paths")? {
357            reader = reader.with_include_file_paths(Some(column.into()));
358        }
359
360        let lf = reader.finish()?;
361        Ok((path, lf))
362    }
363
364    #[allow(dead_code)]
365    fn get_file_path_from_arg(&self, arg: &SQLFunctionArg) -> PolarsResult<PlRefPath> {
366        match arg {
367            SQLFunctionArg::Unnamed(SQLFunctionArgExpr::Expr(SQLExpr::Value(
368                SQLValueWithSpan {
369                    value: SQLValue::SingleQuotedString(s) | SQLValue::EscapedStringLiteral(s),
370                    ..
371                },
372            ))) => Ok(PlRefPath::new(s)),
373            _ => polars_bail!(
374                SQLSyntax:
375                "expected a valid file path as a single-quoted string; found: {}", arg,
376            ),
377        }
378    }
