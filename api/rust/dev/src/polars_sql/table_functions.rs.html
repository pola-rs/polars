32/// Table functions that are supported by Polars
33#[allow(clippy::enum_variant_names)]
34pub(crate) enum PolarsTableFunctions {
35    /// SQL 'read_csv' function, with optional named options as in Python's `scan_csv`.
36    /// ```sql
37    /// SELECT * FROM read_csv('path/to/file.csv')
38    /// SELECT * FROM read_csv('path/to/file.csv', has_header => false, separator => ';')
39    /// ```
40    #[cfg(feature = "csv")]
41    ReadCsv,
42    /// SQL 'read_parquet' function, with optional named options as in Python's `scan_parquet`.
43    /// ```sql
44    /// SELECT * FROM read_parquet('path/to/file.parquet')
45    /// SELECT * FROM read_parquet('path/to/*.parquet', hive_partitioning => false)
46    /// ```
47    #[cfg(feature = "parquet")]
48    ReadParquet,
49    /// SQL 'read_ipc' function, with optional named options as in Python's `scan_ipc`.
50    /// ```sql
51    /// SELECT * FROM read_ipc('path/to/file.ipc')
52    /// SELECT * FROM read_ipc('path/to/*.ipc', hive_partitioning => false)
53    /// ```
54    #[cfg(feature = "ipc")]
55    ReadIpc,
56    /// SQL 'read_json' function: reads newline-delimited JSON from a path only; for named options,
57    /// use `read_ndjson`. It stays path-only so that it can later read JSON documents, as Python's
58    /// `read_json` does, without a change of meaning for its options.
59    /// ```sql
60    /// SELECT * FROM read_json('path/to/file.ndjson')
61    /// ```
62    #[cfg(feature = "json")]
63    ReadJson,
64    /// SQL 'read_ndjson' function, with optional named options as in Python's `scan_ndjson`.
65    /// ```sql
66    /// SELECT * FROM read_ndjson('path/to/file.ndjson')
67    /// SELECT * FROM read_ndjson('path/to/file.ndjson', infer_schema_length => NULL)
68    /// ```
69    #[cfg(feature = "json")]
70    ReadNdjson,
71    /// SQL 'generate_series' function (stop included).
72    /// ```sql
73    /// SELECT * FROM generate_series(1, 10, 2)
74    /// SELECT * FROM generate_series(DATE '2026-01-01', DATE '2026-01-31', INTERVAL '1 day')
75    /// ```
76    GenerateSeries,
77    /// SQL 'range' function (stop not included).
78    /// ```sql
79    /// SELECT * FROM range(0, 10, 2)
80    /// ```
81    Range,
82}

84impl FromStr for PolarsTableFunctions {
85    type Err = PolarsError;
86
87    #[allow(unreachable_code)]
88    fn from_str(s: &str) -> Result<Self, Self::Err> {
89        Ok(match s.to_lowercase().as_str() {
90            #[cfg(feature = "csv")]
91            "read_csv" => PolarsTableFunctions::ReadCsv,
92            #[cfg(feature = "parquet")]
93            "read_parquet" => PolarsTableFunctions::ReadParquet,
94            #[cfg(feature = "ipc")]
95            "read_ipc" => PolarsTableFunctions::ReadIpc,
96            #[cfg(feature = "json")]
97            "read_json" => PolarsTableFunctions::ReadJson,
98            #[cfg(feature = "json")]
99            "read_ndjson" => PolarsTableFunctions::ReadNdjson,
100            "generate_series" => PolarsTableFunctions::GenerateSeries,
101            "range" => PolarsTableFunctions::Range,
102            _ => polars_bail!(SQLInterface: "'{}' is not a supported table function", s),
103        })
104    }
105}

108    #[allow(unreachable_code)]
109    pub(crate) fn execute(
110        &self,
111        args: &[SQLFunctionArg],
112        ctx: &mut SQLContext,
113    ) -> PolarsResult<(String, LazyFrame)> {
114        let (name, lf): (String, LazyFrame) = match self {
115            #[cfg(feature = "csv")]
116            PolarsTableFunctions::ReadCsv => self.read_csv(args)?,
117            #[cfg(feature = "parquet")]
118            PolarsTableFunctions::ReadParquet => self.read_parquet(args)?,
119            #[cfg(feature = "ipc")]
120            PolarsTableFunctions::ReadIpc => self.read_ipc(args)?,
121            #[cfg(feature = "json")]
122            PolarsTableFunctions::ReadJson => self.read_json(args)?,
123            #[cfg(feature = "json")]
124            PolarsTableFunctions::ReadNdjson => self.read_ndjson(args)?,
125            PolarsTableFunctions::GenerateSeries => return self.series(args, ctx, true),
126            PolarsTableFunctions::Range => return self.series(args, ctx, false),
127        };
128        Ok((name, lf))
129    }

131    fn series(
132        &self,
133        args: &[SQLFunctionArg],
134        ctx: &mut SQLContext,
135        include_end: bool,
136    ) -> PolarsResult<(String, LazyFrame)> {
137        let fname = if include_end {
138            "generate_series"
139        } else {
140            "range"
141        };
142        let args = args
143            .iter()
144            .map(|arg| match arg {
145                SQLFunctionArg::Unnamed(SQLFunctionArgExpr::Expr(e)) => Ok(e),
146                _ => polars_bail!(SQLSyntax: "`{}` expects positional arguments; found {}", fname, arg),
147            })
148            .collect::<PolarsResult<Vec<_>>>()?;
149        let mut expr = series_expr(fname, &args, include_end, |e| {
150            let expr = parse_sql_expr(e, ctx, None)?;
151            polars_ensure!(
152                expr_to_leaf_column_names_iter(&expr).next().is_none(),
153                SQLInterface: "`{}` in FROM takes only constant arguments; use UNNEST({}(...)) in the SELECT list for a series per row", fname, fname
154            );
155            Ok(expr)
156        })?;
157        let (lf, _) = ctx.process_subqueries(
158            DataFrame::empty_with_height(1).lazy(),
159            vec![&mut expr],
160            SubqueryShape::Scalar,
161        )?;
162        let lf = lf.select([expr
163            .explode(ExplodeOptions {
164                empty_as_null: false,
165                keep_nulls: false,
166            })
167            .alias(fname)]);
168        Ok((fname.to_string(), lf))
169    }

171    #[cfg(feature = "csv")]
172    fn read_csv(&self, args: &[SQLFunctionArg]) -> PolarsResult<(String, LazyFrame)> {
173        use std::sync::Arc;
174
175        use polars_io::csv::read::{CsvEncoding, NullValues};
176        use polars_lazy::frame::LazyFileListReader;
177        use polars_plan::dsl::{ExtraColumnsPolicy, MissingColumnsPolicy};
178
179        let (path_arg, options) = TableFunctionOptions::parse(
180            "read_csv",
181            args,
182            READ_CSV_OPTIONS,
183            READ_CSV_EXCLUDED_OPTIONS,
184        )?;
185        let (name, paths) = self.get_file_paths_from_arg("read_csv", path_arg)?;
186
187        let has_header = options.bool("has_header")?.unwrap_or(true);
188        let schema = options.schema("schema")?;
189        let mut reader = LazyCsvReader::new_paths(paths.into())
190            .with_has_header(has_header)
191            .with_try_parse_dates(options.bool("try_parse_dates")?.unwrap_or(true))
192            // As in Python: an empty file only errors when there is no schema to return.
193            .with_raise_if_empty(schema.is_none() || has_header)
194            .with_schema(schema.map(Arc::new));
195
196        if let Some(separator) = options.byte("separator")? {
197            reader = reader.with_separator(separator);
198        }
199        if let Some(quote_char) = options.nullable_byte("quote_char")? {
200            reader = reader.with_quote_char(quote_char);
201        }
202        if let Some(prefix) = options.nullable_string("comment_prefix")? {
203            reader = reader.with_comment_prefix(prefix.map(PlSmallStr::from));
204        }
205        if let Some(n) = options.usize("skip_rows")? {
206            reader = reader.with_skip_rows(n);
207        }
208        if let Some(n) = options.usize("skip_rows_after_header")? {
209            reader = reader.with_skip_rows_after_header(n);
210        }
211        if let Some(n) = options.usize("skip_lines")? {
212            reader = reader.with_skip_lines(n);
213        }
214        if let Some(encoding) = options.one_of(
215            "encoding",
216            &[
217                ("utf8", CsvEncoding::Utf8),
218                ("utf8-lossy", CsvEncoding::LossyUtf8),
219            ],
220        )? {
221            reader = reader.with_encoding(encoding);
222        }
223        if let Some(names) = options.strings("new_columns")? {
224            let names: Vec<PlSmallStr> = names.into_iter().map(PlSmallStr::from).collect();
225            reader = reader.with_column_names_overwrite(names.into());
226        }
227        match options.dtype_overrides("schema_overrides")? {
228            Some(DtypeOverrides::ByName(schema)) => {
229                reader = reader.with_dtype_overwrite(Some(Arc::new(schema)));
230            },
231            Some(DtypeOverrides::ByPosition(dtypes)) => {
232                reader = reader.with_dtype_overwrite_by_position(Some(Arc::new(dtypes)));
233            },
234            None => {},
235        }
236        if let Some(n) = options.nullable_usize("infer_schema_length")? {
237            reader = reader.with_infer_schema_length(n);
238        }
239        if let Some(n) = options.positive_usize("infer_schema_files")? {
240            reader = reader.with_infer_schema_files(n);
241        }
242        // As in Python: `infer_schema => false` reads every column as a string.
243        if options.bool("infer_schema")? == Some(false) {
244            reader = reader.with_infer_schema_length(Some(0));
245        }
246        if let Some(decimal_comma) = options.bool("decimal_comma")? {
247            reader = reader.with_decimal_comma(decimal_comma);
248        }
249        let null_values = options
250            .string_values("null_values")?
251            .map(|values| match values {
252                StringValues::One(value) => NullValues::AllColumnsSingle(value.into()),
253                StringValues::List(values) => {
254                    NullValues::AllColumns(values.into_iter().map(Into::into).collect())
255                },
256                StringValues::Dict(pairs) => NullValues::Named(
257                    pairs
258                        .into_iter()
259                        .map(|(k, v)| (k.into(), v.into()))
260                        .collect(),
261                ),
262            });
263        reader = reader.with_null_values(null_values);
264        if let Some(ignore_errors) = options.bool("ignore_errors")? {
265            reader = reader.with_ignore_errors(ignore_errors);
266        }
267        if let Some(policy) = options.one_of(
268            "missing_columns",
269            &[
270                ("insert", MissingColumnsPolicy::Insert),
271                ("raise", MissingColumnsPolicy::Raise),
272            ],
273        )? {
274            reader = reader.with_missing_columns_policy(Some(policy));
275        }
276        let extra_columns = options.one_of(
277            "extra_columns",
278            &[
279                ("ignore", ExtraColumnsPolicy::Ignore),
280                ("raise", ExtraColumnsPolicy::Raise),
281            ],
282        )?;
283        // As in Python: ignoring extra columns also truncates rows with extra fields, unless
284        // `truncate_ragged_lines` is given. The reader doesn't apply this rule itself, so `scan_csv`
285        // and `read_csv` each do, as for `raise_if_empty` above.
286        let ignore_extra_columns = extra_columns == Some(ExtraColumnsPolicy::Ignore);
287        let truncate = options.bool("truncate_ragged_lines")?;
288        polars_ensure!(
289            !(ignore_extra_columns && truncate == Some(false)),
290            SQLSyntax: "`read_csv` cannot set truncate_ragged_lines => false with extra_columns => 'ignore'"
291        );
292        if let Some(truncate) = truncate.or(ignore_extra_columns.then_some(true)) {
293            reader = reader.with_truncate_ragged_lines(truncate);
294        }
295        if let Some(policy) = extra_columns {
296            reader = reader.with_extra_columns_policy(policy);
297        }
298        if let Some(column) = options.string("include_file_paths")? {
299            reader = reader.with_include_file_paths(Some(column.into()));
300        }
301
302        let lf = reader.finish()?;
303        Ok((name, lf))
304    }

306    #[cfg(feature = "parquet")]
307    fn read_parquet(&self, args: &[SQLFunctionArg]) -> PolarsResult<(String, LazyFrame)> {
308        use std::sync::Arc;
309
310        use polars_io::HiveOptions;
311        use polars_io::prelude::ParquetOptions;
312        use polars_plan::dsl::{
313            DslBuilder, ExtraColumnsPolicy, MissingColumnsPolicy, ScanSources, UnifiedScanArgs,
314        };
315
316        let (path_arg, options) = TableFunctionOptions::parse(
317            "read_parquet",
318            args,
319            READ_PARQUET_OPTIONS,
320            READ_PARQUET_EXCLUDED_OPTIONS,
321        )?;
322        let (name, paths) = self.get_file_paths_from_arg("read_parquet", path_arg)?;
323
324        let parquet_options = ParquetOptions {
325            schema: options.schema("schema")?.map(Arc::new),
326            ..Default::default()
327        };
328        // `ScanArgsParquet` has no `extra_columns`, so the scan is built as in Python.
329        let scan_args = UnifiedScanArgs {
330            hive_options: HiveOptions {
331                // Unlike `scan_parquet`, SQL reads the hive partitions of one path unless told not
332                // to. A list of paths gets the automatic choice, as `hive_options` says.
333                enabled: options
334                    .bool("hive_partitioning")?
335                    .or(hive_options(path_arg).enabled),
336                hive_start_idx: 0,
337                schema: options.schema("hive_schema")?.map(Arc::new),
338                try_parse_dates: options.bool("try_parse_hive_dates")?.unwrap_or(true),
339            },
340            missing_columns_policy: options
341                .one_of(
342                    "missing_columns",
343                    &[
344                        ("insert", MissingColumnsPolicy::Insert),
345                        ("raise", MissingColumnsPolicy::Raise),
346                    ],
347                )?
348                .unwrap_or_default(),
349            extra_columns_policy: options
350                .one_of(
351                    "extra_columns",
352                    &[
353                        ("ignore", ExtraColumnsPolicy::Ignore),
354                        ("raise", ExtraColumnsPolicy::Raise),
355                    ],
356                )?
357                .unwrap_or_default(),
358            include_file_paths: options.string("include_file_paths")?.map(Into::into),
359            cache: true,
360            ..Default::default()
361        };
362        let sources = ScanSources::Paths(paths.into());
363        let lf: LazyFrame = DslBuilder::scan_parquet(sources, parquet_options, scan_args)?
364            .build()
365            .into();
366        Ok((name, lf))
367    }

453    /// The paths given as one string or as a list of strings, with the name of the table they
454    /// form: the path itself, or the function name for a list.
455    #[allow(dead_code)]
456    fn get_file_paths_from_arg(
457        &self,
458        function: &str,
459        arg: &SQLFunctionArg,
460    ) -> PolarsResult<(String, Vec<PlRefPath>)> {
461        let path = |expr: &SQLExpr| match expr {
462            SQLExpr::Value(SQLValueWithSpan {
463                value: SQLValue::SingleQuotedString(s) | SQLValue::EscapedStringLiteral(s),
464                ..
465            }) => Some(PlRefPath::new(s)),
466            _ => None,
467        };
468        let SQLFunctionArg::Unnamed(SQLFunctionArgExpr::Expr(expr)) = arg else {
469            polars_bail!(
470                SQLSyntax:
471                "expected a valid file path as a single-quoted string; found: {}", arg,
472            );
473        };
474        if let Some(path) = path(expr) {
475            return Ok((path.to_string(), vec![path]));
476        }
477        if let SQLExpr::Array(SQLArray { elem, .. }) = expr {
478            if let Some(paths) = elem.iter().map(path).collect::<Option<Vec<_>>>() {
479                return Ok((function.to_string(), paths));
480            }
481        }
482        polars_bail!(
483            SQLSyntax:
484            "expected a valid file path as a single-quoted string, or a list of them; found: {}", arg,
485        )
486    }
487}

489/// As in Python: a list of paths only reads Hive partitions when it is a single directory. One path
490/// always reads them.
491#[cfg(any(feature = "parquet", feature = "ipc"))]
492fn hive_options(path_arg: &SQLFunctionArg) -> polars_io::HiveOptions {
493    let is_list = matches!(
494        path_arg,
495        SQLFunctionArg::Unnamed(SQLFunctionArgExpr::Expr(SQLExpr::Array(_)))
496    );
497    polars_io::HiveOptions {
498        enabled: if is_list { None } else { Some(true) },
499        ..Default::default()
500    }
501}
