14polars_utils::regex_cache::cached_regex! {
15    static DATETIME_LITERAL_RE = r"^\d{4}-[01]\d-[0-3]\d[ T](?:[01][0-9]|2[0-3]):[0-5][0-9](?::[0-5][0-9](\.\d{1,9})?)?(?:Z|[+-](?:[01][0-9]|2[0-3])(?::?[0-5][0-9])?)?$";
16    static DATE_LITERAL_RE = r"^\d{4}-[01]\d-[0-3]\d$";
17    static UTC_OFFSET_RE = r"[ T]\d{2}:\d{2}(?::\d{2}(?:\.\d+)?)?(Z|[+-]\d{2}(?::?\d{2})?)$";
18    static TIME_LITERAL_RE = r"^(?:[01][0-9]|2[0-3]):[0-5][0-9](?::[0-5][0-9](\.\d{1,9})?)?$";
19}

21pub fn bitstring_to_bytes_literal(b: &String) -> PolarsResult<Expr> {
22    let n_bits = b.len();
23    if !b.chars().all(|c| c == '0' || c == '1') || n_bits > 64 {
24        polars_bail!(
25            SQLSyntax:
26            "bit string literal should contain only 0s and 1s and have length <= 64; found '{}' with length {}", b, n_bits
27        )
28    }
29    let s = b.as_str();
30    Ok(lit(match n_bits {
31        0 => b"".to_vec(),
32        1..=8 => u8::from_str_radix(s, 2).unwrap().to_be_bytes().to_vec(),
33        9..=16 => u16::from_str_radix(s, 2).unwrap().to_be_bytes().to_vec(),
34        17..=32 => u32::from_str_radix(s, 2).unwrap().to_be_bytes().to_vec(),
35        _ => u64::from_str_radix(s, 2).unwrap().to_be_bytes().to_vec(),
36    }))
37}

39pub fn is_iso_datetime(value: &str) -> bool {
40    DATETIME_LITERAL_RE.is_match(value)
41}

43/// `value` without a UTC offset after its time, if it has one.
44pub fn strip_utc_offset(value: &str) -> &str {
45    match UTC_OFFSET_RE.captures(value).and_then(|c| c.get(1)) {
46        Some(offset) => &value[..offset.start()],
47        None => value,
48    }
49}

51pub fn is_iso_date(value: &str) -> bool {
52    DATE_LITERAL_RE.is_match(value)
53}

55pub fn is_iso_time(value: &str) -> bool {
56    TIME_LITERAL_RE.is_match(value)

59pub(crate) fn timeunit_from_precision(prec: &Option<u64>) -> PolarsResult<TimeUnit> {
60    Ok(match prec {
61        None => TimeUnit::Microseconds,
62        Some(n) if (1u64..=3u64).contains(n) => TimeUnit::Milliseconds,
63        Some(n) if (4u64..=6u64).contains(n) => TimeUnit::Microseconds,
64        Some(n) if (7u64..=9u64).contains(n) => TimeUnit::Nanoseconds,
65        Some(n) => {
66            polars_bail!(SQLSyntax: "invalid temporal type precision (expected 1-9, found {})", n)
67        },
68    })
69}

71pub(crate) fn map_sql_dtype_to_polars(dtype: &SQLDataType) -> PolarsResult<DataType> {
72    Ok(match dtype {
73        // ---------------------------------
74        // array/list
75        // ---------------------------------
76        SQLDataType::Array(ArrayElemTypeDef::AngleBracket(inner_type))
77        | SQLDataType::Array(ArrayElemTypeDef::SquareBracket(inner_type, _)) => {
78            DataType::List(Box::new(map_sql_dtype_to_polars(inner_type)?))
79        },
80
81        // ---------------------------------
82        // struct/map
83        // ---------------------------------
84        SQLDataType::Struct(fields, _) => {
85            let mut names = PlHashSet::with_capacity(fields.len());
86            let fields = fields
87                .iter()
88                .map(|field| {
89                    let Some(name) = &field.field_name else {
90                        polars_bail!(SQLInterface: "STRUCT fields must be named; found: {}", field)
91                    };
92                    polars_ensure!(
93                        field.options.is_none(),
94                        SQLInterface: "STRUCT field OPTIONS are not supported; found: {}", field
95                    );
96                    polars_ensure!(
97                        names.insert(name.value.as_str()),
98                        SQLSyntax: "STRUCT field '{}' is given more than once", name.value
99                    );
100                    let dtype = map_sql_dtype_to_polars(&field.field_type)?;
101                    Ok(Field::new(name.value.as_str().into(), dtype))
102                })
103                .collect::<PolarsResult<Vec<_>>>()?;
104            polars_ensure!(!fields.is_empty(), SQLSyntax: "STRUCT type requires at least one field");
105            DataType::Struct(fields)
106        },
107        #[cfg(feature = "dtype-map")]
108        SQLDataType::Map(key, value) => {
109            let dtype = DataType::Map(
110                Box::new(map_sql_dtype_to_polars(key)?),
111                Box::new(map_sql_dtype_to_polars(value)?),
112            );
113            dtype.ensure_valid_map_dtype()?;
114            dtype
115        },
116        #[cfg(not(feature = "dtype-map"))]
117        SQLDataType::Map(_, _) => {
118            polars_bail!(SQLInterface: "MAP requires the `dtype-map` feature")
119        },
120
121        // ---------------------------------
122        // binary
123        // ---------------------------------
124        SQLDataType::Bytea
125        | SQLDataType::Bytes(_)
126        | SQLDataType::Binary(_)
127        | SQLDataType::Blob(_)
128        | SQLDataType::Varbinary(_) => DataType::Binary,
129
130        // ---------------------------------
131        // boolean
132        // ---------------------------------
133        SQLDataType::Boolean | SQLDataType::Bool => DataType::Boolean,
134
135        // ---------------------------------
136        // signed integer
137        // ---------------------------------
138        SQLDataType::TinyInt(_) => DataType::Int8,
139        SQLDataType::Int16 | SQLDataType::Int2(_) | SQLDataType::SmallInt(_) => DataType::Int16,
140        SQLDataType::Int32
141        | SQLDataType::Int4(_)
142        | SQLDataType::MediumInt(_)
143        | SQLDataType::Integer(_)
144        | SQLDataType::Int(_) => DataType::Int32,
145        SQLDataType::Int64 | SQLDataType::Int8(_) | SQLDataType::BigInt(_) => DataType::Int64,
146        SQLDataType::Int128 | SQLDataType::HugeInt => DataType::Int128,
147
148        // ---------------------------------
149        // unsigned integer
150        // ---------------------------------
151        SQLDataType::UTinyInt | SQLDataType::TinyIntUnsigned(_) => DataType::UInt8,
152        SQLDataType::Int2Unsigned(_)
153        | SQLDataType::SmallIntUnsigned(_)
154        | SQLDataType::USmallInt
155        | SQLDataType::UInt16 => DataType::UInt16,
156        SQLDataType::Int4Unsigned(_) | SQLDataType::MediumIntUnsigned(_) | SQLDataType::UInt32 => {
157            DataType::UInt32
158        },
159        SQLDataType::Int8Unsigned(_)
160        | SQLDataType::BigIntUnsigned(_)
161        | SQLDataType::UBigInt
162        | SQLDataType::UInt64
163        | SQLDataType::UInt8 => DataType::UInt64,
164        SQLDataType::IntUnsigned(_) | SQLDataType::UnsignedInteger => DataType::UInt32,
165        SQLDataType::UHugeInt => DataType::UInt128,
166
167        // ---------------------------------
168        // float
169        // ---------------------------------
170        SQLDataType::Float4 | SQLDataType::Real => DataType::Float32,
171        SQLDataType::Double(_) | SQLDataType::DoublePrecision | SQLDataType::Float8 => {
172            DataType::Float64
173        },
174        SQLDataType::Float(n_bytes) => match n_bytes {
175            ExactNumberInfo::Precision(n) if (1u64..=24u64).contains(n) => DataType::Float32,
176            ExactNumberInfo::Precision(n) if (25u64..=53u64).contains(n) => DataType::Float64,
177            ExactNumberInfo::Precision(n) => {
178                polars_bail!(SQLSyntax: "unsupported `float` size (expected a value between 1 and 53, found {})", n)
179            },
180            ExactNumberInfo::None => DataType::Float64,
181            ExactNumberInfo::PrecisionAndScale(_, _) => {
182                polars_bail!(SQLSyntax: "FLOAT does not support scale parameter")
183            },
184        },
185
186        // ---------------------------------
187        // decimal
188        // ---------------------------------
189        #[cfg(feature = "dtype-decimal")]
190        SQLDataType::Dec(info)
191        | SQLDataType::Decimal(info)
192        | SQLDataType::BigDecimal(info)
193        | SQLDataType::Numeric(info) => match *info {
194            ExactNumberInfo::PrecisionAndScale(p, s) => DataType::Decimal(p as usize, s as usize),
195            ExactNumberInfo::Precision(p) => DataType::Decimal(p as usize, 0),
196            ExactNumberInfo::None => DataType::Decimal(38, 9),
197        },
198
199        // ---------------------------------
200        // temporal
201        // ---------------------------------
202        SQLDataType::Date => DataType::Date,
203        SQLDataType::Interval { fields, precision } => {
204            if !fields.is_none() {
205                // eg: "YEARS TO MONTH"
206                polars_bail!(SQLInterface: "`interval` definition with fields={:?} is not supported", fields)
207            }
208            let time_unit = match precision {
209                Some(p) if (1u64..=3u64).contains(p) => TimeUnit::Milliseconds,
210                Some(p) if (4u64..=6u64).contains(p) => TimeUnit::Microseconds,
211                Some(p) if (7u64..=9u64).contains(p) => TimeUnit::Nanoseconds,
212                Some(p) => {
213                    polars_bail!(SQLSyntax: "invalid `interval` precision (expected 1-9, found {})", p)
214                },
215                None => TimeUnit::Microseconds,
216            };
217            DataType::Duration(time_unit)
218        },
219        SQLDataType::Time(_, tz) => match tz {
220            TimezoneInfo::None | TimezoneInfo::WithoutTimeZone => DataType::Time,
221            _ => {
222                polars_bail!(SQLInterface: "`time` with timezone is not supported; found tz={}", tz)
223            },
224        },
225        SQLDataType::Datetime(prec) => DataType::Datetime(timeunit_from_precision(prec)?, None),
226        SQLDataType::Timestamp(prec, tz) => {
227            let time_zone = match tz {
228                TimezoneInfo::None | TimezoneInfo::WithoutTimeZone => None,
229                TimezoneInfo::WithTimeZone | TimezoneInfo::Tz => Some(TimeZone::UTC),
230            };
231            DataType::Datetime(timeunit_from_precision(prec)?, time_zone)
232        },
233
234        // ---------------------------------
235        // string
236        // ---------------------------------
237        SQLDataType::Char(_)
238        | SQLDataType::CharVarying(_)
239        | SQLDataType::Character(_)
240        | SQLDataType::CharacterVarying(_)
241        | SQLDataType::Clob(_)
242        | SQLDataType::String(_)
243        | SQLDataType::Text
244        | SQLDataType::Uuid
245        | SQLDataType::Varchar(_) => DataType::String,
246
247        // ---------------------------------
248        // custom
249        // ---------------------------------
250        SQLDataType::Custom(ObjectName(idents), _) => match idents.as_slice() {
251            [ObjectNamePart::Identifier(Ident { value, .. })] => {
252                match value.to_lowercase().as_str() {
253                    // these integer types are not supported by the PostgreSQL core distribution,
254                    // but they ARE available via `pguint` (https://github.com/petere/pguint), an
255                    // extension maintained by one of the PostgreSQL core developers, and/or DuckDB.
256                    "int1" => DataType::Int8,
257                    "uint1" | "utinyint" => DataType::UInt8,
258                    "uint2" | "usmallint" => DataType::UInt16,
259                    "uint4" | "uinteger" | "uint" => DataType::UInt32,
260                    "uint8" | "ubigint" => DataType::UInt64,
261                    _ => {
262                        polars_bail!(SQLInterface: "datatype {:?} is not currently supported", value)
263                    },
264                }
265            },
266            _ => {
267                polars_bail!(SQLInterface: "datatype {:?} is not currently supported", idents)
268            },
269        },
270        _ => {
271            polars_bail!(SQLInterface: "datatype {:?} is not currently supported", dtype)
272        },
273    })
