4/// Attaches `span` unless a more precise location is attached.
5pub(crate) fn locate(error: PolarsError, kind: SqlLocationKind, span: Span) -> PolarsError {
6    if error.sql_location().is_some() || span.start.line == 0 {
7        return error;
8    }
9    error.with_attachment(ErrorAttachment::SqlLocation(SqlLocation {
10        kind,
11        start: position(span.start),
12        end: position(span.end),
13        line: String::new(),
14    }))
15}

17/// Locates `error` at the statement, unless it is located already: for an error about a part
18/// of an expression that has no position of its own.
19pub(crate) fn at_statement(error: PolarsError) -> PolarsError {
20    if error.sql_location().is_some() {
21        return error;
22    }
23    let start = SqlPosition { line: 0, column: 0 };
24    error.with_attachment(ErrorAttachment::SqlLocation(SqlLocation {
25        kind: SqlLocationKind::Statement,
26        start,
27        end: start,
28        line: String::new(),
29    }))
30}

32/// `complete_location`, or the statement when nothing more precise is known.
33pub(crate) fn locate_in_statement(error: PolarsError, sql: &str) -> PolarsError {
34    match error.sql_location() {
35        Some(_) => complete_location(error, sql),
36        None => error.with_attachment(ErrorAttachment::SqlLocation(SqlLocation::statement(sql))),
37    }
38}

40/// Fills in the text of the located line from the statement `sql`.
41pub(crate) fn complete_location(error: PolarsError, sql: &str) -> PolarsError {
42    let location = match error.sql_location() {
43        Some(location)
44            if location.line.is_empty() && location.kind == SqlLocationKind::Statement =>
45        {
46            SqlLocation::statement(sql)
47        },
48        Some(location) if location.line.is_empty() => {
49            let line = (location.start.line as usize).saturating_sub(1);
50            SqlLocation {
51                line: sql.lines().nth(line).unwrap_or_default().to_string(),
52                ..location.clone()
53            }
54        },
55        _ => return error,
56    };
57    without_positions(error).with_attachment(ErrorAttachment::SqlLocation(location))
58}

60/// `error` without the positions of macro bodies, if `expanded` put any into the statement:
61/// they are in the text that created the macro.
62pub(crate) fn without_macro_positions(error: PolarsError, expanded: bool) -> PolarsError {
63    match expanded {
64        true => without_positions(error),
65        false => error,
66    }
67}

69/// Removes the SQL locations of `error` that have no line text yet: positions in a text that
70/// is not known here.
71pub(crate) fn without_positions(error: PolarsError) -> PolarsError {
72    error.retain_attachments(|attachment| {
73        !matches!(attachment, ErrorAttachment::SqlLocation(location) if location.line.is_empty())
74    })
75}

77fn position(location: Location) -> SqlPosition {
