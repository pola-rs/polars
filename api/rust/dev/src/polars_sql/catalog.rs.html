39/// A DDL or DML statement on a catalog table or schema, for the caller to perform (see
40/// [`SQLContext::catalog_statement`](crate::SQLContext::catalog_statement)).
41#[derive(Clone, Debug, PartialEq, Eq)]
42pub enum CatalogStatement {
43    /// `DROP TABLE [IF EXISTS] <name> [PURGE]`.
44    DropTable {
45        /// The table name as the statement wrote it.
46        name: String,
47        /// The catalog table it resolves to.
48        table: CatalogTable,
49        /// `IF EXISTS`.
50        if_exists: bool,
51        /// `PURGE`: also delete the data files.
52        purge: bool,
53    },
54    /// `CREATE TABLE [IF NOT EXISTS] <name> (<column>, ...)`, or `... <name> AS <query>`.
55    CreateTable {
56        /// The table name as the statement wrote it.
57        name: String,
58        /// The catalog table to create.
59        table: CatalogTable,
60        /// `IF NOT EXISTS`.
61        if_not_exists: bool,
62        /// The columns; empty with a query.
63        schema: Schema,
64        /// The query that fills the table, as SQL.
65        query: Option<String>,
66        /// The columns declared `NOT NULL`.
67        required: Vec<String>,
68        /// The table location, from `LOCATION` or the `location` option.
69        location: Option<String>,
70        /// The other `WITH` options, as table properties.
71        properties: Vec<(String, String)>,
72        /// The partition fields, in the order written; empty for an unpartitioned table.
73        partition_by: Vec<CatalogPartitionField>,
74    },
75    /// `CREATE SCHEMA [IF NOT EXISTS] <name> [WITH (...)]`.
76    CreateSchema {
77        /// The schema name as the statement wrote it.
78        name: String,
79        /// The catalog namespace to create.
80        namespace: CatalogNamespace,
81        /// `IF NOT EXISTS`.
82        if_not_exists: bool,
83        /// The `WITH` options, as namespace properties.
84        properties: Vec<(String, String)>,
85    },
86    /// `DROP SCHEMA [IF EXISTS] <name> [RESTRICT]`.
87    DropSchema {
88        /// The schema name as the statement wrote it.
89        name: String,
90        /// The catalog namespace to drop.
91        namespace: CatalogNamespace,
92        /// `IF EXISTS`.
93        if_exists: bool,
94    },
95    /// `INSERT INTO <name> [(<column>, ...)] <query>`, or `INSERT OVERWRITE ...`.
96    Insert {
97        /// The table name as the statement wrote it.
98        name: String,
99        /// The catalog table to write.
100        table: CatalogTable,
101        /// The columns written, as listed; empty for all, in the table's order.
102        columns: Vec<String>,
103        /// The query giving the rows, as SQL.
104        query: String,
105        /// `OVERWRITE`: replace the table's rows.
106        overwrite: bool,
107    },
108    /// `DELETE FROM <name> [WHERE <condition>]`.
109    Delete {
110        /// The table name as the statement wrote it.
111        name: String,
112        /// The catalog table to delete rows from.
113        table: CatalogTable,
114        /// The `WHERE` condition, as SQL, for messages.
115        condition: Option<String>,
116        /// Whether the condition has a subquery, which a catalog filter can't express.
117        subquery: bool,
118        /// The first part of the condition a catalog filter likely can't express, for messages.
119        hint_unsupported_part: Option<String>,
120    },
121    /// `UPDATE <name> SET <column> = <value>, ... [WHERE <condition>]`.
122    Update {
123        /// The table name as the statement wrote it.
124        name: String,
125        /// The catalog table to update.
126        table: CatalogTable,
127        /// The `SET` columns, in the order written.
128        columns: Vec<String>,
129        /// The `WHERE` condition, as SQL, for messages.
130        condition: Option<String>,
131        /// Whether the condition has a subquery, which a catalog filter can't express.
132        subquery: bool,
133        /// The first part of the condition a catalog filter likely can't express, for messages.
134        hint_unsupported_part: Option<String>,
135    },
136    /// `MERGE INTO <name> USING <source> ON <condition> WHEN ...`.
137    Merge {
138        /// The table name as the statement wrote it.
139        name: String,
140        /// The catalog table to merge into.
141        table: CatalogTable,
142    },
143    /// `TRUNCATE [TABLE] [IF EXISTS] <name>`.
144    Truncate {
145        /// The table name as the statement wrote it.
146        name: String,
147        /// The catalog table to empty.
148        table: CatalogTable,
149        /// `IF EXISTS`.
150        if_exists: bool,
151    },
152    /// `ALTER TABLE [IF EXISTS] <name> <operation>`.
153    AlterTable {
154        /// The table name as the statement wrote it.
155        name: String,
156        /// The catalog table to change.
157        table: CatalogTable,
158        /// `IF EXISTS`.
159        if_exists: bool,
160        /// The change, one per statement.
161        operation: CatalogAlter,
162    },

277/// The catalogs attached to a context.
278#[derive(Clone, Debug, Default)]
279pub(crate) struct CatalogRegistry {
280    catalogs: Vec<AttachedCatalog>,
281    /// The catalog of table names that don't start with an attached catalog.
282    default: Option<String>,
283}
284
285impl CatalogRegistry {
286    pub(crate) fn is_empty(&self) -> bool {
287        self.catalogs.is_empty()
288    }
289
290    /// Attach a catalog; the same name replaces it, with the new `default_namespace`, and keeps
291    /// its default status (`default` only ever sets the default). A name that differs from an
292    /// attached one only in case is an error, so matching names ignoring case is never ambiguous.
293    pub(crate) fn register(
294        &mut self,
295        name: &str,
296        default_namespace: Option<Vec<String>>,
297        default: bool,
298    ) -> PolarsResult<()> {
299        polars_ensure!(!name.is_empty(), InvalidOperation: "catalog name must not be empty");
300        polars_ensure!(
301            default_namespace.as_ref().is_none_or(|ns| !ns.is_empty() && ns.iter().all(|p| !p.is_empty())),
302            InvalidOperation: "the default namespace of catalog '{}' must have non-empty parts", name
303        );
304        if let Some(other) = self
305            .catalogs
306            .iter()
307            .find(|c| c.name != name && c.name.eq_ignore_ascii_case(name))
308        {
309            polars_bail!(
310                InvalidOperation: "catalog '{}' differs from attached catalog '{}' only in case; use that name, or unregister it first",
311                name, other.name
312            );
313        }
314        let keep_default = self.default.as_deref() == Some(name);
315        self.unregister(name);
316        self.catalogs.push(AttachedCatalog {
317            name: name.to_string(),
318            default_namespace,
319        });
320        if default || keep_default {
321            self.default = Some(name.to_string());
322        }
323        Ok(())
324    }
325
326    /// The name of the default catalog.
327    pub(crate) fn default_catalog(&self) -> Option<&str> {
328        self.default.as_deref()
329    }
330
331    /// Make an attached catalog (exact name, else ignoring case) the default.
332    pub(crate) fn set_default(&mut self, name: &str) -> PolarsResult<()> {
333        let Some(name) = self.catalog_name(name).map(str::to_string) else {
334            polars_bail!(SQLInterface: "catalog '{}' is not attached", name);
335        };
336        self.default = Some(name);
337        Ok(())
338    }
339
340    /// Detach a catalog (exact name, else ignoring case), returning the name it had.
341    pub(crate) fn unregister(&mut self, name: &str) -> Option<String> {
342        let idx = self.position(name)?;
343        let removed = self.catalogs.remove(idx).name;
344        if self.default.as_deref() == Some(removed.as_str()) {
345            self.default = None;
346        }
347        Some(removed)
348    }
349
350    fn position(&self, name: &str) -> Option<usize> {
351        self.catalogs
352            .iter()
353            .position(|c| c.name == name)
354            .or_else(|| {
355                self.catalogs
356                    .iter()
357                    .position(|c| c.name.eq_ignore_ascii_case(name))
358            })
359    }
360
361    /// The attached catalog called `name` (exact, else ignoring case).
362    pub(crate) fn catalog_name(&self, name: &str) -> Option<&str> {
363        self.position(name).map(|i| self.catalogs[i].name.as_str())
364    }
365
366    /// The attached catalog a table name explicitly starts with.
367    pub(crate) fn named_catalog(&self, parts: &[String]) -> Option<&str> {
368        match parts {
369            [first, _, ..] => self.catalog_name(first),
370            _ => None,
371        }
372    }
373
374    /// The catalog table a table name refers to: a leading part naming an attached catalog
375    /// selects it, else the default catalog; a name without namespace parts gets the catalog's
376    /// default namespace. `None` if no catalog applies.
377    pub(crate) fn resolve(&self, parts: &[String]) -> Option<CatalogTable> {
378        let (catalog, rest) = self.select(parts)?;
379        let identifier = match rest {
380            [table] => {
381                let mut identifier = catalog.default_namespace.clone()?;
382                identifier.push(table.clone());
383                identifier
384            },
385            _ => rest.to_vec(),
386        };
387        Some(CatalogTable {
388            catalog: catalog.name.clone(),
389            identifier,
390        })
391    }
392
393    /// The catalog namespace a schema name refers to: a leading part naming an attached catalog
394    /// selects it, else the default catalog; the other parts are the namespace (never relative
395    /// to a default namespace). `None` if no catalog applies.
396    pub(crate) fn resolve_namespace(&self, parts: &[String]) -> Option<CatalogNamespace> {
397        let (catalog, rest) = self.select(parts)?;
398        Some(CatalogNamespace {
399            catalog: catalog.name.clone(),
400            namespace: rest.to_vec(),
401        })
402    }
403
404    /// The catalog a name selects, and the name's other parts.
405    fn select<'a>(&self, parts: &'a [String]) -> Option<(&AttachedCatalog, &'a [String])> {
406        let named = match parts {
407            [first, rest @ ..] if !rest.is_empty() => self.position(first).map(|idx| (idx, rest)),
408            _ => None,
409        };
410        let (idx, rest) = match named {
411            Some(named) => named,
412            None => (self.position(self.default.as_deref()?)?, parts),
413        };
414        Some((&self.catalogs[idx], rest))
415    }
416
417    /// Why a table name that nothing resolved is not a catalog table, if a catalog could have
418    /// applied.
419    pub(crate) fn unresolved_hint(&self, parts: &[String]) -> Option<String> {
420        if self.is_empty() || self.resolve(parts).is_some() {
421            return None;
422        }
423        let catalog = match self.named_catalog(parts) {
424            Some(named) => named,
425            None => match &self.default {
426                Some(default) => default.as_str(),
427                None => return Some("; no default catalog is set".to_string()),
428            },
429        };
430        Some(format!("; catalog '{catalog}' has no default namespace"))
431    }
432}

455/// Rewrite the names of catalog tables in `stmt` to their full names, `lineitem` to
456/// `onelake.tpch.lineitem`, so the rest of the translation knows them by that name. Names of
457/// CTEs in scope and of local tables (`is_local`) are left alone. Returns each rewritten name
458/// as written with its full name, and where each name was written.
459pub(crate) fn rewrite_catalog_names(
460    stmt: &mut Statement,
461    resolve: impl Fn(&[String]) -> Option<Vec<String>>,
462    is_local: impl Fn(&str) -> bool,
463) -> PolarsResult<(Vec<RewrittenName>, CatalogSpellings)> {
464    let mut registered = RegisteredAliases(Vec::new());
465    let _ = dyn_visit::visit(&*stmt, &mut registered);
466    let mut rewriter = Rewriter {
467        resolve,
468        is_local,
469        registered_aliases: registered.0,
470        cte_scopes: Vec::new(),
471        with_stack: Vec::new(),
472        rewritten: Vec::new(),
473        spellings: CatalogSpellings::default(),
474        desugared: 0,
475    };
476    if let ControlFlow::Break(err) = dyn_visit::visit_mut(stmt, &mut rewriter) {
477        return Err(err);
478    }
479    Ok((rewriter.rewritten, rewriter.spellings))
480}

482/// The aliases of the statement's derived tables, table functions and `UNNEST`s, which are
483/// registered as tables while the statement runs (#30070), wherever they are written.
484struct RegisteredAliases(Vec<String>);
485
486impl Visitor for RegisteredAliases {
487    type Break = ();
488
489    fn pre_visit_table_factor(&mut self, factor: &TableFactor) -> ControlFlow<()> {
490        let alias = match factor {
491            TableFactor::Table {
492                args: Some(_),
493                alias,
494                ..
495            }
496            | TableFactor::Derived { alias, .. }
497            | TableFactor::UNNEST { alias, .. } => alias.as_ref(),
498            _ => None,
499        };
500        self.0.extend(alias.map(|alias| alias.name.value.clone()));
501        ControlFlow::Continue(())
502    }
503}

505struct Rewriter<R, L> {
506    resolve: R,
507    is_local: L,
508    /// Aliases that tables of the same name would be looked up as (see [`RegisteredAliases`]).
509    registered_aliases: Vec<String>,
510    /// The CTE names visible in each enclosing query.
511    cte_scopes: Vec<Vec<String>>,
512    /// The `WITH` clauses of the queries being visited, taken out while their bodies are visited.
513    with_stack: Vec<Option<With>>,
514    rewritten: Vec<RewrittenName>,
515    spellings: CatalogSpellings,
516    /// The number of `TABLE x` desugared, which numbers their (made-up) name positions.
517    desugared: u64,
518}
519
520impl<R, L> Rewriter<R, L>
521where
522    R: Fn(&[String]) -> Option<Vec<String>>,
523    L: Fn(&str) -> bool,
524{
525    fn is_cte(&self, name: &str) -> bool {
526        self.cte_scopes
527            .iter()
528            .flatten()
529            .any(|cte| cte.eq_ignore_ascii_case(name))
530    }
531
532    /// The full name of the catalog table that `parts` refers to: not a CTE or a local table.
533    fn resolve_name(&self, parts: &[String]) -> Option<Vec<String>> {
534        let written = parts.join(".");
535        if self.is_cte(&written) || (self.is_local)(&written) {
536            return None;
537        }
538        (self.resolve)(parts)
539    }
540
541    /// Record `parts` as referring to the catalog table `full`. Tables are looked up by their
542    /// dotted name, so it must be the catalog table's alone.
543    fn record(&mut self, parts: &[String], full: &[String]) -> PolarsResult<()> {
544        let key = full.join(".");
545        let is_alias = self
546            .registered_aliases
547            .iter()
548            .any(|alias| alias.eq_ignore_ascii_case(&key));
549        if is_alias || self.is_cte(&key) || (self.is_local)(&key) {
550            polars_bail!(
551                SQLInterface: "table '{}' refers to catalog table '{}', which is also the name of a registered table, CTE or alias; rename one of them",
552                display_name(parts), display_name(full)
553            );
554        }
555        if let Some((_, other)) = self
556            .rewritten
557            .iter()
558            .find(|(_, other)| other.join(".") == key && other != full)
559        {
560            polars_bail!(
561                SQLInterface: "catalog tables {} and {} have the same dotted name, which SQL cannot tell apart",
562                display_name(other), display_name(full)
563            );
564        }
565        if !self.rewritten.iter().any(|(written, _)| written == parts) {
566            self.rewritten.push((parts.to_vec(), full.to_vec()));
567        }
568        Ok(())
569    }
570
571    /// Replace `TABLE x` of a catalog table by `SELECT * FROM x`, whose table name the visit
572    /// then rewrites (sqlparser's `TABLE` holds at most two name parts).
573    fn desugar_table(&mut self, body: &mut SetExpr) {
574        match body {
575            SetExpr::SetOperation { left, right, .. } => {
576                self.desugar_table(left);
577                self.desugar_table(right);
578            },
579            SetExpr::Table(table) => {
580                let parts: Vec<String> = table
581                    .schema_name
582                    .iter()
583                    .chain(&table.table_name)
584                    .cloned()
585                    .collect();
586                if !parts.is_empty() && self.resolve_name(&parts).is_some() {
587                    // Line 0 is never a position in the query text.
588                    self.desugared += 1;
589                    let at = Location::new(0, self.desugared);
590                    *body = select_star_from(&parts, Span::new(at, at));
591                }
592            },
593            _ => {},
594        }
595    }
596}

598impl<R, L> VisitorMut for Rewriter<R, L>
599where
600    R: Fn(&[String]) -> Option<Vec<String>>,
601    L: Fn(&str) -> bool,
602{
603    type Break = PolarsError;
604
605    fn pre_visit_query(&mut self, query: &mut Query) -> ControlFlow<PolarsError> {
606        // A non-recursive CTE is visible after its definition, as `register_ctes` registers them.
607        let mut with = query.with.take();
608        let mut names = Vec::new();
609        if let Some(with) = with.as_mut() {
610            let all: Vec<String> = with
611                .cte_tables
612                .iter()
613                .map(|cte| cte.alias.name.value.clone())
614                .collect();
615            let recursive = with.recursive;
616            for cte in &mut with.cte_tables {
617                self.cte_scopes.push(if recursive {
618                    all.clone()
619                } else {
620                    names.clone()
621                });
622                let flow = dyn_visit::visit_mut(&mut *cte.query, self);
623                self.cte_scopes.pop();
624                flow?;
625                names.push(cte.alias.name.value.clone());
626            }
627        }
628        self.cte_scopes.push(names);
629        self.with_stack.push(with);
630        self.desugar_table(&mut query.body);
631        ControlFlow::Continue(())
632    }
633
634    fn post_visit_query(&mut self, query: &mut Query) -> ControlFlow<PolarsError> {
635        self.cte_scopes.pop();
636        query.with = self.with_stack.pop().flatten();
637        ControlFlow::Continue(())
638    }
639
640    fn pre_visit_table_factor(&mut self, factor: &mut TableFactor) -> ControlFlow<PolarsError> {
641        // A name with arguments is a table function.
642        let TableFactor::Table {
643            name, args: None, ..
644        } = factor
645        else {
646            return ControlFlow::Continue(());
647        };
648        let Some(parts) = name_parts(name) else {
649            return ControlFlow::Continue(());
650        };
651        if let Some(full) = self.resolve_name(&parts) {
652            if let Err(err) = self.record(&parts, &full) {
653                return ControlFlow::Break(err);
654            }
655            let span = name.0[0]
656                .as_ident()
657                .map_or_else(Span::empty, |ident| ident.span);
658            if let Some(position) = name_position(name) {
659                self.spellings.insert(position, parts);
660            }
661            *name = object_name(&full, span);
662        }
663        ControlFlow::Continue(())
664    }
665}

667/// A table name whose parts all have `span`, so the name keeps the position it was written at.
668fn object_name(parts: &[String], span: Span) -> ObjectName {
669    let ident = |part: &String| Ident {
670        value: part.clone(),
671        quote_style: None,
672        span,
673    };
674    ObjectName(
675        parts
676            .iter()
677            .map(|part| ObjectNamePart::Identifier(ident(part)))
678            .collect(),
679    )
680}

682/// `SELECT * FROM <parts>`, with the name at `span`.
683fn select_star_from(parts: &[String], span: Span) -> SetExpr {
684    let mut stmts = Parser::parse_sql(&GenericDialect, "SELECT * FROM t").unwrap();
685    let Statement::Query(query) = stmts.remove(0) else {
686        unreachable!()
687    };
688    let mut body = *query.body;
689    if let SetExpr::Select(select) = &mut body
690        && let TableFactor::Table { name, .. } = &mut select.from[0].relation
691    {
692        *name = object_name(parts, span);
693    }
694    body
695}

697/// The schema of a catalog table from its column definitions, and its `NOT NULL` columns.
698pub(crate) fn create_table_columns(columns: &[ColumnDef]) -> PolarsResult<(Schema, Vec<String>)> {
699    let mut schema = Schema::with_capacity(columns.len());
700    let mut required = Vec::new();
701    for col in columns {
702        let name = col.name.value.as_str();
703        let mut not_null = None;
704        for option in &col.options {
705            let is_not_null = match &option.option {
706                ColumnOption::NotNull => true,
707                ColumnOption::Null => false,
708                // the default of a nullable column anyway
709                ColumnOption::Default(expr) if is_null(expr) => continue,
710                _ => polars_bail!(
711                    SQLInterface: "only NOT NULL constraints are supported for catalog tables; found `{}` on column '{}'",
712                    option, name
713                ),
714            };
715            polars_ensure!(
716                not_null.is_none_or(|n| n == is_not_null),
717                SQLInterface: "conflicting NULL/NOT NULL declarations for column '{}'", name
718            );
719            not_null = Some(is_not_null);
720        }
721        let dtype = map_sql_dtype_to_polars(&col.data_type)?;
722        polars_ensure!(
723            schema.insert(name.into(), dtype).is_none(),
724            Duplicate: "column '{}' is defined more than once", name
725        );
726        if not_null == Some(true) {
727            required.push(name.to_string());
728        }
729    }
730    Ok((schema, required))
731}

733/// The name and type of a column added by `ALTER TABLE ... ADD COLUMN`, which is nullable.
734pub(crate) fn alter_table_column(col: &ColumnDef) -> PolarsResult<(String, DataType)> {
735    let name = col.name.value.as_str();
736    for option in &col.options {
737        match &option.option {
738            ColumnOption::Null => {},
739            ColumnOption::Default(expr) if is_null(expr) => {},
740            ColumnOption::NotNull => polars_bail!(
741                SQLInterface: "`ADD COLUMN ... NOT NULL` is not supported: the table's rows have no value for column '{}'",
742                name
743            ),
744            _ => polars_bail!(
745                SQLInterface: "`ADD COLUMN` takes only NULL or DEFAULT NULL; found `{}` on column '{}'",
746                option, name
747            ),
748        }
749    }
750    Ok((name.to_string(), map_sql_dtype_to_polars(&col.data_type)?))
751}

801/// The location and properties of a catalog table, from `LOCATION` and `WITH (...)`.
802pub(crate) fn create_table_options(
803    options: &CreateTableOptions,
804    location_clause: Option<&str>,
805) -> PolarsResult<(Option<String>, TableProperties)> {
806    let options = match options {
807        CreateTableOptions::None => &[][..],
808        CreateTableOptions::With(options) => options.as_slice(),
809        CreateTableOptions::TableProperties(_) => {
810            polars_bail!(SQLInterface: "`TBLPROPERTIES` is not supported; use `WITH (...)` instead")
811        },
812        _ => {
813            polars_bail!(SQLInterface: "table `OPTIONS` clauses are not supported; use `WITH (...)` instead")
814        },
815    };
816    let mut location = location_clause.map(str::to_string);
817    let mut properties = TableProperties::new();
818    for option in options {
819        let (key, value) = option_key_value(option, "table", &["location", "format-version"])?;
820        polars_ensure!(
821            !RESERVED_TABLE_PROPERTIES.contains(&key),
822            SQLInterface: "table option '{}' is reserved by Iceberg", key
823        );
824        if key == "location" {
825            polars_ensure!(
826                location_clause.is_none(),
827                SQLInterface: "`LOCATION` and table option 'location' cannot be combined"
828            );
829            polars_ensure!(location.is_none(), SQLInterface: "duplicate table option 'location'");
830            location = Some(value);
831        } else {
832            polars_ensure!(
833                properties.iter().all(|(k, _)| k != key),
834                SQLInterface: "duplicate table option '{}'", key
835            );
836            properties.push((key.to_string(), value));
837        }
838    }
839    if let Some(location) = &location {
840        polars_ensure!(
841            is_absolute_location(location),
842            SQLInterface: "location must be an absolute path or a URI; found '{}'", location
843        );
844    }
845    Ok((location, properties))
846}

848/// The partition fields of a catalog table, from `PARTITION BY (...)` or `PARTITIONED BY (...)`.
849pub(crate) fn create_table_partitioning(
850    partition_by: Option<&Expr>,
851    hive_distribution: &HiveDistributionStyle,
852) -> PolarsResult<Vec<CatalogPartitionField>> {
853    // Each field with its SQL, for messages.
854    let fields: Vec<(CatalogPartitionField, String)> = match (partition_by, hive_distribution) {
855        (Some(_), HiveDistributionStyle::PARTITIONED { .. }) => {
856            polars_bail!(SQLInterface: "`PARTITION BY` and `PARTITIONED BY` cannot be combined")
857        },
858        (Some(expr), _) => {
859            let exprs = match unparenthesized(expr) {
860                Expr::Tuple(exprs) => exprs.as_slice(),
861                expr => std::slice::from_ref(expr),
862            };
863            exprs
864                .iter()
865                .map(|expr| Ok((partition_field(expr)?, expr.to_string())))
866                .collect::<PolarsResult<_>>()?
867        },
868        (None, HiveDistributionStyle::PARTITIONED { columns }) => columns
869            .iter()
870            .map(|col| {
871                // Hive's `PARTITIONED BY (p INT)` adds a column.
872                polars_ensure!(
873                    col.data_type == SQLDataType::Unspecified && col.options.is_empty(),
874                    SQLInterface: "`PARTITIONED BY` takes column names only; found `{}`; types and constraints belong in the column list",
875                    col
876                );
877                let field = CatalogPartitionField {
878                    column: col.name.value.clone(),
879                    transform: "identity".to_string(),
880                };
881                Ok((field, col.name.value.clone()))
882            })
883            .collect::<PolarsResult<_>>()?,
884        (None, _) => Vec::new(),
885    };
886    // Iceberg allows one time transform per column.
887    let is_time = |f: &CatalogPartitionField| {
888        matches!(f.transform.as_str(), "year" | "month" | "day" | "hour")
889    };
890    for (i, (field, sql)) in fields.iter().enumerate() {
891        for (earlier, earlier_sql) in &fields[..i] {
892            polars_ensure!(earlier != field, SQLInterface: "duplicate partition field `{}`", sql);
893            polars_ensure!(
894                earlier.column != field.column || !is_time(earlier) || !is_time(field),
895                SQLInterface: "partition fields `{}` and `{}` both partition column '{}' by time; use one of them",
896                earlier_sql, sql, field.column
897            );
898        }
899    }
900    Ok(fields.into_iter().map(|(field, _)| field).collect())
901}

903/// A partition field: a column, `year|month|day|hour(column)` (with Iceberg's aliases), or
904/// `bucket|truncate(N, column)`.
905fn partition_field(expr: &Expr) -> PolarsResult<CatalogPartitionField> {
906    let unsupported = || {
907        polars_err!(
908            SQLInterface: "unsupported partition field `{}`; expected a column, or year, month, day, hour, bucket(N, column) or truncate(W, column) of a column",
909            expr
910        )
911    };
912    let column = |expr: &Expr| match unparenthesized(expr) {
913        Expr::Identifier(ident) => Some(ident.value.clone()),
914        _ => None,
915    };
916    // A qualified column or a struct field: without the schema, these look the same.
917    let is_compound = |expr: &Expr| matches!(unparenthesized(expr), Expr::CompoundIdentifier(_));
918    let compound = |expr: &Expr| {
919        polars_err!(
920            SQLInterface: "`PARTITION BY` takes unqualified column names; found `{}`; partitioning by a nested field is not yet supported",
921            unparenthesized(expr)
922        )
923    };
924    if let Some(column) = column(expr) {
925        return Ok(CatalogPartitionField {
926            column,
927            transform: "identity".to_string(),
928        });
929    }
930    if is_compound(expr) {
931        return Err(compound(expr));
932    }
933    let Expr::Function(Function {
934        name,
935        uses_odbc_syntax: false,
936        parameters: FunctionArguments::None,
937        args: FunctionArguments::List(list),
938        filter: None,
939        null_treatment: None,
940        over: None,
941        within_group,
942    }) = unparenthesized(expr)
943    else {
944        return Err(unsupported());
945    };
946    let fname = match name_parts(name).as_deref() {
947        Some([fname])
948            if within_group.is_empty()
949                && list.duplicate_treatment.is_none()
950                && list.clauses.is_empty() =>
951        {
952            fname.to_ascii_lowercase()
953        },
954        _ => return Err(unsupported()),
955    };
956    let args = list
957        .args
958        .iter()
959        .map(|arg| match arg {
960            FunctionArg::Unnamed(FunctionArgExpr::Expr(expr)) => Some(expr),
961            _ => None,
962        })
963        .collect::<Option<Vec<_>>>()
964        .ok_or_else(unsupported)?;
965    let positive_int = |expr: &Expr| match unparenthesized(expr) {
966        Expr::Value(ValueWithSpan {
967            value: Value::Number(n, false),
968            ..
969        }) => n.parse::<i32>().ok().filter(|n| *n > 0),
970        _ => None,
971    };
972    let (transform, source) = match (fname.as_str(), args.as_slice()) {
973        ("year" | "years", [source]) => ("year".to_string(), source),
974        ("month" | "months", [source]) => ("month".to_string(), source),
975        ("day" | "days", [source]) => ("day".to_string(), source),
976        ("hour" | "hours", [source]) => ("hour".to_string(), source),
977        ("bucket" | "truncate", [n, source]) => {
978            let Some(n) = positive_int(n) else {
979                if (column(n).is_some() || is_compound(n)) && positive_int(source).is_some() {
980                    polars_bail!(SQLInterface: "`{}` takes the number first, as in `{}({}, {})`", expr, fname, source, n)
981                }
982                polars_bail!(SQLInterface: "the first argument of `{}` must be an integer from 1 to {}", expr, i32::MAX)
983            };
984            (format!("{fname}[{n}]"), source)
985        },
986        _ => return Err(unsupported()),
987    };
988    let column = match column(source) {
989        Some(column) => column,
990        None if is_compound(source) => return Err(compound(source)),
991        None => return Err(unsupported()),
992    };
993    Ok(CatalogPartitionField { column, transform })
994}

996/// `expr` without enclosing parentheses.
997fn unparenthesized(mut expr: &Expr) -> &Expr {
998    while let Expr::Nested(inner) = expr {
999        expr = inner;
1000    }
1001    expr
1002}

1022/// The key and value of a `'key' = value` option of a `kind` ("table", "schema"), with keys that
1023/// differ from a `known` one only in case rejected.
1024fn option_key_value<'a>(
1025    option: &'a SqlOption,
1026    kind: &str,
1027    known: &[&str],
1028) -> PolarsResult<(&'a str, String)> {
1029    let SqlOption::KeyValue { key, value } = option else {
1030        polars_bail!(SQLInterface: "{} options must be `'key' = value` pairs; found `{}`", kind, option)
1031    };
1032    let key = key.value.as_str();
1033    polars_ensure!(!key.is_empty(), SQLInterface: "{} option names must not be empty", kind);
1034    for known in known {
1035        polars_ensure!(
1036            key == *known || !key.eq_ignore_ascii_case(known),
1037            SQLInterface: "'{}' is not a {} option; did you mean '{}'?", key, kind, known
1038        );
1039    }
1040    let Some(value) = option_value(value) else {
1041        polars_bail!(
1042            SQLInterface: "{} option values must be string, number or boolean literals; found `{}` for '{}'",
1043            kind, value, key
1044        )
1045    };
1046    Ok((key, value))
1047}

1049/// A table option value as a property string, if it is a string, number or boolean literal.
1050fn option_value(value: &Expr) -> Option<String> {
1051    match value {
1052        Expr::Value(v) => match &v.value {
1053            Value::SingleQuotedString(s) => Some(s.clone()),
1054            Value::Number(n, _) => Some(n.clone()),
1055            Value::Boolean(b) => Some(b.to_string()),
1056            _ => None,
1057        },
1058        Expr::UnaryOp {
1059            op: UnaryOperator::Minus,
1060            expr,
1061        } => match expr.as_ref() {
1062            Expr::Value(ValueWithSpan {
1063                value: Value::Number(n, _),
1064                ..
1065            }) => Some(format!("-{n}")),
1066            _ => None,
1067        },
1068        _ => None,
1069    }
1070}

1072/// Whether a location is an absolute path (POSIX or Windows) or has a URI scheme.
1073fn is_absolute_location(location: &str) -> bool {
1074    let bytes = location.as_bytes();
1075    let windows_path = bytes.len() > 2
1076        && bytes[0].is_ascii_alphabetic()
1077        && bytes[1] == b':'
1078        && matches!(bytes[2], b'/' | b'\\');
1079    let has_scheme = location.split_once("://").is_some_and(|(scheme, _)| {
1080        scheme.starts_with(|c: char| c.is_ascii_alphabetic())
1081            && scheme
1082                .chars()
1083                .all(|c| c.is_ascii_alphanumeric() || matches!(c, '+' | '-' | '.'))
1084    });
1085    location.starts_with('/') || windows_path || has_scheme
1086}

1100/// The first part of a `DELETE` condition that a catalog filter can't express, described for a
1101/// message. Only a hint: converting the planned predicate decides.
1102pub(crate) fn hint_unsupported_part(expr: &Expr) -> Option<String> {
1103    use BinaryOperator as Op;
1104    if !expr_references_any_column(expr) {
1105        return None;
1106    }
1107    match expr {
1108        Expr::Nested(e)
1109        | Expr::UnaryOp {
1110            op: UnaryOperator::Not,
1111            expr: e,
1112        } => hint_unsupported_part(e),
1113        Expr::BinaryOp {
1114            left,
1115            op: Op::And | Op::Or,
1116            right,
1117        } => hint_unsupported_part(left).or_else(|| hint_unsupported_part(right)),
1118        Expr::BinaryOp {
1119            left,
1120            op: Op::Eq | Op::NotEq | Op::Lt | Op::LtEq | Op::Gt | Op::GtEq,
1121            right,
1122        } => match expr_references_any_column(left) && expr_references_any_column(right) {
1123            true => Some(format!("comparison of two columns `{expr}`")),
1124            false => unsupported_operand(left)
1125                .or_else(|| unsupported_operand(right))
1126                .or_else(|| {
1127                    unsupported_string(left, false).or_else(|| unsupported_string(right, false))
1128                }),
1129        },
1130        Expr::IsDistinctFrom(left, right) | Expr::IsNotDistinctFrom(left, right) => {
1131            match expr_references_any_column(left) && expr_references_any_column(right) {
1132                true => Some(format!("comparison of two columns `{expr}`")),
1133                false => unsupported_operand(left)
1134                    .or_else(|| unsupported_operand(right))
1135                    .or_else(|| {
1136                        unsupported_string(left, false).or_else(|| unsupported_string(right, false))
1137                    }),
1138            }
1139        },
1140        Expr::IsNull(e)
1141        | Expr::IsNotNull(e)
1142        | Expr::IsTrue(e)
1143        | Expr::IsNotTrue(e)
1144        | Expr::IsFalse(e)
1145        | Expr::IsNotFalse(e) => unsupported_operand(e),
1146        Expr::InList { expr: e, list, .. } => unsupported_operand(e).or_else(|| {
1147            match list.iter().any(|item| {
1148                matches!(
1149                    item,
1150                    Expr::Value(ValueWithSpan {
1151                        value: Value::Null,
1152                        ..
1153                    })
1154                )
1155            }) {
1156                true => Some("NULL in the IN list".to_string()),
1157                false => list
1158                    .iter()
1159                    .find_map(|item| column_in_bound(item, expr))
1160                    .or_else(|| list.iter().find_map(|item| unsupported_string(item, false))),
1161            }
1162        }),
1163        Expr::Between {
1164            expr: e, low, high, ..
1165        } => unsupported_operand(e)
1166            .or_else(|| column_in_bound(low, expr))
1167            .or_else(|| column_in_bound(high, expr))
1168            .or_else(|| unsupported_string(low, false))
1169            .or_else(|| unsupported_string(high, false)),
1170        Expr::Like {
1171            any,
1172            expr: e,
1173            pattern,
1174            escape_char,
1175            ..
1176        } => unsupported_operand(e).or_else(|| {
1177            let prefix = !any
1178                && escape_char.is_none()
1179                && matches!(pattern.as_ref(), Expr::Value(ValueWithSpan {
1180                    value: Value::SingleQuotedString(p), ..
1181                }) if is_prefix_pattern(p));
1182            match prefix {
1183                true => unsupported_string(pattern, true),
1184                false => Some(format!("LIKE pattern {pattern} (only 'prefix%')")),
1185            }
1186        }),
1187        Expr::ILike { .. } => Some("ILIKE".to_string()),
1188        Expr::Identifier(_) | Expr::CompoundIdentifier(_) => None,
1189        _ => Some(describe_operand(expr)),
1190    }
1191}

1193fn unsupported_operand(expr: &Expr) -> Option<String> {
1194    match expr {
1195        Expr::Identifier(_) | Expr::CompoundIdentifier(_) => None,
1196        Expr::Nested(e) => unsupported_operand(e),
1197        _ if !expr_references_any_column(expr) => None,
1198        _ => Some(describe_operand(expr)),
1199    }
1200}

1202/// With `like`, a trailing '%' is the pattern's wildcard, not part of the constant.
1203fn unsupported_string(expr: &Expr, like: bool) -> Option<String> {
1204    let Expr::Value(ValueWithSpan {
1205        value: Value::SingleQuotedString(s),
1206        ..
1207    }) = expr
1208    else {
1209        return None;
1210    };
1211    let text = match like {
1212        true => s.strip_suffix('%').unwrap_or(s),
1213        false => s,
1214    };
1215    // as `sanitize` in polars-plan's pyarrow predicates
1216    let supported = |c: char| c.is_alphanumeric() || matches!(c, ' ' | '-' | '_');
1217    (!text.chars().all(supported)).then(|| {
1218        format!("string constant {expr} (only letters, digits, space, '-' and '_' for now)")
1219    })
1220}

1222fn column_in_bound(bound: &Expr, expr: &Expr) -> Option<String> {
1223    expr_references_any_column(bound).then(|| format!("comparison of two columns `{expr}`"))
1224}

1226fn describe_operand(expr: &Expr) -> String {
1227    use BinaryOperator as Op;
1228    match expr {
1229        Expr::Function(_) => format!("function `{expr}`"),
1230        Expr::Cast { .. } => format!("cast `{expr}`"),
1231        Expr::BinaryOp {
1232            op: Op::Plus | Op::Minus | Op::Multiply | Op::Divide | Op::Modulo,
1233            ..
1234        }
1235        | Expr::UnaryOp {
1236            op: UnaryOperator::Minus | UnaryOperator::Plus,
1237            ..
1238        } => format!("arithmetic `{expr}`"),
1239        _ => format!("expression `{expr}`"),
1240    }
1241}

1243fn is_prefix_pattern(pattern: &str) -> bool {
1244    match pattern.strip_suffix('%') {
1245        Some(prefix) => !prefix.is_empty() && !prefix.contains(['%', '_']),
1246        None => !pattern.contains(['%', '_']),
1247    }
1248}
