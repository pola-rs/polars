38/// A statement changing a catalog table, for the caller to perform (see
39/// [`SQLContext::catalog_ddl`](crate::SQLContext::catalog_ddl)).
40#[derive(Clone, Debug, PartialEq, Eq)]
41pub enum CatalogDdl {
42    /// `DROP TABLE [IF EXISTS] <name> [PURGE]`.
43    DropTable {
44        /// The table name as the statement wrote it.
45        name: String,
46        /// The catalog table it resolves to.
47        table: CatalogTable,
48        /// `IF EXISTS`.
49        if_exists: bool,
50        /// `PURGE`: also delete the data files.
51        purge: bool,
52    },
53    /// `CREATE TABLE [IF NOT EXISTS] <name> (<column>, ...)`, or `... <name> AS <query>`.
54    CreateTable {
55        /// The table name as the statement wrote it.
56        name: String,
57        /// The catalog table to create.
58        table: CatalogTable,
59        /// `IF NOT EXISTS`.
60        if_not_exists: bool,
61        /// The columns; empty with a query.
62        schema: Schema,
63        /// The query that fills the table, as SQL.
64        query: Option<String>,
65        /// The columns declared `NOT NULL`.
66        required: Vec<String>,
67        /// The table location, from `LOCATION` or the `location` option.
68        location: Option<String>,
69        /// The other `WITH` options, as table properties.
70        properties: Vec<(String, String)>,
71        /// The partition fields, in the order written; empty for an unpartitioned table.
72        partition_by: Vec<CatalogPartitionField>,
73    },
74    /// `CREATE SCHEMA [IF NOT EXISTS] <name> [WITH (...)]`.
75    CreateSchema {
76        /// The schema name as the statement wrote it.
77        name: String,
78        /// The catalog namespace to create.
79        namespace: CatalogNamespace,
80        /// `IF NOT EXISTS`.
81        if_not_exists: bool,
82        /// The `WITH` options, as namespace properties.
83        properties: Vec<(String, String)>,
84    },
85    /// `DROP SCHEMA [IF EXISTS] <name> [RESTRICT]`.
86    DropSchema {
87        /// The schema name as the statement wrote it.
88        name: String,
89        /// The catalog namespace to drop.
90        namespace: CatalogNamespace,
91        /// `IF EXISTS`.
92        if_exists: bool,
93    },
94    /// `INSERT INTO <name> [(<column>, ...)] <query>`, or `INSERT OVERWRITE ...`.
95    Insert {
96        /// The table name as the statement wrote it.
97        name: String,
98        /// The catalog table to write.
99        table: CatalogTable,
100        /// The columns written, as listed; empty for all, in the table's order.
101        columns: Vec<String>,
102        /// The query giving the rows, as SQL.
103        query: String,
104        /// `OVERWRITE`: replace the table's rows.
105        overwrite: bool,
106    },
107    /// `ALTER TABLE [IF EXISTS] <name> <operation>`.
108    AlterTable {
109        /// The table name as the statement wrote it.
110        name: String,
111        /// The catalog table to change.
112        table: CatalogTable,
113        /// `IF EXISTS`.
114        if_exists: bool,
115        /// The change, one per statement.
116        operation: CatalogAlter,
117    },
118}

200/// The catalogs attached to a context.
201#[derive(Clone, Debug, Default)]
202pub(crate) struct CatalogRegistry {
203    catalogs: Vec<AttachedCatalog>,
204    /// The catalog of table names that don't start with an attached catalog.
205    default: Option<String>,
206}
207
208impl CatalogRegistry {
209    pub(crate) fn is_empty(&self) -> bool {
210        self.catalogs.is_empty()
211    }
212
213    /// Attach a catalog; the same name replaces it, with the new `default_namespace`, and keeps
214    /// its default status (`default` only ever sets the default). A name that differs from an
215    /// attached one only in case is an error, so matching names ignoring case is never ambiguous.
216    pub(crate) fn register(
217        &mut self,
218        name: &str,
219        default_namespace: Option<Vec<String>>,
220        default: bool,
221    ) -> PolarsResult<()> {
222        polars_ensure!(!name.is_empty(), InvalidOperation: "catalog name must not be empty");
223        polars_ensure!(
224            default_namespace.as_ref().is_none_or(|ns| !ns.is_empty() && ns.iter().all(|p| !p.is_empty())),
225            InvalidOperation: "the default namespace of catalog '{}' must have non-empty parts", name
226        );
227        if let Some(other) = self
228            .catalogs
229            .iter()
230            .find(|c| c.name != name && c.name.eq_ignore_ascii_case(name))
231        {
232            polars_bail!(
233                InvalidOperation: "catalog '{}' differs from attached catalog '{}' only in case; use that name, or unregister it first",
234                name, other.name
235            );
236        }
237        let keep_default = self.default.as_deref() == Some(name);
238        self.unregister(name);
239        self.catalogs.push(AttachedCatalog {
240            name: name.to_string(),
241            default_namespace,
242        });
243        if default || keep_default {
244            self.default = Some(name.to_string());
245        }
246        Ok(())
247    }
248
249    /// The name of the default catalog.
250    pub(crate) fn default_catalog(&self) -> Option<&str> {
251        self.default.as_deref()
252    }
253
254    /// Make an attached catalog (exact name, else ignoring case) the default.
255    pub(crate) fn set_default(&mut self, name: &str) -> PolarsResult<()> {
256        let Some(name) = self.catalog_name(name).map(str::to_string) else {
257            polars_bail!(SQLInterface: "catalog '{}' is not attached", name);
258        };
259        self.default = Some(name);
260        Ok(())
261    }
262
263    /// Detach a catalog (exact name, else ignoring case), returning the name it had.
264    pub(crate) fn unregister(&mut self, name: &str) -> Option<String> {
265        let idx = self.position(name)?;
266        let removed = self.catalogs.remove(idx).name;
267        if self.default.as_deref() == Some(removed.as_str()) {
268            self.default = None;
269        }
270        Some(removed)
271    }
272
273    fn position(&self, name: &str) -> Option<usize> {
274        self.catalogs
275            .iter()
276            .position(|c| c.name == name)
277            .or_else(|| {
278                self.catalogs
279                    .iter()
280                    .position(|c| c.name.eq_ignore_ascii_case(name))
281            })
282    }
283
284    /// The attached catalog called `name` (exact, else ignoring case).
285    pub(crate) fn catalog_name(&self, name: &str) -> Option<&str> {
286        self.position(name).map(|i| self.catalogs[i].name.as_str())
287    }
288
289    /// The attached catalog a table name explicitly starts with.
290    pub(crate) fn named_catalog(&self, parts: &[String]) -> Option<&str> {
291        match parts {
292            [first, _, ..] => self.catalog_name(first),
293            _ => None,
294        }
295    }
296
297    /// The catalog table a table name refers to: a leading part naming an attached catalog
298    /// selects it, else the default catalog; a name without namespace parts gets the catalog's
299    /// default namespace. `None` if no catalog applies.
300    pub(crate) fn resolve(&self, parts: &[String]) -> Option<CatalogTable> {
301        let (catalog, rest) = self.select(parts)?;
302        let identifier = match rest {
303            [table] => {
304                let mut identifier = catalog.default_namespace.clone()?;
305                identifier.push(table.clone());
306                identifier
307            },
308            _ => rest.to_vec(),
309        };
310        Some(CatalogTable {
311            catalog: catalog.name.clone(),
312            identifier,
313        })
314    }
315
316    /// The catalog namespace a schema name refers to: a leading part naming an attached catalog
317    /// selects it, else the default catalog; the other parts are the namespace (never relative
318    /// to a default namespace). `None` if no catalog applies.
319    pub(crate) fn resolve_namespace(&self, parts: &[String]) -> Option<CatalogNamespace> {
320        let (catalog, rest) = self.select(parts)?;
321        Some(CatalogNamespace {
322            catalog: catalog.name.clone(),
323            namespace: rest.to_vec(),
324        })
325    }
326
327    /// The catalog a name selects, and the name's other parts.
328    fn select<'a>(&self, parts: &'a [String]) -> Option<(&AttachedCatalog, &'a [String])> {
329        let named = match parts {
330            [first, rest @ ..] if !rest.is_empty() => self.position(first).map(|idx| (idx, rest)),
331            _ => None,
332        };
333        let (idx, rest) = match named {
334            Some(named) => named,
335            None => (self.position(self.default.as_deref()?)?, parts),
336        };
337        Some((&self.catalogs[idx], rest))
338    }
339
340    /// Why a table name that nothing resolved is not a catalog table, if a catalog could have
341    /// applied.
342    pub(crate) fn unresolved_hint(&self, parts: &[String]) -> Option<String> {
343        if self.is_empty() || self.resolve(parts).is_some() {
344            return None;
345        }
346        let catalog = match self.named_catalog(parts) {
347            Some(named) => named,
348            None => match &self.default {
349                Some(default) => default.as_str(),
350                None => return Some("; no default catalog is set".to_string()),
351            },
352        };
353        Some(format!("; catalog '{catalog}' has no default namespace"))
354    }
355}

378/// Rewrite the names of catalog tables in `stmt` to their full names, `lineitem` to
379/// `onelake.tpch.lineitem`, so the rest of the translation knows them by that name. Names of
380/// CTEs in scope and of local tables (`is_local`) are left alone. Returns each rewritten name
381/// as written with its full name, and where each name was written.
382pub(crate) fn rewrite_catalog_names(
383    stmt: &mut Statement,
384    resolve: impl Fn(&[String]) -> Option<Vec<String>>,
385    is_local: impl Fn(&str) -> bool,
386) -> PolarsResult<(Vec<RewrittenName>, CatalogSpellings)> {
387    let mut registered = RegisteredAliases(Vec::new());
388    let _ = dyn_visit::visit(&*stmt, &mut registered);
389    let mut rewriter = Rewriter {
390        resolve,
391        is_local,
392        registered_aliases: registered.0,
393        cte_scopes: Vec::new(),
394        with_stack: Vec::new(),
395        rewritten: Vec::new(),
396        spellings: CatalogSpellings::default(),
397        desugared: 0,
398    };
399    if let ControlFlow::Break(err) = dyn_visit::visit_mut(stmt, &mut rewriter) {
400        return Err(err);
401    }
402    Ok((rewriter.rewritten, rewriter.spellings))
403}

405/// The aliases of the statement's derived tables, table functions and `UNNEST`s, which are
406/// registered as tables while the statement runs (#30070), wherever they are written.
407struct RegisteredAliases(Vec<String>);
408
409impl Visitor for RegisteredAliases {
410    type Break = ();
411
412    fn pre_visit_table_factor(&mut self, factor: &TableFactor) -> ControlFlow<()> {
413        let alias = match factor {
414            TableFactor::Table {
415                args: Some(_),
416                alias,
417                ..
418            }
419            | TableFactor::Derived { alias, .. }
420            | TableFactor::UNNEST { alias, .. } => alias.as_ref(),
421            _ => None,
422        };
423        self.0.extend(alias.map(|alias| alias.name.value.clone()));
424        ControlFlow::Continue(())
425    }
426}

428struct Rewriter<R, L> {
429    resolve: R,
430    is_local: L,
431    /// Aliases that tables of the same name would be looked up as (see [`RegisteredAliases`]).
432    registered_aliases: Vec<String>,
433    /// The CTE names visible in each enclosing query.
434    cte_scopes: Vec<Vec<String>>,
435    /// The `WITH` clauses of the queries being visited, taken out while their bodies are visited.
436    with_stack: Vec<Option<With>>,
437    rewritten: Vec<RewrittenName>,
438    spellings: CatalogSpellings,
439    /// The number of `TABLE x` desugared, which numbers their (made-up) name positions.
440    desugared: u64,
441}
442
443impl<R, L> Rewriter<R, L>
444where
445    R: Fn(&[String]) -> Option<Vec<String>>,
446    L: Fn(&str) -> bool,
447{
448    fn is_cte(&self, name: &str) -> bool {
449        self.cte_scopes
450            .iter()
451            .flatten()
452            .any(|cte| cte.eq_ignore_ascii_case(name))
453    }
454
455    /// The full name of the catalog table that `parts` refers to: not a CTE or a local table.
456    fn resolve_name(&self, parts: &[String]) -> Option<Vec<String>> {
457        let written = parts.join(".");
458        if self.is_cte(&written) || (self.is_local)(&written) {
459            return None;
460        }
461        (self.resolve)(parts)
462    }
463
464    /// Record `parts` as referring to the catalog table `full`. Tables are looked up by their
465    /// dotted name, so it must be the catalog table's alone.
466    fn record(&mut self, parts: &[String], full: &[String]) -> PolarsResult<()> {
467        let key = full.join(".");
468        let is_alias = self
469            .registered_aliases
470            .iter()
471            .any(|alias| alias.eq_ignore_ascii_case(&key));
472        if is_alias || self.is_cte(&key) || (self.is_local)(&key) {
473            polars_bail!(
474                SQLInterface: "table '{}' refers to catalog table '{}', which is also the name of a registered table, CTE or alias; rename one of them",
475                display_name(parts), display_name(full)
476            );
477        }
478        if let Some((_, other)) = self
479            .rewritten
480            .iter()
481            .find(|(_, other)| other.join(".") == key && other != full)
482        {
483            polars_bail!(
484                SQLInterface: "catalog tables {} and {} have the same dotted name, which SQL cannot tell apart",
485                display_name(other), display_name(full)
486            );
487        }
488        if !self.rewritten.iter().any(|(written, _)| written == parts) {
489            self.rewritten.push((parts.to_vec(), full.to_vec()));
490        }
491        Ok(())
492    }
493
494    /// Replace `TABLE x` of a catalog table by `SELECT * FROM x`, whose table name the visit
495    /// then rewrites (sqlparser's `TABLE` holds at most two name parts).
496    fn desugar_table(&mut self, body: &mut SetExpr) {
497        match body {
498            SetExpr::SetOperation { left, right, .. } => {
499                self.desugar_table(left);
500                self.desugar_table(right);
501            },
502            SetExpr::Table(table) => {
503                let parts: Vec<String> = table
504                    .schema_name
505                    .iter()
506                    .chain(&table.table_name)
507                    .cloned()
508                    .collect();
509                if !parts.is_empty() && self.resolve_name(&parts).is_some() {
510                    // Line 0 is never a position in the query text.
511                    self.desugared += 1;
512                    let at = Location::new(0, self.desugared);
513                    *body = select_star_from(&parts, Span::new(at, at));
514                }
515            },
516            _ => {},
517        }
518    }
519}

521impl<R, L> VisitorMut for Rewriter<R, L>
522where
523    R: Fn(&[String]) -> Option<Vec<String>>,
524    L: Fn(&str) -> bool,
525{
526    type Break = PolarsError;
527
528    fn pre_visit_query(&mut self, query: &mut Query) -> ControlFlow<PolarsError> {
529        // A non-recursive CTE is visible after its definition, as `register_ctes` registers them.
530        let mut with = query.with.take();
531        let mut names = Vec::new();
532        if let Some(with) = with.as_mut() {
533            let all: Vec<String> = with
534                .cte_tables
535                .iter()
536                .map(|cte| cte.alias.name.value.clone())
537                .collect();
538            let recursive = with.recursive;
539            for cte in &mut with.cte_tables {
540                self.cte_scopes.push(if recursive {
541                    all.clone()
542                } else {
543                    names.clone()
544                });
545                let flow = dyn_visit::visit_mut(&mut *cte.query, self);
546                self.cte_scopes.pop();
547                flow?;
548                names.push(cte.alias.name.value.clone());
549            }
550        }
551        self.cte_scopes.push(names);
552        self.with_stack.push(with);
553        self.desugar_table(&mut query.body);
554        ControlFlow::Continue(())
555    }
556
557    fn post_visit_query(&mut self, query: &mut Query) -> ControlFlow<PolarsError> {
558        self.cte_scopes.pop();
559        query.with = self.with_stack.pop().flatten();
560        ControlFlow::Continue(())
561    }
562
563    fn pre_visit_table_factor(&mut self, factor: &mut TableFactor) -> ControlFlow<PolarsError> {
564        // A name with arguments is a table function.
565        let TableFactor::Table {
566            name, args: None, ..
567        } = factor
568        else {
569            return ControlFlow::Continue(());
570        };
571        let Some(parts) = name_parts(name) else {
572            return ControlFlow::Continue(());
573        };
574        if let Some(full) = self.resolve_name(&parts) {
575            if let Err(err) = self.record(&parts, &full) {
576                return ControlFlow::Break(err);
577            }
578            let span = name.0[0]
579                .as_ident()
580                .map_or_else(Span::empty, |ident| ident.span);
581            if let Some(position) = name_position(name) {
582                self.spellings.insert(position, parts);
583            }
584            *name = object_name(&full, span);
585        }
586        ControlFlow::Continue(())
587    }
588}

590/// A table name whose parts all have `span`, so the name keeps the position it was written at.
591fn object_name(parts: &[String], span: Span) -> ObjectName {
592    let ident = |part: &String| Ident {
593        value: part.clone(),
594        quote_style: None,
595        span,
596    };
597    ObjectName(
598        parts
599            .iter()
600            .map(|part| ObjectNamePart::Identifier(ident(part)))
601            .collect(),
602    )
603}

605/// `SELECT * FROM <parts>`, with the name at `span`.
606fn select_star_from(parts: &[String], span: Span) -> SetExpr {
607    let mut stmts = Parser::parse_sql(&GenericDialect, "SELECT * FROM t").unwrap();
608    let Statement::Query(query) = stmts.remove(0) else {
609        unreachable!()
610    };
611    let mut body = *query.body;
612    if let SetExpr::Select(select) = &mut body
613        && let TableFactor::Table { name, .. } = &mut select.from[0].relation
614    {
615        *name = object_name(parts, span);
616    }
617    body
618}

620/// The schema of a catalog table from its column definitions, and its `NOT NULL` columns.
621pub(crate) fn create_table_columns(columns: &[ColumnDef]) -> PolarsResult<(Schema, Vec<String>)> {
622    let mut schema = Schema::with_capacity(columns.len());
623    let mut required = Vec::new();
624    for col in columns {
625        let name = col.name.value.as_str();
626        let mut not_null = None;
627        for option in &col.options {
628            let is_not_null = match &option.option {
629                ColumnOption::NotNull => true,
630                ColumnOption::Null => false,
631                // the default of a nullable column anyway
632                ColumnOption::Default(expr) if is_null(expr) => continue,
633                _ => polars_bail!(
634                    SQLInterface: "only NOT NULL constraints are supported for catalog tables; found `{}` on column '{}'",
635                    option, name
636                ),
637            };
638            polars_ensure!(
639                not_null.is_none_or(|n| n == is_not_null),
640                SQLInterface: "conflicting NULL/NOT NULL declarations for column '{}'", name
641            );
642            not_null = Some(is_not_null);
643        }
644        let dtype = map_sql_dtype_to_polars(&col.data_type)?;
645        polars_ensure!(
646            schema.insert(name.into(), dtype).is_none(),
647            Duplicate: "column '{}' is defined more than once", name
648        );
649        if not_null == Some(true) {
650            required.push(name.to_string());
651        }
652    }
653    Ok((schema, required))
654}

656/// The name and type of a column added by `ALTER TABLE ... ADD COLUMN`, which is nullable.
657pub(crate) fn alter_table_column(col: &ColumnDef) -> PolarsResult<(String, DataType)> {
658    let name = col.name.value.as_str();
659    for option in &col.options {
660        match &option.option {
661            ColumnOption::Null => {},
662            ColumnOption::Default(expr) if is_null(expr) => {},
663            ColumnOption::NotNull => polars_bail!(
664                SQLInterface: "`ADD COLUMN ... NOT NULL` is not supported: the table's rows have no value for column '{}'",
665                name
666            ),
667            _ => polars_bail!(
668                SQLInterface: "`ADD COLUMN` takes only NULL or DEFAULT NULL; found `{}` on column '{}'",
669                option, name
670            ),
671        }
672    }
673    Ok((name.to_string(), map_sql_dtype_to_polars(&col.data_type)?))
674}

676/// Whether `expr` is the literal `NULL`.
677fn is_null(expr: &Expr) -> bool {
678    matches!(
679        expr,
680        Expr::Value(ValueWithSpan {
681            value: Value::Null,
682            ..
683        })
684    )
685}

687/// The properties of an `ALTER TABLE ... SET TBLPROPERTIES (...)`.
688pub(crate) fn alter_table_properties(options: &[SqlOption]) -> PolarsResult<Vec<(String, String)>> {
689    let mut properties: Vec<(String, String)> = Vec::new();
690    for option in options {
691        let (key, value) = option_key_value(option, "table", &["location", "format-version"])?;
692        polars_ensure!(
693            !RESERVED_TABLE_PROPERTIES.contains(&key) && key != "format-version",
694            SQLInterface: "table property '{}' is reserved by Iceberg", key
695        );
696        polars_ensure!(
697            key != "location",
698            SQLInterface: "a table's location is not a property and cannot be set"
699        );
700        polars_ensure!(
701            properties.iter().all(|(k, _)| k != key),
702            SQLInterface: "duplicate table property '{}'", key
703        );
704        properties.push((key.to_string(), value));
705    }
706    Ok(properties)
707}

709/// Iceberg's reserved table properties, other than `format-version`, which a create may set.
710const RESERVED_TABLE_PROPERTIES: [&str; 8] = [
711    "uuid",
712    "snapshot-count",
713    "current-snapshot-id",
714    "current-snapshot-summary",
715    "current-snapshot-timestamp-ms",
716    "current-schema",
717    "default-partition-spec",
718    "default-sort-order",
719];

721/// Table properties, in the order written.
722type TableProperties = Vec<(String, String)>;

724/// The location and properties of a catalog table, from `LOCATION` and `WITH (...)`.
725pub(crate) fn create_table_options(
726    options: &CreateTableOptions,
727    location_clause: Option<&str>,
728) -> PolarsResult<(Option<String>, TableProperties)> {
729    let options = match options {
730        CreateTableOptions::None => &[][..],
731        CreateTableOptions::With(options) => options.as_slice(),
732        CreateTableOptions::TableProperties(_) => {
733            polars_bail!(SQLInterface: "`TBLPROPERTIES` is not supported; use `WITH (...)` instead")
734        },
735        _ => {
736            polars_bail!(SQLInterface: "table `OPTIONS` clauses are not supported; use `WITH (...)` instead")
737        },
738    };
739    let mut location = location_clause.map(str::to_string);
740    let mut properties = TableProperties::new();
741    for option in options {
742        let (key, value) = option_key_value(option, "table", &["location", "format-version"])?;
743        polars_ensure!(
744            !RESERVED_TABLE_PROPERTIES.contains(&key),
745            SQLInterface: "table option '{}' is reserved by Iceberg", key
746        );
747        if key == "location" {
748            polars_ensure!(
749                location_clause.is_none(),
750                SQLInterface: "`LOCATION` and table option 'location' cannot be combined"
751            );
752            polars_ensure!(location.is_none(), SQLInterface: "duplicate table option 'location'");
753            location = Some(value);
754        } else {
755            polars_ensure!(
756                properties.iter().all(|(k, _)| k != key),
757                SQLInterface: "duplicate table option '{}'", key
758            );
759            properties.push((key.to_string(), value));
760        }
761    }
762    if let Some(location) = &location {
763        polars_ensure!(
764            is_absolute_location(location),
765            SQLInterface: "location must be an absolute path or a URI; found '{}'", location
766        );
767    }
768    Ok((location, properties))
769}

771/// The partition fields of a catalog table, from `PARTITION BY (...)` or `PARTITIONED BY (...)`.
772pub(crate) fn create_table_partitioning(
773    partition_by: Option<&Expr>,
774    hive_distribution: &HiveDistributionStyle,
775) -> PolarsResult<Vec<CatalogPartitionField>> {
776    // Each field with its SQL, for messages.
777    let fields: Vec<(CatalogPartitionField, String)> = match (partition_by, hive_distribution) {
778        (Some(_), HiveDistributionStyle::PARTITIONED { .. }) => {
779            polars_bail!(SQLInterface: "`PARTITION BY` and `PARTITIONED BY` cannot be combined")
780        },
781        (Some(expr), _) => {
782            let exprs = match unparenthesized(expr) {
783                Expr::Tuple(exprs) => exprs.as_slice(),
784                expr => std::slice::from_ref(expr),
785            };
786            exprs
787                .iter()
788                .map(|expr| Ok((partition_field(expr)?, expr.to_string())))
789                .collect::<PolarsResult<_>>()?
790        },
791        (None, HiveDistributionStyle::PARTITIONED { columns }) => columns
792            .iter()
793            .map(|col| {
794                // Hive's `PARTITIONED BY (p INT)` adds a column.
795                polars_ensure!(
796                    col.data_type == SQLDataType::Unspecified && col.options.is_empty(),
797                    SQLInterface: "`PARTITIONED BY` takes column names only; found `{}`; types and constraints belong in the column list",
798                    col
799                );
800                let field = CatalogPartitionField {
801                    column: col.name.value.clone(),
802                    transform: "identity".to_string(),
803                };
804                Ok((field, col.name.value.clone()))
805            })
806            .collect::<PolarsResult<_>>()?,
807        (None, _) => Vec::new(),
808    };
809    // Iceberg allows one time transform per column.
810    let is_time = |f: &CatalogPartitionField| {
811        matches!(f.transform.as_str(), "year" | "month" | "day" | "hour")
812    };
813    for (i, (field, sql)) in fields.iter().enumerate() {
814        for (earlier, earlier_sql) in &fields[..i] {
815            polars_ensure!(earlier != field, SQLInterface: "duplicate partition field `{}`", sql);
816            polars_ensure!(
817                earlier.column != field.column || !is_time(earlier) || !is_time(field),
818                SQLInterface: "partition fields `{}` and `{}` both partition column '{}' by time; use one of them",
819                earlier_sql, sql, field.column
820            );
821        }
822    }
823    Ok(fields.into_iter().map(|(field, _)| field).collect())
824}

826/// A partition field: a column, `year|month|day|hour(column)` (with Iceberg's aliases), or
827/// `bucket|truncate(N, column)`.
828fn partition_field(expr: &Expr) -> PolarsResult<CatalogPartitionField> {
829    let unsupported = || {
830        polars_err!(
831            SQLInterface: "unsupported partition field `{}`; expected a column, or year, month, day, hour, bucket(N, column) or truncate(W, column) of a column",
832            expr
833        )
834    };
835    let column = |expr: &Expr| match unparenthesized(expr) {
836        Expr::Identifier(ident) => Some(ident.value.clone()),
837        _ => None,
838    };
839    // A qualified column or a struct field: without the schema, these look the same.
840    let is_compound = |expr: &Expr| matches!(unparenthesized(expr), Expr::CompoundIdentifier(_));
841    let compound = |expr: &Expr| {
842        polars_err!(
843            SQLInterface: "`PARTITION BY` takes unqualified column names; found `{}`; partitioning by a nested field is not yet supported",
844            unparenthesized(expr)
845        )
846    };
847    if let Some(column) = column(expr) {
848        return Ok(CatalogPartitionField {
849            column,
850            transform: "identity".to_string(),
851        });
852    }
853    if is_compound(expr) {
854        return Err(compound(expr));
855    }
856    let Expr::Function(Function {
857        name,
858        uses_odbc_syntax: false,
859        parameters: FunctionArguments::None,
860        args: FunctionArguments::List(list),
861        filter: None,
862        null_treatment: None,
863        over: None,
864        within_group,
865    }) = unparenthesized(expr)
866    else {
867        return Err(unsupported());
868    };
869    let fname = match name_parts(name).as_deref() {
870        Some([fname])
871            if within_group.is_empty()
872                && list.duplicate_treatment.is_none()
873                && list.clauses.is_empty() =>
874        {
875            fname.to_ascii_lowercase()
876        },
877        _ => return Err(unsupported()),
878    };
879    let args = list
880        .args
881        .iter()
882        .map(|arg| match arg {
883            FunctionArg::Unnamed(FunctionArgExpr::Expr(expr)) => Some(expr),
884            _ => None,
885        })
886        .collect::<Option<Vec<_>>>()
887        .ok_or_else(unsupported)?;
888    let positive_int = |expr: &Expr| match unparenthesized(expr) {
889        Expr::Value(ValueWithSpan {
890            value: Value::Number(n, false),
891            ..
892        }) => n.parse::<i32>().ok().filter(|n| *n > 0),
893        _ => None,
894    };
895    let (transform, source) = match (fname.as_str(), args.as_slice()) {
896        ("year" | "years", [source]) => ("year".to_string(), source),
897        ("month" | "months", [source]) => ("month".to_string(), source),
898        ("day" | "days", [source]) => ("day".to_string(), source),
899        ("hour" | "hours", [source]) => ("hour".to_string(), source),
900        ("bucket" | "truncate", [n, source]) => {
901            let Some(n) = positive_int(n) else {
902                if (column(n).is_some() || is_compound(n)) && positive_int(source).is_some() {
903                    polars_bail!(SQLInterface: "`{}` takes the number first, as in `{}({}, {})`", expr, fname, source, n)
904                }
905                polars_bail!(SQLInterface: "the first argument of `{}` must be an integer from 1 to {}", expr, i32::MAX)
906            };
907            (format!("{fname}[{n}]"), source)
908        },
909        _ => return Err(unsupported()),
910    };
911    let column = match column(source) {
912        Some(column) => column,
913        None if is_compound(source) => return Err(compound(source)),
914        None => return Err(unsupported()),
915    };
916    Ok(CatalogPartitionField { column, transform })
917}

919/// `expr` without enclosing parentheses.
920fn unparenthesized(mut expr: &Expr) -> &Expr {
921    while let Expr::Nested(inner) = expr {
922        expr = inner;
923    }
924    expr
925}

937            properties.iter().all(|(k, _)| k != key),

939        );
940        properties.push((key.to_string(), value));
941    }
942    Ok(properties)

945/// The key and value of a `'key' = value` option of a `kind` ("table", "schema"), with keys that
946/// differ from a `known` one only in case rejected.
947fn option_key_value<'a>(
948    option: &'a SqlOption,
949    kind: &str,
950    known: &[&str],
951) -> PolarsResult<(&'a str, String)> {
952    let SqlOption::KeyValue { key, value } = option else {
953        polars_bail!(SQLInterface: "{} options must be `'key' = value` pairs; found `{}`", kind, option)
954    };
955    let key = key.value.as_str();
956    polars_ensure!(!key.is_empty(), SQLInterface: "{} option names must not be empty", kind);
957    for known in known {
958        polars_ensure!(
959            key == *known || !key.eq_ignore_ascii_case(known),
960            SQLInterface: "'{}' is not a {} option; did you mean '{}'?", key, kind, known
961        );
962    }
963    let Some(value) = option_value(value) else {
964        polars_bail!(
965            SQLInterface: "{} option values must be string, number or boolean literals; found `{}` for '{}'",
966            kind, value, key
967        )
968    };
969    Ok((key, value))
970}

972/// A table option value as a property string, if it is a string, number or boolean literal.
973fn option_value(value: &Expr) -> Option<String> {
974    match value {
975        Expr::Value(v) => match &v.value {
976            Value::SingleQuotedString(s) => Some(s.clone()),
977            Value::Number(n, _) => Some(n.clone()),
978            Value::Boolean(b) => Some(b.to_string()),
979            _ => None,
980        },
981        Expr::UnaryOp {
982            op: UnaryOperator::Minus,
983            expr,
984        } => match expr.as_ref() {
985            Expr::Value(ValueWithSpan {
986                value: Value::Number(n, _),
987                ..
988            }) => Some(format!("-{n}")),
989            _ => None,
990        },
991        _ => None,
992    }
993}

995/// Whether a location is an absolute path (POSIX or Windows) or has a URI scheme.
996fn is_absolute_location(location: &str) -> bool {
997    let bytes = location.as_bytes();
998    let windows_path = bytes.len() > 2
999        && bytes[0].is_ascii_alphabetic()
1000        && bytes[1] == b':'
1001        && matches!(bytes[2], b'/' | b'\\');
1002    let has_scheme = location.split_once("://").is_some_and(|(scheme, _)| {
1003        scheme.starts_with(|c: char| c.is_ascii_alphabetic())
1004            && scheme
1005                .chars()
1006                .all(|c| c.is_ascii_alphanumeric() || matches!(c, '+' | '-' | '.'))
1007    });
1008    location.starts_with('/') || windows_path || has_scheme
1009}

1011/// A table name for messages, with parts containing dots quoted.
1012pub(crate) fn display_name(parts: &[String]) -> String {
1013    let parts: Vec<String> = parts
1014        .iter()
1015        .map(|part| match part.contains('.') {
1016            true => format!("\"{part}\""),
1017            false => part.clone(),
1018        })
1019        .collect();
1020    parts.join(".")
