239/// The procedure and arguments of a `CALL`. Checks the procedure and argument names, not the
240/// values.
241pub(crate) fn call_parts(function: &Function) -> PolarsResult<CallParts> {
242    let Some(parts) = name_parts(&function.name) else {
243        polars_bail!(SQLInterface: "CALL takes a procedure name, found '{}'", function.name);
244    };
245    let procedure = parts.last().map(String::as_str).unwrap_or_default();
246    let Some(&(procedure, signature)) = CALL_PROCEDURES
247        .iter()
248        .find(|(name, _)| name.eq_ignore_ascii_case(procedure))
249    else {
250        let supported: Vec<&str> = CALL_PROCEDURES.iter().map(|(name, _)| *name).collect();
251        polars_bail!(
252            SQLInterface: "unknown procedure '{}' in CALL; supported: {}",
253            function.name, supported.join(", ")
254        );
255    };
256    polars_ensure!(
257        function.filter.is_none()
258            && function.over.is_none()
259            && function.null_treatment.is_none()
260            && function.within_group.is_empty()
261            && matches!(function.parameters, FunctionArguments::None),
262        SQLInterface: "CALL takes a procedure and its arguments only: `{}`", function
263    );
264    let list = match &function.args {
265        FunctionArguments::List(list)
266            if list.duplicate_treatment.is_none() && list.clauses.is_empty() =>
267        {
268            list
269        },
270        FunctionArguments::None => polars_bail!(
271            SQLInterface: "CALL {} needs an argument list, e.g. `{}(table => 'ns.t')`",
272            function.name, function.name
273        ),
274        _ => polars_bail!(SQLInterface: "CALL takes literal arguments only: `{}`", function),
275    };
276    let mut args: Vec<(String, Option<CallArg>)> = Vec::new();
277    let mut named = false;
278    for (position, arg) in list.args.iter().enumerate() {
279        let (name, value) = match arg {
280            FunctionArg::Named {
281                name,
282                arg: FunctionArgExpr::Expr(value),
283                operator: FunctionArgOperator::RightArrow,
284            } => {
285                named = true;
286                (name.value.clone(), value)
287            },
288            FunctionArg::ExprNamed {
289                name: Expr::Identifier(name),
290                arg: FunctionArgExpr::Expr(value),
291                operator: FunctionArgOperator::RightArrow,
292            } => {
293                named = true;
294                (name.value.clone(), value)
295            },
296            FunctionArg::Unnamed(FunctionArgExpr::Expr(value)) => {
297                polars_ensure!(
298                    !named,
299                    SQLInterface: "CALL {}: a positional argument cannot follow a named one",
300                    function.name
301                );
302                let Some(name) = signature.get(position) else {
303                    polars_bail!(
304                        SQLInterface: "CALL {} takes at most {} arguments",
305                        function.name, signature.len()
306                    );
307                };
308                (name.to_string(), value)
309            },
310            _ => polars_bail!(
311                SQLInterface: "CALL arguments are `value` or `name => value`, found `{}`", arg
312            ),
313        };
314        let Some(name) = signature
315            .iter()
316            .find(|known| known.eq_ignore_ascii_case(&name))
317        else {
318            polars_bail!(
319                SQLInterface: "unknown argument '{}' of procedure '{}'; supported: {}",
320                name, procedure, signature.join(", ")
321            );
322        };
323        polars_ensure!(
324            args.iter().all(|(given, _)| given.as_str() != *name),
325            SQLInterface: "argument '{}' of procedure '{}' is given twice", name, procedure
326        );
327        args.push((name.to_string(), call_arg(value)?));
328    }
329    args.sort_by_key(|(name, _)| signature.iter().position(|known| known == name));
330    let args = args
331        .into_iter()
332        .filter_map(|(name, value)| value.map(|value| (name, value)))
333        .collect();
334    Ok(CallParts {
335        name: parts,
336        procedure,
337        args,
338    })
339}

349/// A `CALL` argument value; `None` for `NULL`.
350fn call_arg(value: &Expr) -> PolarsResult<Option<CallArg>> {
351    let literal = |value: &Expr| match value {
352        Expr::Value(v) => match &v.value {
353            Value::SingleQuotedString(s) => Some(s.clone()),
354            _ => None,
355        },
356        _ => None,
357    };
358    let arg = match value {
359        Expr::Value(v) => match &v.value {
360            Value::SingleQuotedString(s) => Some(CallArg::String(s.clone())),
361            Value::Boolean(b) => Some(CallArg::Bool(*b)),
362            Value::Null => return Ok(None),
363            _ => option_value(value)
364                .and_then(|n| n.parse().ok())
365                .map(CallArg::Int),
366        },
367        Expr::UnaryOp { .. } => option_value(value)
368            .and_then(|n| n.parse().ok())
369            .map(CallArg::Int),
370        Expr::Function(function) => {
371            let name = function.name.to_string();
372            let values = match &function.args {
373                FunctionArguments::List(list)
374                    if list.duplicate_treatment.is_none() && list.clauses.is_empty() =>
375                {
376                    list.args
377                        .iter()
378                        .map(|arg| match arg {
379                            FunctionArg::Unnamed(FunctionArgExpr::Expr(value)) => literal(value),
380                            _ => None,
381                        })
382                        .collect::<Option<Vec<String>>>()
383                },
384                _ => None,
385            };
386            match values {
387                Some(values) if name.eq_ignore_ascii_case("array") => Some(CallArg::Array(values)),
388                Some(values) if name.eq_ignore_ascii_case("map") && values.len() % 2 == 0 => {
389                    Some(CallArg::Map(
390                        values
391                            .chunks(2)
392                            .map(|kv| (kv[0].clone(), kv[1].clone()))
393                            .collect(),
394                    ))
395                },
396                _ => None,
397            }
398        },
399        _ => None,
400    };
401    match arg {
402        Some(arg) => Ok(Some(arg)),
403        None => polars_bail!(
404            SQLInterface: "CALL arguments must be literals: strings, integers, booleans, NULL, \
405            map('k', 'v', ...) or array('a', ...); found `{}`",
406            value
407        ),
408    }
409}

607impl CatalogRegistry {
608    pub(crate) fn is_empty(&self) -> bool {
609        self.catalogs.is_empty()
610    }
611
612    /// Attach a catalog; the same name replaces it, with the new `default_namespace`, and keeps
613    /// its default status (`default` only ever sets the default). A name that differs from an
614    /// attached one only in case is an error, so matching names ignoring case is never ambiguous.
615    pub(crate) fn register(
616        &mut self,
617        name: &str,
618        default_namespace: Option<Vec<String>>,
619        default: bool,
620    ) -> PolarsResult<()> {
621        polars_ensure!(!name.is_empty(), InvalidOperation: "catalog name must not be empty");
622        polars_ensure!(
623            default_namespace.as_ref().is_none_or(|ns| !ns.is_empty() && ns.iter().all(|p| !p.is_empty())),
624            InvalidOperation: "the default namespace of catalog '{}' must have non-empty parts", name
625        );
626        if let Some(other) = self
627            .catalogs
628            .iter()
629            .find(|c| c.name != name && c.name.eq_ignore_ascii_case(name))
630        {
631            polars_bail!(
632                InvalidOperation: "catalog '{}' differs from attached catalog '{}' only in case; use that name, or unregister it first",
633                name, other.name
634            );
635        }
636        let keep_default = self.default.as_deref() == Some(name);
637        self.unregister(name);
638        self.catalogs.push(AttachedCatalog {
639            name: name.to_string(),
640            default_namespace,
641        });
642        if default || keep_default {
643            self.default = Some(name.to_string());
644        }
645        Ok(())
646    }
647
648    /// The name of the default catalog.

664        let idx = self.position(name)?;

679                    .position(|c| c.name.eq_ignore_ascii_case(name))

696    /// The catalog table a table name refers to: a leading part naming an attached catalog
697    /// selects it, else the default catalog; a name without namespace parts gets the catalog's
698    /// default namespace. `None` if no catalog applies.
699    pub(crate) fn resolve(&self, parts: &[String]) -> Option<CatalogTable> {
700        let (catalog, rest) = self.select(parts)?;
701        let identifier = match rest {
702            [table] => {
703                let mut identifier = catalog.default_namespace.clone()?;
704                identifier.push(table.clone());
705                identifier
706            },
707            _ => rest.to_vec(),
708        };
709        Some(CatalogTable {
710            catalog: catalog.name.clone(),
711            identifier,
712        })
713    }

715    /// The catalog namespace a schema name refers to: a leading part naming an attached catalog
716    /// selects it, else the default catalog; the other parts are the namespace (never relative
717    /// to a default namespace). `None` if no catalog applies.
718    pub(crate) fn resolve_namespace(&self, parts: &[String]) -> Option<CatalogNamespace> {
719        let (catalog, rest) = self.select(parts)?;
720        Some(CatalogNamespace {
721            catalog: catalog.name.clone(),
722            namespace: rest.to_vec(),
723        })
724    }

726    /// The catalog a name selects, and the name's other parts.
727    fn select<'a>(&self, parts: &'a [String]) -> Option<(&AttachedCatalog, &'a [String])> {
728        let named = match parts {
729            [first, rest @ ..] if !rest.is_empty() => self.position(first).map(|idx| (idx, rest)),
730            _ => None,
731        };
732        let (idx, rest) = match named {
733            Some(named) => named,
734            None => (self.position(self.default.as_deref()?)?, parts),
735        };
736        Some((&self.catalogs[idx], rest))
737    }

739    /// Why a table name that nothing resolved is not a catalog table, if a catalog could have
740    /// applied.
741    pub(crate) fn unresolved_hint(&self, parts: &[String]) -> Option<String> {
742        if self.is_empty() || self.resolve(parts).is_some() {
743            return None;
744        }
745        let catalog = match self.named_catalog(parts) {
746            Some(named) => named,
747            None => match &self.default {
748                Some(default) => default.as_str(),
749                None => return Some("; no default catalog is set".to_string()),
750            },
751        };
752        Some(format!("; catalog '{catalog}' has no default namespace"))
753    }

777/// Rewrite the names of catalog tables in `stmt` to their full names, `lineitem` to
778/// `onelake.tpch.lineitem`, so the rest of the translation knows them by that name. Names of
779/// CTEs in scope and of local tables (`is_local`) are left alone. Returns each rewritten name
780/// as written with its full name, and where each name was written.
781pub(crate) fn rewrite_catalog_names(
782    stmt: &mut Statement,
783    resolve: impl Fn(&[String]) -> Option<Vec<String>>,
784    is_local: impl Fn(&str) -> bool,
785) -> PolarsResult<(Vec<RewrittenName>, CatalogSpellings)> {
786    let mut registered = RegisteredAliases(Vec::new());
787    let _ = dyn_visit::visit(&*stmt, &mut registered);
788    let mut rewriter = Rewriter {
789        resolve,
790        is_local,
791        registered_aliases: registered.0,
792        cte_scopes: Vec::new(),
793        with_stack: Vec::new(),
794        rewritten: Vec::new(),
795        spellings: CatalogSpellings::default(),
796        desugared: 0,
797    };
798    if let ControlFlow::Break(err) = dyn_visit::visit_mut(stmt, &mut rewriter) {
799        return Err(err);
800    }
801    Ok((rewriter.rewritten, rewriter.spellings))
802}

804/// The aliases of the statement's derived tables, table functions and `UNNEST`s, which are
805/// registered as tables while the statement runs (#30070), wherever they are written.
806struct RegisteredAliases(Vec<String>);
807
808impl Visitor for RegisteredAliases {
809    type Break = ();
810
811    fn pre_visit_table_factor(&mut self, factor: &TableFactor) -> ControlFlow<()> {
812        let alias = match factor {
813            TableFactor::Table {
814                args: Some(_),
815                alias,
816                ..
817            }
818            | TableFactor::Derived { alias, .. }
819            | TableFactor::UNNEST { alias, .. }
820            | TableFactor::Pivot { alias, .. }
821            | TableFactor::Unpivot { alias, .. } => alias.as_ref(),
822            _ => None,
823        };
824        self.0.extend(alias.map(|alias| alias.name.value.clone()));
825        ControlFlow::Continue(())
826    }
827}

829struct Rewriter<R, L> {
830    resolve: R,
831    is_local: L,
832    /// Aliases that tables of the same name would be looked up as (see [`RegisteredAliases`]).
833    registered_aliases: Vec<String>,
834    /// The CTE names visible in each enclosing query.
835    cte_scopes: Vec<Vec<String>>,
836    /// The `WITH` clauses of the queries being visited, taken out while their bodies are visited.
837    with_stack: Vec<Option<With>>,
838    rewritten: Vec<RewrittenName>,
839    spellings: CatalogSpellings,
840    /// The number of `TABLE x` desugared, which numbers their (made-up) name positions.
841    desugared: u64,
842}
843
844impl<R, L> Rewriter<R, L>
845where
846    R: Fn(&[String]) -> Option<Vec<String>>,
847    L: Fn(&str) -> bool,
848{
849    fn is_cte(&self, name: &str) -> bool {
850        self.cte_scopes
851            .iter()
852            .flatten()
853            .any(|cte| cte.eq_ignore_ascii_case(name))
854    }
855
856    /// The full name of the catalog table that `parts` refers to: not a CTE or a local table.
857    fn resolve_name(&self, parts: &[String]) -> Option<Vec<String>> {
858        let written = parts.join(".");
859        if self.is_cte(&written) || (self.is_local)(&written) {
860            return None;
861        }
862        (self.resolve)(parts)
863    }
864
865    /// Record `parts` as referring to the catalog table `full`. Tables are looked up by their
866    /// dotted name, so it must be the catalog table's alone.
867    fn record(&mut self, parts: &[String], full: &[String]) -> PolarsResult<()> {
868        let key = full.join(".");
869        let is_alias = self
870            .registered_aliases
871            .iter()
872            .any(|alias| alias.eq_ignore_ascii_case(&key));
873        if is_alias || self.is_cte(&key) || (self.is_local)(&key) {
874            polars_bail!(
875                SQLInterface: "table '{}' refers to catalog table '{}', which is also the name of a registered table, CTE or alias; rename one of them",
876                display_name(parts), display_name(full)
877            );
878        }
879        if let Some((_, other)) = self
880            .rewritten
881            .iter()
882            .find(|(_, other)| other.join(".") == key && other != full)
883        {
884            polars_bail!(
885                SQLInterface: "catalog tables {} and {} have the same dotted name, which SQL cannot tell apart",
886                display_name(other), display_name(full)
887            );
888        }
889        if !self.rewritten.iter().any(|(written, _)| written == parts) {
890            self.rewritten.push((parts.to_vec(), full.to_vec()));
891        }
892        Ok(())
893    }
894
895    /// Replace `TABLE x` of a catalog table by `SELECT * FROM x`, whose table name the visit
896    /// then rewrites (sqlparser's `TABLE` holds at most two name parts).
897    fn desugar_table(&mut self, body: &mut SetExpr) {
898        match body {
899            SetExpr::SetOperation { left, right, .. } => {
900                self.desugar_table(left);
901                self.desugar_table(right);
902            },
903            SetExpr::Table(table) => {
904                let parts: Vec<String> = table
905                    .schema_name
906                    .iter()
907                    .chain(&table.table_name)
908                    .cloned()
909                    .collect();
910                if !parts.is_empty() && self.resolve_name(&parts).is_some() {
911                    // Line 0 is never a position in the query text.
912                    self.desugared += 1;
913                    let at = Location::new(0, self.desugared);
914                    *body = select_star_from(&parts, Span::new(at, at));
915                }
916            },
917            _ => {},
918        }
919    }
920}
921
922impl<R, L> VisitorMut for Rewriter<R, L>
923where
924    R: Fn(&[String]) -> Option<Vec<String>>,
925    L: Fn(&str) -> bool,
926{
927    type Break = PolarsError;
928
929    fn pre_visit_query(&mut self, query: &mut Query) -> ControlFlow<PolarsError> {
930        // A non-recursive CTE is visible after its definition, as `register_ctes` registers them.
931        let mut with = query.with.take();
932        let mut names = Vec::new();
933        if let Some(with) = with.as_mut() {
934            let all: Vec<String> = with
935                .cte_tables
936                .iter()
937                .map(|cte| cte.alias.name.value.clone())
938                .collect();
939            let recursive = with.recursive;
940            for cte in &mut with.cte_tables {
941                self.cte_scopes.push(if recursive {
942                    all.clone()
943                } else {
944                    names.clone()
945                });
946                let flow = dyn_visit::visit_mut(&mut *cte.query, self);
947                self.cte_scopes.pop();
948                flow?;
949                names.push(cte.alias.name.value.clone());
950            }
951        }
952        self.cte_scopes.push(names);
953        self.with_stack.push(with);
954        self.desugar_table(&mut query.body);
955        ControlFlow::Continue(())
956    }
957
958    fn post_visit_query(&mut self, query: &mut Query) -> ControlFlow<PolarsError> {
959        self.cte_scopes.pop();
960        query.with = self.with_stack.pop().flatten();
961        ControlFlow::Continue(())
962    }
963
964    fn pre_visit_statement(&mut self, stmt: &mut Statement) -> ControlFlow<PolarsError> {
965        // `DESCRIBE <table>` names its table as `FROM` does.
966        if let Statement::ExplainTable { table_name, .. } = stmt
967            && let SetExpr::Select(mut select) = select_star_from(&[], Span::empty())
968            && let TableFactor::Table { name, .. } = &mut select.from[0].relation
969        {
970            *name = table_name.clone();
971            self.pre_visit_table_factor(&mut select.from[0].relation)?;
972            if let TableFactor::Table { name, .. } = &select.from[0].relation {
973                *table_name = name.clone();
974            }
975        }
976        ControlFlow::Continue(())
977    }
978
979    fn pre_visit_table_factor(&mut self, factor: &mut TableFactor) -> ControlFlow<PolarsError> {
980        // A name with arguments is a table function.
981        let TableFactor::Table {
982            name,
983            args: None,
984            version,
985            ..
986        } = factor
987        else {
988            return ControlFlow::Continue(());
989        };
990        let Some(parts) = name_parts(name) else {
991            return ControlFlow::Continue(());
992        };
993        if parts.iter().any(|part| is_version_part(part)) {
994            return ControlFlow::Break(polars_err!(
995                SQLInterface: "invalid table name {}", name
996            ));
997        }
998        let written = parts.join(".");
999        if self.is_cte(&written) || (self.is_local)(&written) {
1000            // a time travel clause is refused where the table is read
1001            return ControlFlow::Continue(());
1002        }
1003        let spec = match version
1004            .as_ref()
1005            .map(VersionSpec::from_table_version)
1006            .transpose()
1007        {
1008            Ok(spec) => spec,
1009            Err(err) => return ControlFlow::Break(err),
1010        };
1011        // A versioned read is its own relation, known by the name and the version.
1012        let key = with_version(parts.clone(), spec.as_ref());
1013        if let Some(full) = (self.resolve)(&key) {
1014            if let Err(err) = self.record(&key, &full) {
1015                return ControlFlow::Break(err);
1016            }
1017            let span = name.0[0]
1018                .as_ident()
1019                .map_or_else(Span::empty, |ident| ident.span);
1020            if let Some(position) = name_position(name) {
1021                self.spellings.insert(position, parts);
1022            }
1023            *name = object_name(&full, span);
1024            *version = None;
1025        }
1026        ControlFlow::Continue(())
1027    }
1028}

1045/// `SELECT * FROM <parts>`, with the name at `span`.
1046fn select_star_from(parts: &[String], span: Span) -> SetExpr {
1047    let mut stmts = Parser::parse_sql(&GenericDialect, "SELECT * FROM t").unwrap();
1048    let Statement::Query(query) = stmts.remove(0) else {

1050    };

1060/// The schema of a catalog table from its column definitions, and its `NOT NULL` columns.
1061pub(crate) fn create_table_columns(columns: &[ColumnDef]) -> PolarsResult<(Schema, Vec<String>)> {
1062    let mut schema = Schema::with_capacity(columns.len());
1063    let mut required = Vec::new();
1064    for col in columns {
1065        let name = col.name.value.as_str();
1066        let mut not_null = None;
1067        for option in &col.options {
1068            let is_not_null = match &option.option {
1069                ColumnOption::NotNull => true,
1070                ColumnOption::Null => false,
1071                // the default of a nullable column anyway
1072                ColumnOption::Default(expr) if is_null(expr) => continue,
1073                _ => polars_bail!(
1074                    SQLInterface: "only NOT NULL constraints are supported for catalog tables; found `{}` on column '{}'",
1075                    option, name
1076                ),
1077            };
1078            polars_ensure!(
1079                not_null.is_none_or(|n| n == is_not_null),
1080                SQLInterface: "conflicting NULL/NOT NULL declarations for column '{}'", name
1081            );
1082            not_null = Some(is_not_null);
1083        }
1084        let dtype = map_sql_dtype_to_polars(&col.data_type)?;
1085        polars_ensure!(
1086            schema.insert(name.into(), dtype).is_none(),
1087            Duplicate: "column '{}' is defined more than once", name
1088        );
1089        if not_null == Some(true) {
1090            required.push(name.to_string());
1091        }
1092    }
1093    Ok((schema, required))
1094}

1211/// The partition fields of a catalog table, from `PARTITION BY (...)` or `PARTITIONED BY (...)`.
1212pub(crate) fn create_table_partitioning(
1213    partition_by: Option<&Expr>,
1214    hive_distribution: &HiveDistributionStyle,
1215) -> PolarsResult<Vec<CatalogPartitionField>> {
1216    // Each field with its SQL, for messages.
1217    let fields: Vec<(CatalogPartitionField, String)> = match (partition_by, hive_distribution) {
1218        (Some(_), HiveDistributionStyle::PARTITIONED { .. }) => {
1219            polars_bail!(SQLInterface: "`PARTITION BY` and `PARTITIONED BY` cannot be combined")
1220        },
1221        (Some(expr), _) => {
1222            let exprs = match strip_nested(expr) {
1223                Expr::Tuple(exprs) => exprs.as_slice(),
1224                expr => std::slice::from_ref(expr),
1225            };
1226            partition_fields(exprs)?
1227        },
1228        (None, HiveDistributionStyle::PARTITIONED { columns }) => columns
1229            .iter()
1230            .map(|col| {
1231                // Hive's `PARTITIONED BY (p INT)` adds a column.
1232                polars_ensure!(
1233                    col.data_type == SQLDataType::Unspecified && col.options.is_empty(),
1234                    SQLInterface: "`PARTITIONED BY` takes column names only; found `{}`; types and constraints belong in the column list",
1235                    col
1236                );
1237                let field = CatalogPartitionField {
1238                    path: vec![col.name.value.clone()],
1239                    transform: "identity".to_string(),
1240                };
1241                Ok((field, col.name.value.clone()))
1242            })
1243            .collect::<PolarsResult<_>>()?,
1244        (None, _) => Vec::new(),
1245    };
1246    ensure_unique_fields(&fields)?;
1247    // A new spec takes one time transform per column, as Iceberg's spec builders do.
1248    let is_time = |f: &CatalogPartitionField| {
1249        matches!(f.transform.as_str(), "year" | "month" | "day" | "hour")
1250    };
1251    for (i, (field, sql)) in fields.iter().enumerate() {
1252        for (earlier, earlier_sql) in &fields[..i] {
1253            polars_ensure!(
1254                earlier.path != field.path || !is_time(earlier) || !is_time(field),
1255                SQLInterface: "partition fields `{}` and `{}` both partition column '{}' by time; use one of them",
1256                earlier_sql, sql, field.path.join(".")
1257            );
1258        }
1259    }
1260    Ok(fields.into_iter().map(|(field, _)| field).collect())
1261}

1316/// The path of a layout source: a column or a struct field (`s.f` or `(s).f`). Checked against
1317/// the schema later: a struct field, never a table-qualified column.
1318fn layout_source(expr: &Expr) -> Option<Vec<String>> {
1319    match strip_nested(expr) {
1320        Expr::Identifier(ident) => Some(vec![ident.value.clone()]),
1321        Expr::CompoundIdentifier(idents) => Some(idents.iter().map(|i| i.value.clone()).collect()),
1322        Expr::CompoundFieldAccess { root, access_chain } => {
1323            let mut path = layout_source(root)?;
1324            for access in access_chain {
1325                let AccessExpr::Dot(Expr::Identifier(field)) = access else {
1326                    return None;
1327                };
1328                path.push(field.value.clone());
1329            }
1330            Some(path)
1331        },
1332        _ => None,
1333    }
1334}

1336/// The source path and transform of a partition or sort field (`kind`): a source,
1337/// `year|month|day|hour(source)` (with Iceberg's aliases), or `bucket|truncate(N, source)`. A
1338/// source is a column or a struct field (`s.f`).
1339fn layout_field(expr: &Expr, kind: &str) -> PolarsResult<(Vec<String>, String)> {
1340    let unsupported = || {
1341        polars_err!(
1342            SQLInterface: "unsupported {} field `{}`; expected a column or struct field (`s.f`), or year, month, day, hour, bucket(N, ...) or truncate(W, ...) of one",
1343            kind, expr
1344        )
1345    };
1346    if let Some(path) = layout_source(expr) {
1347        return Ok((path, "identity".to_string()));
1348    }
1349    let Expr::Function(Function {
1350        name,
1351        uses_odbc_syntax: false,
1352        parameters: FunctionArguments::None,
1353        args: FunctionArguments::List(list),
1354        filter: None,
1355        null_treatment: None,
1356        over: None,
1357        within_group,
1358    }) = strip_nested(expr)
1359    else {
1360        return Err(unsupported());
1361    };
1362    let fname = match name_parts(name).as_deref() {
1363        Some([fname])
1364            if within_group.is_empty()
1365                && list.duplicate_treatment.is_none()
1366                && list.clauses.is_empty() =>
1367        {
1368            fname.to_ascii_lowercase()
1369        },
1370        _ => return Err(unsupported()),
1371    };
1372    let args = list
1373        .args
1374        .iter()
1375        .map(|arg| match arg {
1376            FunctionArg::Unnamed(FunctionArgExpr::Expr(expr)) => Some(expr),
1377            _ => None,
1378        })
1379        .collect::<Option<Vec<_>>>()
1380        .ok_or_else(unsupported)?;
1381    let positive_int = |expr: &Expr| match strip_nested(expr) {
1382        Expr::Value(ValueWithSpan {
1383            value: Value::Number(n, false),
1384            ..
1385        }) => n.parse::<i32>().ok().filter(|n| *n > 0),
1386        _ => None,
1387    };
1388    let (transform, source) = match (fname.as_str(), args.as_slice()) {
1389        ("year" | "years", [source]) => ("year".to_string(), source),
1390        ("month" | "months", [source]) => ("month".to_string(), source),
1391        ("day" | "days", [source]) => ("day".to_string(), source),
1392        ("hour" | "hours", [source]) => ("hour".to_string(), source),
1393        ("bucket" | "truncate", [n, source]) => {
1394            let Some(n) = positive_int(n) else {
1395                if layout_source(n).is_some() && positive_int(source).is_some() {
1396                    polars_bail!(SQLInterface: "`{}` takes the number first, as in `{}({}, {})`", expr, fname, source, n)
1397                }
1398                polars_bail!(SQLInterface: "the first argument of `{}` must be an integer from 1 to {}", expr, i32::MAX)
1399            };
1400            (format!("{fname}[{n}]"), source)
1401        },
1402        _ => return Err(unsupported()),
1403    };
1404    let path = layout_source(source).ok_or_else(unsupported)?;
1405    Ok((path, transform))
1406}

1408/// The properties of a catalog namespace, from `WITH (...)`.
1409pub(crate) fn create_schema_options(options: &[SqlOption]) -> PolarsResult<Vec<(String, String)>> {
1410    let mut properties: Vec<(String, String)> = Vec::new();
1411    for option in options {
1412        let (key, value) = option_key_value(option, "schema", &["location"])?;
1413        polars_ensure!(
1414            key != "location" || is_absolute_location(&value),
1415            SQLInterface: "location must be an absolute path or a URI; found '{}'", value
1416        );
1417        polars_ensure!(
1418            properties.iter().all(|(k, _)| k != key),
1419            SQLInterface: "duplicate schema option '{}'", key
1420        );
1421        properties.push((key.to_string(), value));
1422    }
1423    Ok(properties)
1424}

1426/// The key and value of a `'key' = value` option of a `kind` ("table", "schema"), with keys that
1427/// differ from a `known` one only in case rejected.
1428fn option_key_value<'a>(
1429    option: &'a SqlOption,
1430    kind: &str,
1431    known: &[&str],
1432) -> PolarsResult<(&'a str, String)> {
1433    let SqlOption::KeyValue { key, value } = option else {
1434        polars_bail!(SQLInterface: "{} options must be `'key' = value` pairs; found `{}`", kind, option)
1435    };
1436    let key = key.value.as_str();
1437    polars_ensure!(!key.is_empty(), SQLInterface: "{} option names must not be empty", kind);
1438    for known in known {
1439        polars_ensure!(
1440            key == *known || !key.eq_ignore_ascii_case(known),
1441            SQLInterface: "'{}' is not a {} option; did you mean '{}'?", key, kind, known
1442        );
1443    }
1444    let Some(value) = option_value(value) else {
1445        polars_bail!(
1446            SQLInterface: "{} option values must be string, number or boolean literals; found `{}` for '{}'",
1447            kind, value, key
1448        )
1449    };
1450    Ok((key, value))
1451}

1453/// A table option value as a property string, if it is a string, number or boolean literal.
1454fn option_value(value: &Expr) -> Option<String> {
1455    match strip_nested(value) {
1456        Expr::Value(v) => match &v.value {
1457            Value::SingleQuotedString(s) => Some(s.clone()),
1458            Value::Number(n, _) => Some(n.clone()),
1459            Value::Boolean(b) => Some(b.to_string()),
1460            _ => None,
1461        },
1462        Expr::UnaryOp {
1463            op: UnaryOperator::Minus,
1464            expr,
1465        } => match strip_nested(expr) {
1466            Expr::Value(ValueWithSpan {
1467                value: Value::Number(n, _),
1468                ..
1469            }) => Some(format!("-{n}")),
1470            _ => None,
1471        },
1472        _ => None,
1473    }
1474}

1476/// Whether a location is an absolute path (POSIX or Windows) or has a URI scheme.
1477fn is_absolute_location(location: &str) -> bool {
1478    let bytes = location.as_bytes();
1479    let windows_path = bytes.len() > 2
1480        && bytes[0].is_ascii_alphabetic()
1481        && bytes[1] == b':'
1482        && matches!(bytes[2], b'/' | b'\\');
1483    let has_scheme = location.split_once("://").is_some_and(|(scheme, _)| {
1484        scheme.starts_with(|c: char| c.is_ascii_alphabetic())
1485            && scheme
1486                .chars()
1487                .all(|c| c.is_ascii_alphanumeric() || matches!(c, '+' | '-' | '.'))
1488    });
1489    location.starts_with('/') || windows_path || has_scheme
1490}

1508/// The first part of a `DELETE` condition that a catalog filter can't express, described for a
1509/// message. Only a hint: converting the planned predicate decides.
1510pub(crate) fn hint_unsupported_part(expr: &Expr) -> Option<String> {
1511    use BinaryOperator as Op;
1512    if !expr_references_any_column(expr) {
1513        return None;
1514    }
1515    match expr {
1516        Expr::Nested(e)
1517        | Expr::UnaryOp {
1518            op: UnaryOperator::Not,
1519            expr: e,
1520        } => hint_unsupported_part(e),
1521        Expr::BinaryOp {
1522            left,
1523            op: Op::And | Op::Or,
1524            right,
1525        } => hint_unsupported_part(left).or_else(|| hint_unsupported_part(right)),
1526        Expr::BinaryOp {
1527            left,
1528            op: Op::Eq | Op::NotEq | Op::Lt | Op::LtEq | Op::Gt | Op::GtEq,
1529            right,
1530        } => match expr_references_any_column(left) && expr_references_any_column(right) {
1531            true => Some(format!("comparison of two columns `{expr}`")),
1532            false => unsupported_operand(left)
1533                .or_else(|| unsupported_operand(right))
1534                .or_else(|| {
1535                    unsupported_string(left, false).or_else(|| unsupported_string(right, false))
1536                }),
1537        },
1538        Expr::IsDistinctFrom(left, right) | Expr::IsNotDistinctFrom(left, right) => {
1539            match expr_references_any_column(left) && expr_references_any_column(right) {
1540                true => Some(format!("comparison of two columns `{expr}`")),
1541                false => unsupported_operand(left)
1542                    .or_else(|| unsupported_operand(right))
1543                    .or_else(|| {
1544                        unsupported_string(left, false).or_else(|| unsupported_string(right, false))
1545                    }),
1546            }
1547        },
1548        Expr::IsNull(e)
1549        | Expr::IsNotNull(e)
1550        | Expr::IsTrue(e)
1551        | Expr::IsNotTrue(e)
1552        | Expr::IsFalse(e)
1553        | Expr::IsNotFalse(e) => unsupported_operand(e),
1554        Expr::InList { expr: e, list, .. } => unsupported_operand(e).or_else(|| {
1555            match list.iter().any(|item| {
1556                matches!(
1557                    item,
1558                    Expr::Value(ValueWithSpan {
1559                        value: Value::Null,
1560                        ..
1561                    })
1562                )
1563            }) {
1564                true => Some("NULL in the IN list".to_string()),
1565                false => list
1566                    .iter()
1567                    .find_map(|item| column_in_bound(item, expr))
1568                    .or_else(|| list.iter().find_map(|item| unsupported_string(item, false))),
1569            }
1570        }),
1571        Expr::Between {
1572            expr: e, low, high, ..
1573        } => unsupported_operand(e)
1574            .or_else(|| column_in_bound(low, expr))
1575            .or_else(|| column_in_bound(high, expr))
1576            .or_else(|| unsupported_string(low, false))
1577            .or_else(|| unsupported_string(high, false)),
1578        Expr::Like {
1579            any,
1580            expr: e,
1581            pattern,
1582            escape_char,
1583            ..
1584        } => unsupported_operand(e).or_else(|| {
1585            let prefix = !any
1586                && escape_char.is_none()
1587                && matches!(pattern.as_ref(), Expr::Value(ValueWithSpan {
1588                    value: Value::SingleQuotedString(p), ..
1589                }) if is_prefix_pattern(p));
1590            match prefix {
1591                true => unsupported_string(pattern, true),
1592                false => Some(format!("LIKE pattern {pattern} (only 'prefix%')")),
1593            }
1594        }),
1595        Expr::ILike { .. } => Some("ILIKE".to_string()),
1596        Expr::Identifier(_) | Expr::CompoundIdentifier(_) => None,
1597        _ => Some(describe_operand(expr)),
1598    }
1599}

1603        Expr::Identifier(_) | Expr::CompoundIdentifier(_) => None,
1604        Expr::Nested(e) => unsupported_operand(e),
1605        _ if !expr_references_any_column(expr) => None,
1606        _ => Some(describe_operand(expr)),

1610/// With `like`, a trailing '%' is the pattern's wildcard, not part of the constant.
1611fn unsupported_string(expr: &Expr, like: bool) -> Option<String> {
1612    let Expr::Value(ValueWithSpan {
1613        value: Value::SingleQuotedString(s),
1614        ..
1615    }) = expr
1616    else {
1617        return None;
1618    };
1619    let text = match like {
1620        true => s.strip_suffix('%').unwrap_or(s),
1621        false => s,
1622    };
1623    // as `sanitize` in polars-plan's pyarrow predicates
1624    let supported = |c: char| c.is_alphanumeric() || matches!(c, ' ' | '-' | '_');
1625    (!text.chars().all(supported)).then(|| {
1626        format!("string constant {expr} (only letters, digits, space, '-' and '_' for now)")
1627    })
1628}
