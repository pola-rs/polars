66/// Visits the bodies and parameter defaults of a created macro.
67impl VisitMut for MacroStatement {
68    fn visit<V: VisitorMut>(&mut self, visitor: &mut V) -> ControlFlow<V::Break> {
69        let MacroStatement::Create { overloads, .. } = self else {
70            return ControlFlow::Continue(());
71        };
72        for overload in overloads {
73            for default in overload
74                .params
75                .iter_mut()
76                .filter_map(|p| p.default.as_mut())
77            {
78                default.visit(visitor)?;
79            }
80            match &mut overload.body {
81                MacroBody::Expr(expr) => expr.visit(visitor)?,
82                MacroBody::Table(query) => query.visit(visitor)?,
83            }
84        }
85        ControlFlow::Continue(())
86    }
87}

89/// The macros of a context, by lowercase name.
90#[derive(Clone, Default)]
91pub(crate) struct Macros {
92    scalar: Arc<RwLock<PlHashMap<String, Arc<[MacroOverload]>>>>,
93    table: Arc<RwLock<PlHashMap<String, Arc<[MacroOverload]>>>>,
94}

96impl Macros {
97    fn map(&self, table: bool) -> &RwLock<PlHashMap<String, Arc<[MacroOverload]>>> {
98        if table { &self.table } else { &self.scalar }
99    }
100
101    pub(crate) fn create(
102        &self,
103        name: &Ident,
104        table: bool,
105        or_replace: bool,
106        overloads: Vec<MacroOverload>,
107    ) -> PolarsResult<()> {
108        let mut map = self.map(table).write().unwrap();
109        let key = name.value.to_lowercase();
110        polars_ensure!(
111            or_replace || !map.contains_key(&key),
112            SQLInterface: "macro '{}' already exists", name.value
113        );
114        map.insert(key, overloads.into());
115        Ok(())
116    }
117
118    pub(crate) fn drop(&self, name: &Ident, table: bool, if_exists: bool) -> PolarsResult<()> {
119        let removed = self
120            .map(table)
121            .write()
122            .unwrap()
123            .remove(&name.value.to_lowercase());
124        polars_ensure!(
125            removed.is_some() || if_exists,
126            SQLInterface: "{}macro '{}' does not exist", if table { "table " } else { "" }, name.value
127        );
128        Ok(())
129    }
130
131    fn get(&self, name: &ObjectName, table: bool) -> Option<(String, Arc<[MacroOverload]>)> {
132        let [part] = name.0.as_slice() else {
133            return None;
134        };
135        let name = &part.as_ident()?.value;
136        let map = self.map(table).read().unwrap();
137        if map.is_empty() {
138            return None;
139        }
140        let overloads = map.get(&name.to_lowercase())?.clone();
141        Some((name.clone(), overloads))
142    }
143
144    pub(crate) fn is_empty(&self) -> bool {
145        self.scalar.read().unwrap().is_empty() && self.table.read().unwrap().is_empty()
146    }

148    /// Replace the macro calls in `node` by the macro bodies. Returns whether there were any.
149    pub(crate) fn expand<T: Expandable>(&self, node: &mut T) -> PolarsResult<bool> {
150        if self.is_empty() {
151            return Ok(false);
152        }
153        let mut expander = MacroExpander {
154            macros: self,
155            expanding: vec![],
156            expanded: false,
157            size: ExpandedSize::default(),
158            lambdas: LambdaScopes::default(),
159        };
160        if let ControlFlow::Break(err) = dyn_visit::visit_mut(node, &mut expander) {
161            return Err(err);
162        }
163        if expander.expanded {
164            // The expanded SQL may not be nested deeper than the parser accepts.
165            let sql = node.to_string();
166            let too_deep = tokenize(&sql).is_ok_and(|(tokens, _)| {
167                let reparse = |parser: &mut Parser| {
168                    T::reparse(parser)?;
169                    parser.expect_token(&Token::EOF)
170                };
171                reparse(&mut parser_of(tokens.clone()))
172                    .is_err_and(|err| hit_recursion_limit(&err, tokens, reparse))
173            });
174            polars_ensure!(
175                !too_deep,
176                SQLInterface: "the query is nested too deeply after expanding its macros"
177            );
178        }
179        Ok(expander.expanded)
180    }
181
182    /// `node` with its macro calls replaced by the macro bodies, or `None` if it has none.
183    pub(crate) fn expanded<T: Expandable + Clone>(&self, node: &T) -> PolarsResult<Option<T>> {
184        if self.is_empty() {
185            return Ok(None);
186        }
187        let mut node = node.clone();
188        Ok(self.expand(&mut node)?.then_some(node))
189    }
190}

221/// The macro statement at the start of `parser`; if there is none, `parser` stays at the start.
222pub(crate) fn parse_macro_statement(parser: &mut Parser) -> PolarsResult<Option<MacroStatement>> {
223    let head = parser
224        .maybe_parse(|p| {
225            while p.consume_token(&Token::SemiColon) {}
226            match p.parse_one_of_keywords(&[Keyword::CREATE, Keyword::DROP]) {
227                Some(Keyword::CREATE) => {
228                    let or_replace = p.parse_keywords(&[Keyword::OR, Keyword::REPLACE]);
229                    let _ = p.parse_one_of_keywords(&[Keyword::TEMP, Keyword::TEMPORARY]);
230                    p.expect_one_of_keywords(&[Keyword::MACRO, Keyword::FUNCTION])?;
231                    Ok((true, or_replace, false))
232                },
233                Some(_) => {
234                    let table =
235                        match p.expect_one_of_keywords(&[Keyword::MACRO, Keyword::FUNCTION])? {
236                            Keyword::MACRO => p.parse_keyword(Keyword::TABLE),
237                            _ => false,
238                        };
239                    Ok((false, false, table))
240                },
241                None => Err(ParserError::ParserError(String::new())),
242            }
243        })
244        .map_err(to_sql_interface_err)?;
245    let Some((create, or_replace, table)) = head else {
246        return Ok(None);
247    };
248    let stmt = if create {
249        parse_create_macro(parser, or_replace)
250    } else {
251        parse_drop_macro(parser, table)
252    }
253    .map_err(to_sql_interface_err)?;
254    while parser.consume_token(&Token::SemiColon) {}
255    if parser.peek_token().token != Token::EOF {
256        polars_bail!(SQLInterface: "one (and only one) statement can be parsed at a time");
257    }
258    Ok(Some(stmt))
259}

261fn parse_create_macro(
262    parser: &mut Parser,
263    or_replace: bool,
264) -> Result<MacroStatement, ParserError> {
265    let name = single_name(parser.parse_object_name(false)?)?;
266    let mut overloads = vec![];
267    let mut table = false;
268    loop {
269        let params = parse_macro_params(parser)?;
270        parser.expect_keyword_is(Keyword::AS)?;
271        let body = if parser.parse_keyword(Keyword::TABLE) {
272            MacroBody::Table(parse_table_body(parser)?)
273        } else {
274            MacroBody::Expr(Box::new(parser.parse_expr()?))
275        };
276        let is_table = matches!(body, MacroBody::Table(_));
277        if !overloads.is_empty() && is_table != table {
278            return Err(ParserError::ParserError(
279                "the definitions of a macro must all be scalar or all be TABLE".into(),
280            ));
281        }
282        table = is_table;
283        overloads.push(MacroOverload { params, body });
284        if !parser.consume_token(&Token::Comma) {
285            break;
286        }
287    }
288    Ok(MacroStatement::Create {
289        or_replace,
290        name,
291        table,
292        overloads,
293    })
294}

296/// The parameters of a definition, in parentheses.
297fn parse_macro_params(parser: &mut Parser) -> Result<Vec<MacroParam>, ParserError> {
298    parser.expect_token(&Token::LParen)?;
299    let mut params = vec![];
300    if parser.consume_token(&Token::RParen) {
301        return Ok(params);
302    }
303    loop {
304        let name = parser.parse_identifier()?;
305        let default =
306            if parser.consume_token(&Token::Assignment) || parser.consume_token(&Token::RArrow) {
307                Some(parser.parse_expr()?)
308            } else {
309                None
310            };
311        params.push(MacroParam { name, default });
312        if !parser.consume_token(&Token::Comma) {
313            break;
314        }
315    }
316    parser.expect_token(&Token::RParen)?;
317    Ok(params)
318}

320/// The query of a TABLE definition, which ends before the `, (...) AS TABLE` of the next
321/// definition.
322fn parse_table_body(parser: &mut Parser) -> Result<Box<Query>, ParserError> {
323    // The parser is past `AS TABLE`, so the next token is at the current index + 1.
324    let tokens: Vec<TokenWithSpan> = (parser.get_current_index() + 1..)
325        .map(|i| parser.token_at(i))
326        .take_while(|t| !matches!(t.token, Token::EOF | Token::SemiColon))
327        .cloned()
328        .collect();
329    let (query, len) = split_table_body(&tokens)?;
330    let n_parsed = tokens[..len]
331        .iter()
332        .filter(|t| !matches!(t.token, Token::Whitespace(_)))
333        .count();
334    for _ in 0..n_parsed {
335        parser.next_token();
336    }
337    Ok(query)
338}

340/// The query at the start of `tokens` and its number of tokens. It ends at a comma that
341/// starts another TABLE definition. The longest query that parses is taken, so a definition
342/// that is also a valid continuation of the query must be put in parentheses.
343fn split_table_body(tokens: &[TokenWithSpan]) -> Result<(Box<Query>, usize), ParserError> {
344    let words: Vec<usize> = (0..tokens.len())
345        .filter(|&i| !matches!(tokens[i].token, Token::Whitespace(_)))
346        .collect();
347    let mut boundaries = vec![];
348    let mut depth = 0usize;
349    for (k, &i) in words.iter().enumerate() {
350        match tokens[i].token {
351            Token::LParen | Token::LBracket | Token::LBrace => depth += 1,
352            Token::RParen | Token::RBracket | Token::RBrace => depth = depth.saturating_sub(1),
353            Token::Comma if depth == 0 && starts_table_definition(tokens, &words[k + 1..]) => {
354                boundaries.push(i)
355            },
356            _ => {},
357        }
358    }
359    let whole = parse_whole_query(tokens);
360    if whole.is_err() {
361        for &i in boundaries.iter().rev() {
362            if let Ok(query) = parse_whole_query(&tokens[..i]) {
363                return Ok((query, i));
364            }
365        }
366    }
367    Ok((whole?, tokens.len()))
368}

370/// Whether the tokens at `words` start with `(params) AS TABLE`.
371fn starts_table_definition(tokens: &[TokenWithSpan], words: &[usize]) -> bool {
372    let token = |k: usize| words.get(k).map(|&i| &tokens[i].token);
373    let is_keyword =
374        |k: usize, keyword| matches!(token(k), Some(Token::Word(w)) if w.keyword == keyword);
375    if token(0) != Some(&Token::LParen) {
376        return false;
377    }
378    let mut depth = 0usize;
379    let Some(close) = (0..words.len()).find(|&k| {
380        match token(k) {
381            Some(Token::LParen) => depth += 1,
382            Some(Token::RParen) => depth -= 1,
383            _ => {},
384        }
385        depth == 0
386    }) else {
387        return false;
388    };
389    if !is_keyword(close + 1, Keyword::AS) || !is_keyword(close + 2, Keyword::TABLE) {
390        return false;
391    }
392    let mut parser = parser_of(tokens[words[0]..=words[close + 2]].to_vec());
393    parse_macro_params(&mut parser).is_ok()
394        && parser.parse_keywords(&[Keyword::AS, Keyword::TABLE])
395        && parser.peek_token().token == Token::EOF
396}

398fn parse_whole_query(tokens: &[TokenWithSpan]) -> Result<Box<Query>, ParserError> {
399    let mut parser = parser_of(tokens.to_vec());
400    let query = parser.parse_query()?;
401    let rest = parser.peek_token();
402    if rest.token != Token::EOF {
403        return parser.expected("the end of the macro body", rest);
404    }
405    Ok(query)
406}

408fn parse_drop_macro(parser: &mut Parser, table: bool) -> Result<MacroStatement, ParserError> {
409    let if_exists = parser.parse_keywords(&[Keyword::IF, Keyword::EXISTS]);
410    let name = single_name(parser.parse_object_name(false)?)?;
411    Ok(MacroStatement::Drop {
412        table,
413        if_exists,
414        name,
415    })
416}

418fn single_name(name: ObjectName) -> Result<Ident, ParserError> {
419    match name.0.as_slice() {
420        [part] => part
421            .as_ident()
422            .cloned()
423            .ok_or_else(|| ParserError::ParserError(format!("invalid macro name {name}"))),
424        _ => Err(ParserError::ParserError(format!(
425            "a macro name cannot be qualified; found {name}"
426        ))),
427    }
428}

430/// Check the definitions of macro `name`.
431pub(crate) fn check_macro(name: &Ident, overloads: &[MacroOverload]) -> PolarsResult<()> {
432    for overload in overloads {
433        let params = &overload.params;
434        let param = match &overload.body {
435            MacroBody::Expr(body) => first_param(body.as_ref()),
436            MacroBody::Table(query) => first_param(query.as_ref()),
437        };
438        if let Some(p) = param.or_else(|| params.iter().find_map(|p| first_param(&p.default))) {
439            polars_bail!(
440                SQLInterface: "query parameters are not supported in CREATE MACRO; found {}", p
441            );
442        }
443        for (i, param) in params.iter().enumerate() {
444            polars_ensure!(
445                !params[..i].iter().any(|p| same_name(&p.name, &param.name.value)),
446                SQLInterface: "macro '{}' has a duplicate parameter '{}'", name.value, param.name.value
447            );
448            polars_ensure!(
449                param.default.as_ref().is_none_or(|d| !expr_references_any_column(d)),
450                SQLInterface: "the default of parameter '{}' of macro '{}' cannot read columns",
451                param.name.value, name.value
452            );
453            polars_ensure!(
454                param.default.is_some() || params[..i].iter().all(|p| p.default.is_none()),
455                SQLInterface: "in macro '{}', parameter '{}' without a default follows a parameter with a default",
456                name.value, param.name.value
457            );
458        }
459        if let MacroBody::Expr(body) = &overload.body {
460            let mut check = FreeIdentifiers {
461                params,
462                query_depth: 0,
463            };
464            if let ControlFlow::Break(ident) = dyn_visit::visit(body, &mut check) {
465                polars_bail!(
466                    SQLInterface: "the body of macro '{}' can only use its parameters; found '{}'",
467                    name.value, ident
468                );
469            }
470        }
471    }
472    Ok(())
473}

475/// Whether a call of `name` parses as a call of a function (or table function), rather than
476/// as special syntax like `TRIM(...)` or `UNNEST(...)`, which a macro cannot replace.
477pub(crate) fn is_callable_name(name: &Ident, table: bool) -> bool {
478    let sql = if table {
479        format!("SELECT * FROM {name}(1)")
480    } else {
481        format!("SELECT {name}(1)")
482    };
483    let Ok(statements) = Parser::parse_sql(&GenericDialect, &sql) else {
484        return false;
485    };
486    let [Statement::Query(query)] = statements.as_slice() else {
487        return false;
488    };
489    let SetExpr::Select(select) = query.body.as_ref() else {
490        return false;
491    };
492    if table {
493        matches!(
494            select.from.as_slice(),
495            [TableWithJoins {
496                relation: TableFactor::Table { args: Some(_), .. },
497                ..
498            }]
499        )
500    } else {
501        matches!(
502            select.projection.as_slice(),
503            [SelectItem::UnnamedExpr(SQLExpr::Function(_))]
504        )
505    }
506}

512/// Finds an identifier outside subqueries that is not a parameter.
513struct FreeIdentifiers<'a> {
514    params: &'a [MacroParam],
515    query_depth: usize,
516}
517
518impl Visitor for FreeIdentifiers<'_> {
519    type Break = String;
520
521    fn pre_visit_query(&mut self, _query: &Query) -> ControlFlow<String> {
522        self.query_depth += 1;
523        ControlFlow::Continue(())
524    }
525
526    fn post_visit_query(&mut self, _query: &Query) -> ControlFlow<String> {
527        self.query_depth -= 1;
528        ControlFlow::Continue(())
529    }
530
531    fn pre_visit_expr(&mut self, expr: &SQLExpr) -> ControlFlow<String> {
532        if self.query_depth > 0 {
533            return ControlFlow::Continue(());
534        }
535        let first = match expr {
536            SQLExpr::Identifier(ident) => ident,
537            SQLExpr::CompoundIdentifier(idents) => &idents[0],
538            _ => return ControlFlow::Continue(()),
539        };
540        if self.params.iter().any(|p| same_name(&p.name, &first.value)) {
541            ControlFlow::Continue(())
542        } else {
543            ControlFlow::Break(expr.to_string())
544        }
545    }
546}

548/// The deepest nesting of macro calls in macro bodies.
549const MAX_MACRO_DEPTH: usize = 64;
550
551/// The most nodes that expanding the macros of a statement may copy.
552const MAX_EXPANDED_EXPRS: usize = 100_000;
553
554/// The number of nodes copied by an expansion.
555#[derive(Default)]
556struct ExpandedSize(Cell<usize>);
557
558impl ExpandedSize {
559    /// Counts the expressions, queries and relations of `node`, which is about to be copied.
560    fn add<T: Visit>(&self, node: &T) -> ControlFlow<PolarsError> {
561        struct Count(usize);
562        impl Visitor for Count {
563            type Break = ();
564            fn pre_visit_expr(&mut self, _expr: &SQLExpr) -> ControlFlow<()> {
565                self.0 += 1;
566                ControlFlow::Continue(())
567            }
568
569            fn pre_visit_query(&mut self, _query: &Query) -> ControlFlow<()> {
570                self.0 += 1;
571                ControlFlow::Continue(())
572            }
573
574            fn pre_visit_select(&mut self, _select: &Select) -> ControlFlow<()> {
575                self.0 += 1;
576                ControlFlow::Continue(())
577            }
578
579            fn pre_visit_table_factor(&mut self, _factor: &TableFactor) -> ControlFlow<()> {
580                self.0 += 1;
581                ControlFlow::Continue(())
582            }
583        }
584        let mut count = Count(0);
585        let _ = dyn_visit::visit(node, &mut count);
586        let size = self.0.get() + count.0;
587        if size > MAX_EXPANDED_EXPRS {
588            return ControlFlow::Break(polars_err!(
589                SQLInterface: "the query is too large after expanding its macros"
590            ));
591        }
592        self.0.set(size);
593        ControlFlow::Continue(())
594    }
595}

597struct MacroExpander<'a> {
598    macros: &'a Macros,
599    /// The macros whose bodies are being expanded, as (lowercase name, whether a table macro).
600    expanding: Vec<(String, bool)>,
601    expanded: bool,
602    size: ExpandedSize,
603    lambdas: LambdaScopes,
604}

606/// The lambdas around the visited expression.
607#[derive(Default)]
608struct LambdaScopes {
609    depth: usize,
610    /// The bodies of the lambda arguments of the functions being visited.
611    bodies: Vec<*const SQLExpr>,
612}
613
614impl LambdaScopes {
615    /// Called before the children of `expr` are visited.
616    fn enter(&mut self, expr: &SQLExpr) {
617        if self.bodies.contains(&(expr as *const SQLExpr)) {
618            self.depth += 1;
619        }
620        if let SQLExpr::Function(func) = expr {
621            self.bodies
622                .extend(lambda_args(func).map(|body| body as *const SQLExpr));
623        }
624    }
625
626    /// Called after the children of `expr` are visited. Returns whether `expr` is a lambda
627    /// body, which `leave_body` leaves once `expr` is visited.
628    fn leave_children(&mut self, expr: &SQLExpr) -> bool {
629        if let SQLExpr::Function(func) = expr {
630            let n_lambdas = lambda_args(func).count();
631            self.bodies.truncate(self.bodies.len() - n_lambdas);
632        }
633        self.bodies.contains(&(expr as *const SQLExpr))
634    }
635
636    fn leave_body(&mut self, is_body: bool) {
637        if is_body {
638            self.depth -= 1;
639        }
640    }
641}

643impl MacroExpander<'_> {
644    /// An error if macro `name` is being expanded: a macro that calls itself never ends.
645    fn check_not_expanding(&self, name: &str, table: bool) -> ControlFlow<PolarsError> {
646        let key = (name.to_lowercase(), table);
647        if self.expanding.contains(&key) {
648            return ControlFlow::Break(polars_err!(
649                SQLInterface: "macro '{}' calls itself", name
650            ));
651        }
652        ControlFlow::Continue(())
653    }
654
655    /// Expand the macro calls in the defaults of a call of macro `name`. The other arguments
656    /// were expanded where the call is.
657    fn expand_args(
658        &mut self,
659        bound: Vec<BoundArg>,
660        name: &str,
661        table: bool,
662    ) -> ControlFlow<PolarsError, Vec<(Ident, SQLExpr)>> {
663        let mut params = Vec::with_capacity(bound.len());
664        for mut arg in bound {
665            if arg.is_default {
666                // The lambdas of a default count from the call.
667                shift_lambda_elements(&mut arg.value, 0, self.lambdas.depth);
668                self.expand_nested(&mut arg.value, name, table)?;
669                // As written in the definition, a default cannot read columns.
670                if expr_references_any_column(&arg.value) {
671                    return ControlFlow::Break(polars_err!(
672                        SQLInterface: "the default of parameter '{}' of macro '{}' cannot read columns",
673                        arg.name.value, name
674                    ));
675                }
676            }
677            params.push((arg.name, arg.value));
678        }
679        ControlFlow::Continue(params)
680    }
681
682    /// Expand the macro calls in the body of macro `name`, which replaced a call.
683    fn expand_nested<T: VisitMut>(
684        &mut self,
685        node: &mut T,
686        name: &str,
687        table: bool,
688    ) -> ControlFlow<PolarsError> {
689        if self.expanding.len() >= MAX_MACRO_DEPTH {
690            return ControlFlow::Break(polars_err!(
691                SQLInterface: "macro '{}' is nested more than {} macros deep", name, MAX_MACRO_DEPTH
692            ));
693        }
694        self.expanding.push((name.to_lowercase(), table));
695        let res = dyn_visit::visit_mut(node, self);
696        self.expanding.pop();
697        res
698    }
699}

723impl MacroExpander<'_> {
724    /// Replaces `factor` by the body of the table macro it calls, if it calls one.
725    fn expand_table_call(&mut self, factor: &mut TableFactor) -> ControlFlow<PolarsError> {
726        let (macro_name, lateral) = match &*factor {
727            TableFactor::Table {
728                name,
729                args: Some(_),
730                ..
731            } => (name, false),
732            TableFactor::Function { name, lateral, .. } => (name, *lateral),
733            _ => return ControlFlow::Continue(()),
734        };
735        let Some((name, overloads)) = self.macros.get(macro_name, true) else {
736            return ControlFlow::Continue(());
737        };
738        let default_alias = macro_name
739            .0
740            .last()
741            .and_then(|part| part.as_ident())
742            .cloned();
743        self.check_not_expanding(&name, true)?;
744        let (args, alias, sample) = match factor {
745            TableFactor::Table {
746                alias,
747                args: Some(args),
748                sample,
749                version,
750                with_hints,
751                partitions,
752                json_path,
753                index_hints,
754                with_ordinality,
755                ..
756            } if version.is_none()
757                && with_hints.is_empty()
758                && partitions.is_empty()
759                && json_path.is_none()
760                && index_hints.is_empty()
761                && !*with_ordinality =>
762            {
763                (&args.args, alias, sample.take())
764            },
765            TableFactor::Function {
766                args,
767                alias,
768                with_ordinality: false,
769                ..
770            } => (&*args, alias, None),
771            _ => {
772                return ControlFlow::Break(polars_err!(
773                    SQLInterface: "table macro '{}' takes no table modifiers", name
774                ));
775            },
776        };
777        let (overload, bound) = flow(bind_args(&name, &overloads, args))?;
778        let alias = alias.take();
779        let bound = self.expand_args(bound, &name, true)?;
780        let MacroBody::Table(query) = &overload.body else {
781            unreachable!()
782        };
783        self.size.add(&**query)?;
784        let mut subquery = query.clone();
785        dyn_visit::visit_mut(
786            &mut *subquery,
787            &mut ParamReplacer::new(self.macros, &name, &bound, &self.size, self.lambdas.depth),
788        )?;
789        self.expand_nested(&mut subquery, &name, true)?;
790        let alias = alias.unwrap_or_else(|| TableAlias {
791            explicit: false,
792            name: default_alias.unwrap_or_else(|| Ident::new(name)),
793            columns: vec![],
794            at: None,
795        });
796        *factor = TableFactor::Derived {
797            lateral,
798            subquery,
799            alias: Some(alias),
800            sample,
801        };
802        self.expanded = true;
803        ControlFlow::Continue(())
804    }
805
806    /// Replaces `expr` by the body of the macro it calls, if it calls one.
807    fn expand_call(&mut self, expr: &mut SQLExpr) -> ControlFlow<PolarsError> {
808        unparenthesize_operands(expr, self.macros);
809        let SQLExpr::Function(func) = expr else {
810            return ControlFlow::Continue(());
811        };
812        let Some((name, overloads)) = self.macros.get(&func.name, false) else {
813            return ControlFlow::Continue(());
814        };
815        self.check_not_expanding(&name, false)?;
816        let args = flow(macro_call_args(&name, func))?;
817        let (overload, bound) = flow(bind_args(&name, &overloads, args))?;
818        let bound = self.expand_args(bound, &name, false)?;
819        let MacroBody::Expr(body) = &overload.body else {
820            unreachable!()
821        };
822        self.size.add(body.as_ref())?;
823        let mut body = body.as_ref().clone();
824        shift_lambda_elements(&mut body, 0, self.lambdas.depth);
825        let mut replacer =
826            ParamReplacer::new(self.macros, &name, &bound, &self.size, self.lambdas.depth);
827        dyn_visit::visit_mut(&mut body, &mut replacer)?;
828        self.expand_nested(&mut body, &name, false)?;
829        // The value of a call is an expression, not a position or keyword in ORDER BY or
830        // GROUP BY.
831        *expr = match body {
832            SQLExpr::Value(_) => SQLExpr::Nested(Box::new(body)),
833            body => parenthesize(body),
834        };
835        self.expanded = true;
836        ControlFlow::Continue(())
837    }
838}

840/// The bodies of the lambda arguments of `func`.
841fn lambda_args(func: &SQLFunction) -> impl Iterator<Item = &SQLExpr> {
842    let args = match &func.args {
843        FunctionArguments::List(list) => list.args.as_slice(),
844        _ => &[],
845    };
846    args.iter().filter_map(|arg| Some(lambda_arg(arg)?.1))
847}

849/// `res` as the result of a visit.
850fn flow<T>(res: PolarsResult<T>) -> ControlFlow<PolarsError, T> {
851    match res {
852        Ok(value) => ControlFlow::Continue(value),
853        Err(err) => ControlFlow::Break(err),
854    }

857/// The arguments of the scalar macro call `func`.
858fn macro_call_args<'a>(name: &str, func: &'a SQLFunction) -> PolarsResult<&'a [FunctionArg]> {
859    polars_ensure!(
860        func.over.is_none()
861            && func.filter.is_none()
862            && func.within_group.is_empty()
863            && func.null_treatment.is_none()
864            && matches!(func.parameters, FunctionArguments::None),
865        SQLInterface: "macro '{}' does not support OVER, FILTER, WITHIN GROUP or IGNORE/RESPECT NULLS", name
866    );
867    match &func.args {
868        FunctionArguments::None => Ok(&[]),
869        FunctionArguments::List(FunctionArgumentList {
870            args,
871            duplicate_treatment: None,
872            clauses,
873        }) if clauses.is_empty() => Ok(args),
874        _ => polars_bail!(
875            SQLInterface: "macro '{}' takes only arguments, without DISTINCT, ORDER BY or other clauses", name
876        ),
877    }

880/// The definition of macro `name` for `args`, with each parameter and its value.
881fn bind_args<'a>(
882    name: &str,
883    overloads: &'a [MacroOverload],
884    args: &[FunctionArg],
885) -> PolarsResult<(&'a MacroOverload, Vec<BoundArg>)> {
886    let mut positional = vec![];
887    let mut named: Vec<(&Ident, &SQLExpr)> = vec![];
888    for arg in args {
889        match arg {
890            FunctionArg::Unnamed(FunctionArgExpr::Expr(e)) => {
891                polars_ensure!(
892                    named.is_empty(),
893                    SQLInterface: "in the call of macro '{}', a positional argument follows a named argument", name
894                );
895                positional.push(e);
896            },
897            FunctionArg::Named {
898                name: arg_name,
899                arg: FunctionArgExpr::Expr(e),
900                ..
901            } => named.push((arg_name, e)),
902            _ => polars_bail!(SQLInterface: "invalid argument of macro '{}': {}", name, arg),
903        }
904    }
905    let mut given_twice = None;
906    let mut bind = |params: &[MacroParam]| -> Option<Vec<BoundArg>> {
907        if positional.len() > params.len() {
908            return None;
909        }
910        let mut values: Vec<Option<SQLExpr>> = positional
911            .iter()
912            .map(|e| Some((*e).clone()))
913            .chain(std::iter::repeat_n(None, params.len() - positional.len()))
914            .collect();
915        for (arg_name, e) in &named {
916            let i = params
917                .iter()
918                .position(|p| same_name(&p.name, &arg_name.value))?;
919            if values[i].is_some() {
920                given_twice.get_or_insert(arg_name.value.clone());
921                return None;
922            }
923            values[i] = Some((*e).clone());
924        }
925        params
926            .iter()
927            .zip(values)
928            .map(|(param, value)| {
929                let (value, is_default) = match value {
930                    Some(value) => (value, false),
931                    None => (param.default.clone()?, true),
932                };
933                Some(BoundArg {
934                    name: param.name.clone(),
935                    value,
936                    is_default,
937                })
938            })
939            .collect()
940    };
941    let mut matches = overloads
942        .iter()
943        .filter_map(|overload| Some((overload, bind(&overload.params)?)));
944    let Some(binding) = matches.next() else {
945        if let Some(param) = given_twice {
946            polars_bail!(
947                SQLInterface: "in the call of macro '{}', parameter '{}' is given twice", name, param
948            );
949        }
950        polars_bail!(
951            SQLInterface: "macro '{}' has no definition for the given arguments", name
952        )
953    };
954    polars_ensure!(
955        matches.next().is_none(),
956        SQLInterface: "the call of macro '{}' matches more than one of its definitions", name
957    );
958    Ok(binding)
959}

961/// A parameter of a macro call and its value.
962struct BoundArg {
963    name: Ident,
964    value: SQLExpr,
965    /// Whether the value is the parameter's default.
966    is_default: bool,
967}

969/// The names of the relations of a query or SELECT, with each part of a qualified name.
970struct Relations {
971    names: Vec<RelationName>,
972    /// A CTE or derived table, which does not see the relations of the query it is part of.
973    starts_scope: bool,
974    /// The relations before a derived table in its FROM, which it reads as if LATERAL.
975    preceding: Vec<RelationName>,
976    is_query: bool,
977}

979/// Replaces the parameters of a macro body by their values.
980struct ParamReplacer<'a> {
981    macros: &'a Macros,
982    name: &'a str,
983    params: &'a [(Ident, SQLExpr)],
984    size: &'a ExpandedSize,
985    /// The number of lambdas around the call, which the elements of the values count.
986    call_depth: usize,
987    /// The lambdas of the body around the visited expression.
988    lambdas: LambdaScopes,
989    /// The relations of each query and SELECT around the visited expression.
990    relations: Vec<Relations>,
991    /// The CTEs and derived tables of the visited queries, with the relations before each
992    /// derived table in its FROM.
993    scoped_queries: PlHashMap<*const Query, Vec<RelationName>>,
994    /// The join conditions and table function arguments of the visited SELECT statements, with the
995    /// relations each sees.
996    narrowed_exprs: PlHashMap<*const SQLExpr, Vec<RelationName>>,
997    /// The narrowed expressions being visited.
998    open_narrowed: Vec<*const SQLExpr>,
999}
1000
1001impl<'a> ParamReplacer<'a> {
1002    fn new(
1003        macros: &'a Macros,
1004        name: &'a str,
1005        params: &'a [(Ident, SQLExpr)],
1006        size: &'a ExpandedSize,
1007        call_depth: usize,
1008    ) -> Self {
1009        Self {
1010            macros,
1011            name,
1012            params,
1013            size,
1014            call_depth,
1015            lambdas: LambdaScopes::default(),
1016            relations: Vec::new(),
1017            scoped_queries: PlHashMap::default(),
1018            narrowed_exprs: PlHashMap::default(),
1019            open_narrowed: vec![],
1020        }
1021    }
1022
1023    /// Whether a leading run of `idents` names a relation that the visited expression sees.
1024    fn names_relation(&self, idents: &[Ident]) -> bool {
1025        let mut hidden = false;
1026        for relations in self.relations.iter().rev() {
1027            if hidden {
1028                // The relations of an enclosing query can still be correlated, unless the
1029                // query is itself a CTE or derived table. A derived table still sees the
1030                // relations before it.
1031                if relations.is_query {
1032                    if match_qualifier(idents, relations.preceding.iter()).is_some() {
1033                        return true;
1034                    }
1035                    hidden = relations.starts_scope;
1036                }
1037                continue;
1038            }
1039            if match_qualifier(idents, relations.names.iter()).is_some()
1040                || match_qualifier(idents, relations.preceding.iter()).is_some()
1041            {
1042                return true;
1043            }
1044            hidden = relations.starts_scope;
1045        }
1046        false
1047    }
1048
1049    fn value(&self, name: &str) -> ControlFlow<PolarsError, Option<SQLExpr>> {
1050        let Some((_, value)) = self.params.iter().find(|(p, _)| same_name(p, name)) else {
1051            return ControlFlow::Continue(None);
1052        };
1053        // A column of the caller would be read from the tables of the subquery.
1054        if !self.relations.is_empty() && expr_references_any_column(value) {
1055            return ControlFlow::Break(polars_err!(
1056                SQLInterface: "macro '{}' uses parameter '{}' in a query, where its value cannot read columns",
1057                self.name, name
1058            ));
1059        }
1060        self.size.add(value)?;
1061        let mut value = parenthesize(value.clone());
1062        // The lambdas of the value are placed in the lambdas of the body around it.
1063        shift_lambda_elements(&mut value, self.call_depth, self.lambdas.depth);
1064        ControlFlow::Continue(Some(value))
1065    }
1066}
1067
1068/// A compound expression in parentheses, so that it keeps its meaning where it is placed.
1069fn parenthesize(expr: SQLExpr) -> SQLExpr {
1070    match expr {
1071        SQLExpr::Value(_)
1072        | SQLExpr::Interval(_)
1073        | SQLExpr::Identifier(_)
1074        | SQLExpr::CompoundIdentifier(_)
1075        | SQLExpr::Function(_)
1076        | SQLExpr::Nested(_)
1077        | SQLExpr::Subquery(_) => expr,
1078        _ => SQLExpr::Nested(Box::new(expr)),
1079    }
1080}

1082impl VisitorMut for ParamReplacer<'_> {
1083    type Break = PolarsError;
1084
1085    fn pre_visit_select(&mut self, select: &mut Select) -> ControlFlow<PolarsError> {
1086        // A SELECT item that is just a parameter keeps the parameter name.
1087        for item in &mut select.projection {
1088            if let SelectItem::UnnamedExpr(SQLExpr::Identifier(ident)) = item
1089                && self.params.iter().any(|(p, _)| same_name(p, &ident.value))
1090            {
1091                let alias = ident.clone();
1092                *item = SelectItem::ExprWithAlias {
1093                    expr: SQLExpr::Identifier(ident.clone()),
1094                    alias,
1095                };
1096            }
1097        }
1098        let mut preceding = Vec::new();
1099        let mut names = Vec::new();
1100        for table in &select.from {
1101            self.add_relations(table, &mut preceding);
1102            table_names(table, &mut names);
1103        }
1104        self.relations.push(Relations {
1105            names,
1106            starts_scope: false,
1107            preceding: vec![],
1108            is_query: false,
1109        });
1110        ControlFlow::Continue(())
1111    }
1112
1113    fn post_visit_select(&mut self, _select: &mut Select) -> ControlFlow<PolarsError> {
1114        self.relations.pop();
1115        ControlFlow::Continue(())
1116    }
1117
1118    fn pre_visit_query(&mut self, query: &mut Query) -> ControlFlow<PolarsError> {
1119        let preceding = self.scoped_queries.remove(&(&*query as *const Query));
1120        if let Some(with) = &query.with {
1121            self.scoped_queries.extend(
1122                with.cte_tables
1123                    .iter()
1124                    .map(|cte| (&*cte.query as *const Query, vec![])),
1125            );
1126        }
1127        let mut names = Vec::new();
1128        relation_names(&query.body, &mut names);
1129        self.relations.push(Relations {
1130            names,
1131            starts_scope: preceding.is_some(),
1132            preceding: preceding.unwrap_or_default(),
1133            is_query: true,
1134        });
1135        ControlFlow::Continue(())
1136    }
1137
1138    fn post_visit_query(&mut self, _query: &mut Query) -> ControlFlow<PolarsError> {
1139        self.relations.pop();
1140        ControlFlow::Continue(())
1141    }
1142
1143    fn pre_visit_table_factor(&mut self, factor: &mut TableFactor) -> ControlFlow<PolarsError> {
1144        if let TableFactor::Derived { subquery, .. } = factor {
1145            self.scoped_queries
1146                .entry(&**subquery as *const Query)
1147                .or_default();
1148        }
1149        // The conditions inside an aliased join see the relations that the alias hides, but
1150        // not the alias or the other relations of the FROM.
1151        if let Some(names) = hidden_by_alias(factor) {
1152            self.relations.push(Relations {
1153                names,
1154                starts_scope: true,
1155                preceding: vec![],
1156                is_query: false,
1157            });
1158        }
1159        ControlFlow::Continue(())
1160    }
1161
1162    fn pre_visit_expr(&mut self, expr: &mut SQLExpr) -> ControlFlow<PolarsError> {
1163        self.lambdas.enter(expr);
1164        // A join condition sees the relations of its join, not those joined after it. The
1165        // arguments of a table function see the relations before it.
1166        if let Some(names) = self.narrowed_exprs.remove(&(expr as *const SQLExpr)) {
1167            self.relations.push(Relations {
1168                names,
1169                starts_scope: true,
1170                preceding: vec![],
1171                is_query: false,
1172            });
1173            self.open_narrowed.push(expr as *const SQLExpr);
1174        }
1175        ControlFlow::Continue(())
1176    }
1177
1178    fn post_visit_expr(&mut self, expr: &mut SQLExpr) -> ControlFlow<PolarsError> {
1179        let ptr = expr as *const SQLExpr;
1180        let is_body = self.lambdas.leave_children(expr);
1181        self.replace_param(expr)?;
1182        self.lambdas.leave_body(is_body);
1183        if self.open_narrowed.last() == Some(&ptr) {
1184            self.open_narrowed.pop();
1185            self.relations.pop();
1186        }
1187        ControlFlow::Continue(())
1188    }
1189
1190    fn post_visit_table_factor(&mut self, factor: &mut TableFactor) -> ControlFlow<PolarsError> {
1191        if hidden_by_alias(factor).is_some() {
1192            self.relations.pop();
1193        }
1194        unparenthesize_table_args(factor, self.macros);
1195        ControlFlow::Continue(())
1196    }
1197}

1199impl ParamReplacer<'_> {
1200    /// Adds the names of the relations of `table` to `names`, the relations before it. Records
1201    /// the relations before each derived table, which it reads as if LATERAL.
1202    /// Records the relations each join condition sees.
1203    fn add_relations(&mut self, table: &TableWithJoins, names: &mut Vec<RelationName>) {
1204        let start = names.len();
1205        let mut joined = Vec::new();
1206        self.add_factor(&table.relation, names, &mut joined);
1207        for join in &table.joins {
1208            let (left_names, left_joined) = (names.len(), joined.len());
1209            self.add_factor(&join.relation, names, &mut joined);
1210            for condition in root_exprs(&join.join_operator) {
1211                self.narrowed_exprs.insert(condition, joined.clone());
1212            }
1213            // After a semi or anti join, the relations of its dropped input are gone.
1214            let (keep_left, keep_right) = join_keeps_inputs(&join.join_operator);
1215            if !keep_right {
1216                names.truncate(left_names);
1217                joined.truncate(left_joined);
1218            }
1219            if !keep_left {
1220                names.drain(start..left_names);
1221                joined.drain(..left_joined);
1222            }
1223        }
1224    }
1225
1226    /// As `add_relations`, for `factor`, which also adds its names to `joined`.
1227    fn add_factor(
1228        &mut self,
1229        factor: &TableFactor,
1230        names: &mut Vec<RelationName>,
1231        joined: &mut Vec<RelationName>,
1232    ) {
1233        factor_names(factor, joined);
1234        match factor {
1235            TableFactor::Derived { subquery, .. } => {
1236                self.scoped_queries
1237                    .insert(&**subquery as *const Query, names.clone());
1238            },
1239            TableFactor::NestedJoin {
1240                table_with_joins, ..
1241            } => self.add_relations(table_with_joins, &mut names.clone()),
1242            TableFactor::Pivot { table, .. } | TableFactor::Unpivot { table, .. } => {
1243                self.add_factor(table, &mut names.clone(), &mut vec![])
1244            },
1245            TableFactor::Table { args: Some(_), .. }
1246            | TableFactor::Function { .. }
1247            | TableFactor::TableFunction { .. }
1248            | TableFactor::UNNEST { .. } => {
1249                for arg in root_exprs(factor) {
1250                    self.narrowed_exprs.insert(arg, names.clone());
1251                }
1252            },
1253            _ => {},
1254        }
1255        factor_names(factor, names);
1256    }
1257
1258    fn replace_param(&mut self, expr: &mut SQLExpr) -> ControlFlow<PolarsError> {
1259        match expr {
1260            SQLExpr::Identifier(ident) => {
1261                if let Some(value) = self.value(&ident.value)? {
1262                    *expr = value;
1263                }
1264            },
1265            // A qualifier naming a relation of the body reads that relation.
1266            SQLExpr::CompoundIdentifier(idents) if self.names_relation(idents) => {},
1267            SQLExpr::CompoundIdentifier(idents) => {
1268                if let Some(value) = self.value(&idents[0].value)? {
1269                    // The fields of the value, also when it names a column like a table.
1270                    *expr = field_access(SQLExpr::Nested(Box::new(value)), &idents[1..]);
1271                }
1272            },
1273            _ => unparenthesize_operands(expr, self.macros),
1274        }
1275        ControlFlow::Continue(())
1276    }
1277}

1279/// The names of the relations inside `factor` that its alias hides.
1280fn hidden_by_alias(factor: &TableFactor) -> Option<Vec<RelationName>> {
1281    let mut names = Vec::new();
1282    match factor {
1283        TableFactor::NestedJoin {
1284            table_with_joins,
1285            alias: Some(_),
1286        } => table_names(table_with_joins, &mut names),
1287        TableFactor::Pivot {
1288            table,
1289            alias: Some(_),
1290            ..
1291        }
1292        | TableFactor::Unpivot {
1293            table,
1294            alias: Some(_),
1295            ..
1296        } => factor_names(table, &mut names),
1297        _ => return None,
1298    }
1299    Some(names)
1300}

1302/// Adds the names of the relations that the ORDER BY of a query with this body can qualify
1303/// columns with to `names`. Each SELECT of a set
1304/// operation has its own relations.
1305fn relation_names(body: &SetExpr, names: &mut Vec<RelationName>) {
1306    match body {
1307        SetExpr::Select(select) => {
1308            for table in &select.from {
1309                table_names(table, names);
1310            }
1311        },
1312        SetExpr::Query(query) => relation_names(&query.body, names),
1313        _ => {},
1314    }
1315}

1317/// Adds the names of the relations whose columns `table` returns to `names`.
1318fn table_names(table: &TableWithJoins, names: &mut Vec<RelationName>) {

1324/// The outermost expressions of `node`.
1325fn root_exprs<T: Visit>(node: &T) -> Vec<*const SQLExpr> {
1326    struct Roots {
1327        depth: usize,
1328        roots: Vec<*const SQLExpr>,
1329    }
1330    impl Visitor for Roots {
1331        type Break = ();
1332
1333        fn pre_visit_expr(&mut self, expr: &SQLExpr) -> ControlFlow<()> {
1334            if self.depth == 0 {
1335                self.roots.push(expr);
1336            }
1337            self.depth += 1;
1338            ControlFlow::Continue(())
1339        }
1340
1341        fn post_visit_expr(&mut self, _expr: &SQLExpr) -> ControlFlow<()> {
1342            self.depth -= 1;
1343            ControlFlow::Continue(())
1344        }
1345    }
1346    let mut roots = Roots {
1347        depth: 0,
1348        roots: vec![],
1349    };
1350    let _ = dyn_visit::visit(node, &mut roots);
1351    roots.roots
1352}

1354fn factor_names(factor: &TableFactor, names: &mut Vec<RelationName>) {
1355    match factor {
1356        // An alias hides the names of the relations inside.
1357        TableFactor::NestedJoin {
1358            table_with_joins,
1359            alias: None,
1360        } => table_names(table_with_joins, names),
1361        TableFactor::Pivot {
1362            table, alias: None, ..
1363        }
1364        | TableFactor::Unpivot {
1365            table, alias: None, ..
1366        } => factor_names(table, names),
1367        TableFactor::Function { name, alias, .. } => names.push(RelationName::new(match alias {
1368            Some(alias) => vec![alias.name.value.clone()],
1369            None => name
1370                .0
1371                .iter()
1372                .filter_map(|p| Some(p.as_ident()?.value.clone()))
1373                .collect(),
1374        })),
1375        _ => {},
1376    }
1377    names.extend(relation_name(factor));
1378}

1380/// Drops the parentheses around function arguments. A substituted value keeps its
1381/// parentheses only where it is an operand.
1382fn unparenthesize_operands(expr: &mut SQLExpr, macros: &Macros) {
1383    match expr {
1384        // The value of a macro argument keeps its meaning, as a number that is no position.
1385        SQLExpr::Function(function) if macros.get(&function.name, false).is_some() => {},
1386        SQLExpr::Cast { kind, expr, .. } if *kind != CastKind::DoubleColon => unparenthesize(expr),
1387        SQLExpr::Interval(interval)
1388            if matches!(strip_nested(&interval.value), SQLExpr::Value(_)) =>
1389        {
1390            unparenthesize(&mut interval.value)
1391        },
1392        SQLExpr::Function(function) => {
1393            if let FunctionArguments::List(list) = &mut function.args {
1394                unparenthesize_args(&mut list.args);
1395            }
1396        },
1397        _ => {},
1398    }
1399}

1401/// Drops the parentheses around the arguments of a table function or `UNNEST`.
1402fn unparenthesize_table_args(factor: &mut TableFactor, macros: &Macros) {
1403    match factor {
1404        TableFactor::Table { name, .. } if macros.get(name, true).is_some() => {},
1405        TableFactor::Table {
1406            args: Some(args), ..
1407        } => unparenthesize_args(&mut args.args),
1408        TableFactor::UNNEST { array_exprs, .. } => array_exprs.iter_mut().for_each(unparenthesize),
1409        _ => {},
1410    }
1411}
1412
1413fn unparenthesize_args(args: &mut [FunctionArg]) {
1414    for arg in args {
1415        if let FunctionArg::Unnamed(FunctionArgExpr::Expr(expr))
1416        | FunctionArg::Named {
1417            arg: FunctionArgExpr::Expr(expr),
1418            ..
1419        }
1420        | FunctionArg::ExprNamed {
1421            arg: FunctionArgExpr::Expr(expr),
1422            ..
1423        } = arg
1424        {
1425            unparenthesize(expr);
1426        }
1427    }
1428}
1429
1430fn unparenthesize(expr: &mut SQLExpr) {
1431    while let SQLExpr::Nested(inner) = expr {
1432        *expr = std::mem::replace(inner.as_mut(), SQLExpr::value(SQLValue::Null));
1433    }
