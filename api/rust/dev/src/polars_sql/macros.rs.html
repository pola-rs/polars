66/// Visits the bodies and parameter defaults of a created macro.
67impl VisitMut for MacroStatement {
68    fn visit<V: VisitorMut>(&mut self, visitor: &mut V) -> ControlFlow<V::Break> {
69        let MacroStatement::Create { overloads, .. } = self else {
70            return ControlFlow::Continue(());
71        };
72        for overload in overloads {
73            for default in overload
74                .params
75                .iter_mut()
76                .filter_map(|p| p.default.as_mut())
77            {
78                default.visit(visitor)?;
79            }
80            match &mut overload.body {
81                MacroBody::Expr(expr) => expr.visit(visitor)?,
82                MacroBody::Table(query) => query.visit(visitor)?,
83            }
84        }
85        ControlFlow::Continue(())
86    }
87}

89/// The macros of a context, by lowercase name.
90#[derive(Clone, Default)]
91pub(crate) struct Macros {
92    scalar: Arc<RwLock<PlHashMap<String, Arc<[MacroOverload]>>>>,
93    table: Arc<RwLock<PlHashMap<String, Arc<[MacroOverload]>>>>,
94}
95
96impl Macros {
97    fn map(&self, table: bool) -> &RwLock<PlHashMap<String, Arc<[MacroOverload]>>> {
98        if table { &self.table } else { &self.scalar }
99    }
100
101    pub(crate) fn create(
102        &self,
103        name: &Ident,
104        table: bool,
105        or_replace: bool,
106        overloads: Vec<MacroOverload>,
107    ) -> PolarsResult<()> {
108        let mut map = self.map(table).write().unwrap();
109        let key = name.value.to_lowercase();
110        polars_ensure!(
111            or_replace || !map.contains_key(&key),
112            SQLInterface: "macro '{}' already exists", name.value
113        );
114        map.insert(key, overloads.into());
115        Ok(())
116    }
117
118    pub(crate) fn drop(&self, name: &Ident, table: bool, if_exists: bool) -> PolarsResult<()> {
119        let removed = self
120            .map(table)
121            .write()
122            .unwrap()
123            .remove(&name.value.to_lowercase());
124        polars_ensure!(
125            removed.is_some() || if_exists,
126            SQLInterface: "{}macro '{}' does not exist", if table { "table " } else { "" }, name.value
127        );
128        Ok(())
129    }
130
131    fn get(&self, name: &ObjectName, table: bool) -> Option<(String, Arc<[MacroOverload]>)> {
132        let [part] = name.0.as_slice() else {
133            return None;
134        };
135        let name = &part.as_ident()?.value;
136        let map = self.map(table).read().unwrap();
137        if map.is_empty() {
138            return None;
139        }
140        let overloads = map.get(&name.to_lowercase())?.clone();
141        Some((name.clone(), overloads))
142    }
143
144    pub(crate) fn is_empty(&self) -> bool {
145        self.scalar.read().unwrap().is_empty() && self.table.read().unwrap().is_empty()
146    }
147
148    /// Replace the macro calls in `node` by the macro bodies. Returns whether there were any.
149    pub(crate) fn expand<T: Expandable>(&self, node: &mut T) -> PolarsResult<bool> {
150        if self.is_empty() {
151            return Ok(false);
152        }
153        let mut expander = MacroExpander {
154            macros: self,
155            expanding: vec![],
156            expanded: false,
157            size: ExpandedSize::default(),
158            lambdas: LambdaScopes::default(),
159        };
160        if let ControlFlow::Break(err) = dyn_visit::visit_mut(node, &mut expander) {
161            return Err(err);
162        }
163        if expander.expanded {
164            // The expanded SQL may not be nested deeper than the parser accepts.
165            let sql = node.to_string();
166            let reparsed = Parser::new(&GenericDialect)
167                .try_with_sql(&sql)
168                .and_then(|mut parser| T::reparse(&mut parser));
169            polars_ensure!(
170                !matches!(reparsed, Err(ParserError::RecursionLimitExceeded)),
171                SQLInterface: "the query is nested too deeply after expanding its macros"
172            );
173        }
174        Ok(expander.expanded)
175    }
176
177    /// `node` with its macro calls replaced by the macro bodies, or `None` if it has none.
178    pub(crate) fn expanded<T: Expandable + Clone>(&self, node: &T) -> PolarsResult<Option<T>> {
179        if self.is_empty() {
180            return Ok(None);
181        }
182        let mut node = node.clone();
183        Ok(self.expand(&mut node)?.then_some(node))
184    }
185}

216/// The macro statement at the start of `parser`; if there is none, `parser` stays at the start.
217pub(crate) fn parse_macro_statement(parser: &mut Parser) -> PolarsResult<Option<MacroStatement>> {
218    let head = parser
219        .maybe_parse(|p| {
220            while p.consume_token(&Token::SemiColon) {}
221            match p.parse_one_of_keywords(&[Keyword::CREATE, Keyword::DROP]) {
222                Some(Keyword::CREATE) => {
223                    let or_replace = p.parse_keywords(&[Keyword::OR, Keyword::REPLACE]);
224                    let _ = p.parse_one_of_keywords(&[Keyword::TEMP, Keyword::TEMPORARY]);
225                    p.expect_one_of_keywords(&[Keyword::MACRO, Keyword::FUNCTION])?;
226                    Ok((true, or_replace, false))
227                },
228                Some(_) => {
229                    let table =
230                        match p.expect_one_of_keywords(&[Keyword::MACRO, Keyword::FUNCTION])? {
231                            Keyword::MACRO => p.parse_keyword(Keyword::TABLE),
232                            _ => false,
233                        };
234                    Ok((false, false, table))
235                },
236                None => Err(ParserError::ParserError(String::new())),
237            }
238        })
239        .map_err(to_sql_interface_err)?;
240    let Some((create, or_replace, table)) = head else {
241        return Ok(None);
242    };
243    let stmt = if create {
244        parse_create_macro(parser, or_replace)
245    } else {
246        parse_drop_macro(parser, table)
247    }
248    .map_err(to_sql_interface_err)?;
249    while parser.consume_token(&Token::SemiColon) {}
250    if parser.peek_token().token != Token::EOF {
251        polars_bail!(SQLInterface: "one (and only one) statement can be parsed at a time");
252    }
253    Ok(Some(stmt))
254}

256fn parse_create_macro(
257    parser: &mut Parser,
258    or_replace: bool,
259) -> Result<MacroStatement, ParserError> {
260    let name = single_name(parser.parse_object_name(false)?)?;
261    let mut overloads = vec![];
262    let mut table = false;
263    loop {
264        let params = parse_macro_params(parser)?;
265        parser.expect_keyword_is(Keyword::AS)?;
266        let body = if parser.parse_keyword(Keyword::TABLE) {
267            MacroBody::Table(parse_table_body(parser)?)
268        } else {
269            MacroBody::Expr(Box::new(parser.parse_expr()?))
270        };
271        let is_table = matches!(body, MacroBody::Table(_));
272        if !overloads.is_empty() && is_table != table {
273            return Err(ParserError::ParserError(
274                "the definitions of a macro must all be scalar or all be TABLE".into(),
275            ));
276        }
277        table = is_table;
278        overloads.push(MacroOverload { params, body });
279        if !parser.consume_token(&Token::Comma) {
280            break;
281        }
282    }
283    Ok(MacroStatement::Create {
284        or_replace,
285        name,
286        table,
287        overloads,
288    })
289}

291/// The parameters of a definition, in parentheses.
292fn parse_macro_params(parser: &mut Parser) -> Result<Vec<MacroParam>, ParserError> {
293    parser.expect_token(&Token::LParen)?;
294    let mut params = vec![];
295    if parser.consume_token(&Token::RParen) {
296        return Ok(params);
297    }
298    loop {
299        let name = parser.parse_identifier()?;
300        let default =
301            if parser.consume_token(&Token::Assignment) || parser.consume_token(&Token::RArrow) {
302                Some(parser.parse_expr()?)
303            } else {
304                None
305            };
306        params.push(MacroParam { name, default });
307        if !parser.consume_token(&Token::Comma) {
308            break;
309        }
310    }
311    parser.expect_token(&Token::RParen)?;
312    Ok(params)
313}

315/// The query of a TABLE definition, which ends before the `, (...) AS TABLE` of the next
316/// definition.
317fn parse_table_body(parser: &mut Parser) -> Result<Box<Query>, ParserError> {
318    // The parser is past `AS TABLE`, so the next token is at the current index + 1.
319    let tokens: Vec<TokenWithSpan> = (parser.get_current_index() + 1..)
320        .map(|i| parser.token_at(i))
321        .take_while(|t| !matches!(t.token, Token::EOF | Token::SemiColon))
322        .cloned()
323        .collect();
324    let (query, len) = split_table_body(&tokens)?;
325    let n_parsed = tokens[..len]
326        .iter()
327        .filter(|t| !matches!(t.token, Token::Whitespace(_)))
328        .count();
329    for _ in 0..n_parsed {
330        parser.next_token();
331    }
332    Ok(query)
333}

335/// The query at the start of `tokens` and its number of tokens. It ends at a comma that
336/// starts another TABLE definition. The longest query that parses is taken, so a definition
337/// that is also a valid continuation of the query must be put in parentheses.
338fn split_table_body(tokens: &[TokenWithSpan]) -> Result<(Box<Query>, usize), ParserError> {
339    let words: Vec<usize> = (0..tokens.len())
340        .filter(|&i| !matches!(tokens[i].token, Token::Whitespace(_)))
341        .collect();
342    let mut boundaries = vec![];
343    let mut depth = 0usize;
344    for (k, &i) in words.iter().enumerate() {
345        match tokens[i].token {
346            Token::LParen | Token::LBracket | Token::LBrace => depth += 1,
347            Token::RParen | Token::RBracket | Token::RBrace => depth = depth.saturating_sub(1),
348            Token::Comma if depth == 0 && starts_table_definition(tokens, &words[k + 1..]) => {
349                boundaries.push(i)
350            },
351            _ => {},
352        }
353    }
354    let whole = parse_whole_query(tokens);
355    if whole.is_err() {
356        for &i in boundaries.iter().rev() {
357            if let Ok(query) = parse_whole_query(&tokens[..i]) {
358                return Ok((query, i));
359            }
360        }
361    }
362    Ok((whole?, tokens.len()))
363}

365/// Whether the tokens at `words` start with `(params) AS TABLE`.
366fn starts_table_definition(tokens: &[TokenWithSpan], words: &[usize]) -> bool {
367    let token = |k: usize| words.get(k).map(|&i| &tokens[i].token);
368    let is_keyword =
369        |k: usize, keyword| matches!(token(k), Some(Token::Word(w)) if w.keyword == keyword);
370    if token(0) != Some(&Token::LParen) {
371        return false;
372    }
373    let mut depth = 0usize;
374    let Some(close) = (0..words.len()).find(|&k| {
375        match token(k) {
376            Some(Token::LParen) => depth += 1,
377            Some(Token::RParen) => depth -= 1,
378            _ => {},
379        }
380        depth == 0
381    }) else {
382        return false;
383    };
384    if !is_keyword(close + 1, Keyword::AS) || !is_keyword(close + 2, Keyword::TABLE) {
385        return false;
386    }
387    let mut parser = parser_of(tokens[words[0]..=words[close + 2]].to_vec());
388    parse_macro_params(&mut parser).is_ok()
389        && parser.parse_keywords(&[Keyword::AS, Keyword::TABLE])
390        && parser.peek_token().token == Token::EOF
391}

393fn parse_whole_query(tokens: &[TokenWithSpan]) -> Result<Box<Query>, ParserError> {
394    let mut parser = parser_of(tokens.to_vec());
395    let query = parser.parse_query()?;
396    let rest = parser.peek_token();
397    if rest.token != Token::EOF {
398        return parser.expected("the end of the macro body", rest);
399    }
400    Ok(query)
401}

403fn parse_drop_macro(parser: &mut Parser, table: bool) -> Result<MacroStatement, ParserError> {
404    let if_exists = parser.parse_keywords(&[Keyword::IF, Keyword::EXISTS]);
405    let name = single_name(parser.parse_object_name(false)?)?;
406    Ok(MacroStatement::Drop {
407        table,
408        if_exists,
409        name,
410    })
411}

413fn single_name(name: ObjectName) -> Result<Ident, ParserError> {
414    match name.0.as_slice() {
415        [part] => part
416            .as_ident()
417            .cloned()
418            .ok_or_else(|| ParserError::ParserError(format!("invalid macro name {name}"))),
419        _ => Err(ParserError::ParserError(format!(
420            "a macro name cannot be qualified; found {name}"
421        ))),
422    }
423}

425/// Check the definitions of macro `name`.
426pub(crate) fn check_macro(name: &Ident, overloads: &[MacroOverload]) -> PolarsResult<()> {
427    for overload in overloads {
428        let params = &overload.params;
429        let param = match &overload.body {
430            MacroBody::Expr(body) => first_param(body.as_ref()),
431            MacroBody::Table(query) => first_param(query.as_ref()),
432        };
433        if let Some(p) = param.or_else(|| params.iter().find_map(|p| first_param(&p.default))) {
434            polars_bail!(
435                SQLInterface: "query parameters are not supported in CREATE MACRO; found {}", p
436            );
437        }
438        for (i, param) in params.iter().enumerate() {
439            polars_ensure!(
440                !params[..i].iter().any(|p| same_name(&p.name, &param.name.value)),
441                SQLInterface: "macro '{}' has a duplicate parameter '{}'", name.value, param.name.value
442            );
443            polars_ensure!(
444                param.default.as_ref().is_none_or(|d| !expr_references_any_column(d)),
445                SQLInterface: "the default of parameter '{}' of macro '{}' cannot read columns",
446                param.name.value, name.value
447            );
448            polars_ensure!(
449                param.default.is_some() || params[..i].iter().all(|p| p.default.is_none()),
450                SQLInterface: "in macro '{}', parameter '{}' without a default follows a parameter with a default",
451                name.value, param.name.value
452            );
453        }
454        if let MacroBody::Expr(body) = &overload.body {
455            let mut check = FreeIdentifiers {
456                params,
457                query_depth: 0,
458            };
459            if let ControlFlow::Break(ident) = dyn_visit::visit(body, &mut check) {
460                polars_bail!(
461                    SQLInterface: "the body of macro '{}' can only use its parameters; found '{}'",
462                    name.value, ident
463                );
464            }
465        }
466    }
467    Ok(())
468}

470/// Whether a call of `name` parses as a call of a function (or table function), rather than
471/// as special syntax like `TRIM(...)` or `UNNEST(...)`, which a macro cannot replace.
472pub(crate) fn is_callable_name(name: &Ident, table: bool) -> bool {
473    let sql = if table {
474        format!("SELECT * FROM {name}(1)")
475    } else {
476        format!("SELECT {name}(1)")
477    };
478    let Ok(statements) = Parser::parse_sql(&GenericDialect, &sql) else {
479        return false;
480    };
481    let [Statement::Query(query)] = statements.as_slice() else {
482        return false;
483    };
484    let SetExpr::Select(select) = query.body.as_ref() else {
485        return false;
486    };
487    if table {
488        matches!(
489            select.from.as_slice(),
490            [TableWithJoins {
491                relation: TableFactor::Table { args: Some(_), .. },
492                ..
493            }]
494        )
495    } else {
496        matches!(
497            select.projection.as_slice(),
498            [SelectItem::UnnamedExpr(SQLExpr::Function(_))]
499        )
500    }
501}

507/// Finds an identifier outside subqueries that is not a parameter.
508struct FreeIdentifiers<'a> {
509    params: &'a [MacroParam],
510    query_depth: usize,
511}
512
513impl Visitor for FreeIdentifiers<'_> {
514    type Break = String;
515
516    fn pre_visit_query(&mut self, _query: &Query) -> ControlFlow<String> {
517        self.query_depth += 1;
518        ControlFlow::Continue(())
519    }
520
521    fn post_visit_query(&mut self, _query: &Query) -> ControlFlow<String> {
522        self.query_depth -= 1;
523        ControlFlow::Continue(())
524    }
525
526    fn pre_visit_expr(&mut self, expr: &SQLExpr) -> ControlFlow<String> {
527        if self.query_depth > 0 {
528            return ControlFlow::Continue(());
529        }
530        let first = match expr {
531            SQLExpr::Identifier(ident) => ident,
532            SQLExpr::CompoundIdentifier(idents) => &idents[0],
533            _ => return ControlFlow::Continue(()),
534        };
535        if self.params.iter().any(|p| same_name(&p.name, &first.value)) {
536            ControlFlow::Continue(())
537        } else {
538            ControlFlow::Break(expr.to_string())
539        }
540    }
541}

543/// The deepest nesting of macro calls in macro bodies.
544const MAX_MACRO_DEPTH: usize = 64;
545
546/// The most nodes that expanding the macros of a statement may copy.
547const MAX_EXPANDED_EXPRS: usize = 100_000;
548
549/// The number of nodes copied by an expansion.
550#[derive(Default)]
551struct ExpandedSize(Cell<usize>);
552
553impl ExpandedSize {
554    /// Counts the expressions, queries and relations of `node`, which is about to be copied.
555    fn add<T: Visit>(&self, node: &T) -> ControlFlow<PolarsError> {
556        struct Count(usize);
557        impl Visitor for Count {
558            type Break = ();
559            fn pre_visit_expr(&mut self, _expr: &SQLExpr) -> ControlFlow<()> {
560                self.0 += 1;
561                ControlFlow::Continue(())
562            }
563
564            fn pre_visit_query(&mut self, _query: &Query) -> ControlFlow<()> {
565                self.0 += 1;
566                ControlFlow::Continue(())
567            }
568
569            fn pre_visit_select(&mut self, _select: &Select) -> ControlFlow<()> {
570                self.0 += 1;
571                ControlFlow::Continue(())
572            }
573
574            fn pre_visit_table_factor(&mut self, _factor: &TableFactor) -> ControlFlow<()> {
575                self.0 += 1;
576                ControlFlow::Continue(())
577            }
578        }
579        let mut count = Count(0);
580        let _ = dyn_visit::visit(node, &mut count);
581        let size = self.0.get() + count.0;
582        if size > MAX_EXPANDED_EXPRS {
583            return ControlFlow::Break(polars_err!(
584                SQLInterface: "the query is too large after expanding its macros"
585            ));
586        }
587        self.0.set(size);
588        ControlFlow::Continue(())
589    }
590}

592struct MacroExpander<'a> {
593    macros: &'a Macros,
594    /// The macros whose bodies are being expanded, as (lowercase name, whether a table macro).
595    expanding: Vec<(String, bool)>,
596    expanded: bool,
597    size: ExpandedSize,
598    lambdas: LambdaScopes,
599}

601/// The lambdas around the visited expression.
602#[derive(Default)]
603struct LambdaScopes {
604    depth: usize,
605    /// The bodies of the lambda arguments of the functions being visited.
606    bodies: Vec<*const SQLExpr>,
607}
608
609impl LambdaScopes {
610    /// Called before the children of `expr` are visited.
611    fn enter(&mut self, expr: &SQLExpr) {
612        if self.bodies.contains(&(expr as *const SQLExpr)) {
613            self.depth += 1;
614        }
615        if let SQLExpr::Function(func) = expr {
616            self.bodies
617                .extend(lambda_args(func).map(|body| body as *const SQLExpr));
618        }
619    }
620
621    /// Called after the children of `expr` are visited. Returns whether `expr` is a lambda
622    /// body, which `leave_body` leaves once `expr` is visited.
623    fn leave_children(&mut self, expr: &SQLExpr) -> bool {
624        if let SQLExpr::Function(func) = expr {
625            let n_lambdas = lambda_args(func).count();
626            self.bodies.truncate(self.bodies.len() - n_lambdas);
627        }
628        self.bodies.contains(&(expr as *const SQLExpr))
629    }
630
631    fn leave_body(&mut self, is_body: bool) {
632        if is_body {
633            self.depth -= 1;
634        }
635    }
636}

638impl MacroExpander<'_> {
639    /// An error if macro `name` is being expanded: a macro that calls itself never ends.
640    fn check_not_expanding(&self, name: &str, table: bool) -> ControlFlow<PolarsError> {
641        let key = (name.to_lowercase(), table);
642        if self.expanding.contains(&key) {
643            return ControlFlow::Break(polars_err!(
644                SQLInterface: "macro '{}' calls itself", name
645            ));
646        }
647        ControlFlow::Continue(())
648    }
649
650    /// Expand the macro calls in the defaults of a call of macro `name`. The other arguments
651    /// were expanded where the call is.
652    fn expand_args(
653        &mut self,
654        bound: Vec<BoundArg>,
655        name: &str,
656        table: bool,
657    ) -> ControlFlow<PolarsError, Vec<(Ident, SQLExpr)>> {
658        let mut params = Vec::with_capacity(bound.len());
659        for mut arg in bound {
660            if arg.is_default {
661                // The lambdas of a default count from the call.
662                shift_lambda_elements(&mut arg.value, 0, self.lambdas.depth);
663                self.expand_nested(&mut arg.value, name, table)?;
664                // As written in the definition, a default cannot read columns.
665                if expr_references_any_column(&arg.value) {
666                    return ControlFlow::Break(polars_err!(
667                        SQLInterface: "the default of parameter '{}' of macro '{}' cannot read columns",
668                        arg.name.value, name
669                    ));
670                }
671            }
672            params.push((arg.name, arg.value));
673        }
674        ControlFlow::Continue(params)
675    }
676
677    /// Expand the macro calls in the body of macro `name`, which replaced a call.
678    fn expand_nested<T: VisitMut>(
679        &mut self,
680        node: &mut T,
681        name: &str,
682        table: bool,
683    ) -> ControlFlow<PolarsError> {
684        if self.expanding.len() >= MAX_MACRO_DEPTH {
685            return ControlFlow::Break(polars_err!(
686                SQLInterface: "macro '{}' is nested more than {} macros deep", name, MAX_MACRO_DEPTH
687            ));
688        }
689        self.expanding.push((name.to_lowercase(), table));
690        let res = dyn_visit::visit_mut(node, self);
691        self.expanding.pop();
692        res
693    }
694}

718impl MacroExpander<'_> {
719    /// Replaces `factor` by the body of the table macro it calls, if it calls one.
720    fn expand_table_call(&mut self, factor: &mut TableFactor) -> ControlFlow<PolarsError> {
721        let (macro_name, lateral) = match &*factor {
722            TableFactor::Table {
723                name,
724                args: Some(_),
725                ..
726            } => (name, false),
727            TableFactor::Function { name, lateral, .. } => (name, *lateral),
728            _ => return ControlFlow::Continue(()),
729        };
730        let Some((name, overloads)) = self.macros.get(macro_name, true) else {
731            return ControlFlow::Continue(());
732        };
733        let default_alias = macro_name
734            .0
735            .last()
736            .and_then(|part| part.as_ident())
737            .cloned();
738        self.check_not_expanding(&name, true)?;
739        let (args, alias, sample) = match factor {
740            TableFactor::Table {
741                alias,
742                args: Some(args),
743                sample,
744                version,
745                with_hints,
746                partitions,
747                json_path,
748                index_hints,
749                with_ordinality,
750                ..
751            } if version.is_none()
752                && with_hints.is_empty()
753                && partitions.is_empty()
754                && json_path.is_none()
755                && index_hints.is_empty()
756                && !*with_ordinality =>
757            {
758                (&args.args, alias, sample.take())
759            },
760            TableFactor::Function {
761                args,
762                alias,
763                with_ordinality: false,
764                ..
765            } => (&*args, alias, None),
766            _ => {
767                return ControlFlow::Break(polars_err!(
768                    SQLInterface: "table macro '{}' takes no table modifiers", name
769                ));
770            },
771        };
772        let (overload, bound) = flow(bind_args(&name, &overloads, args))?;
773        let alias = alias.take();
774        let bound = self.expand_args(bound, &name, true)?;
775        let MacroBody::Table(query) = &overload.body else {
776            unreachable!()
777        };
778        self.size.add(&**query)?;
779        let mut subquery = query.clone();
780        dyn_visit::visit_mut(
781            &mut *subquery,
782            &mut ParamReplacer::new(self.macros, &name, &bound, &self.size, self.lambdas.depth),
783        )?;
784        self.expand_nested(&mut subquery, &name, true)?;
785        let alias = alias.unwrap_or_else(|| TableAlias {
786            explicit: false,
787            name: default_alias.unwrap_or_else(|| Ident::new(name)),
788            columns: vec![],
789            at: None,
790        });
791        *factor = TableFactor::Derived {
792            lateral,
793            subquery,
794            alias: Some(alias),
795            sample,
796        };
797        self.expanded = true;
798        ControlFlow::Continue(())
799    }
800
801    /// Replaces `expr` by the body of the macro it calls, if it calls one.
802    fn expand_call(&mut self, expr: &mut SQLExpr) -> ControlFlow<PolarsError> {
803        unparenthesize_operands(expr, self.macros);
804        let SQLExpr::Function(func) = expr else {
805            return ControlFlow::Continue(());
806        };
807        let Some((name, overloads)) = self.macros.get(&func.name, false) else {
808            return ControlFlow::Continue(());
809        };
810        self.check_not_expanding(&name, false)?;
811        let args = flow(macro_call_args(&name, func))?;
812        let (overload, bound) = flow(bind_args(&name, &overloads, args))?;
813        let bound = self.expand_args(bound, &name, false)?;
814        let MacroBody::Expr(body) = &overload.body else {
815            unreachable!()
816        };
817        self.size.add(body.as_ref())?;
818        let mut body = body.as_ref().clone();
819        shift_lambda_elements(&mut body, 0, self.lambdas.depth);
820        let mut replacer =
821            ParamReplacer::new(self.macros, &name, &bound, &self.size, self.lambdas.depth);
822        dyn_visit::visit_mut(&mut body, &mut replacer)?;
823        self.expand_nested(&mut body, &name, false)?;
824        // The value of a call is an expression, not a position or keyword in ORDER BY or
825        // GROUP BY.
826        *expr = match body {
827            SQLExpr::Value(_) => SQLExpr::Nested(Box::new(body)),
828            body => parenthesize(body),
829        };
830        self.expanded = true;
831        ControlFlow::Continue(())
832    }
833}

835/// The bodies of the lambda arguments of `func`.
836fn lambda_args(func: &SQLFunction) -> impl Iterator<Item = &SQLExpr> {
837    let args = match &func.args {
838        FunctionArguments::List(list) => list.args.as_slice(),
839        _ => &[],

844/// `res` as the result of a visit.
845fn flow<T>(res: PolarsResult<T>) -> ControlFlow<PolarsError, T> {
846    match res {
847        Ok(value) => ControlFlow::Continue(value),
848        Err(err) => ControlFlow::Break(err),

852/// The arguments of the scalar macro call `func`.
853fn macro_call_args<'a>(name: &str, func: &'a SQLFunction) -> PolarsResult<&'a [FunctionArg]> {
854    polars_ensure!(
855        func.over.is_none()
856            && func.filter.is_none()
857            && func.within_group.is_empty()
858            && func.null_treatment.is_none()
859            && matches!(func.parameters, FunctionArguments::None),
860        SQLInterface: "macro '{}' does not support OVER, FILTER, WITHIN GROUP or IGNORE/RESPECT NULLS", name
861    );
862    match &func.args {
863        FunctionArguments::None => Ok(&[]),
864        FunctionArguments::List(FunctionArgumentList {
865            args,
866            duplicate_treatment: None,
867            clauses,
868        }) if clauses.is_empty() => Ok(args),
869        _ => polars_bail!(
870            SQLInterface: "macro '{}' takes only arguments, without DISTINCT, ORDER BY or other clauses", name
871        ),
872    }
873}

875/// The definition of macro `name` for `args`, with each parameter and its value.
876fn bind_args<'a>(
877    name: &str,
878    overloads: &'a [MacroOverload],
879    args: &[FunctionArg],
880) -> PolarsResult<(&'a MacroOverload, Vec<BoundArg>)> {
881    let mut positional = vec![];
882    let mut named: Vec<(&Ident, &SQLExpr)> = vec![];
883    for arg in args {
884        match arg {
885            FunctionArg::Unnamed(FunctionArgExpr::Expr(e)) => {
886                polars_ensure!(
887                    named.is_empty(),
888                    SQLInterface: "in the call of macro '{}', a positional argument follows a named argument", name
889                );
890                positional.push(e);
891            },
892            FunctionArg::Named {
893                name: arg_name,
894                arg: FunctionArgExpr::Expr(e),
895                ..
896            } => named.push((arg_name, e)),
897            _ => polars_bail!(SQLInterface: "invalid argument of macro '{}': {}", name, arg),
898        }
899    }
900    let mut given_twice = None;
901    let mut bind = |params: &[MacroParam]| -> Option<Vec<BoundArg>> {
902        if positional.len() > params.len() {
903            return None;
904        }
905        let mut values: Vec<Option<SQLExpr>> = positional
906            .iter()
907            .map(|e| Some((*e).clone()))
908            .chain(std::iter::repeat_n(None, params.len() - positional.len()))
909            .collect();
910        for (arg_name, e) in &named {
911            let i = params
912                .iter()
913                .position(|p| same_name(&p.name, &arg_name.value))?;
914            if values[i].is_some() {
915                given_twice.get_or_insert(arg_name.value.clone());
916                return None;
917            }
918            values[i] = Some((*e).clone());
919        }
920        params
921            .iter()
922            .zip(values)
923            .map(|(param, value)| {
924                let (value, is_default) = match value {
925                    Some(value) => (value, false),
926                    None => (param.default.clone()?, true),
927                };
928                Some(BoundArg {
929                    name: param.name.clone(),
930                    value,
931                    is_default,
932                })
933            })
934            .collect()
935    };
936    let mut matches = overloads
937        .iter()
938        .filter_map(|overload| Some((overload, bind(&overload.params)?)));
939    let Some(binding) = matches.next() else {
940        if let Some(param) = given_twice {
941            polars_bail!(
942                SQLInterface: "in the call of macro '{}', parameter '{}' is given twice", name, param
943            );
944        }
945        polars_bail!(
946            SQLInterface: "macro '{}' has no definition for the given arguments", name
947        )
948    };
949    polars_ensure!(
950        matches.next().is_none(),
951        SQLInterface: "the call of macro '{}' matches more than one of its definitions", name
952    );
953    Ok(binding)
954}

956/// A parameter of a macro call and its value.
957struct BoundArg {
958    name: Ident,
959    value: SQLExpr,
960    /// Whether the value is the parameter's default.
961    is_default: bool,
962}

964/// The names of the relations of a query or SELECT, with each part of a qualified name.
965struct Relations {
966    names: Vec<RelationName>,
967    /// A CTE or derived table, which does not see the relations of the query it is part of.
968    starts_scope: bool,
969    /// The relations before a derived table in its FROM, which it reads as if LATERAL.
970    preceding: Vec<RelationName>,
971    is_query: bool,
972}

974/// Replaces the parameters of a macro body by their values.
975struct ParamReplacer<'a> {
976    macros: &'a Macros,
977    name: &'a str,
978    params: &'a [(Ident, SQLExpr)],
979    size: &'a ExpandedSize,
980    /// The number of lambdas around the call, which the elements of the values count.
981    call_depth: usize,
982    /// The lambdas of the body around the visited expression.
983    lambdas: LambdaScopes,
984    /// The relations of each query and SELECT around the visited expression.
985    relations: Vec<Relations>,
986    /// The CTEs and derived tables of the visited queries, with the relations before each
987    /// derived table in its FROM.
988    scoped_queries: PlHashMap<*const Query, Vec<RelationName>>,
989    /// The join conditions and table function arguments of the visited SELECT statements, with the
990    /// relations each sees.
991    narrowed_exprs: PlHashMap<*const SQLExpr, Vec<RelationName>>,
992    /// The narrowed expressions being visited.
993    open_narrowed: Vec<*const SQLExpr>,
994}
995
996impl<'a> ParamReplacer<'a> {
997    fn new(
998        macros: &'a Macros,
999        name: &'a str,
1000        params: &'a [(Ident, SQLExpr)],
1001        size: &'a ExpandedSize,
1002        call_depth: usize,
1003    ) -> Self {
1004        Self {
1005            macros,
1006            name,
1007            params,
1008            size,
1009            call_depth,
1010            lambdas: LambdaScopes::default(),
1011            relations: Vec::new(),
1012            scoped_queries: PlHashMap::default(),
1013            narrowed_exprs: PlHashMap::default(),
1014            open_narrowed: vec![],
1015        }
1016    }
1017
1018    /// Whether a leading run of `idents` names a relation that the visited expression sees.
1019    fn names_relation(&self, idents: &[Ident]) -> bool {
1020        let mut hidden = false;
1021        for relations in self.relations.iter().rev() {
1022            if hidden {
1023                // The relations of an enclosing query can still be correlated, unless the
1024                // query is itself a CTE or derived table. A derived table still sees the
1025                // relations before it.
1026                if relations.is_query {
1027                    if match_qualifier(idents, relations.preceding.iter()).is_some() {
1028                        return true;
1029                    }
1030                    hidden = relations.starts_scope;
1031                }
1032                continue;
1033            }
1034            if match_qualifier(idents, relations.names.iter()).is_some()
1035                || match_qualifier(idents, relations.preceding.iter()).is_some()
1036            {
1037                return true;
1038            }
1039            hidden = relations.starts_scope;
1040        }
1041        false
1042    }
1043
1044    fn value(&self, name: &str) -> ControlFlow<PolarsError, Option<SQLExpr>> {
1045        let Some((_, value)) = self.params.iter().find(|(p, _)| same_name(p, name)) else {
1046            return ControlFlow::Continue(None);
1047        };
1048        // A column of the caller would be read from the tables of the subquery.
1049        if !self.relations.is_empty() && expr_references_any_column(value) {
1050            return ControlFlow::Break(polars_err!(
1051                SQLInterface: "macro '{}' uses parameter '{}' in a query, where its value cannot read columns",
1052                self.name, name
1053            ));
1054        }
1055        self.size.add(value)?;
1056        let mut value = parenthesize(value.clone());
1057        // The lambdas of the value are placed in the lambdas of the body around it.
1058        shift_lambda_elements(&mut value, self.call_depth, self.lambdas.depth);
1059        ControlFlow::Continue(Some(value))
1060    }
1061}
1062
1063/// A compound expression in parentheses, so that it keeps its meaning where it is placed.
1064fn parenthesize(expr: SQLExpr) -> SQLExpr {
1065    match expr {
1066        SQLExpr::Value(_)
1067        | SQLExpr::Interval(_)
1068        | SQLExpr::Identifier(_)
1069        | SQLExpr::CompoundIdentifier(_)
1070        | SQLExpr::Function(_)
1071        | SQLExpr::Nested(_)
1072        | SQLExpr::Subquery(_) => expr,
1073        _ => SQLExpr::Nested(Box::new(expr)),
1074    }
1075}

1077impl VisitorMut for ParamReplacer<'_> {
1078    type Break = PolarsError;
1079
1080    fn pre_visit_select(&mut self, select: &mut Select) -> ControlFlow<PolarsError> {
1081        // A SELECT item that is just a parameter keeps the parameter name.
1082        for item in &mut select.projection {
1083            if let SelectItem::UnnamedExpr(SQLExpr::Identifier(ident)) = item
1084                && self.params.iter().any(|(p, _)| same_name(p, &ident.value))
1085            {
1086                let alias = ident.clone();
1087                *item = SelectItem::ExprWithAlias {
1088                    expr: SQLExpr::Identifier(ident.clone()),
1089                    alias,
1090                };
1091            }
1092        }
1093        let mut preceding = Vec::new();
1094        let mut names = Vec::new();
1095        for table in &select.from {
1096            self.add_relations(table, &mut preceding);
1097            table_names(table, &mut names);
1098        }
1099        self.relations.push(Relations {
1100            names,
1101            starts_scope: false,
1102            preceding: vec![],
1103            is_query: false,
1104        });
1105        ControlFlow::Continue(())
1106    }
1107
1108    fn post_visit_select(&mut self, _select: &mut Select) -> ControlFlow<PolarsError> {
1109        self.relations.pop();
1110        ControlFlow::Continue(())
1111    }
1112
1113    fn pre_visit_query(&mut self, query: &mut Query) -> ControlFlow<PolarsError> {
1114        let preceding = self.scoped_queries.remove(&(&*query as *const Query));
1115        if let Some(with) = &query.with {
1116            self.scoped_queries.extend(
1117                with.cte_tables
1118                    .iter()
1119                    .map(|cte| (&*cte.query as *const Query, vec![])),
1120            );
1121        }
1122        let mut names = Vec::new();
1123        relation_names(&query.body, &mut names);
1124        self.relations.push(Relations {
1125            names,
1126            starts_scope: preceding.is_some(),
1127            preceding: preceding.unwrap_or_default(),
1128            is_query: true,
1129        });
1130        ControlFlow::Continue(())
1131    }
1132
1133    fn post_visit_query(&mut self, _query: &mut Query) -> ControlFlow<PolarsError> {
1134        self.relations.pop();
1135        ControlFlow::Continue(())
1136    }
1137
1138    fn pre_visit_table_factor(&mut self, factor: &mut TableFactor) -> ControlFlow<PolarsError> {
1139        if let TableFactor::Derived { subquery, .. } = factor {
1140            self.scoped_queries
1141                .entry(&**subquery as *const Query)
1142                .or_default();
1143        }
1144        // The conditions inside an aliased join see the relations that the alias hides, but
1145        // not the alias or the other relations of the FROM.
1146        if let Some(names) = hidden_by_alias(factor) {
1147            self.relations.push(Relations {
1148                names,
1149                starts_scope: true,
1150                preceding: vec![],
1151                is_query: false,
1152            });
1153        }
1154        ControlFlow::Continue(())
1155    }
1156
1157    fn pre_visit_expr(&mut self, expr: &mut SQLExpr) -> ControlFlow<PolarsError> {
1158        self.lambdas.enter(expr);
1159        // A join condition sees the relations of its join, not those joined after it. The
1160        // arguments of a table function see the relations before it.
1161        if let Some(names) = self.narrowed_exprs.remove(&(expr as *const SQLExpr)) {
1162            self.relations.push(Relations {
1163                names,
1164                starts_scope: true,
1165                preceding: vec![],
1166                is_query: false,
1167            });
1168            self.open_narrowed.push(expr as *const SQLExpr);
1169        }
1170        ControlFlow::Continue(())
1171    }
1172
1173    fn post_visit_expr(&mut self, expr: &mut SQLExpr) -> ControlFlow<PolarsError> {
1174        let ptr = expr as *const SQLExpr;
1175        let is_body = self.lambdas.leave_children(expr);
1176        self.replace_param(expr)?;
1177        self.lambdas.leave_body(is_body);
1178        if self.open_narrowed.last() == Some(&ptr) {
1179            self.open_narrowed.pop();
1180            self.relations.pop();
1181        }
1182        ControlFlow::Continue(())
1183    }
1184
1185    fn post_visit_table_factor(&mut self, factor: &mut TableFactor) -> ControlFlow<PolarsError> {
1186        if hidden_by_alias(factor).is_some() {
1187            self.relations.pop();
1188        }
1189        unparenthesize_table_args(factor, self.macros);
1190        ControlFlow::Continue(())
1191    }
1192}

1194impl ParamReplacer<'_> {
1195    /// Adds the names of the relations of `table` to `names`, the relations before it. Records
1196    /// the relations before each derived table, which it reads as if LATERAL.
1197    /// Records the relations each join condition sees.
1198    fn add_relations(&mut self, table: &TableWithJoins, names: &mut Vec<RelationName>) {
1199        let start = names.len();
1200        let mut joined = Vec::new();
1201        self.add_factor(&table.relation, names, &mut joined);
1202        for join in &table.joins {
1203            let (left_names, left_joined) = (names.len(), joined.len());
1204            self.add_factor(&join.relation, names, &mut joined);
1205            for condition in root_exprs(&join.join_operator) {
1206                self.narrowed_exprs.insert(condition, joined.clone());
1207            }
1208            // After a semi or anti join, the relations of its dropped input are gone.
1209            let (keep_left, keep_right) = join_keeps_inputs(&join.join_operator);
1210            if !keep_right {
1211                names.truncate(left_names);
1212                joined.truncate(left_joined);
1213            }
1214            if !keep_left {
1215                names.drain(start..left_names);
1216                joined.drain(..left_joined);
1217            }
1218        }
1219    }
1220
1221    /// As `add_relations`, for `factor`, which also adds its names to `joined`.
1222    fn add_factor(
1223        &mut self,
1224        factor: &TableFactor,
1225        names: &mut Vec<RelationName>,
1226        joined: &mut Vec<RelationName>,
1227    ) {
1228        factor_names(factor, joined);
1229        match factor {
1230            TableFactor::Derived { subquery, .. } => {
1231                self.scoped_queries
1232                    .insert(&**subquery as *const Query, names.clone());
1233            },
1234            TableFactor::NestedJoin {
1235                table_with_joins, ..
1236            } => self.add_relations(table_with_joins, &mut names.clone()),
1237            TableFactor::Pivot { table, .. } | TableFactor::Unpivot { table, .. } => {
1238                self.add_factor(table, &mut names.clone(), &mut vec![])
1239            },
1240            TableFactor::Table { args: Some(_), .. }
1241            | TableFactor::Function { .. }
1242            | TableFactor::TableFunction { .. }
1243            | TableFactor::UNNEST { .. } => {
1244                for arg in root_exprs(factor) {
1245                    self.narrowed_exprs.insert(arg, names.clone());
1246                }
1247            },
1248            _ => {},
1249        }
1250        factor_names(factor, names);
1251    }
1252
1253    fn replace_param(&mut self, expr: &mut SQLExpr) -> ControlFlow<PolarsError> {
1254        match expr {
1255            SQLExpr::Identifier(ident) => {
1256                if let Some(value) = self.value(&ident.value)? {
1257                    *expr = value;
1258                }
1259            },
1260            // A qualifier naming a relation of the body reads that relation.
1261            SQLExpr::CompoundIdentifier(idents) if self.names_relation(idents) => {},
1262            SQLExpr::CompoundIdentifier(idents) => {
1263                if let Some(value) = self.value(&idents[0].value)? {
1264                    // The fields of the value, also when it names a column like a table.
1265                    *expr = field_access(SQLExpr::Nested(Box::new(value)), &idents[1..]);
1266                }
1267            },
1268            _ => unparenthesize_operands(expr, self.macros),
1269        }
1270        ControlFlow::Continue(())
1271    }
1272}

1274/// The names of the relations inside `factor` that its alias hides.
1275fn hidden_by_alias(factor: &TableFactor) -> Option<Vec<RelationName>> {
1276    let mut names = Vec::new();
1277    match factor {
1278        TableFactor::NestedJoin {
1279            table_with_joins,
1280            alias: Some(_),
1281        } => table_names(table_with_joins, &mut names),
1282        TableFactor::Pivot {
1283            table,
1284            alias: Some(_),
1285            ..
1286        }
1287        | TableFactor::Unpivot {
1288            table,
1289            alias: Some(_),
1290            ..
1291        } => factor_names(table, &mut names),
1292        _ => return None,
1293    }
1294    Some(names)
1295}

1297/// Adds the names of the relations that the ORDER BY of a query with this body can qualify
1298/// columns with to `names`. Each SELECT of a set
1299/// operation has its own relations.
1300fn relation_names(body: &SetExpr, names: &mut Vec<RelationName>) {
1301    match body {
1302        SetExpr::Select(select) => {
1303            for table in &select.from {
1304                table_names(table, names);
1305            }
1306        },
1307        SetExpr::Query(query) => relation_names(&query.body, names),
1308        _ => {},
1309    }
1310}

1319/// The outermost expressions of `node`.
1320fn root_exprs<T: Visit>(node: &T) -> Vec<*const SQLExpr> {
1321    struct Roots {
1322        depth: usize,
1323        roots: Vec<*const SQLExpr>,
1324    }
1325    impl Visitor for Roots {
1326        type Break = ();
1327
1328        fn pre_visit_expr(&mut self, expr: &SQLExpr) -> ControlFlow<()> {
1329            if self.depth == 0 {
1330                self.roots.push(expr);
1331            }
1332            self.depth += 1;
1333            ControlFlow::Continue(())
1334        }
1335
1336        fn post_visit_expr(&mut self, _expr: &SQLExpr) -> ControlFlow<()> {
1337            self.depth -= 1;
1338            ControlFlow::Continue(())
1339        }
1340    }
1341    let mut roots = Roots {
1342        depth: 0,
1343        roots: vec![],
1344    };
1345    let _ = dyn_visit::visit(node, &mut roots);
1346    roots.roots
1347}

1349fn factor_names(factor: &TableFactor, names: &mut Vec<RelationName>) {
1350    match factor {
1351        // An alias hides the names of the relations inside.
1352        TableFactor::NestedJoin {
1353            table_with_joins,
1354            alias: None,
1355        } => table_names(table_with_joins, names),
1356        TableFactor::Pivot {
1357            table, alias: None, ..
1358        }
1359        | TableFactor::Unpivot {
1360            table, alias: None, ..
1361        } => factor_names(table, names),
1362        TableFactor::Function { name, alias, .. } => names.push(RelationName::new(match alias {
1363            Some(alias) => vec![alias.name.value.clone()],
1364            None => name
1365                .0
1366                .iter()
1367                .filter_map(|p| Some(p.as_ident()?.value.clone()))
1368                .collect(),
1369        })),
1370        _ => {},
1371    }
1372    names.extend(relation_name(factor));
1373}

1375/// Drops the parentheses around function arguments. A substituted value keeps its
1376/// parentheses only where it is an operand.
1377fn unparenthesize_operands(expr: &mut SQLExpr, macros: &Macros) {
1378    match expr {
1379        // The value of a macro argument keeps its meaning, as a number that is no position.
1380        SQLExpr::Function(function) if macros.get(&function.name, false).is_some() => {},
1381        SQLExpr::Cast { kind, expr, .. } if *kind != CastKind::DoubleColon => unparenthesize(expr),
1382        SQLExpr::Interval(interval)
1383            if matches!(strip_nested(&interval.value), SQLExpr::Value(_)) =>
1384        {
1385            unparenthesize(&mut interval.value)
1386        },
1387        SQLExpr::Function(function) => {
1388            if let FunctionArguments::List(list) = &mut function.args {
1389                unparenthesize_args(&mut list.args);
1390            }
1391        },
1392        _ => {},
1393    }
1394}

1396/// Drops the parentheses around the arguments of a table function or `UNNEST`.
1397fn unparenthesize_table_args(factor: &mut TableFactor, macros: &Macros) {
1398    match factor {
1399        TableFactor::Table { name, .. } if macros.get(name, true).is_some() => {},
1400        TableFactor::Table {
1401            args: Some(args), ..
1402        } => unparenthesize_args(&mut args.args),
1403        TableFactor::UNNEST { array_exprs, .. } => array_exprs.iter_mut().for_each(unparenthesize),
1404        _ => {},
1405    }
1406}
1407
1408fn unparenthesize_args(args: &mut [FunctionArg]) {
1409    for arg in args {
1410        if let FunctionArg::Unnamed(FunctionArgExpr::Expr(expr))
1411        | FunctionArg::Named {
1412            arg: FunctionArgExpr::Expr(expr),
1413            ..
1414        }
1415        | FunctionArg::ExprNamed {
1416            arg: FunctionArgExpr::Expr(expr),
1417            ..
1418        } = arg
1419        {
1420            unparenthesize(expr);
1421        }
1422    }
1423}
1424
1425fn unparenthesize(expr: &mut SQLExpr) {
1426    while let SQLExpr::Nested(inner) = expr {
1427        *expr = std::mem::replace(inner.as_mut(), SQLExpr::value(SQLValue::Null));
1428    }
