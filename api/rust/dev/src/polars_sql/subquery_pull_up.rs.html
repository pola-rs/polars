1//! Moves a subquery that reads a relation of a query further out than the query
2//! directly around it into the FROM of that query, which then reads the outer
3//! relation itself. The decorrelation in `subquery.rs` handles that query. A shape
4//! that isn't moved keeps its deeper references.
5//!
6//! The query `q` below and its subquery `s` read the outer relation `o`:
7//!
8//! ```sql
9//! SELECT .. FROM f WHERE f.x IN (SELECT g.y FROM g WHERE g.c = o.k AND <rest>)
10//! ```
11//!
12//! For a fixed row of `o`, `q` keeps the rows of `f` that have a match in `s`.
13//! Those are the rows of `f` joined with the distinct `(g.y, g.c)` pairs of
14//! `s` without the outer predicate, where the pair also matches `o.k`:
15//!
16//! ```sql
17//! SELECT .. FROM f, (SELECT DISTINCT g.y AS v, g.c AS k0 FROM g WHERE <rest>) AS n
18//! WHERE f.x = n.v AND n.k0 = o.k
19//! ```
20//!
21//! Each row of `f` matches at most one distinct pair, so `q` returns the same rows.
22//! The derived table is planned with `strict_relations`, so that it can only read
23//! the relations it declares itself.

51/// The columns of the frame a subquery is decorrelated against.
52#[derive(Clone, Copy)]
53pub(crate) enum ScopeColumns<'a> {
54    /// A SELECT without FROM, which declares nothing.
55    None,
56    Frame(&'a Schema),

61/// A query that a column of a subquery may read: its context resolves the qualifiers.
62#[derive(Clone, Copy)]
63struct Scope<'a> {

107impl SQLContext {
108    /// `subquery`, about to be decorrelated against `columns` of this context, with the
109    /// subqueries of its WHERE that read this context or further out moved into its FROM.
110    /// `None` if nothing moved.
111    pub(crate) fn pull_up_nested(
112        &self,
113        columns: ScopeColumns<'_>,
114        subquery: &Query,
115    ) -> Option<Query> {
116        // An error while preparing leaves the subquery unchanged.
117        prepare(&[Scope { ctx: self, columns }], subquery).unwrap_or(None)
118    }
119
120    /// `expr` with `pull_up_nested` applied to each of its subqueries, or `None` if nothing
121    /// moved.
122    pub(crate) fn pull_up_nested_in(
123        &self,
124        columns: ScopeColumns<'_>,
125        expr: &SQLExpr,
126    ) -> Option<SQLExpr> {
127        struct Prepare<'a, 'b> {
128            ctx: &'a SQLContext,
129            columns: ScopeColumns<'b>,
130            query_depth: usize,
131            prepared: Vec<Option<Query>>,
132        }
133
134        impl Visitor for Prepare<'_, '_> {
135            type Break = ();
136
137            fn pre_visit_query(&mut self, _query: &Query) -> ControlFlow<()> {
138                self.query_depth += 1;
139                ControlFlow::Continue(())
140            }
141
142            fn post_visit_query(&mut self, _query: &Query) -> ControlFlow<()> {
143                self.query_depth -= 1;
144                ControlFlow::Continue(())
145            }
146
147            fn pre_visit_expr(&mut self, expr: &SQLExpr) -> ControlFlow<()> {
148                if self.query_depth == 0
149                    && let Some(subquery) = expr_subquery(expr)
150                {
151                    let prepared = self.ctx.pull_up_nested(self.columns, subquery);
152                    self.prepared.push(prepared);
153                }
154                ControlFlow::Continue(())
155            }
156        }
157
158        struct Replace {
159            query_depth: usize,
160            prepared: std::vec::IntoIter<Option<Query>>,
161        }
162
163        impl VisitorMut for Replace {
164            type Break = ();
165
166            fn pre_visit_query(&mut self, _query: &mut Query) -> ControlFlow<()> {
167                self.query_depth += 1;
168                ControlFlow::Continue(())
169            }
170
171            fn post_visit_query(&mut self, _query: &mut Query) -> ControlFlow<()> {
172                self.query_depth -= 1;
173                ControlFlow::Continue(())
174            }
175
176            fn pre_visit_expr(&mut self, expr: &mut SQLExpr) -> ControlFlow<()> {
177                if self.query_depth == 0 && is_subquery_expr(expr) {
178                    let prepared = self.prepared.next().flatten();
179                    if let (Some(prepared), Some(subquery)) = (prepared, expr_subquery_mut(expr)) {
180                        *subquery = prepared;
181                    }
182                }
183                ControlFlow::Continue(())
184            }
185        }
186
187        if !expr_contains_subquery(expr) {
188            return None;
189        }
190        let mut prepare = Prepare {
191            ctx: self,
192            columns,
193            query_depth: 0,
194            prepared: Vec::new(),
195        };
196        let _ = dyn_visit::visit(expr, &mut prepare);
197        if prepare.prepared.iter().all(Option::is_none) {
198            return None;
199        }
200        let mut expr = expr.clone();
201        let _ = dyn_visit::visit_mut(
202            &mut expr,
203            &mut Replace {
204                query_depth: 0,
205                prepared: prepare.prepared.into_iter(),
206            },
207        );
208        Some(expr)
209    }
210}

212fn expr_subquery(expr: &SQLExpr) -> Option<&Query> {
213    match expr {
214        SQLExpr::Subquery(subquery)
215        | SQLExpr::Exists { subquery, .. }
216        | SQLExpr::InSubquery { subquery, .. } => Some(subquery),
217        _ => None,
218    }
219}

226        _ => None,
227    }

230/// `x` with the subqueries of its WHERE that read one of `scopes` (the queries around
231/// `x`, outermost first) moved into its FROM, or `None` if nothing moved. Inner queries
232/// are handled first.
233fn prepare(scopes: &[Scope<'_>], x: &Query) -> PolarsResult<Option<Query>> {
234    let Some(select) = eligible_query_block(x) else {
235        return Ok(None);
236    };
237    let Some(selection) = &select.selection else {
238        return Ok(None);
239    };
240    let conjuncts = flatten_and_conditions(selection);
241    if !conjuncts.iter().any(|conj| candidate(conj).is_some()) {
242        return Ok(None);
243    }
244    // The added columns are excluded from `SELECT *`, which can't be combined with
245    // these.
246    if select.projection.iter().any(|item| {
247        matches!(item, SelectItem::Wildcard(options) if options.opt_except.is_some()
248            || options.opt_ilike.is_some()
249            || options.opt_replace.is_some()
250            || options.opt_rename.is_some())
251    }) {
252        return Ok(None);
253    }
254    // The FROM of `x` is resolved as it is when `x` is planned. Only what moves into the
255    // derived tables is planned strictly, and so is everything that reads it here. A FROM
256    // without named relations lets a qualifier name any table outside strict planning.
257    let mut x_ctx = scopes.last().unwrap().ctx.isolated();
258    let x_frame = match select.from.is_empty() {
259        true => None,
260        false => {
261            let Some((names, lf, schema)) = x_ctx.resolve_subquery_from(select)? else {
262                return Ok(None);
263            };
264            if names.is_empty() {
265                return Ok(None);
266            }
267            Some((lf, schema))
268        },
269    };
270    x_ctx.strict_relations = true;
271
272    let mut kept = Vec::new();
273    let mut moved = Vec::new();
274    for conj in conjuncts {
275        let Some((compare, y)) = candidate(conj) else {
276            kept.push(conj.clone());
277            continue;
278        };
279        let x_frame = x_frame.as_ref().map(|(lf, schema)| (lf, schema.as_ref()));
280        let prepared_y = {
281            let mut inner = scopes.to_vec();
282            inner.push(Scope {
283                ctx: &x_ctx,
284                columns: frame_columns(x_frame),
285            });
286            prepare(&inner, y).unwrap_or(None)
287        };
288        let y = prepared_y.as_ref().unwrap_or(y);
289        match try_move(scopes, &mut x_ctx, x_frame, compare, y).unwrap_or(None) {
290            Some(m) => {
291                kept.extend(m.replaced);
292                moved.push((m.table, m.columns));
293            },
294            None => kept.push(conj.clone()),
295        }
296    }
297    if moved.is_empty() {
298        return Ok(None);
299    }
300
301    let mut select = select.clone();
302    // The added columns must not show up in `SELECT *`.
303    let added: Vec<ObjectName> = moved
304        .iter()
305        .flat_map(|(_, columns)| columns)
306        .map(|name| ObjectName::from(vec![Ident::new(name)]))
307        .collect();
308    for item in &mut select.projection {
309        if let SelectItem::Wildcard(options) = item {
310            let mut excluded = match options.opt_exclude.take() {
311                None => Vec::new(),
312                Some(ExcludeSelectItem::Single(name)) => vec![name],
313                Some(ExcludeSelectItem::Multiple(names)) => names,
314            };
315            excluded.extend(added.iter().cloned());
316            options.opt_exclude = Some(ExcludeSelectItem::Multiple(excluded));
317        }
318    }
319    select
320        .from
321        .extend(moved.into_iter().map(|(table, _)| table));
322    select.selection = combine_conditions(kept, SQLBinaryOperator::And);
323    let mut x = x.clone();
324    x.body = Box::new(SetExpr::Select(Box::new(select)));
325    Ok(Some(x))
326}

335/// The comparison and subquery of a conjunct `lhs IN (y)`, `lhs <op> (y)` or `EXISTS (y)`.
336fn candidate(conj: &SQLExpr) -> Option<(Option<Compare<'_>>, &Query)> {
337    use SQLBinaryOperator::*;
338    let (compare, subquery) = match strip_nested(conj) {
339        SQLExpr::InSubquery {
340            expr,
341            subquery,
342            negated: false,
343        } => (Some((expr.as_ref(), Eq, false)), subquery.as_ref()),
344        SQLExpr::Exists {
345            subquery,
346            negated: false,
347        } => (None, subquery.as_ref()),
348        SQLExpr::BinaryOp {
349            left,
350            op: op @ (Eq | NotEq | Lt | LtEq | Gt | GtEq),
351            right,
352        } => match (strip_nested(left), strip_nested(right)) {
353            (SQLExpr::Subquery(subquery), lhs) => {
354                (Some((lhs, op.clone(), true)), subquery.as_ref())
355            },
356            (lhs, SQLExpr::Subquery(subquery)) => {
357                (Some((lhs, op.clone(), false)), subquery.as_ref())
358            },
359            _ => return None,
360        },
361        _ => return None,
362    };
363    let lhs = compare.as_ref().map(|(lhs, ..)| *lhs);
364    if lhs.is_some_and(|lhs| expr_contains_subquery(lhs) || matches!(lhs, SQLExpr::Tuple(_))) {
365        return None;
366    }
367    Some((compare, subquery))
368}

370/// Move `y`, compared by `compare` (or under EXISTS), into the FROM of `x`, whose context
371/// is `x_ctx` and whose frame is `x_frame` (`None` without FROM). `scopes` are the queries
372/// around `x`. `None` if `y` reads none of `scopes`, or isn't a shape that moves.
373fn try_move(
374    scopes: &[Scope<'_>],
375    x_ctx: &mut SQLContext,
376    x_frame: Option<(&LazyFrame, &Schema)>,
377    compare: Option<Compare<'_>>,
378    y: &Query,
379) -> PolarsResult<Option<Moved>> {
380    use SQLBinaryOperator::*;
381    let Some(first_row) = first_row_limit(y) else {
382        return Ok(None);
383    };
384    let Some(select) = eligible_subquery_body(y) else {
385        return Ok(None);
386    };
387    if matches!(select.distinct, Some(Distinct::On(_))) || !select.named_window.is_empty() {
388        return Ok(None);
389    }
390    let order_by_exprs: &[OrderByExpr] = match y.order_by.as_ref().map(|o| &o.kind) {
391        None => &[],
392        Some(OrderByKind::Expressions(exprs)) => exprs,
393        Some(_) => return Ok(None),
394    };
395    let order_aggregates = order_by_exprs
396        .iter()
397        .any(|ob| has_aggregate(x_ctx, &ob.expr));
398    let value = match compare {
399        Some(_) => {
400            let [SelectItem::UnnamedExpr(value) | SelectItem::ExprWithAlias { expr: value, .. }] =
401                select.projection.as_slice()
402            else {
403                return Ok(None);
404            };
405            if expr_has_window_functions(value) {
406                return Ok(None);
407            }
408            Some(value)
409        },
410        // An aggregate makes EXISTS return one row whatever the rows are. The projection
411        // and the order are otherwise ignored.
412        None if order_aggregates
413            || select
414                .projection
415                .iter()
416                .any(|item| has_aggregate(x_ctx, item)) =>
417        {
418            return Ok(None);
419        },
420        None => None,
421    };
422    // An aggregate is taken per value of the correlation keys. A key with no rows then
423    // has no group, which is only the same as a NULL value.
424    let aggregates = value.is_some_and(|value| has_aggregate(x_ctx, value));
425    if aggregates && !null_over_no_rows(x_ctx, value.unwrap()) {
426        return Ok(None);
427    }
428    // An aggregate in the order alone also makes the subquery return one row.
429    if value.is_some() && !aggregates && order_aggregates {
430        return Ok(None);
431    }
432    // Without a limit, the distinct values of a key can be several. Only an equality
433    // matches each row of the query around it with at most one of them. A limit doesn't
434    // change whether EXISTS finds a row, nor how many rows an aggregate returns.
435    let first_row = first_row && value.is_some() && !aggregates;
436    if compare.as_ref().is_some_and(|(_, op, _)| *op != Eq) && !aggregates && !first_row {
437        return Ok(None);
438    }
439    let order_by = match first_row {
440        true => match first_row_order(y, select, order_by_exprs) {
441            Some(order_by) => order_by,
442            None => return Ok(None),
443        },
444        false => Vec::new(),
445    };
446
447    let mut y_ctx = x_ctx.isolated();
448    let Some((y_names, _, y_schema)) = y_ctx.resolve_subquery_from(select)? else {
449        return Ok(None);
450    };
451    // Dropping the projection of EXISTS must not change its rows, as `UNNEST` does.
452    if compare.is_none() && exists_aggregates(&y_ctx, y, select, &y_names, &y_schema) != Some(false)
453    {
454        return Ok(None);
455    }
456    let mut kept = Vec::new();
457    let mut keys = Vec::new();
458    let mut reads_outer = false;
459    if let Some(selection) = &select.selection {
460        // Innermost first: `y`, `x`, then the queries around `x`.
461        let chain: Vec<Scope> = [
462            Scope {
463                ctx: &y_ctx,
464                columns: ScopeColumns::Frame(&y_schema),
465            },
466            Scope {
467                ctx: &*x_ctx,
468                columns: frame_columns(x_frame),
469            },
470        ]
471        .into_iter()
472        .chain(scopes.iter().rev().copied())
473        .collect();
474        // A conjunct with a subquery stays: planning the derived table rejects it if it
475        // reads anything around `y`.
476        for c in flatten_and_conditions(selection) {
477            if expr_contains_subquery(c) || reads_local_only(&chain, c)? == Some(true) {
478                kept.push(c.clone());
479                continue;
480            }
481            let Some((key, depth)) = correlation_key(&chain, c)? else {
482                return Ok(None);
483            };
484            reads_outer |= depth >= 2;
485            keys.push(key);
486        }
487    }
488    if !reads_outer {
489        return Ok(None);
490    }
491
492    let name = format!("{NESTED_PREFIX}{}", unique_column_name());
493    let value_name = format!("{name}_v");
494    let key_names: Vec<String> = (0..keys.len()).map(|i| format!("{name}_k{i}")).collect();
495    let mut pulled = select.clone();
496    pulled.projection = value
497        .map(|value| (value.clone(), value_name.clone()))
498        .into_iter()
499        .chain(
500            keys.iter()
501                .map(|k| k.inner.clone())
502                .zip(key_names.iter().cloned()),
503        )
504        .map(|(expr, alias)| SelectItem::ExprWithAlias {
505            expr,
506            alias: Ident::new(alias),
507        })
508        .collect();
509    pulled.selection = combine_conditions(kept, And);
510    let partition_by: Vec<SQLExpr> = keys.iter().map(|k| k.inner.clone()).collect();
511    let distinct = !aggregates && !first_row;
512    if aggregates {
513        pulled.group_by = GroupByExpr::Expressions(partition_by, vec![]);
514    } else if first_row {
515        pulled.distinct = None;
516        pulled.qualify = Some(SQLExpr::BinaryOp {
517            left: Box::new(row_number(partition_by, order_by)),
518            op: Eq,
519            right: Box::new(SQLExpr::value(SQLValue::Number("1".to_string(), false))),
520        });
521    } else {
522        pulled.distinct = Some(Distinct::Distinct);
523    }
524    let mut pulled_query = y.clone();
525    pulled_query.body = Box::new(SetExpr::Select(Box::new(pulled)));
526    pulled_query.order_by = None;
527    pulled_query.limit_clause = None;
528    pulled_query.fetch = None;
529
530    // Planning the derived table as the FROM will (strict, as `x_ctx` is) shows that it
531    // reads only what it declares, and gives the dtypes of its columns.
532    let mut pulled_ctx = x_ctx.isolated();
533    let mut pulled_lf = pulled_ctx.execute_query(&pulled_query)?;
534    let pulled_schema = pulled_ctx.get_frame_schema(&mut pulled_lf)?;
535    // Deduplicating or grouping keeps one row per value that the comparison tells apart
536    // only when the comparison doesn't merge values of the derived table.
537    for (key, key_name) in keys.iter().zip(&key_names) {
538        let Some(dtype) = pulled_schema.get(key_name.as_str()) else {
539            return Ok(None);
540        };
541        if !compares_exactly(dtype, &key.dtype) {
542            return Ok(None);
543        }
544    }
545    if distinct && let Some((lhs, ..)) = compare {
546        let Some(value_dtype) = pulled_schema.get(value_name.as_str()) else {
547            return Ok(None);
548        };
549        let Some(lhs_dtype) = local_dtype(scopes, x_ctx, x_frame, lhs)? else {
550            return Ok(None);
551        };
552        if !compares_exactly(value_dtype, &lhs_dtype) {
553            return Ok(None);
554        }
555    }
556
557    let table = TableWithJoins {
558        relation: TableFactor::Derived {
559            lateral: false,
560            subquery: Box::new(pulled_query),
561            alias: Some(TableAlias {
562                explicit: true,
563                name: Ident::new(&name),
564                columns: vec![],
565                at: None,
566            }),
567            sample: None,
568        },
569        joins: vec![],
570    };
571    let column =
572        |column: &str| SQLExpr::CompoundIdentifier(vec![Ident::new(&name), Ident::new(column)]);
573    let binary = |left: SQLExpr, op: SQLBinaryOperator, right: SQLExpr| SQLExpr::BinaryOp {
574        left: Box::new(left),
575        op,
576        right: Box::new(right),
577    };
578    let replaced = compare
579        .map(|(lhs, op, subquery_left)| match subquery_left {
580            true => binary(column(&value_name), op, lhs.clone()),
581            false => binary(lhs.clone(), op, column(&value_name)),
582        })
583        .into_iter()
584        .chain(
585            keys.into_iter()
586                .zip(&key_names)
587                .map(|(key, key_name)| binary(column(key_name), Eq, key.foreign)),
588        )
589        .collect();
590    let columns = value
591        .map(|_| value_name)
592        .into_iter()
593        .chain(key_names)
594        .collect();
595    Ok(Some(Moved {
596        table,
597        columns,
598        replaced,
599    }))
600}

602/// The `ORDER BY` of a first-row subquery, to order the rows of each key by. `None` for an
603/// order that isn't read from the row alone.
604fn first_row_order(y: &Query, select: &Select, exprs: &[OrderByExpr]) -> Option<Vec<OrderByExpr>> {
605    if y.order_by.as_ref().is_some_and(|o| o.interpolate.is_some()) {
606        return None;
607    }
608    let alias = match &select.projection[0] {
609        SelectItem::ExprWithAlias { alias, .. } => Some(alias),
610        _ => None,
611    };
612    // An order by the output column, by name or by position, isn't kept. A window
613    // function would see the rows of the other keys.
614    let unsupported = |e: &SQLExpr| match strip_nested(e) {
615        SQLExpr::Identifier(ident) => alias.is_some_and(|a| a.value == ident.value),
616        SQLExpr::Value(_) => true,
617        e => expr_has_window_functions(e),
618    };
619    if exprs
620        .iter()
621        .any(|ob| ob.with_fill.is_some() || unsupported(&ob.expr))
622    {
623        return None;
624    }
625    Some(exprs.to_vec())
626}

628/// Call `f` on each column that `expr` reads outside of its subqueries.
629fn visit_columns<E>(expr: &SQLExpr, f: impl FnMut(&SQLExpr) -> ControlFlow<E>) -> ControlFlow<E> {
630    struct Columns<F> {
631        query_depth: usize,
632        f: F,
633    }
634
635    impl<E, F: FnMut(&SQLExpr) -> ControlFlow<E>> Visitor for Columns<F> {
636        type Break = E;
637
638        fn pre_visit_query(&mut self, _query: &Query) -> ControlFlow<E> {
639            self.query_depth += 1;
640            ControlFlow::Continue(())
641        }
642
643        fn post_visit_query(&mut self, _query: &Query) -> ControlFlow<E> {
644            self.query_depth -= 1;
645            ControlFlow::Continue(())
646        }
647
648        fn pre_visit_expr(&mut self, expr: &SQLExpr) -> ControlFlow<E> {
649            match expr {
650                SQLExpr::Identifier(_) | SQLExpr::CompoundIdentifier(_)
651                    if self.query_depth == 0 =>
652                {
653                    (self.f)(expr)
654                },
655                _ => ControlFlow::Continue(()),
656            }
657        }
658    }
659
660    dyn_visit::visit(expr, &mut Columns { query_depth: 0, f })
661}

663/// The scope in `chain` (innermost first) that a column binds to: the first one with a
664/// relation the qualifier names, or with a struct column of the qualifier's name. An
665/// unqualified column, and one no scope binds, is left to the innermost scope. `None` for
666/// a column that its relation doesn't have.
667fn bind(chain: &[Scope<'_>], column: &SQLExpr) -> PolarsResult<Option<Binding>> {
668    let SQLExpr::CompoundIdentifier(parts) = column else {
669        return Ok(Some(Binding::Local));
670    };
671    for (depth, scope) in chain.iter().enumerate() {
672        let Some(found) = scope.find(parts)? else {
673            continue;
674        };
675        return Ok(match (found, depth) {
676            (Found::Missing, _) => None,
677            (_, 0) => Some(Binding::Local),
678            (Found::Column(dtype), depth) => Some(Binding::Foreign {
679                depth,
680                dtype: Some(dtype),
681            }),
682            (Found::Field, depth) => Some(Binding::Foreign { depth, dtype: None }),
683        });
684    }
685    Ok(Some(Binding::Local))
686}
687
688impl Scope<'_> {
689    fn find(&self, parts: &[Ident]) -> PolarsResult<Option<Found>> {
690        if matches!(self.columns, ScopeColumns::None) {
691            return Ok(None);
692        }
693        if let Some((relation, len)) = self.ctx.resolve_qualifier(parts)? {
694            if len + 1 != parts.len() {
695                return Ok(Some(Found::Field));
696            }
697            return Ok(Some(
698                match self.column_dtype(&relation, &parts[len].value) {
699                    Some(dtype) => Found::Column(dtype),
700                    None => Found::Missing,
701                },
702            ));
703        }
704        let is_struct =
705            |schema: &Schema| matches!(schema.get(&parts[0].value), Some(DataType::Struct(_)));
706        let has_struct = match self.columns {
707            ScopeColumns::None => false,
708            ScopeColumns::Frame(schema) => is_struct(schema),
709            ScopeColumns::Join(left, right) => is_struct(&left.schema) || is_struct(&right.schema),
710        };
711        Ok(has_struct.then_some(Found::Field))
712    }
713
714    fn column_dtype(&self, relation: &str, column: &str) -> Option<DataType> {
715        let schema = match self.columns {
716            ScopeColumns::None => return None,
717            ScopeColumns::Frame(schema) => schema,
718            ScopeColumns::Join(left, right) => {
719                if self.ctx.is_right_relation(relation, right) {
720                    return right.schema.get(column).cloned();
721                }
722                if !self.ctx.is_joined_relation(relation, left) {
723                    return None;
724                }
725                &left.schema
726            },
727        };
728        schema
729            .get(&self.ctx.resolve_name(relation, column))
730            .cloned()
731    }
732}

734/// Whether every column `expr` reads binds to the innermost scope of `chain`, or `None`
735/// for a column that its relation doesn't have.
736fn reads_local_only(chain: &[Scope<'_>], expr: &SQLExpr) -> PolarsResult<Option<bool>> {
737    let mut local = true;
738    let flow = visit_columns(expr, |column| match bind(chain, column) {
739        Ok(Some(Binding::Local)) => ControlFlow::Continue(()),
740        Ok(Some(Binding::Foreign { .. })) => {
741            local = false;
742            ControlFlow::Continue(())
743        },
744        Ok(None) => ControlFlow::Break(None),
745        Err(e) => ControlFlow::Break(Some(e)),
746    });
747    match flow {
748        ControlFlow::Break(Some(e)) => Err(e),
749        ControlFlow::Break(None) => Ok(None),
750        ControlFlow::Continue(()) => Ok(Some(local)),
751    }
752}

754/// `conj` as an equality between an expression over the subquery alone and a column of
755/// a query around it, with the depth of that query in `chain`.
756fn correlation_key(
757    chain: &[Scope<'_>],
758    conj: &SQLExpr,
759) -> PolarsResult<Option<(CorrelationKey, usize)>> {
760    let SQLExpr::BinaryOp {
761        left,
762        op: SQLBinaryOperator::Eq,
763        right,
764    } = strip_nested(conj)
765    else {
766        return Ok(None);
767    };
768    for (inner, foreign) in [(left, right), (right, left)] {
769        let foreign = strip_nested(foreign);
770        if expr_has_window_functions(inner)
771            || has_aggregate(chain[0].ctx, inner.as_ref())
772            || !expr_references_any_column(inner)
773            || reads_local_only(chain, inner)? != Some(true)
774        {
775            continue;
776        }
777        if let Some(Binding::Foreign {
778            depth,
779            dtype: Some(dtype),
780        }) = bind(chain, foreign)?
781        {
782            let key = CorrelationKey {
783                inner: inner.as_ref().clone(),
784                foreign: foreign.clone(),
785                dtype,
786            };
787            return Ok(Some((key, depth)));
788        }
789    }
790    Ok(None)
791}

793/// The dtype of `expr` over the frame of `x`, or `None` if it reads anything else.
794fn local_dtype(
795    scopes: &[Scope<'_>],
796    x_ctx: &mut SQLContext,
797    x_frame: Option<(&LazyFrame, &Schema)>,
798    expr: &SQLExpr,
799) -> PolarsResult<Option<DataType>> {
800    let Some((lf, schema)) = x_frame else {
801        return Ok(None);
802    };
803    let local = {
804        let chain: Vec<Scope> = std::iter::once(Scope {
805            ctx: &*x_ctx,
806            columns: ScopeColumns::Frame(schema),
807        })
808        .chain(scopes.iter().rev().copied())
809        .collect();
810        reads_local_only(&chain, expr)?
811    };
812    if local != Some(true) {
813        return Ok(None);
814    }
815    let parsed = parse_sql_expr(expr, x_ctx, Some(schema))?;
816    let mut probe = lf.clone().select([parsed]);
817    let probe_schema = x_ctx.get_frame_schema(&mut probe)?;
818    Ok(probe_schema.get_at_index(0).map(|(_, dtype)| dtype.clone()))
819}

821/// Whether comparing a value of `inner` with one of `other` casts it without merging
822/// values, so that distinct values of `inner` never equal the same value of `other`.
823pub(crate) fn compares_exactly(inner: &DataType, other: &DataType) -> bool {
824    if inner == other {
825        return true;
826    }
827    get_supertype(inner, other).is_some_and(|to| holds_values(inner, &to))
828}

830/// Whether `to` holds every value of the numeric dtype `from`.
831fn holds_values(from: &DataType, to: &DataType) -> bool {
832    let bits = |dtype: &DataType| match dtype {
833        DataType::Int8 | DataType::UInt8 => 8,
834        DataType::Int16 | DataType::UInt16 => 16,
835        DataType::Int32 | DataType::UInt32 | DataType::Float32 => 32,
836        DataType::Int64 | DataType::UInt64 | DataType::Float64 => 64,
837        DataType::Int128 | DataType::UInt128 => 128,
838        _ => 0,
839    };
840    match (from, to) {
841        _ if from == to => true,
842        (DataType::Float32, DataType::Float64) => true,
843        // The mantissa holds every integer of up to 24 or 53 bits.
844        (from, DataType::Float32) if from.is_integer() => bits(from) <= 16,
845        (from, DataType::Float64) if from.is_integer() => bits(from) <= 32,
846        (from, to) if from.is_integer() && to.is_integer() => {
847            match (from.is_signed_integer(), to.is_signed_integer()) {
848                (true, true) | (false, false) => bits(from) <= bits(to),
849                (false, true) => bits(from) < bits(to),
850                (true, false) => false,
851            }
852        },
853        _ => false,
854    }
855}

857/// Whether `node` calls an aggregate outside of its subqueries. A function that isn't
858/// known counts as one.
859pub(crate) fn has_aggregate(ctx: &SQLContext, node: &(impl Visit + ?Sized)) -> bool {
860    struct Aggregates<'a> {
861        ctx: &'a SQLContext,
862        query_depth: usize,
863    }
864
865    impl Visitor for Aggregates<'_> {
866        type Break = ();
867
868        fn pre_visit_query(&mut self, _query: &Query) -> ControlFlow<()> {
869            self.query_depth += 1;
870            ControlFlow::Continue(())
871        }
872
873        fn post_visit_query(&mut self, _query: &Query) -> ControlFlow<()> {
874            self.query_depth -= 1;
875            ControlFlow::Continue(())
876        }
877
878        fn pre_visit_expr(&mut self, expr: &SQLExpr) -> ControlFlow<()> {
879            match expr {
880                SQLExpr::Function(function) if self.query_depth == 0 && function.over.is_none() => {
881                    match PolarsSQLFunctions::try_from_sql(function, self.ctx) {
882                        Ok(PolarsSQLFunctions::Udf(_)) | Err(_) => ControlFlow::Break(()),
883                        Ok(f) if f.is_builtin_aggregate() => ControlFlow::Break(()),
884                        Ok(_) => ControlFlow::Continue(()),
885                    }
886                },
887                _ => ControlFlow::Continue(()),
888            }
889        }
890    }
891
892    dyn_visit::visit(
893        node,
894        &mut Aggregates {
895            ctx,
896            query_depth: 0,
897        },
898    )
899    .is_break()
900}

902/// Whether `expr` is NULL when it aggregates no rows: an aggregate other than a count,
903/// possibly cast, negated or combined with a constant by arithmetic.
904fn null_over_no_rows(ctx: &SQLContext, expr: &SQLExpr) -> bool {
905    use PolarsSQLFunctions::*;
906    match strip_nested(expr) {
907        SQLExpr::Function(function) if function.over.is_none() => matches!(
908            PolarsSQLFunctions::try_from_sql(function, ctx),
909            Ok(Avg
910                | Corr
911                | CovarPop
912                | CovarSamp
913                | First
914                | FirstValue
915                | Last
916                | Max
917                | Median
918                | Min
919                | QuantileCont
920                | QuantileDisc
921                | StdDev
922                | Sum
923                | Variance)
924        ),
925        SQLExpr::Cast { expr, .. }
926        | SQLExpr::UnaryOp {
927            op: SQLUnaryOperator::Minus,
928            expr,
929        } => null_over_no_rows(ctx, expr),
930        SQLExpr::BinaryOp {
931            left,
932            op:
933                SQLBinaryOperator::Plus
934                | SQLBinaryOperator::Minus
935                | SQLBinaryOperator::Multiply
936                | SQLBinaryOperator::Divide
937                | SQLBinaryOperator::Modulo,
938            right,
939        } => {
940            let constant = |e: &SQLExpr| !expr_references_any_column(e) && !has_aggregate(ctx, e);
941            (null_over_no_rows(ctx, left) && constant(right))
942                || (constant(left) && null_over_no_rows(ctx, right))
943        },
944        _ => false,
945    }
946}
