1//! Moves a subquery that reads a relation of a query further out than the query
2//! directly around it into the FROM of that query, which then reads the outer
3//! relation itself. The decorrelation in `subquery.rs` handles that query. A shape
4//! that isn't moved keeps its deeper references.
5//!
6//! The query `q` below and its subquery `s` read the outer relation `o`:
7//!
8//! ```sql
9//! SELECT .. FROM f WHERE f.x IN (SELECT g.y FROM g WHERE g.c = o.k AND <rest>)
10//! ```
11//!
12//! For a fixed row of `o`, `q` keeps the rows of `f` that have a match in `s`.
13//! Those are the rows of `f` joined with the distinct `(g.y, g.c)` pairs of
14//! `s` without the outer predicate, where the pair also matches `o.k`:
15//!
16//! ```sql
17//! SELECT .. FROM f, (SELECT DISTINCT g.y AS v, g.c AS k0 FROM g WHERE <rest>) AS n
18//! WHERE f.x = n.v AND n.k0 = o.k
19//! ```
20//!
21//! Each row of `f` matches at most one distinct pair, so `q` returns the same rows.
22//! The derived table is planned with `strict_relations`, so that it can only read
23//! the relations it declares itself.

55/// How the query around a subquery reads the rows of the subquery.
56#[derive(Clone, Copy)]
57pub(crate) enum ReadAs {

59    Set,
60    /// The value of its row, as for a scalar subquery.

122impl SQLContext {
123    /// `subquery`, about to be decorrelated against `columns` of this context, with the
124    /// subqueries of its WHERE that read this context or further out moved into its FROM.
125    /// `None` if nothing moved.
126    pub(crate) fn pull_up_nested(
127        &self,
128        columns: ScopeColumns<'_>,
129        subquery: &Query,
130        read_as: ReadAs,
131    ) -> Option<Query> {
132        // An error while preparing leaves the subquery unchanged.
133        prepare(&[Scope { ctx: self, columns }], subquery, read_as).unwrap_or(None)
134    }
135
136    /// `expr` with `pull_up_nested` applied to each of its subqueries, or `None` if nothing
137    /// moved.
138    pub(crate) fn pull_up_nested_in(
139        &self,
140        columns: ScopeColumns<'_>,
141        expr: &SQLExpr,
142    ) -> Option<SQLExpr> {
143        struct Prepare<'a, 'b> {
144            ctx: &'a SQLContext,
145            columns: ScopeColumns<'b>,
146            query_depth: usize,
147            prepared: Vec<Option<Query>>,
148        }
149
150        impl Visitor for Prepare<'_, '_> {
151            type Break = ();
152
153            fn pre_visit_query(&mut self, _query: &Query) -> ControlFlow<()> {
154                self.query_depth += 1;
155                ControlFlow::Continue(())
156            }
157
158            fn post_visit_query(&mut self, _query: &Query) -> ControlFlow<()> {
159                self.query_depth -= 1;
160                ControlFlow::Continue(())
161            }
162
163            fn pre_visit_expr(&mut self, expr: &SQLExpr) -> ControlFlow<()> {
164                if self.query_depth == 0
165                    && let Some(subquery) = expr_subquery(expr)
166                {
167                    let read_as = match expr {
168                        SQLExpr::Subquery(_) => ReadAs::Value,
169                        _ => ReadAs::Set,
170                    };
171                    let prepared = self.ctx.pull_up_nested(self.columns, subquery, read_as);
172                    self.prepared.push(prepared);
173                }
174                ControlFlow::Continue(())
175            }
176        }
177
178        struct Replace {
179            query_depth: usize,
180            prepared: std::vec::IntoIter<Option<Query>>,
181        }
182
183        impl VisitorMut for Replace {
184            type Break = ();
185
186            fn pre_visit_query(&mut self, _query: &mut Query) -> ControlFlow<()> {
187                self.query_depth += 1;
188                ControlFlow::Continue(())
189            }
190
191            fn post_visit_query(&mut self, _query: &mut Query) -> ControlFlow<()> {
192                self.query_depth -= 1;
193                ControlFlow::Continue(())
194            }
195
196            fn pre_visit_expr(&mut self, expr: &mut SQLExpr) -> ControlFlow<()> {
197                if self.query_depth == 0 && is_subquery_expr(expr) {
198                    let prepared = self.prepared.next().flatten();
199                    if let (Some(prepared), Some(subquery)) = (prepared, expr_subquery_mut(expr)) {
200                        *subquery = prepared;
201                    }
202                }
203                ControlFlow::Continue(())
204            }
205        }
206
207        if !expr_contains_subquery(expr) {
208            return None;
209        }
210        let mut prepare = Prepare {
211            ctx: self,
212            columns,
213            query_depth: 0,
214            prepared: Vec::new(),
215        };
216        let _ = dyn_visit::visit(expr, &mut prepare);
217        if prepare.prepared.iter().all(Option::is_none) {
218            return None;
219        }
220        let mut expr = expr.clone();
221        let _ = dyn_visit::visit_mut(
222            &mut expr,
223            &mut Replace {
224                query_depth: 0,
225                prepared: prepare.prepared.into_iter(),
226            },
227        );
228        Some(expr)
229    }
230}

232fn expr_subquery(expr: &SQLExpr) -> Option<&Query> {
233    match expr {
234        SQLExpr::Subquery(subquery)
235        | SQLExpr::Exists { subquery, .. }
236        | SQLExpr::InSubquery { subquery, .. } => Some(subquery),
237        _ => None,
238    }
239}

241fn expr_subquery_mut(expr: &mut SQLExpr) -> Option<&mut Query> {
242    match expr {
243        SQLExpr::Subquery(subquery)
244        | SQLExpr::Exists { subquery, .. }
245        | SQLExpr::InSubquery { subquery, .. } => Some(subquery),
246        _ => None,
247    }

250/// `x` with the subqueries of its WHERE that read one of `scopes` (the queries around
251/// `x`, outermost first) moved into its FROM, or `None` if nothing moved. Inner queries
252/// are handled first. The conjuncts of an inner join condition filter the joined rows as
253/// those of the WHERE do, and their subqueries move as well.
254fn prepare(scopes: &[Scope<'_>], x: &Query, read_as: ReadAs) -> PolarsResult<Option<Query>> {
255    let Some(select) = eligible_query_block(x) else {
256        return Ok(None);
257    };
258    let (mut split_from, on_conjuncts) = match split_join_candidates(&select.from) {
259        Some((from, conjuncts)) => (Some(from), conjuncts),
260        None => (None, Vec::new()),
261    };
262    let inlined = select.selection.as_ref().and_then(inline_from_less);
263    let unchanged = || {
264        inlined
265            .clone()
266            .map(|selection| with_selection(x, select, Some(selection)))
267    };
268    let conjuncts = inlined
269        .as_ref()
270        .or(select.selection.as_ref())
271        .map(flatten_and_conditions)
272        .unwrap_or_default();
273    if on_conjuncts.is_empty()
274        && !conjuncts
275            .iter()
276            .any(|conj| candidate(conj).is_some() || expr_contains_scalar_subquery(conj))
277    {
278        return Ok(unchanged());
279    }
280    // The added columns are excluded from `SELECT *`, which can't be combined with
281    // these.
282    if select.projection.iter().any(|item| {
283        matches!(item, SelectItem::Wildcard(options) if options.opt_except.is_some()
284            || options.opt_ilike.is_some()
285            || options.opt_replace.is_some()
286            || options.opt_rename.is_some())
287    }) {
288        return Ok(unchanged());
289    }
290    // The FROM of `x` is resolved as it is when `x` is planned. Only what moves into the
291    // derived tables is planned strictly, and so is everything that reads it here. A FROM
292    // without named relations lets a qualifier name any table outside strict planning.
293    let mut x_ctx = scopes.last().unwrap().ctx.isolated();
294    let x_frame = match select.from.is_empty() {
295        true => None,
296        false => {
297            let resolved = match &split_from {
298                None => x_ctx.resolve_subquery_from(select)?,
299                Some(from) => {
300                    let mut stripped = select.clone();
301                    stripped.from = from.clone();
302                    x_ctx.resolve_subquery_from(&stripped)?
303                },
304            };
305            let Some((names, lf, schema)) = resolved else {
306                return Ok(unchanged());
307            };
308            if names.is_empty() {
309                return Ok(unchanged());
310            }
311            Some((lf, schema))
312        },
313    };
314    x_ctx.strict_relations = true;
315
316    let repeats_harmless = repeats_harmless(x, select, read_as);
317    let mut kept = Vec::new();
318    let mut moved = Vec::new();
319    let conjuncts = conjuncts
320        .into_iter()
321        .map(|conj| (conj, false))
322        .chain(on_conjuncts.iter().map(|conj| (conj, true)));
323    for (conj, in_join) in conjuncts {
324        let Some((compare, y)) = candidate(conj) else {
325            kept.push(conj.clone());
326            continue;
327        };
328        let x_frame = x_frame.as_ref().map(|(lf, schema)| (lf, schema.as_ref()));
329        let y_read_as = match strip_nested(conj) {
330            SQLExpr::Exists { .. } | SQLExpr::InSubquery { .. } => ReadAs::Set,
331            _ => ReadAs::Value,
332        };
333        let prepared_y = {
334            let mut inner = scopes.to_vec();
335            inner.push(Scope {
336                ctx: &x_ctx,
337                columns: frame_columns(x_frame),
338            });
339            prepare(&inner, y, y_read_as).unwrap_or(None)
340        };
341        let y = prepared_y.as_ref().unwrap_or(y);
342        // A filter can match a row of `x` with several values of a scalar `y`, which must
343        // instead raise.
344        let filters_allowed = repeats_harmless && matches!(y_read_as, ReadAs::Set);
345        match try_move(scopes, &mut x_ctx, x_frame, compare, y, filters_allowed).unwrap_or(None) {
346            Some(m) => {
347                kept.extend(m.replaced);
348                moved.push((m.table, m.columns));
349            },
350            None => match (in_join, &mut split_from) {
351                (true, Some(from)) => restore_join_conjunct(from, conj.clone()),
352                _ => kept.push(conj.clone()),
353            },
354        }
355    }
356    // A conjunct that reads both `x` and a query around it is decorrelated as a whole,
357    // which a subquery in it prevents. An uncorrelated scalar subquery in it is read from
358    // a derived table of one row instead.
359    let x_frame = x_frame.as_ref().map(|(lf, schema)| (lf, schema.as_ref()));
360    let x_chain: Vec<Scope> = std::iter::once(Scope {
361        ctx: &x_ctx,
362        columns: frame_columns(x_frame),
363    })
364    .chain(scopes.iter().rev().copied())
365    .collect();
366    for conj in &mut kept {
367        if !expr_contains_scalar_subquery(conj) || !reads_local_and_foreign(&x_chain, conj)? {
368            continue;
369        }
370        if let Some((hoisted, tables)) = hoist_uncorrelated(&x_ctx, conj) {
371            *conj = hoisted;
372            moved.extend(tables);
373        }
374    }
375    if moved.is_empty() {
376        return Ok(unchanged());
377    }
378
379    let mut select = select.clone();
380    // The added columns must not show up in `SELECT *`.
381    let added: Vec<ObjectName> = moved
382        .iter()
383        .flat_map(|(_, columns)| columns)
384        .map(|name| ObjectName::from(vec![Ident::new(name)]))
385        .collect();
386    for item in &mut select.projection {
387        if let SelectItem::Wildcard(options) = item {
388            let mut excluded = match options.opt_exclude.take() {
389                None => Vec::new(),
390                Some(ExcludeSelectItem::Single(name)) => vec![name],
391                Some(ExcludeSelectItem::Multiple(names)) => names,
392            };
393            excluded.extend(added.iter().cloned());
394            options.opt_exclude = Some(ExcludeSelectItem::Multiple(excluded));
395        }
396    }
397    if let Some(from) = split_from {
398        select.from = from;
399    }
400    select
401        .from
402        .extend(moved.into_iter().map(|(table, _)| table));
403    Ok(Some(with_selection(
404        x,
405        &select,
406        combine_conditions(kept, SQLBinaryOperator::And),
407    )))
408}

410/// `conj` with each of its scalar subqueries read from a derived table of one row that
411/// holds its value, and those tables with their column. `None` if a subquery reads a
412/// relation it doesn't declare, which `x_ctx` (strict) rejects.
413#[allow(clippy::type_complexity)]
414fn hoist_uncorrelated(
415    x_ctx: &SQLContext,
416    conj: &SQLExpr,
417) -> Option<(SQLExpr, Vec<(TableWithJoins, Vec<String>)>)> {
418    struct Hoist<'a> {
419        x_ctx: &'a SQLContext,
420        query_depth: usize,
421        tables: Vec<(TableWithJoins, Vec<String>)>,
422    }
423
424    impl VisitorMut for Hoist<'_> {
425        type Break = ();
426
427        fn pre_visit_query(&mut self, _query: &mut Query) -> ControlFlow<()> {
428            self.query_depth += 1;
429            ControlFlow::Continue(())
430        }
431
432        fn post_visit_query(&mut self, _query: &mut Query) -> ControlFlow<()> {
433            self.query_depth -= 1;
434            ControlFlow::Continue(())
435        }
436
437        fn pre_visit_expr(&mut self, expr: &mut SQLExpr) -> ControlFlow<()> {
438            if self.query_depth > 0 {
439                return ControlFlow::Continue(());
440            }
441            let SQLExpr::Subquery(subquery) = expr else {
442                return ControlFlow::Continue(());
443            };
444            let name = format!("{NESTED_PREFIX}{}", unique_column_name());
445            let value_name = format!("{name}_v");
446            let Ok(query) = select_value(SQLExpr::Subquery(subquery.clone()), &value_name) else {
447                return ControlFlow::Break(());
448            };
449            let mut ctx = self.x_ctx.isolated();
450            let planned = ctx
451                .execute_query(&query)
452                .and_then(|mut lf| ctx.get_frame_schema(&mut lf));
453            if planned.is_err() {
454                return ControlFlow::Break(());
455            }
456            *expr = SQLExpr::CompoundIdentifier(vec![Ident::new(&name), Ident::new(&value_name)]);
457            self.tables
458                .push((derived_table(query, &name), vec![value_name]));
459            ControlFlow::Continue(())
460        }
461    }
462
463    let mut conj = conj.clone();
464    let mut hoist = Hoist {
465        x_ctx,
466        query_depth: 0,
467        tables: Vec::new(),
468    };
469    if dyn_visit::visit_mut(&mut conj, &mut hoist).is_break() {
470        return None;
471    }
472    Some((conj, hoist.tables))
473}

475/// `SELECT <value> AS <alias>`.
476fn select_value(value: SQLExpr, alias: &str) -> PolarsResult<Query> {
477    let mut query = Parser::new(&GenericDialect)
478        .try_with_sql("SELECT NULL")
479        .and_then(|mut parser| parser.parse_query())
480        .map_err(to_sql_interface_err)?;
481    let SetExpr::Select(select) = query.body.as_mut() else {
482        unreachable!()
483    };
484    select.projection = vec![SelectItem::ExprWithAlias {
485        expr: value,
486        alias: Ident::new(alias),
487    }];
488    Ok(*query)
489}

509/// `from` without the conjuncts of the condition of its last join that hold a subquery,
510/// and those conjuncts, or `None` if there are none. Only a single FROM item whose joins
511/// are all inner gives them up: they then filter the rows as WHERE does, and read the
512/// same relations.
513#[allow(clippy::type_complexity)]
514fn split_join_candidates(from: &[TableWithJoins]) -> Option<(Vec<TableWithJoins>, Vec<SQLExpr>)> {
515    let [item] = from else {
516        return None;
517    };
518    let all_inner = item.joins.iter().all(|join| {
519        matches!(
520            join.join_operator,
521            JoinOperator::Join(_) | JoinOperator::Inner(_) | JoinOperator::CrossJoin(_)
522        )
523    });
524    let (JoinOperator::Join(JoinConstraint::On(on)) | JoinOperator::Inner(JoinConstraint::On(on))) =
525        &item.joins.last()?.join_operator
526    else {
527        return None;
528    };
529    if !all_inner {
530        return None;
531    }
532    let (moving, rest): (Vec<&SQLExpr>, Vec<&SQLExpr>) = flatten_and_conditions(on)
533        .into_iter()
534        .partition(|conj| candidate(conj).is_some());
535    if moving.is_empty() {
536        return None;
537    }
538    let moving = moving.into_iter().cloned().collect();
539    let rest = combine_conditions(rest.into_iter().cloned().collect(), SQLBinaryOperator::And);
540    let mut from = from.to_vec();
541    from[0].joins.last_mut().unwrap().join_operator = match rest {
542        Some(on) => JoinOperator::Inner(JoinConstraint::On(on)),
543        None => JoinOperator::CrossJoin(JoinConstraint::None),
544    };
545    Some((from, moving))
546}

548/// Add `conj` back to the condition of the last join of `from`.
549fn restore_join_conjunct(from: &mut [TableWithJoins], conj: SQLExpr) {
550    let join = from[0].joins.last_mut().unwrap();
551    let on = match std::mem::replace(
552        &mut join.join_operator,
553        JoinOperator::CrossJoin(JoinConstraint::None),
554    ) {
555        JoinOperator::Inner(JoinConstraint::On(on)) => SQLExpr::BinaryOp {
556            left: Box::new(on),
557            op: SQLBinaryOperator::And,
558            right: Box::new(conj),
559        },
560        _ => conj,
561    };
562    join.join_operator = JoinOperator::Inner(JoinConstraint::On(on));
563}

565/// `x`, whose body is `select`, with `selection` as the WHERE of `select`.
566fn with_selection(x: &Query, select: &Select, selection: Option<SQLExpr>) -> Query {
567    let mut select = select.clone();
568    select.selection = selection;
569    let mut x = x.clone();
570    x.body = Box::new(SetExpr::Select(Box::new(select)));
571    x
572}

574/// `selection` with each scalar subquery that selects one expression without FROM
575/// replaced by that expression, which it is on one row. `None` if there is none. A
576/// function may be an aggregate, which reads the rows of the query around it.
577fn inline_from_less(selection: &SQLExpr) -> Option<SQLExpr> {
578    fn value(subquery: &Query) -> Option<&SQLExpr> {
579        let select = eligible_query_block(subquery)?;
580        first_row_limit(subquery)?;
581        // A clause that is dropped must not be left unchecked.
582        if !select.from.is_empty()
583            || select.selection.is_some()
584            || subquery.order_by.is_some()
585            || matches!(select.distinct, Some(Distinct::On(_)))
586            || select.exclude.is_some()
587            || !select.named_window.is_empty()
588            || !select.cluster_by.is_empty()
589            || !select.distribute_by.is_empty()
590            || !select.sort_by.is_empty()
591        {
592            return None;
593        }
594        let [SelectItem::UnnamedExpr(value) | SelectItem::ExprWithAlias { expr: value, .. }] =
595            select.projection.as_slice()
596        else {
597            return None;
598        };
599        let plain = visit_expressions(value, |e| match e {
600            SQLExpr::Function(_) => ControlFlow::Break(()),
601            e if is_subquery_expr(e) => ControlFlow::Break(()),
602            _ => ControlFlow::Continue(()),
603        })
604        .is_continue();
605        plain.then_some(value)
606    }
607
608    struct Inline {
609        query_depth: usize,
610        inlined: bool,
611    }
612
613    impl VisitorMut for Inline {
614        type Break = ();
615
616        fn pre_visit_query(&mut self, _query: &mut Query) -> ControlFlow<()> {
617            self.query_depth += 1;
618            ControlFlow::Continue(())
619        }
620
621        fn post_visit_query(&mut self, _query: &mut Query) -> ControlFlow<()> {
622            self.query_depth -= 1;
623            ControlFlow::Continue(())
624        }
625
626        fn pre_visit_expr(&mut self, expr: &mut SQLExpr) -> ControlFlow<()> {
627            if self.query_depth == 0
628                && let SQLExpr::Subquery(subquery) = &*expr
629                && let Some(value) = value(subquery)
630            {
631                *expr = SQLExpr::Nested(Box::new(value.clone()));
632                self.inlined = true;
633            }
634            ControlFlow::Continue(())
635        }
636    }
637
638    let mut selection = selection.clone();
639    let mut inline = Inline {
640        query_depth: 0,
641        inlined: false,
642    };
643    let _ = dyn_visit::visit_mut(&mut selection, &mut inline);
644    inline.inlined.then_some(selection)
645}

654/// The comparison and subquery of a conjunct `lhs IN (y)`, `lhs <op> (y)` or `EXISTS (y)`.
655fn candidate(conj: &SQLExpr) -> Option<(Option<Compare<'_>>, &Query)> {
656    use SQLBinaryOperator::*;
657    let (compare, subquery) = match strip_nested(conj) {
658        SQLExpr::InSubquery {
659            expr,
660            subquery,
661            negated: false,
662        } => (Some((expr.as_ref(), Eq, false)), subquery.as_ref()),
663        SQLExpr::Exists {
664            subquery,
665            negated: false,
666        } => (None, subquery.as_ref()),
667        SQLExpr::BinaryOp {
668            left,
669            op: op @ (Eq | NotEq | Lt | LtEq | Gt | GtEq),
670            right,
671        } => match (strip_nested(left), strip_nested(right)) {
672            (SQLExpr::Subquery(subquery), lhs) => {
673                (Some((lhs, op.clone(), true)), subquery.as_ref())
674            },
675            (lhs, SQLExpr::Subquery(subquery)) => {
676                (Some((lhs, op.clone(), false)), subquery.as_ref())
677            },
678            _ => return None,
679        },
680        _ => return None,
681    };
682    let lhs = compare.as_ref().map(|(lhs, ..)| *lhs);
683    if lhs.is_some_and(|lhs| expr_contains_subquery(lhs) || matches!(lhs, SQLExpr::Tuple(_))) {
684        return None;
685    }
686    Some((compare, subquery))
687}

689/// Move `y`, compared by `compare` (or under EXISTS), into the FROM of `x`, whose context
690/// is `x_ctx` and whose frame is `x_frame` (`None` without FROM). `scopes` are the queries
691/// around `x`. `filters_allowed` is whether a row of `x` may match several rows of `y`.
692/// `None` if `y` reads none of `scopes`, or isn't a shape that moves.
693fn try_move(
694    scopes: &[Scope<'_>],
695    x_ctx: &mut SQLContext,
696    x_frame: Option<(&LazyFrame, &Schema)>,
697    compare: Option<Compare<'_>>,
698    y: &Query,
699    filters_allowed: bool,
700) -> PolarsResult<Option<Moved>> {
701    use SQLBinaryOperator::*;
702    let Some(first_row) = first_row_limit(y) else {
703        return Ok(None);
704    };
705    let Some(select) = eligible_subquery_body(y) else {
706        return Ok(None);
707    };
708    if matches!(select.distinct, Some(Distinct::On(_))) || !select.named_window.is_empty() {
709        return Ok(None);
710    }
711    let order_by_exprs: &[OrderByExpr] = match y.order_by.as_ref().map(|o| &o.kind) {
712        None => &[],
713        Some(OrderByKind::Expressions(exprs)) => exprs,
714        Some(_) => return Ok(None),
715    };
716    let order_aggregates = order_by_exprs
717        .iter()
718        .any(|ob| has_aggregate(x_ctx, &ob.expr));
719    let value = match compare {
720        Some(_) => {
721            let [SelectItem::UnnamedExpr(value) | SelectItem::ExprWithAlias { expr: value, .. }] =
722                select.projection.as_slice()
723            else {
724                return Ok(None);
725            };
726            if expr_has_window_functions(value) {
727                return Ok(None);
728            }
729            Some(value)
730        },
731        // An aggregate makes EXISTS return one row whatever the rows are. The projection
732        // and the order are otherwise ignored.
733        None if order_aggregates
734            || select
735                .projection
736                .iter()
737                .any(|item| has_aggregate(x_ctx, item)) =>
738        {
739            return Ok(None);
740        },
741        None => None,
742    };
743    // An aggregate is taken per value of the correlation keys. A key with no rows then
744    // has no group, which is only the same as a NULL value.
745    let aggregates = value.is_some_and(|value| has_aggregate(x_ctx, value));
746    if aggregates && !null_over_no_rows(x_ctx, value.unwrap()) {
747        return Ok(None);
748    }
749    // An aggregate in the order alone also makes the subquery return one row.
750    if value.is_some() && !aggregates && order_aggregates {
751        return Ok(None);
752    }
753    // Without a limit, the distinct values of a key can be several. Only an equality
754    // matches each row of the query around it with at most one of them. A limit doesn't
755    // change whether EXISTS finds a row, nor how many rows an aggregate returns.
756    let first_row = first_row && value.is_some() && !aggregates;
757    if compare.as_ref().is_some_and(|(_, op, _)| *op != Eq) && !aggregates && !first_row {
758        return Ok(None);
759    }
760    let order_by = match first_row {
761        true => match first_row_order(y, select, order_by_exprs) {
762            Some(order_by) => order_by,
763            None => return Ok(None),
764        },
765        false => Vec::new(),
766    };
767    // The moved table reads the rows of every key at once.
768    if value
769        .into_iter()
770        .chain(&select.selection)
771        .chain(order_by.iter().map(|ob| &ob.expr))
772        .any(|e| numbers_rows(x_ctx, e))
773    {
774        return Ok(None);
775    }
776
777    let mut y_ctx = x_ctx.isolated();
778    let Some((y_names, _, y_schema)) = y_ctx.resolve_subquery_from(select)? else {
779        return Ok(None);
780    };
781    // Dropping the projection of EXISTS must not change its rows, as `UNNEST` does.
782    if compare.is_none() && exists_aggregates(&y_ctx, y, select, &y_names, &y_schema) != Some(false)
783    {
784        return Ok(None);
785    }
786    // The compared operand may be a column of a query around `x`, which `x` then reads
787    // next to the moved subquery. That only helps if `y` reads `x`: otherwise the
788    // comparison already reads no column of `x`.
789    let foreign_lhs = match compare {
790        Some((lhs, ..)) => {
791            let x_chain: Vec<Scope> = std::iter::once(Scope {
792                ctx: &*x_ctx,
793                columns: frame_columns(x_frame),
794            })
795            .chain(scopes.iter().rev().copied())
796            .collect();
797            match bind(&x_chain, lhs)? {
798                Some(Binding::Foreign {
799                    dtype: Some(dtype), ..
800                }) => Some(dtype),
801                _ => None,
802            }
803        },
804        None => None,
805    };
806    let mut kept = Vec::new();
807    let mut keys = Vec::new();
808    // Other conjuncts that read a query around `y`. They move into `x`, and read the
809    // columns of `y` that they need from the moved table. A row of `x` can then match
810    // several of its rows.
811    let mut filters = Vec::new();
812    let mut locals = Vec::new();
813    let mut reads_outer = false;
814    if let Some(selection) = &select.selection {
815        // Innermost first: `y`, `x`, then the queries around `x`.
816        let chain: Vec<Scope> = [
817            Scope {
818                ctx: &y_ctx,
819                columns: ScopeColumns::Frame(&y_schema),
820            },
821            Scope {
822                ctx: &*x_ctx,
823                columns: frame_columns(x_frame),
824            },
825        ]
826        .into_iter()
827        .chain(scopes.iter().rev().copied())
828        .collect();
829        // A conjunct with a subquery stays: planning the derived table rejects it if it
830        // reads anything around `y`.
831        for c in flatten_and_conditions(selection) {
832            if expr_contains_subquery(c) || reads_local_only(&chain, c)? == Some(true) {
833                kept.push(c.clone());
834                continue;
835            }
836            if let Some((key, depth)) = correlation_key(&chain, c)? {
837                reads_outer |= depth >= 2;
838                keys.push(key);
839                continue;
840            }
841            let Some((columns, depth)) = foreign_filter(&chain, c)? else {
842                return Ok(None);
843            };
844            reads_outer |= depth >= 2;
845            filters.push(c);
846            for column in columns {
847                if !locals.contains(&column) {
848                    locals.push(column);
849                }
850            }
851        }
852    }
853    reads_outer |= foreign_lhs.is_some() && !(keys.is_empty() && filters.is_empty());
854    if !reads_outer {
855        return Ok(None);
856    }
857    // A limit or an aggregate per key must see the rows that the filters drop.
858    let distinct = !aggregates && !first_row;
859    if !filters.is_empty() && (!filters_allowed || !distinct) {
860        return Ok(None);
861    }
862    // The moved table computes the value before the filters drop rows, so the value must
863    // not be able to fail on a row that a filter drops: it is a column or a literal.
864    let plain = |value: &SQLExpr| {
865        matches!(
866            strip_nested(value),
867            SQLExpr::Identifier(_) | SQLExpr::CompoundIdentifier(_) | SQLExpr::Value(_)
868        )
869    };
870    if !filters.is_empty() && value.is_some_and(|value| !plain(value)) {
871        return Ok(None);
872    }
873    // A moved table without columns would have no rows.
874    if value.is_none() && keys.is_empty() && locals.is_empty() {
875        return Ok(None);
876    }
877
878    let name = format!("{NESTED_PREFIX}{}", unique_column_name());
879    let value_name = format!("{name}_v");
880    let key_names: Vec<String> = (0..keys.len()).map(|i| format!("{name}_k{i}")).collect();
881    let local_names: Vec<String> = (0..locals.len()).map(|i| format!("{name}_c{i}")).collect();
882    let mut pulled = select.clone();
883    pulled.projection = value
884        .map(|value| (value.clone(), value_name.clone()))
885        .into_iter()
886        .chain(
887            keys.iter()
888                .map(|k| k.inner.clone())
889                .zip(key_names.iter().cloned()),
890        )
891        .chain(locals.iter().cloned().zip(local_names.iter().cloned()))
892        .map(|(expr, alias)| SelectItem::ExprWithAlias {
893            expr,
894            alias: Ident::new(alias),
895        })
896        .collect();
897    pulled.selection = combine_conditions(kept, And);
898    let partition_by: Vec<SQLExpr> = keys.iter().map(|k| k.inner.clone()).collect();
899    if aggregates {
900        pulled.group_by = GroupByExpr::Expressions(partition_by, vec![]);
901    } else if first_row {
902        pulled.distinct = None;
903        pulled.qualify = Some(SQLExpr::BinaryOp {
904            left: Box::new(row_number(partition_by, order_by)),
905            op: Eq,
906            right: Box::new(SQLExpr::value(SQLValue::Number("1".to_string(), false))),
907        });
908    } else {
909        pulled.distinct = Some(Distinct::Distinct);
910    }
911    let mut pulled_query = y.clone();
912    pulled_query.body = Box::new(SetExpr::Select(Box::new(pulled)));
913    pulled_query.order_by = None;
914    pulled_query.limit_clause = None;
915    pulled_query.fetch = None;
916
917    // Planning the derived table as the FROM will (strict, as `x_ctx` is) shows that it
918    // reads only what it declares, and gives the dtypes of its columns.
919    let mut pulled_ctx = x_ctx.isolated();
920    let mut pulled_lf = pulled_ctx.execute_query(&pulled_query)?;
921    let pulled_schema = pulled_ctx.get_frame_schema(&mut pulled_lf)?;
922    // Deduplicating or grouping keeps one row per value that the comparison tells apart
923    // only when the comparison doesn't merge values of the derived table.
924    for (key, key_name) in keys.iter().zip(&key_names) {
925        let Some(dtype) = pulled_schema.get(key_name.as_str()) else {
926            return Ok(None);
927        };
928        if !compares_exactly(dtype, &key.dtype) {
929            return Ok(None);
930        }
931    }
932    if distinct && let Some((lhs, ..)) = compare {
933        let Some(value_dtype) = pulled_schema.get(value_name.as_str()) else {
934            return Ok(None);
935        };
936        let lhs_dtype = match foreign_lhs {
937            Some(dtype) => dtype,
938            None => match local_dtype(scopes, x_ctx, x_frame, lhs)? {
939                Some(dtype) => dtype,
940                None => return Ok(None),
941            },
942        };
943        if !compares_exactly(value_dtype, &lhs_dtype) {
944            return Ok(None);
945        }
946    }
947
948    let table = derived_table(pulled_query, &name);
949    let column =
950        |column: &str| SQLExpr::CompoundIdentifier(vec![Ident::new(&name), Ident::new(column)]);
951    let binary = |left: SQLExpr, op: SQLBinaryOperator, right: SQLExpr| SQLExpr::BinaryOp {
952        left: Box::new(left),
953        op,
954        right: Box::new(right),
955    };
956    let filters = filters.into_iter().map(|filter| {
957        let mut filter = filter.clone();
958        let _ = visit_expressions_mut(&mut filter, |e| {
959            if let Some(i) = locals.iter().position(|local| local == e) {
960                *e = column(&local_names[i]);
961            }
962            ControlFlow::<()>::Continue(())
963        });
964        filter
965    });
966    let replaced = compare
967        .map(|(lhs, op, subquery_left)| match subquery_left {
968            true => binary(column(&value_name), op, lhs.clone()),
969            false => binary(lhs.clone(), op, column(&value_name)),
970        })
971        .into_iter()
972        .chain(
973            keys.into_iter()
974                .zip(&key_names)
975                .map(|(key, key_name)| binary(column(key_name), Eq, key.foreign)),
976        )
977        .chain(filters)
978        .collect();
979    let columns = value
980        .map(|_| value_name)
981        .into_iter()
982        .chain(key_names)
983        .chain(local_names)
984        .collect();
985    Ok(Some(Moved {
986        table,
987        columns,
988        replaced,
989    }))
990}

992/// The `ORDER BY` of a first-row subquery, to order the rows of each key by. `None` for an
993/// order that isn't read from the row alone.
994fn first_row_order(y: &Query, select: &Select, exprs: &[OrderByExpr]) -> Option<Vec<OrderByExpr>> {
995    if y.order_by.as_ref().is_some_and(|o| o.interpolate.is_some()) {
996        return None;
997    }
998    let alias = match &select.projection[0] {
999        SelectItem::ExprWithAlias { alias, .. } => Some(alias),
1000        _ => None,
1001    };
1002    // An order by the output column, by name or by position, isn't kept. A window
1003    // function would see the rows of the other keys.
1004    let unsupported = |e: &SQLExpr| match strip_nested(e) {
1005        SQLExpr::Identifier(ident) => alias.is_some_and(|a| a.value == ident.value),
1006        SQLExpr::Value(_) => true,
1007        e => expr_has_window_functions(e),
1008    };
1009    if exprs
1010        .iter()
1011        .any(|ob| ob.with_fill.is_some() || unsupported(&ob.expr))
1012    {
1013        return None;
1014    }
1015    Some(exprs.to_vec())
1016}

1018/// Call `f` on each column that `expr` reads outside of its subqueries.
1019fn visit_columns<E>(expr: &SQLExpr, f: impl FnMut(&SQLExpr) -> ControlFlow<E>) -> ControlFlow<E> {
1020    struct Columns<F> {
1021        query_depth: usize,
1022        f: F,
1023    }
1024
1025    impl<E, F: FnMut(&SQLExpr) -> ControlFlow<E>> Visitor for Columns<F> {
1026        type Break = E;
1027
1028        fn pre_visit_query(&mut self, _query: &Query) -> ControlFlow<E> {
1029            self.query_depth += 1;
1030            ControlFlow::Continue(())
1031        }
1032
1033        fn post_visit_query(&mut self, _query: &Query) -> ControlFlow<E> {
1034            self.query_depth -= 1;
1035            ControlFlow::Continue(())
1036        }
1037
1038        fn pre_visit_expr(&mut self, expr: &SQLExpr) -> ControlFlow<E> {
1039            match expr {
1040                SQLExpr::Identifier(_) | SQLExpr::CompoundIdentifier(_)
1041                    if self.query_depth == 0 =>
1042                {
1043                    (self.f)(expr)
1044                },
1045                _ => ControlFlow::Continue(()),
1046            }
1047        }
1048    }
1049
1050    dyn_visit::visit(expr, &mut Columns { query_depth: 0, f })
1051}

1053/// The scope in `chain` (innermost first) that a column binds to: the first one with a
1054/// relation the qualifier names, or with a struct column of the qualifier's name. An
1055/// unqualified column, and one no scope binds, is left to the innermost scope. `None` for
1056/// a column that its relation doesn't have.
1057fn bind(chain: &[Scope<'_>], column: &SQLExpr) -> PolarsResult<Option<Binding>> {
1058    let SQLExpr::CompoundIdentifier(parts) = column else {
1059        return Ok(Some(Binding::Local));
1060    };
1061    for (depth, scope) in chain.iter().enumerate() {
1062        let Some(found) = scope.find(parts)? else {
1063            continue;
1064        };
1065        return Ok(match (found, depth) {
1066            (Found::Missing, _) => None,
1067            (_, 0) => Some(Binding::Local),
1068            (Found::Column(dtype), depth) => Some(Binding::Foreign {
1069                depth,
1070                dtype: Some(dtype),
1071            }),
1072            (Found::Field, depth) => Some(Binding::Foreign { depth, dtype: None }),
1073        });
1074    }
1075    Ok(Some(Binding::Local))
1076}
1077
1078impl Scope<'_> {
1079    fn find(&self, parts: &[Ident]) -> PolarsResult<Option<Found>> {
1080        if matches!(self.columns, ScopeColumns::None) {
1081            return Ok(None);
1082        }
1083        if let Some((relation, len)) = self.ctx.resolve_qualifier(parts)? {
1084            if len + 1 != parts.len() {
1085                return Ok(Some(Found::Field));
1086            }
1087            return Ok(Some(
1088                match self.column_dtype(&relation, &parts[len].value) {
1089                    Some(dtype) => Found::Column(dtype),
1090                    None => Found::Missing,
1091                },
1092            ));
1093        }
1094        let is_struct =
1095            |schema: &Schema| matches!(schema.get(&parts[0].value), Some(DataType::Struct(_)));
1096        let has_struct = match self.columns {
1097            ScopeColumns::None => false,
1098            ScopeColumns::Frame(schema) => is_struct(schema),
1099            ScopeColumns::Join(left, right) => is_struct(&left.schema) || is_struct(&right.schema),
1100        };
1101        Ok(has_struct.then_some(Found::Field))
1102    }
1103
1104    fn column_dtype(&self, relation: &str, column: &str) -> Option<DataType> {
1105        let schema = match self.columns {
1106            ScopeColumns::None => return None,
1107            ScopeColumns::Frame(schema) => schema,
1108            ScopeColumns::Join(left, right) => {
1109                if self.ctx.is_right_relation(relation, right) {
1110                    return right.schema.get(column).cloned();
1111                }
1112                if !self.ctx.is_joined_relation(relation, left) {
1113                    return None;
1114                }
1115                &left.schema
1116            },
1117        };
1118        schema
1119            .get(&self.ctx.resolve_name(relation, column))
1120            .cloned()
1121    }
1122}

1124/// Whether every column `expr` reads binds to the innermost scope of `chain`, or `None`
1125/// for a column that its relation doesn't have.
1126fn reads_local_only(chain: &[Scope<'_>], expr: &SQLExpr) -> PolarsResult<Option<bool>> {
1127    let mut local = true;
1128    let flow = visit_columns(expr, |column| match bind(chain, column) {
1129        Ok(Some(Binding::Local)) => ControlFlow::Continue(()),
1130        Ok(Some(Binding::Foreign { .. })) => {
1131            local = false;
1132            ControlFlow::Continue(())
1133        },
1134        Ok(None) => ControlFlow::Break(None),
1135        Err(e) => ControlFlow::Break(Some(e)),
1136    });
1137    match flow {
1138        ControlFlow::Break(Some(e)) => Err(e),
1139        ControlFlow::Break(None) => Ok(None),
1140        ControlFlow::Continue(()) => Ok(Some(local)),
1141    }
1142}

1144/// Whether `expr` reads a column of the innermost scope of `chain` and one of a scope
1145/// around it.
1146fn reads_local_and_foreign(chain: &[Scope<'_>], expr: &SQLExpr) -> PolarsResult<bool> {
1147    let (mut local, mut foreign) = (false, false);
1148    let flow = visit_columns(expr, |column| match bind(chain, column) {
1149        Ok(Some(Binding::Local)) => {
1150            local = true;
1151            ControlFlow::Continue(())
1152        },
1153        Ok(Some(Binding::Foreign { .. })) => {
1154            foreign = true;
1155            ControlFlow::Continue(())
1156        },
1157        Ok(None) => ControlFlow::Break(None),
1158        Err(e) => ControlFlow::Break(Some(e)),
1159    });
1160    match flow {
1161        ControlFlow::Break(Some(e)) => Err(e),
1162        ControlFlow::Break(None) => Ok(false),
1163        ControlFlow::Continue(()) => Ok(local && foreign),
1164    }
1165}

1167/// `conj` as an equality between an expression over the subquery alone and a column of
1168/// a query around it, with the depth of that query in `chain`.
1169fn correlation_key(
1170    chain: &[Scope<'_>],
1171    conj: &SQLExpr,
1172) -> PolarsResult<Option<(CorrelationKey, usize)>> {
1173    let SQLExpr::BinaryOp {
1174        left,
1175        op: SQLBinaryOperator::Eq,
1176        right,
1177    } = strip_nested(conj)
1178    else {
1179        return Ok(None);
1180    };
1181    for (inner, foreign) in [(left, right), (right, left)] {
1182        let foreign = strip_nested(foreign);
1183        if expr_has_window_functions(inner)
1184            || has_aggregate(chain[0].ctx, inner.as_ref())
1185            || !expr_references_any_column(inner)
1186            || reads_local_only(chain, inner)? != Some(true)
1187        {
1188            continue;
1189        }
1190        if let Some(Binding::Foreign {
1191            depth,
1192            dtype: Some(dtype),
1193        }) = bind(chain, foreign)?
1194        {
1195            let key = CorrelationKey {
1196                inner: inner.as_ref().clone(),
1197                foreign: foreign.clone(),
1198                dtype,
1199            };
1200            return Ok(Some((key, depth)));
1201        }
1202    }
1203    Ok(None)
1204}

1206/// The columns of the subquery that `conj` reads, and the depth in `chain` of the
1207/// furthest query around it that it reads. `None` if it calls a function, or reads a
1208/// column that its relation doesn't have.
1209fn foreign_filter(
1210    chain: &[Scope<'_>],
1211    conj: &SQLExpr,
1212) -> PolarsResult<Option<(Vec<SQLExpr>, usize)>> {
1213    // A function may read other rows or draw a random value, which differs once the
1214    // filter runs per pair of rows.
1215    if calls_function(conj) {
1216        return Ok(None);
1217    }
1218    let mut locals = Vec::new();
1219    let mut max_depth = 0;
1220    let flow = visit_columns(conj, |column| match bind(chain, column) {
1221        Ok(Some(Binding::Local)) => {
1222            locals.push(column.clone());
1223            ControlFlow::Continue(())
1224        },
1225        Ok(Some(Binding::Foreign { depth, .. })) => {
1226            max_depth = max_depth.max(depth);
1227            ControlFlow::Continue(())
1228        },
1229        Ok(None) => ControlFlow::Break(None),
1230        Err(e) => ControlFlow::Break(Some(e)),
1231    });
1232    match flow {
1233        ControlFlow::Break(Some(e)) => Err(e),
1234        ControlFlow::Break(None) => Ok(None),
1235        ControlFlow::Continue(()) => Ok((max_depth > 0).then_some((locals, max_depth))),
1236    }
1237}

1239/// Whether repeating a row of `x` (whose body is `select`) keeps what the query around it
1240/// reads: `x` is read as a set or for its first row, and calls no function outside of its
1241/// subqueries. A function may aggregate, number the rows or draw a random value.
1242fn repeats_harmless(x: &Query, select: &Select, read_as: ReadAs) -> bool {
1243    let read_once = match (read_as, first_row_limit(x)) {
1244        (ReadAs::Groups, _) => false,
1245        (_, Some(true)) => true,
1246        (ReadAs::Set, Some(false)) => true,
1247        _ => false,
1248    };
1249    let order_by: &[OrderByExpr] = match x.order_by.as_ref().map(|o| &o.kind) {
1250        None => &[],
1251        Some(OrderByKind::Expressions(exprs)) => exprs,
1252        Some(_) => return false,
1253    };
1254    let projection = select.projection.iter().filter_map(|item| match item {
1255        SelectItem::UnnamedExpr(e) | SelectItem::ExprWithAlias { expr: e, .. } => Some(e),
1256        _ => None,
1257    });
1258    let join_conditions = select
1259        .from
1260        .iter()
1261        .flat_map(|item| &item.joins)
1262        .filter_map(|join| match &join.join_operator {
1263            JoinOperator::Join(JoinConstraint::On(on))
1264            | JoinOperator::Inner(JoinConstraint::On(on)) => Some(on),
1265            _ => None,
1266        });
1267    read_once
1268        && !matches!(select.distinct, Some(Distinct::On(_)))
1269        && !projection
1270            .chain(order_by.iter().map(|ob| &ob.expr))
1271            .chain(&select.selection)
1272            .chain(join_conditions)
1273            .any(calls_function)
1274}

1276/// Whether `expr` calls a function without OVER that reads the other rows of its frame,
1277/// such as `ROW_NUMBER()`, outside of its subqueries. A function that isn't known counts
1278/// as one.
1279fn numbers_rows(ctx: &SQLContext, expr: &SQLExpr) -> bool {
1280    use PolarsSQLFunctions::*;
1281    struct Functions<'a> {
1282        ctx: &'a SQLContext,
1283        query_depth: usize,
1284    }
1285
1286    impl Visitor for Functions<'_> {
1287        type Break = ();
1288
1289        fn pre_visit_query(&mut self, _query: &Query) -> ControlFlow<()> {
1290            self.query_depth += 1;
1291            ControlFlow::Continue(())
1292        }
1293
1294        fn post_visit_query(&mut self, _query: &Query) -> ControlFlow<()> {
1295            self.query_depth -= 1;
1296            ControlFlow::Continue(())
1297        }
1298
1299        fn pre_visit_expr(&mut self, expr: &SQLExpr) -> ControlFlow<()> {
1300            match expr {
1301                SQLExpr::Function(function) if self.query_depth == 0 && function.over.is_none() => {
1302                    match PolarsSQLFunctions::try_from_sql(function, self.ctx) {
1303                        Ok(
1304                            CumeDist | DenseRank | Lag | Lead | NthValue | Ntile | PercentRank
1305                            | Rank | RowNumber | Udf(_),
1306                        )
1307                        | Err(_) => ControlFlow::Break(()),
1308                        Ok(_) => ControlFlow::Continue(()),
1309                    }
1310                },
1311                _ => ControlFlow::Continue(()),
1312            }
1313        }
1314    }
1315
1316    dyn_visit::visit(
1317        expr,
1318        &mut Functions {
1319            ctx,
1320            query_depth: 0,
1321        },
1322    )
1323    .is_break()
1324}

1356/// The dtype of `expr` over the frame of `x`, or `None` if it reads anything else.
1357fn local_dtype(
1358    scopes: &[Scope<'_>],
1359    x_ctx: &mut SQLContext,
1360    x_frame: Option<(&LazyFrame, &Schema)>,
1361    expr: &SQLExpr,
1362) -> PolarsResult<Option<DataType>> {
1363    let Some((lf, schema)) = x_frame else {
1364        return Ok(None);
1365    };
1366    let local = {
1367        let chain: Vec<Scope> = std::iter::once(Scope {
1368            ctx: &*x_ctx,
1369            columns: ScopeColumns::Frame(schema),
1370        })
1371        .chain(scopes.iter().rev().copied())
1372        .collect();
1373        reads_local_only(&chain, expr)?
1374    };
1375    if local != Some(true) {
1376        return Ok(None);
1377    }
1378    let parsed = parse_sql_expr(expr, x_ctx, Some(schema))?;
1379    let mut probe = lf.clone().select([parsed]);
1380    let probe_schema = x_ctx.get_frame_schema(&mut probe)?;
1381    Ok(probe_schema.get_at_index(0).map(|(_, dtype)| dtype.clone()))
1382}

1384/// Whether comparing a value of `inner` with one of `other` casts it without merging
1385/// values, so that distinct values of `inner` never equal the same value of `other`.
1386pub(crate) fn compares_exactly(inner: &DataType, other: &DataType) -> bool {
1387    if inner == other {
1388        return true;
1389    }
1390    get_supertype(inner, other).is_some_and(|to| holds_values(inner, &to))
1391}

1393/// Whether `to` holds every value of the numeric dtype `from`.
1394fn holds_values(from: &DataType, to: &DataType) -> bool {
1395    let bits = |dtype: &DataType| match dtype {
1396        DataType::Int8 | DataType::UInt8 => 8,
1397        DataType::Int16 | DataType::UInt16 => 16,
1398        DataType::Int32 | DataType::UInt32 | DataType::Float32 => 32,
1399        DataType::Int64 | DataType::UInt64 | DataType::Float64 => 64,
1400        DataType::Int128 | DataType::UInt128 => 128,
1401        _ => 0,
1402    };
1403    match (from, to) {
1404        _ if from == to => true,
1405        (DataType::Float32, DataType::Float64) => true,
1406        // The mantissa holds every integer of up to 24 or 53 bits.
1407        (from, DataType::Float32) if from.is_integer() => bits(from) <= 16,
1408        (from, DataType::Float64) if from.is_integer() => bits(from) <= 32,
1409        (from, to) if from.is_integer() && to.is_integer() => {
1410            match (from.is_signed_integer(), to.is_signed_integer()) {
1411                (true, true) | (false, false) => bits(from) <= bits(to),
1412                (false, true) => bits(from) < bits(to),
1413                (true, false) => false,
1414            }
1415        },
1416        _ => false,
1417    }
1418}

1420/// Whether `node` calls an aggregate outside of its subqueries. A function that isn't
1421/// known counts as one.
1422pub(crate) fn has_aggregate(ctx: &SQLContext, node: &(impl Visit + ?Sized)) -> bool {
1423    struct Aggregates<'a> {
1424        ctx: &'a SQLContext,
1425        query_depth: usize,
1426    }
1427
1428    impl Visitor for Aggregates<'_> {
1429        type Break = ();
1430
1431        fn pre_visit_query(&mut self, _query: &Query) -> ControlFlow<()> {
1432            self.query_depth += 1;
1433            ControlFlow::Continue(())
1434        }
1435
1436        fn post_visit_query(&mut self, _query: &Query) -> ControlFlow<()> {
1437            self.query_depth -= 1;
1438            ControlFlow::Continue(())
1439        }
1440
1441        fn pre_visit_expr(&mut self, expr: &SQLExpr) -> ControlFlow<()> {
1442            match expr {
1443                SQLExpr::Function(function) if self.query_depth == 0 && function.over.is_none() => {
1444                    match PolarsSQLFunctions::try_from_sql(function, self.ctx) {
1445                        Ok(PolarsSQLFunctions::Udf(_)) | Err(_) => ControlFlow::Break(()),
1446                        Ok(f) if f.is_builtin_aggregate() => ControlFlow::Break(()),
1447                        Ok(_) => ControlFlow::Continue(()),
1448                    }
1449                },
1450                _ => ControlFlow::Continue(()),
1451            }
1452        }
1453    }
1454
1455    dyn_visit::visit(
1456        node,
1457        &mut Aggregates {
1458            ctx,
1459            query_depth: 0,
1460        },
1461    )
1462    .is_break()
1463}

1465/// Whether `expr` is NULL when it aggregates no rows: an aggregate other than a count,
1466/// possibly cast, negated or combined with a constant by arithmetic.
1467fn null_over_no_rows(ctx: &SQLContext, expr: &SQLExpr) -> bool {
1468    use PolarsSQLFunctions::*;
1469    match strip_nested(expr) {
1470        SQLExpr::Function(function) if function.over.is_none() => matches!(
1471            PolarsSQLFunctions::try_from_sql(function, ctx),
1472            Ok(Avg
1473                | Corr
1474                | CovarPop
1475                | CovarSamp
1476                | First
1477                | FirstValue
1478                | Last
1479                | Max
1480                | Median
1481                | Min
1482                | QuantileCont
1483                | QuantileDisc
1484                | StdDev
1485                | Sum
1486                | Variance)
1487        ),
1488        SQLExpr::Cast { expr, .. }
1489        | SQLExpr::UnaryOp {
1490            op: SQLUnaryOperator::Minus,
1491            expr,
1492        } => null_over_no_rows(ctx, expr),
1493        SQLExpr::BinaryOp {
1494            left,
1495            op:
1496                SQLBinaryOperator::Plus
1497                | SQLBinaryOperator::Minus
1498                | SQLBinaryOperator::Multiply
1499                | SQLBinaryOperator::Divide
1500                | SQLBinaryOperator::Modulo,
1501            right,
1502        } => {
1503            let constant = |e: &SQLExpr| !expr_references_any_column(e) && !has_aggregate(ctx, e);
1504            (null_over_no_rows(ctx, left) && constant(right))
1505                || (constant(left) && null_over_no_rows(ctx, right))
1506        },
1507        _ => false,
1508    }
1509}
