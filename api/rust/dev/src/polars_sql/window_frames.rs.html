25/// The rows of a window frame, in the sorted window partition.
26#[derive(Clone, Copy)]
27enum FrameShape {
28    /// All rows of the partition.
29    Partition,
30    /// From the first row to the current row, or to its last peer if `peers`.
31    ToCurrentRow { peers: bool },
32    /// From the current row, or from its first peer if `peers`, to the last row.
33    FromCurrentRow { peers: bool },
34    /// The current row and its peers.
35    Peers,
36    /// The current row and the given number of rows before it.
37    Preceding(usize),
38    /// The rows whose ORDER BY key is within the bounds of the current row's key. At least
39    /// one bound is an offset.
40    Range { start: RangeBound, end: RangeBound },
41}
42
43/// A bound of a RANGE frame.
44#[derive(Clone, Copy)]
45enum RangeBound {
46    Unbounded,
47    /// The peers of the current row.
48    CurrentRow,
49    /// The rows whose key is at most this far from the current row's key.
50    Offset(Duration),
51}
52
53/// A part of a RANGE frame, in the sorted window partition.
54enum FramePart {
55    /// The rows up to the last peer of the current row.
56    ToLastPeer,
57    /// The rows from the first peer of the current row.
58    FromFirstPeer,
59    /// The rows whose `by` value is within `size` before the current row's, read in reverse
60    /// order if `reversed`.
61    #[cfg_attr(not(feature = "rolling_window_by"), allow(dead_code))]
62    Window {
63        by: Expr,
64        size: Duration,
65        closed: ClosedWindow,
66        reversed: bool,
67    },
68}

70/// A bound of a window frame, as a row of the sorted window partition.
71#[derive(Clone, Copy)]
72enum RowBound {
73    /// The first row of the partition.
74    First,
75    /// The last row of the partition.
76    Last,
77    /// The first peer of the current row.
78    FirstPeer,
79    /// The last peer of the current row.
80    LastPeer,
81    /// The row at this offset from the current row. It can be outside the partition.
82    Offset(i64),
83}
84
85struct FrameBounds {
86    /// The first row of the frame, if it is known.
87    start: Option<RowBound>,
88    /// The last row of the frame, if it is known.
89    end: Option<RowBound>,
90    /// Whether the frame holds the current row, so that it is never empty.
91    has_current_row: bool,
92}

94/// Row offsets are capped at this, so that adding them to a row index can't overflow. No
95/// partition has this many rows.
96pub(crate) const MAX_ROW_OFFSET: i64 = i64::MAX / 4;

98enum Aggregate {
99    Avg(Expr),
100    Count(Expr),
101    /// `COUNT(*)`
102    CountRows,
103    Max(Expr),
104    Min(Expr),
105    Sum(Expr),
106}

108/// Whether the function is an aggregate that is computed over its window frame.
109pub(crate) fn is_frame_aggregate(function: &PolarsSQLFunctions, is_distinct: bool) -> bool {
110    use PolarsSQLFunctions::*;
111    match function {
112        Avg | BoolAnd | BoolOr | CountIf | Max | Min | Sum | Total => true,
113        Count => !is_distinct,
114        _ => false,
115    }
116}

119    /// Lower an aggregate with an OVER clause (see [`is_frame_aggregate`]).
120    pub(crate) fn visit_window_aggregate(
121        &mut self,
122        function: &PolarsSQLFunctions,
123    ) -> PolarsResult<Expr> {
124        use PolarsSQLFunctions::*;
125        let order_keys = self.parse_window_order_keys()?;
126        let frame = self.window_frame_shape(!order_keys.is_empty())?;
127
128        let (args, is_distinct) = extract_args_distinct(self.func)?;
129        let value = match args.as_slice() {
130            [] | [FunctionArgExpr::Wildcard] if matches!(function, Count) => None,
131            [FunctionArgExpr::Expr(e)] if matches!(function, Count) && is_non_null_literal(e) => {
132                None
133            },
134            [FunctionArgExpr::Expr(e)] => Some(parse_sql_expr(e, self.ctx, self.active_schema)?),
135            _ => return self.not_supported_error(),
136        };
137        // Rows that don't pass the FILTER become NULL, which these aggregates skip.
138        let value = match (&self.filter, value) {
139            (Some(predicate), value) => Some(
140                when(predicate.clone())
141                    .then(value.unwrap_or(lit(true)))
142                    .otherwise(Expr::Literal(LiteralValue::untyped_null())),
143            ),
144            (None, value) => value,
145        };
146        let value = match function {
147            BoolAnd | BoolOr | CountIf => value.map(|v| self.boolean_arg(v)).transpose()?,
148            _ => value,
149        };
150        // A scalar counts once per row, as in `SUM(1)`.
151        let is_scalar = value.as_ref().is_some_and(is_constant_key);
152        let value = value.map(|value| match frame {
153            FrameShape::Partition | FrameShape::Peers => value,
154            _ if is_scalar => repeat(value, len()),
155            _ => value,
156        });
157
158        // DISTINCT doesn't change MIN and MAX.
159        let distinct_supported = match function {
160            BoolAnd | BoolOr | Max | Min => true,
161            Avg => matches!(frame, FrameShape::Partition),
162            _ => false,
163        };
164        if is_distinct && !distinct_supported {
165            polars_bail!(
166                SQLInterface: "{}(DISTINCT ...) is not supported with this OVER clause",
167                self.func.name
168            );
169        }
170        let aggregate = match (function, value) {
171            (Count, None) => Aggregate::CountRows,
172            (Count, Some(value)) => Aggregate::Count(value),
173            (Avg, Some(value)) if is_distinct => Aggregate::Avg(value.unique()),
174            (Avg, Some(value)) => Aggregate::Avg(value),
175            (CountIf, Some(value)) => Aggregate::Count(
176                when(value)
177                    .then(lit(true))
178                    .otherwise(Expr::Literal(LiteralValue::untyped_null())),
179            ),
180            (BoolOr | Max, Some(value)) => Aggregate::Max(value),
181            (BoolAnd | Min, Some(value)) => Aggregate::Min(value),
182            (Sum | Total, Some(value)) => Aggregate::Sum(value),
183            _ => return self.not_supported_error(),
184        };
185
186        let keys: Vec<Expr> = order_keys.iter().map(|(key, _)| key.clone()).collect();
187        let (expr, peer_keys, order_keys) = match frame {
188            FrameShape::Partition => (aggregate.whole(is_scalar), Vec::new(), Vec::new()),
189            FrameShape::Peers => (aggregate.whole(is_scalar), keys, Vec::new()),
190            FrameShape::ToCurrentRow { peers } => {
191                let running = aggregate.running(false);
192                let expr = if peers {
193                    at_last_peer(running, &keys)
194                } else {
195                    running
196                };
197                (expr, Vec::new(), order_keys)
198            },
199            FrameShape::FromCurrentRow { peers } => {
200                let running = aggregate.running(true);
201                let expr = if peers {
202                    at_first_peer(running, &keys)
203                } else {
204                    running
205                };
206                (expr, Vec::new(), order_keys)
207            },
208            FrameShape::Preceding(n) => (aggregate.rolling(n)?, Vec::new(), order_keys),
209            FrameShape::Range { start, end } => {
210                let expr = self.range_aggregate(&aggregate, start, end, &order_keys)?;
211                (expr, Vec::new(), order_keys)
212            },
213        };
214        let expr = match function {
215            Total => expr.cast(DataType::Float64).fill_null(lit(0.0)),
216            _ => expr,
217        };
218        self.apply_over(expr, peer_keys, order_keys)
219    }

221    /// The rows of the window frame. Without ORDER BY keys, all rows are peers.
222    fn window_frame_shape(&mut self, has_order_keys: bool) -> PolarsResult<FrameShape> {
223        use WindowFrameBound::{CurrentRow, Following, Preceding};
224        let spec = self.window.as_ref().unwrap();
225        let has_order_by = !spec.order_by.is_empty();
226        let Some(frame) = spec.window_frame.clone() else {
227            return Ok(if has_order_keys {
228                FrameShape::ToCurrentRow { peers: true }
229            } else {
230                FrameShape::Partition
231            });
232        };
233        let end_bound = frame.end_bound.clone().unwrap_or(CurrentRow);
234        let shape = match (&frame.units, &frame.start_bound, &end_bound) {
235            (_, Preceding(None), Following(None)) => Some(FrameShape::Partition),
236            (WindowFrameUnits::Rows, _, _) if !has_order_by => polars_bail!(
237                SQLInterface: "{} with a ROWS window frame but no ORDER BY is not supported",
238                self.func.name
239            ),
240            (WindowFrameUnits::Rows, Preceding(None), CurrentRow) => {
241                Some(FrameShape::ToCurrentRow { peers: false })
242            },
243            (WindowFrameUnits::Rows, CurrentRow, Following(None)) => {
244                Some(FrameShape::FromCurrentRow { peers: false })
245            },
246            (WindowFrameUnits::Rows, CurrentRow, CurrentRow) => Some(FrameShape::Preceding(0)),
247            (WindowFrameUnits::Rows, Preceding(Some(n)), CurrentRow) => {
248                Some(FrameShape::Preceding(self.window_frame_offset(n)?))
249            },
250            (WindowFrameUnits::Rows, _, _) => None,
251            // In RANGE and GROUPS frames, the current row includes its peers.
252            (_, Preceding(None) | CurrentRow, CurrentRow | Following(None)) if !has_order_keys => {
253                Some(FrameShape::Partition)
254            },
255            (_, Preceding(None), CurrentRow) => Some(FrameShape::ToCurrentRow { peers: true }),
256            (_, CurrentRow, Following(None)) => Some(FrameShape::FromCurrentRow { peers: true }),
257            (_, CurrentRow, CurrentRow) => Some(FrameShape::Peers),
258            (WindowFrameUnits::Range, start, end) => {
259                self.range_frame_shape(start, end, has_order_keys)?
260            },
261            _ => None,
262        };
263        shape.ok_or_else(|| self.unsupported_frame())
264    }

266    /// The shape of a RANGE frame with an offset bound; `None` if it is not supported.
267    fn range_frame_shape(
268        &mut self,
269        start: &WindowFrameBound,
270        end: &WindowFrameBound,
271        has_order_keys: bool,
272    ) -> PolarsResult<Option<FrameShape>> {
273        use WindowFrameBound::{CurrentRow, Following, Preceding};
274        let start = match start {
275            Preceding(None) => RangeBound::Unbounded,
276            CurrentRow => RangeBound::CurrentRow,
277            Preceding(Some(offset)) => self.range_offset(offset)?,
278            Following(_) => return Ok(None),
279        };
280        let end = match end {
281            Following(None) => RangeBound::Unbounded,
282            CurrentRow => RangeBound::CurrentRow,
283            Following(Some(offset)) => self.range_offset(offset)?,
284            Preceding(_) => return Ok(None),
285        };
286        polars_ensure!(
287            has_order_keys,
288            SQLInterface: "a RANGE window frame with an offset needs one ORDER BY key"
289        );
290        use RangeBound::{CurrentRow as Current, Unbounded};
291        Ok(Some(match (start, end) {
292            (Unbounded, Unbounded) => FrameShape::Partition,
293            (Unbounded, Current) => FrameShape::ToCurrentRow { peers: true },
294            (Current, Unbounded) => FrameShape::FromCurrentRow { peers: true },
295            (Current, Current) => FrameShape::Peers,
296            (start, end) => FrameShape::Range { start, end },
297        }))
298    }

300    /// An offset of a RANGE frame: a non-negative integer or INTERVAL literal. A zero offset
301    /// is the current row.
302    fn range_offset(&mut self, offset: &SQLExpr) -> PolarsResult<RangeBound> {
303        let duration = match offset {
304            SQLExpr::Interval(interval) => Some(interval_to_duration(interval, false)?),
305            _ => match parse_sql_expr(offset, self.ctx, self.active_schema)? {
306                Expr::Literal(LiteralValue::Dyn(DynLiteralValue::Int(n))) => {
307                    i64::try_from(n).ok().map(Duration::new)
308                },
309                _ => None,
310            },
311        };
312        let Some(duration) = duration.filter(|d| !d.negative()) else {
313            polars_bail!(
314                SQLSyntax: "a RANGE window frame offset must be a non-negative integer or INTERVAL literal; found {}",
315                offset
316            );
317        };
318        Ok(if duration.is_zero() {
319            RangeBound::CurrentRow
320        } else {
321            RangeBound::Offset(duration)
322        })
323    }

325    /// An aggregate over a RANGE frame with an offset bound, from rolling windows over the
326    /// ORDER BY key.
327    fn range_aggregate(
328        &mut self,
329        aggregate: &Aggregate,
330        start: RangeBound,
331        end: RangeBound,
332        order_keys: &[(Expr, SortOptions)],
333    ) -> PolarsResult<Expr> {
334        use RangeBound::{CurrentRow, Offset, Unbounded};
335        let [(key, options)] = order_keys else {
336            polars_bail!(SQLInterface: "a RANGE window frame with an offset needs one ORDER BY key");
337        };
338        let Some(dtype) = self.ctx.expr_dtype(key, self.active_schema) else {
339            polars_bail!(SQLInterface: "cannot determine the type of the ORDER BY key of a RANGE window frame");
340        };
341        if let Aggregate::Sum(value)
342        | Aggregate::Avg(value)
343        | Aggregate::Min(value)
344        | Aggregate::Max(value) = aggregate
345            && let Some(dtype) = self.ctx.expr_dtype(value, self.active_schema)
346        {
347            polars_ensure!(
348                dtype.is_primitive_numeric() || dtype.is_null(),
349                SQLInterface: "{} over a RANGE window frame with an offset needs a number value; found {}",
350                self.func.name, dtype
351            );
352        }
353        let range_key = RangeKey {
354            key,
355            dtype: &dtype,
356            descending: options.descending,
357        };
358        let parts = match (start, end) {
359            (Offset(n), CurrentRow) => vec![range_key.before(n, true)?],
360            (CurrentRow, Offset(m)) => vec![range_key.after(m, true)?],
361            (Offset(n), Offset(m)) => vec![range_key.before(n, true)?, range_key.after(m, false)?],
362            (Unbounded, Offset(m)) => {
363                vec![FramePart::ToLastPeer, range_key.after(m, false)?]
364            },
365            (Offset(n), Unbounded) => {
366                vec![range_key.before(n, false)?, FramePart::FromFirstPeer]
367            },
368            _ => unreachable!(),
369        };
370        let keys = [key.clone()];
371        let expr = aggregate.over_parts(&parts, &keys)?;
372
373        // A NULL key is at no distance from another key: an offset reaches its peers only.
374        let null_part = match (start, end) {
375            (Unbounded, _) => FramePart::ToLastPeer,
376            (_, Unbounded) => FramePart::FromFirstPeer,
377            _ if options.nulls_last => FramePart::FromFirstPeer,
378            _ => FramePart::ToLastPeer,
379        };
380        let null_expr = aggregate.over_parts(&[null_part], &keys)?;
381        Ok(when(key.clone().is_null()).then(null_expr).otherwise(expr))
382    }

384    fn unsupported_frame(&self) -> PolarsError {
385        let spec = self.window.as_ref().unwrap();
386        let frame = spec.window_frame.as_ref().unwrap();
387        polars_err!(
388            SQLInterface: "{} with window frame '{} BETWEEN {} AND {}' is not supported",
389            self.func.name,
390            frame.units,
391            frame.start_bound,
392            frame.end_bound.as_ref().unwrap_or(&WindowFrameBound::CurrentRow)

396    /// Lower FIRST_VALUE, LAST_VALUE or NTH_VALUE with an OVER clause.
397    pub(crate) fn visit_window_value(
398        &mut self,
399        function: &PolarsSQLFunctions,
400    ) -> PolarsResult<Expr> {
401        use PolarsSQLFunctions::*;
402        let args = extract_args(self.func)?;
403        let (value, position) = match (function, args.as_slice()) {
404            (FirstValue | LastValue, [FunctionArgExpr::Expr(value)]) => (value, 1),
405            (
406                NthValue,
407                [
408                    FunctionArgExpr::Expr(value),
409                    FunctionArgExpr::Expr(position),
410                ],
411            ) => (value, self.nth_value_position(position)?),
412            _ => return self.not_supported_error(),
413        };
414        let value = parse_sql_expr(value, self.ctx, self.active_schema)?;
415        let order_keys = self.parse_window_order_keys()?;
416        let keys: Vec<Expr> = order_keys.iter().map(|(key, _)| key.clone()).collect();
417        let FrameBounds {
418            start,
419            end,
420            has_current_row,
421        } = self.window_frame_bounds(!keys.is_empty())?;
422
423        // A scalar, as in `LAST_VALUE(1)`, is the value at every row of the frame.
424        let is_scalar = is_constant_key(&value);
425        let expr = match (function, start, end) {
426            (FirstValue | LastValue, _, _) if is_scalar && has_current_row => value,
427            (FirstValue, Some(RowBound::First), _) if has_current_row => value.first(),
428            (LastValue, _, Some(RowBound::Last)) if has_current_row => value.last(),
429            (NthValue, Some(RowBound::First), Some(RowBound::Last)) if !is_scalar => {
430                value.slice(lit(position - 1), lit(1i64)).first()
431            },
432            (FirstValue, Some(RowBound::Offset(0)), _)
433            | (LastValue, _, Some(RowBound::Offset(0)))
434                if has_current_row =>
435            {
436                value
437            },
438            // A frame with the current row is not empty, so only the bound that is read matters.
439            (FirstValue, Some(start), _) if has_current_row => {
440                value.gather(frame_start_row(start, &keys), false)
441            },
442            (LastValue, _, Some(end)) if has_current_row => {
443                value.gather(frame_end_row(end, &keys), false)
444            },
445            (_, Some(start), Some(end)) => {
446                let first = frame_start_row(start, &keys);
447                let last = frame_end_row(end, &keys);
448                let row = match function {
449                    LastValue => last.clone(),
450                    _ => first.clone() + lit(position - 1),
451                };
452                // NULL if the frame is empty or has fewer rows than NTH_VALUE asks for.
453                let in_frame = first.lt_eq(row.clone()).and(row.clone().lt_eq(last));
454                let null = Expr::Literal(LiteralValue::untyped_null());
455                if is_scalar {
456                    when(in_frame).then(value).otherwise(null)
457                } else {
458                    value.gather(when(in_frame).then(row).otherwise(null), false)
459                }
460            },
461            _ => return Err(self.unsupported_frame()),
462        };
463        self.apply_over(expr, Vec::new(), order_keys)
464    }

466    /// The bounds of the window frame. Without ORDER BY keys, all rows are peers.
467    fn window_frame_bounds(&mut self, has_order_keys: bool) -> PolarsResult<FrameBounds> {
468        use WindowFrameBound::{CurrentRow, Following, Preceding};
469        let spec = self.window.as_ref().unwrap();
470        let has_order_by = !spec.order_by.is_empty();
471        let Some(frame) = spec.window_frame.clone() else {
472            let end = if has_order_keys {
473                RowBound::LastPeer
474            } else {
475                RowBound::Last
476            };
477            return Ok(FrameBounds {
478                start: Some(RowBound::First),
479                end: Some(end),
480                has_current_row: true,
481            });
482        };
483        let rows = matches!(frame.units, WindowFrameUnits::Rows);
484        let end_bound = frame.end_bound.clone().unwrap_or(CurrentRow);
485        // As in Postgres, a frame can't start at a later kind of bound than it ends at.
486        if bound_order(&frame.start_bound) > bound_order(&end_bound)
487            || matches!(frame.start_bound, Following(None))
488            || matches!(end_bound, Preceding(None))
489        {
490            polars_bail!(
491                SQLSyntax: "invalid window frame '{} BETWEEN {} AND {}'",
492                frame.units, frame.start_bound, end_bound
493            );
494        }
495        if rows
496            && !has_order_by
497            && !matches!(
498                (&frame.start_bound, &end_bound),
499                (Preceding(None), Following(None))
500            )
501        {
502            polars_bail!(
503                SQLInterface: "{} with a ROWS window frame but no ORDER BY is not supported",
504                self.func.name
505            );
506        }
507        Ok(FrameBounds {
508            start: self.row_bound(&frame.start_bound, &frame.units, has_order_keys, true)?,
509            end: self.row_bound(&end_bound, &frame.units, has_order_keys, false)?,
510            has_current_row: bound_order(&frame.start_bound) <= bound_order(&CurrentRow)
511                && bound_order(&end_bound) >= bound_order(&CurrentRow),
512        })
513    }

515    /// A bound of a frame. `None` for an offset in a RANGE or GROUPS frame, which is only
516    /// checked.
517    fn row_bound(
518        &mut self,
519        bound: &WindowFrameBound,
520        units: &WindowFrameUnits,
521        has_order_keys: bool,
522        is_start: bool,
523    ) -> PolarsResult<Option<RowBound>> {
524        use WindowFrameBound::{CurrentRow, Following, Preceding};
525        let rows = matches!(units, WindowFrameUnits::Rows);
526        Ok(match bound {
527            Preceding(None) if is_start => Some(RowBound::First),
528            Following(None) if !is_start => Some(RowBound::Last),
529            CurrentRow if rows => Some(RowBound::Offset(0)),
530            // In RANGE and GROUPS frames, the current row includes its peers.
531            CurrentRow if !has_order_keys && is_start => Some(RowBound::First),
532            CurrentRow if !has_order_keys => Some(RowBound::Last),
533            CurrentRow if is_start => Some(RowBound::FirstPeer),
534            CurrentRow => Some(RowBound::LastPeer),
535            Preceding(Some(n)) if rows => {
536                Some(RowBound::Offset(-row_offset(self.window_frame_offset(n)?)))
537            },
538            Following(Some(n)) if rows => {
539                Some(RowBound::Offset(row_offset(self.window_frame_offset(n)?)))
540            },
541            Preceding(Some(n)) | Following(Some(n))
542                if matches!(units, WindowFrameUnits::Groups) =>
543            {
544                self.window_frame_offset(n)?;
545                None
546            },
547            Preceding(Some(n)) | Following(Some(n)) => {
548                self.check_range_frame_offset(n)?;
549                None
550            },
551            _ => None,
552        })
553    }

555    /// Check an offset of a RANGE frame: a literal that is not NULL or negative.
556    fn check_range_frame_offset(&mut self, offset: &SQLExpr) -> PolarsResult<()> {
557        let is_valid = match parse_sql_expr(offset, self.ctx, self.active_schema)? {
558            Expr::Literal(LiteralValue::Dyn(DynLiteralValue::Int(n))) => n >= 0,
559            Expr::Literal(LiteralValue::Dyn(DynLiteralValue::Float(f))) => f >= 0.0,
560            Expr::Literal(LiteralValue::Scalar(scalar)) => {
561                (scalar.dtype().is_numeric() || scalar.dtype().is_duration())
562                    && scalar.value().extract::<f64>().is_some_and(|v| v >= 0.0)
563            },
564            _ => false,
565        };
566        polars_ensure!(
567            is_valid,
568            SQLSyntax: "window frame offset must be a non-negative literal; found {}", offset
569        );
570        Ok(())
571    }

573    /// The position argument of NTH_VALUE, a positive integer literal.
574    fn nth_value_position(&mut self, position: &SQLExpr) -> PolarsResult<i64> {
575        match parse_sql_expr(position, self.ctx, self.active_schema)? {
576            Expr::Literal(LiteralValue::Dyn(DynLiteralValue::Int(n))) if n > 0 => {
577                Ok(n.min(i128::from(MAX_ROW_OFFSET)) as i64)
578            },
579            _ => polars_bail!(
580                SQLSyntax: "NTH_VALUE expects a positive integer literal as the position; found {}",
581                position
582            ),
583        }
584    }
585
586    fn window_frame_offset(&mut self, offset: &SQLExpr) -> PolarsResult<usize> {
587        let n = match parse_sql_expr(offset, self.ctx, self.active_schema)? {
588            Expr::Literal(LiteralValue::Dyn(DynLiteralValue::Int(n))) => usize::try_from(n).ok(),
589            _ => None,
590        };
591        n.ok_or_else(|| {
592            polars_err!(
593                SQLSyntax: "window frame offset must be a non-negative integer; found {}", offset
594            )
595        })
596    }
597}
598
599impl Aggregate {
600    /// The aggregate over all rows. A `scalar` value counts once per row.
601    fn whole(self, scalar: bool) -> Expr {
602        match self {
603            // Scaled by the number of rows, not repeated: a reduction inside a reduction reads
604            // as a nested aggregate (`SUM(SUM(x))`), which gives one output row.
605            Aggregate::Count(value) if scalar => (value.count() * len()).cast(DataType::Int64),
606            Aggregate::Sum(value) if scalar => value * len().cast(DataType::Int64),
607            Aggregate::Avg(value) => value.mean(),
608            Aggregate::Count(value) => value.count().cast(DataType::Int64),
609            Aggregate::CountRows => len().cast(DataType::Int64),
610            Aggregate::Max(value) => value.max(),
611            Aggregate::Min(value) => value.min(),
612            Aggregate::Sum(value) => sql_sum(value),
613        }
614    }
615
616    /// The aggregate over the rows up to each row, or from each row if `reverse`.
617    fn running(&self, reverse: bool) -> Expr {
618        // The running value is NULL where the value is NULL; take it from the row before (or
619        // after, if `reverse`).
620        let fill = if reverse {
621            FillNullStrategy::Backward(None)
622        } else {
623            FillNullStrategy::Forward(None)
624        };
625        match self {
626            Aggregate::Avg(value) => {
627                // Sum in Float64, so that an integer sum can't overflow.
628                let sum = Aggregate::Sum(value.clone().cast(DataType::Float64)).running(reverse);
629                let count = Aggregate::Count(value.clone()).running(reverse);
630                sum / count
631            },
632            Aggregate::Count(value) => value.clone().cum_count(reverse).cast(DataType::Int64),
633            Aggregate::CountRows => {
634                let (row_index, n) = window_row_index();
635                if reverse {
636                    n - row_index
637                } else {
638                    row_index + lit(1i64)
639                }
640            },
641            Aggregate::Max(value) => value.clone().cum_max(reverse).fill_null_with_strategy(fill),
642            Aggregate::Min(value) => value.clone().cum_min(reverse).fill_null_with_strategy(fill),
643            Aggregate::Sum(value) => value.clone().cum_sum(reverse).fill_null_with_strategy(fill),
644        }
645    }
646
647    /// The aggregate over each row and the `n` rows before it.
648    fn rolling(&self, n: usize) -> PolarsResult<Expr> {
649        #[cfg(feature = "rolling_window")]
650        {
651            let window_size = n.saturating_add(1);
652            let options = RollingOptionsFixedWindow {
653                window_size,
654                min_samples: 1,
655                ..Default::default()
656            };
657            Ok(match self {
658                Aggregate::Avg(value) => value.clone().rolling_mean(options),
659                Aggregate::Count(value) => value
660                    .clone()
661                    .is_not_null()
662                    .cast(DataType::Int64)
663                    .rolling_sum(options),
664                Aggregate::CountRows => {
665                    let (row_index, _) = window_row_index();
666                    let window_size = lit(i64::try_from(window_size).unwrap_or(i64::MAX));
667                    when(row_index.clone().lt(window_size.clone()))
668                        .then(row_index + lit(1i64))
669                        .otherwise(window_size)
670                },
671                Aggregate::Max(value) => value.clone().rolling_max(options),
672                Aggregate::Min(value) => value.clone().rolling_min(options),
673                Aggregate::Sum(value) => value.clone().rolling_sum(options),
674            })
675        }
676        #[cfg(not(feature = "rolling_window"))]
677        {
678            let _ = n;
679            polars_bail!(SQLInterface: "'ROWS <n> PRECEDING' window frames require the 'rolling_window' feature")
680        }
681    }
682
683    /// The aggregate over the union of disjoint frame parts.
684    fn over_parts(&self, parts: &[FramePart], keys: &[Expr]) -> PolarsResult<Expr> {
685        let total = |aggregate: Aggregate| -> PolarsResult<Expr> {
686            let mut terms = parts.iter().map(|part| part.apply(&aggregate, keys));
687            let first = terms.next().unwrap()?.fill_null(lit(0i64));
688            terms.try_fold(first, |acc, term| Ok(acc + term?.fill_null(lit(0i64))))
689        };
690        Ok(match self {
691            Aggregate::Sum(value) | Aggregate::Avg(value) => {
692                let avg = matches!(self, Aggregate::Avg(_));
693                let value = match avg {
694                    true => value.clone().cast(DataType::Float64),
695                    false => value.clone(),
696                };
697                let count = total(Aggregate::Count(value.clone()))?;
698                let sum = total(Aggregate::Sum(value))?;
699                let result = match avg {
700                    true => sum / count.clone(),
701                    false => sum,
702                };
703                when(count.gt(lit(0i64)))
704                    .then(result)
705                    .otherwise(Expr::Literal(LiteralValue::untyped_null()))
706            },
707            Aggregate::Count(value) => total(Aggregate::Count(value.clone()))?,
708            Aggregate::CountRows => total(Aggregate::CountRows)?,
709            Aggregate::Max(_) | Aggregate::Min(_) => {
710                let terms = parts
711                    .iter()
712                    .map(|part| part.apply(self, keys))
713                    .collect::<PolarsResult<Vec<_>>>()?;
714                match self {
715                    Aggregate::Max(_) => max_horizontal(terms)?,
716                    _ => min_horizontal(terms)?,
717                }
718            },
719        })
720    }
721}

723/// The ORDER BY key of a RANGE frame.
724struct RangeKey<'a> {
725    key: &'a Expr,
726    dtype: &'a DataType,
727    descending: bool,
728}
729
730impl RangeKey<'_> {
731    /// The rows before the current row in the sorted window whose key is at most `size` away;
732    /// the peers of the current row are included if `peers`.
733    fn before(&self, size: Duration, peers: bool) -> PolarsResult<FramePart> {
734        self.window(size, peers, false)
735    }
736
737    /// The rows after the current row in the sorted window whose key is at most `size` away;
738    /// the peers of the current row are included if `peers`.
739    fn after(&self, size: Duration, peers: bool) -> PolarsResult<FramePart> {
740        // Read in reverse, the rows after the current row come before it.
741        self.window(size, peers, true)
742    }
743
744    /// A window over the key, read in reverse if `reversed`.
745    fn window(&self, size: Duration, peers: bool, reversed: bool) -> PolarsResult<FramePart> {
746        // A key that falls in the reading order is negated, so that it rises.
747        let negate = self.descending != reversed;
748        let key = self.key.clone();
749        let is_temporal = matches!(self.dtype, DataType::Date | DataType::Datetime(_, _));
750        polars_ensure!(
751            size.parsed_int != is_temporal,
752            SQLInterface: "a RANGE window frame takes a number offset over a number ORDER BY key and an INTERVAL offset over a date or timestamp key"
753        );
754        // The key and offset as integers.
755        let (key, units) = match self.dtype {
756            dtype if dtype.is_integer() => (key.strict_cast(DataType::Int64), size.nanoseconds()),
757            DataType::Date if size.is_constant_duration(None) => {
758                let days = size
759                    .weeks()
760                    .checked_mul(7)
761                    .and_then(|days| days.checked_add(size.days()))
762                    .and_then(|days| days.checked_add(size.nanoseconds() / NANOSECONDS_IN_DAY));
763                let Some(days) = days else {
764                    polars_bail!(SQLInterface: "RANGE window frame offset is too large");
765                };
766                (key.to_physical().cast(DataType::Int64), days)
767            },
768            DataType::Datetime(unit, time_zone)
769                if size.is_constant_duration(time_zone.as_ref()) =>
770            {
771                (key.to_physical(), size.duration(*unit))
772            },
773            // Calendar months are only rolled over dates in their own order. Over timestamps
774            // a calendar interval back from one can be before that from an earlier one.
775            DataType::Date if !negate => {
776                let closed = if peers {
777                    ClosedWindow::Both
778                } else {
779                    ClosedWindow::Left
780                };
781                return Ok(FramePart::Window {
782                    by: key,
783                    size,
784                    closed,
785                    reversed,
786                });
787            },
788            DataType::Date | DataType::Datetime(_, _) => polars_bail!(
789                SQLInterface: "this INTERVAL offset is not supported in this RANGE window frame"
790            ),
791            dtype => polars_bail!(
792                SQLInterface: "a RANGE window frame with an offset needs an integer, date or timestamp ORDER BY key; found {}",
793                dtype
794            ),
795        };
796        // `!key` reverses the order of the keys and keeps their distances.
797        let by = if negate { key.not() } else { key };
798        // Over integers, the keys within `units` are those less than `units + 1` away.
799        let Some(size) = units.checked_add(1).map(Duration::new) else {
800            polars_bail!(SQLInterface: "RANGE window frame offset is too large");
801        };
802        let closed = if peers {
803            ClosedWindow::Right
804        } else {
805            ClosedWindow::None
806        };
807        Ok(FramePart::Window {
808            by,
809            size,
810            closed,
811            reversed,
812        })
813    }
814}
815
816impl FramePart {
817    /// The aggregate over this part of the frame.
818    fn apply(&self, aggregate: &Aggregate, keys: &[Expr]) -> PolarsResult<Expr> {
819        match self {
820            FramePart::ToLastPeer => Ok(at_last_peer(aggregate.running(false), keys)),
821            FramePart::FromFirstPeer => Ok(at_first_peer(aggregate.running(true), keys)),
822            #[cfg(feature = "rolling_window_by")]
823            FramePart::Window {
824                by,
825                size,
826                closed,
827                reversed,
828            } => {
829                use polars_defs::time::rolling::RollingOptionsDynamicWindow;
830                let options = |min_samples| RollingOptionsDynamicWindow {
831                    window_size: *size,
832                    min_samples,
833                    closed_window: *closed,
834                    fn_params: None,
835                };
836                let order = |e: Expr| if *reversed { e.reverse() } else { e };
837                let by = order(by.clone());
838                let expr = match aggregate {
839                    Aggregate::Sum(value) => order(value.clone()).rolling_sum_by(by, options(0)),
840                    Aggregate::Count(value) => {
841                        order(value.clone().is_not_null().cast(DataType::Int64))
842                            .rolling_sum_by(by, options(0))
843                    },
844                    Aggregate::CountRows => repeat(lit(1i64), len()).rolling_sum_by(by, options(0)),
845                    Aggregate::Max(value) => order(value.clone()).rolling_max_by(by, options(1)),
846                    Aggregate::Min(value) => order(value.clone()).rolling_min_by(by, options(1)),
847                    Aggregate::Avg(_) => unreachable!(),
848                };
849                Ok(order(expr))
850            },
851            #[cfg(not(feature = "rolling_window_by"))]
852            FramePart::Window { .. } => {
853                polars_bail!(SQLInterface: "RANGE window frames with an offset require the 'rolling_window_by' feature")
854            },
855        }
856    }
857}

859impl RowBound {
860    /// The index of the row in the sorted window.
861    fn row_index(self, keys: &[Expr]) -> Expr {
862        let (row_index, n) = window_row_index();
863        match self {
864            RowBound::First => lit(0i64),
865            RowBound::Last => n - lit(1i64),
866            RowBound::FirstPeer => first_peer_index(keys),
867            RowBound::LastPeer => last_peer_index(keys),
868            RowBound::Offset(offset) => row_index + lit(offset),
869        }
870    }
871}
872
873/// The index of the first row of a frame, at least 0.
874fn frame_start_row(start: RowBound, keys: &[Expr]) -> Expr {
875    let row = start.row_index(keys);
876    match start {
877        RowBound::Offset(offset) if offset < 0 => when(row.clone().lt(lit(0i64)))
878            .then(lit(0i64))
879            .otherwise(row),
880        _ => row,
881    }
882}
883
884/// The index of the last row of a frame, at most the last row of the partition.
885fn frame_end_row(end: RowBound, keys: &[Expr]) -> Expr {
886    let row = end.row_index(keys);
887    match end {
888        RowBound::Offset(offset) if offset > 0 => {
889            let (_, n) = window_row_index();
890            when(row.clone().gt_eq(n.clone()))
891                .then(n - lit(1i64))
892                .otherwise(row)
893        },
894        _ => row,
895    }
896}

898/// The order of the kinds of frame bounds, from UNBOUNDED PRECEDING to UNBOUNDED FOLLOWING.
899fn bound_order(bound: &WindowFrameBound) -> u8 {
900    use WindowFrameBound::{CurrentRow, Following, Preceding};
901    match bound {
902        Preceding(None) => 0,
903        Preceding(Some(_)) => 1,
904        CurrentRow => 2,
905        Following(Some(_)) => 3,
906        Following(None) => 4,
907    }
908}

910/// A frame offset as a number of rows.
911fn row_offset(n: usize) -> i64 {
912    i64::try_from(n).map_or(MAX_ROW_OFFSET, |n| n.min(MAX_ROW_OFFSET))
913}

915/// The index of the last peer of each row, in the sorted window.
916fn last_peer_index(keys: &[Expr]) -> Expr {
917    let (row_index, n) = window_row_index();
918    when(is_last_peer(keys, &row_index, &n))
919        .then(row_index)
920        .otherwise(n - lit(1i64))
921        .cum_min(true)
922}
923
924/// The index of the first peer of each row, in the sorted window.
925fn first_peer_index(keys: &[Expr]) -> Expr {
926    let (row_index, _) = window_row_index();
927    when(is_first_peer(keys, &row_index))
928        .then(row_index)
929        .otherwise(lit(0i64))
930        .cum_max(false)
931}
932
933/// `expr` at the last peer of each row, in the sorted window.
934fn at_last_peer(expr: Expr, keys: &[Expr]) -> Expr {
935    expr.gather(last_peer_index(keys), false)
936}
937
938/// `expr` at the first peer of each row, in the sorted window.
939fn at_first_peer(expr: Expr, keys: &[Expr]) -> Expr {
940    expr.gather(first_peer_index(keys), false)
