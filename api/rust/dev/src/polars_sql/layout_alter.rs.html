1//! `ALTER TABLE ... SET | RESET PARTITIONED BY` and `ALTER TABLE ... SET | RESET SORTED BY`, which
2//! sqlparser does not parse yet.
3//!
4//! Temporary: remove once sqlparser parses these statements.
5
6use polars_core::prelude::*;
7use sqlparser::ast::{Expr, ObjectName, OrderByExpr, Statement};
8use sqlparser::keywords::Keyword;
9use sqlparser::parser::{Parser, ParserError};
10use sqlparser::tokenizer::Token;
11
12use crate::sql_expr::{sql_parser, to_sql_interface_err};

14/// A parsed `ALTER TABLE [IF EXISTS] <name> SET | RESET PARTITIONED BY ...` or
15/// `ALTER TABLE [IF EXISTS] <name> SET | RESET SORTED BY ...`.
16#[derive(Clone, Debug)]
17pub(crate) struct LayoutAlter {
18    pub(crate) name: ObjectName,
19    pub(crate) if_exists: bool,
20    pub(crate) layout: Layout,
21}
22
23/// The new layout, as written; no fields for `RESET`.
24#[derive(Clone, Debug)]
25pub(crate) enum Layout {
26    Partitioning(Vec<Expr>),
27    SortOrder(Vec<OrderByExpr>),
28}
29
30/// The statements of `query`, or its layout change.
31pub(crate) fn parse_statements_or_alter(
32    query: &str,
33) -> PolarsResult<Result<Vec<Statement>, LayoutAlter>> {
34    let (mut parser, versions) = sql_parser(query)?;
35    if let Some(alter) = parse_layout_alter(&mut parser)? {
36        // a time travel clause has no table to read here
37        versions.refuse()?;
38        return Ok(Err(alter));
39    }
40    let mut stmts = parser.parse_statements().map_err(to_sql_interface_err)?;
41    versions.attach(&mut stmts)?;
42    Ok(Ok(stmts))
43}
44
45/// The single statement of `query`, or its layout change.
46pub(crate) fn parse_single_statement_or_alter(
47    query: &str,
48) -> PolarsResult<Result<Statement, LayoutAlter>> {
49    Ok(match parse_statements_or_alter(query)? {
50        Err(alter) => Err(alter),
51        Ok(statements) => {
52            let [stmt]: [Statement; 1] = statements.try_into().map_err(|_| {
53                polars_err!(SQLInterface: "one (and only one) statement can be parsed at a time")
54            })?;
55            Ok(stmt)
56        },
57    })
58}

60/// The layout change at the start of `parser`; if there is none, `parser` stays at the start.
61fn parse_layout_alter(parser: &mut Parser) -> PolarsResult<Option<LayoutAlter>> {
62    // Up to the keywords that make this a layout change. Anything else is parsed as usual.
63    let head = parser
64        .maybe_parse(|p| {
65            // as sqlparser, which skips empty statements
66            while p.consume_token(&Token::SemiColon) {}
67            p.expect_keywords(&[Keyword::ALTER, Keyword::TABLE])?;
68            let if_exists = p.parse_keywords(&[Keyword::IF, Keyword::EXISTS]);
69            let only = p.parse_keyword(Keyword::ONLY);
70            let name = p.parse_object_name(false)?;
71            let reset = match p.parse_one_of_keywords(&[Keyword::SET, Keyword::RESET]) {
72                Some(Keyword::SET) => false,
73                Some(_) => true,
74                None => return Err(ParserError::ParserError(String::new())),
75            };
76            let sort = if p.parse_keywords(&[Keyword::PARTITIONED, Keyword::BY]) {
77                false
78            } else if p.parse_keywords(&[Keyword::SORTED, Keyword::BY]) {
79                true
80            } else {
81                return Err(ParserError::ParserError(String::new()));
82            };
83            Ok((name, if_exists, only, reset, sort))
84        })
85        .map_err(to_sql_interface_err)?;
86    let Some((name, if_exists, only, reset, sort)) = head else {
87        return Ok(None);
88    };
89    polars_ensure!(!only, SQLInterface: "`ALTER TABLE ONLY` is not supported");
90    let layout = match (reset, sort) {
91        (true, false) => Layout::Partitioning(Vec::new()),
92        (true, true) => Layout::SortOrder(Vec::new()),
93        (false, false) => Layout::Partitioning(parse_fields(
94            parser,
95            "`SET PARTITIONED BY ()` needs a partition field; `RESET PARTITIONED BY` removes the partitioning",
96            Parser::parse_expr,
97        )?),
98        (false, true) => Layout::SortOrder(parse_fields(
99            parser,
100            "`SET SORTED BY ()` needs a sort field; `RESET SORTED BY` removes the sort order",
101            Parser::parse_order_by_expr,
102        )?),
103    };
104    polars_ensure!(
105        !parser.consume_token(&Token::Comma),
106        SQLInterface: "`ALTER TABLE` takes one operation per statement"
107    );
108    let ended = parser.consume_token(&Token::SemiColon);
109    while parser.consume_token(&Token::SemiColon) {}
110    let next = parser.peek_token();
111    if next.token != Token::EOF {
112        polars_ensure!(
113            !ended,
114            SQLInterface: "one (and only one) statement can be parsed at a time"
115        );
116        return parser
117            .expected("end of statement", next)
118            .map_err(to_sql_interface_err);
119    }
120    Ok(Some(LayoutAlter {
121        name,
122        if_exists,
123        layout,
124    }))
125}

127/// The parenthesized, non-empty fields at `parser`; `empty` is the message for `()`.
128fn parse_fields<'a, T>(
129    parser: &mut Parser<'a>,
130    empty: &str,
131    field: impl FnMut(&mut Parser<'a>) -> Result<T, ParserError>,
132) -> PolarsResult<Vec<T>> {
133    parser
134        .expect_token(&Token::LParen)
135        .map_err(to_sql_interface_err)?;
136    polars_ensure!(!parser.consume_token(&Token::RParen), SQLInterface: "{}", empty);
137    let fields = parser
138        .parse_comma_separated(field)
139        .map_err(to_sql_interface_err)?;
140    parser
141        .expect_token(&Token::RParen)
142        .map_err(to_sql_interface_err)?;
143    Ok(fields)
144}
