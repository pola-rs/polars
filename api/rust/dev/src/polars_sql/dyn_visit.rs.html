1//! Entry points for running sqlparser visitors through dynamic dispatch.
2//!
3//! sqlparser generates the full AST walk for every visitor type. Calling a visitor through
4//! these functions generates it once per `Break` type instead. Use them instead of
5//! `Visit::visit`, `VisitMut::visit` and sqlparser's `visit_expressions(_mut)`.
6
7use std::ops::ControlFlow;
8
9use sqlparser::ast::{
10    Expr, ObjectName, Query, Select, Statement, TableFactor, ValueWithSpan, Visit, VisitMut,
11    Visitor, VisitorMut,
12};
13
14struct DynVisitor<'a, B>(&'a mut dyn Visitor<Break = B>);
15
16struct DynVisitorMut<'a, B>(&'a mut dyn VisitorMut<Break = B>);
17
18macro_rules! forward {
19    ($($method:ident($ty:ty)),* $(,)?) => {
20        $(
21            fn $method(&mut self, node: $ty) -> ControlFlow<B> {
22                self.0.$method(node)
23            }
24        )*
25    };
26}
27
28impl<B> Visitor for DynVisitor<'_, B> {
29    type Break = B;
30
31    forward!(
32        pre_visit_query(&Query),
33        post_visit_query(&Query),
34        pre_visit_select(&Select),
35        post_visit_select(&Select),
36        pre_visit_relation(&ObjectName),
37        post_visit_relation(&ObjectName),
38        pre_visit_table_factor(&TableFactor),
39        post_visit_table_factor(&TableFactor),
40        pre_visit_expr(&Expr),
41        post_visit_expr(&Expr),
42        pre_visit_statement(&Statement),
43        post_visit_statement(&Statement),
44        pre_visit_value(&ValueWithSpan),
45        post_visit_value(&ValueWithSpan),
46    );
47}
48
49impl<B> VisitorMut for DynVisitorMut<'_, B> {
50    type Break = B;
51
52    forward!(
53        pre_visit_query(&mut Query),
54        post_visit_query(&mut Query),
55        pre_visit_select(&mut Select),
56        post_visit_select(&mut Select),
57        pre_visit_relation(&mut ObjectName),
58        post_visit_relation(&mut ObjectName),
59        pre_visit_table_factor(&mut TableFactor),
60        post_visit_table_factor(&mut TableFactor),
61        pre_visit_expr(&mut Expr),
62        post_visit_expr(&mut Expr),
63        pre_visit_statement(&mut Statement),
64        post_visit_statement(&mut Statement),
65        pre_visit_value(&mut ValueWithSpan),
66        post_visit_value(&mut ValueWithSpan),
67    );
68}

70pub(crate) fn visit<B>(
71    node: &(impl Visit + ?Sized),
72    visitor: &mut dyn Visitor<Break = B>,
73) -> ControlFlow<B> {
74    node.visit(&mut DynVisitor(visitor))
75}
76
77pub(crate) fn visit_mut<B>(
78    node: &mut (impl VisitMut + ?Sized),
79    visitor: &mut dyn VisitorMut<Break = B>,
80) -> ControlFlow<B> {
81    node.visit(&mut DynVisitorMut(visitor))
82}
83
84/// Calls `f` on every expression in `node`, parents before children.
85pub(crate) fn visit_expressions<B>(
86    node: &(impl Visit + ?Sized),
87    mut f: impl FnMut(&Expr) -> ControlFlow<B>,
88) -> ControlFlow<B> {
89    struct ExprFn<'a, B>(&'a mut dyn FnMut(&Expr) -> ControlFlow<B>);
90    impl<B> Visitor for ExprFn<'_, B> {
91        type Break = B;
92        fn pre_visit_expr(&mut self, expr: &Expr) -> ControlFlow<B> {
93            (self.0)(expr)
94        }
95    }
96    visit(node, &mut ExprFn(&mut f))
97}
98
99/// Calls `f` on every expression in `node`, children before parents.
100pub(crate) fn visit_expressions_mut<B>(
101    node: &mut (impl VisitMut + ?Sized),
102    mut f: impl FnMut(&mut Expr) -> ControlFlow<B>,
103) -> ControlFlow<B> {
104    struct ExprFn<'a, B>(&'a mut dyn FnMut(&mut Expr) -> ControlFlow<B>);
105    impl<B> VisitorMut for ExprFn<'_, B> {
106        type Break = B;
107        fn post_visit_expr(&mut self, expr: &mut Expr) -> ControlFlow<B> {
108            (self.0)(expr)
109        }
110    }
111    visit_mut(node, &mut ExprFn(&mut f))
112}
