1//! Lambda arguments of list functions: `list_transform(l, lambda x: x + 1)`.
2//!
3//! sqlparser has no `lambda x: body` syntax in the dialect we use, so the tokens are rewritten
4//! into a named argument (`"<marker>x" => body`) before parsing. After parsing, the uses of the
5//! lambda parameters in the body are replaced by placeholders that stand for the list element.
6use std::ops::ControlFlow;

23const LAMBDA_MARKER: &str = "\u{0}lambda\u{0}";
24const RESOLVED_LAMBDA_MARKER: &str = "\u{0}lambda resolved\u{0}";
25const ELEMENT_MARKER: &str = "\u{0}element\u{0}";
26
27/// Rewrite each `lambda p1, p2: body` into the named argument `"<marker>p1<marker>p2" => body`.
28/// Returns whether there was a lambda.
29pub(crate) fn rewrite_lambda_tokens(tokens: &mut Vec<TokenWithSpan>) -> bool {
30    let mut has_lambdas = false;
31    let next_token = |tokens: &[TokenWithSpan], from: usize| {
32        (from..tokens.len()).find(|&i| !matches!(tokens[i].token, Token::Whitespace(_)))
33    };
34    let mut i = 0;
35    while i < tokens.len() {
36        let is_lambda = matches!(
37            &tokens[i].token,
38            Token::Word(Word {
39                keyword: Keyword::LAMBDA,
40                quote_style: None,
41                ..
42            })
43        );
44        if !is_lambda {
45            i += 1;
46            continue;
47        }
48        let mut params = vec![];
49        let mut j = i + 1;
50        let colon = loop {
51            let Some(p) = next_token(tokens, j) else {
52                break None;
53            };
54            let Token::Word(word) = &tokens[p].token else {
55                break None;
56            };
57            params.push((p, word.value.clone()));
58            let Some(sep) = next_token(tokens, p + 1) else {
59                break None;
60            };
61            match &tokens[sep].token {
62                Token::Comma => j = sep + 1,
63                Token::Colon => break Some((sep, None)),
64                // `lambda x:x + 1` reads `:x` as a placeholder.
65                Token::Placeholder(s) if s.starts_with(':') && s.len() > 1 => {
66                    break Some((sep, Some(s[1..].to_string())));
67                },
68                _ => break None,
69            }
70        };
71        let Some((colon, body_start)) = colon else {
72            i += 1;
73            continue;
74        };
75        let mut name = String::new();
76        for (p, param) in &params {
77            name.push_str(LAMBDA_MARKER);
78            name.push_str(param);
79            tokens[*p].token = Token::Whitespace(Whitespace::Space);
80        }
81        for token in &mut tokens[i + 1..colon] {
82            if matches!(token.token, Token::Comma) {
83                token.token = Token::Whitespace(Whitespace::Space);
84            }
85        }
86        tokens[i].token = Token::Word(Word {
87            value: name,
88            quote_style: Some('"'),
89            keyword: Keyword::NoKeyword,
90        });
91        tokens[colon].token = Token::RArrow;
92        if let Some(word) = body_start {
93            let span = tokens[colon].span;
94            tokens.insert(
95                colon + 1,
96                TokenWithSpan::new(Token::make_word(&word, None), span),
97            );
98        }
99        has_lambdas = true;
100        i = colon + 1;
101    }
102    has_lambdas
103}

105/// The parameters of a lambda argument named `name`, or `None` if `name` is not a lambda.
106fn lambda_params(name: &str) -> Option<Vec<&str>> {
107    let params = name.strip_prefix(LAMBDA_MARKER)?;
108    Some(params.split(LAMBDA_MARKER).collect())
109}

111/// The lambda of a function argument, as the number of parameters and the body.
112pub(crate) fn lambda_arg(arg: &FunctionArg) -> Option<(usize, &SQLExpr)> {
113    let FunctionArg::Named {
114        name,
115        arg: FunctionArgExpr::Expr(body),
116        ..
117    } = arg
118    else {
119        return None;
120    };
121    let n_params = name
122        .value
123        .strip_prefix(RESOLVED_LAMBDA_MARKER)?
124        .parse()
125        .ok()?;
126    Some((n_params, body))
127}
128
129pub(crate) fn is_lambda_arg(arg: &FunctionArg) -> bool {
130    matches!(arg, FunctionArg::Named { name, .. }
131        if name.value.starts_with(RESOLVED_LAMBDA_MARKER) || name.value.starts_with(LAMBDA_MARKER))
132}

134/// Replace the uses of lambda parameters in the lambda bodies of `node` by element
135/// placeholders.
136pub(crate) fn resolve_lambda_params<T: VisitMut>(node: &mut T) {
137    let _ = dyn_visit::visit_mut(node, &mut LambdaResolver { scopes: vec![] });
138}
139
140struct LambdaResolver {
141    /// The parameters of the lambdas around the expression being visited, innermost last.
142    scopes: Vec<Vec<String>>,
143}
144
145impl LambdaResolver {
146    fn element_marker(&self, name: &str) -> Option<SQLExpr> {
147        let depth = self
148            .scopes
149            .iter()
150            .rposition(|params| params.iter().any(|p| p == name))?;
151        Some(SQLExpr::value(SQLValue::Placeholder(format!(
152            "{ELEMENT_MARKER}{}",
153            depth + 1
154        ))))
155    }
156}
157
158impl VisitorMut for LambdaResolver {
159    type Break = ();
160
161    fn pre_visit_expr(&mut self, expr: &mut SQLExpr) -> ControlFlow<()> {
162        match expr {
163            SQLExpr::Identifier(ident) => {
164                if let Some(marker) = self.element_marker(&ident.value) {
165                    *expr = marker;
166                }
167            },
168            SQLExpr::CompoundIdentifier(idents) => {
169                if let Some(marker) = self.element_marker(&idents[0].value) {
170                    *expr = field_access(marker, &idents[1..]);
171                }
172            },
173            SQLExpr::Function(func) => {
174                let FunctionArguments::List(list) = &mut func.args else {
175                    return ControlFlow::Continue(());
176                };
177                for arg in &mut list.args {
178                    let FunctionArg::Named { name, arg, .. } = arg else {
179                        continue;
180                    };
181                    let Some(params) = lambda_params(&name.value) else {
182                        continue;
183                    };
184                    self.scopes
185                        .push(params.into_iter().map(str::to_string).collect());
186                    let _ = dyn_visit::visit_mut(arg, self);
187                    let n_params = self.scopes.pop().unwrap().len();
188                    *name = Ident::with_quote('"', format!("{RESOLVED_LAMBDA_MARKER}{n_params}"));
189                }
190            },
191            _ => {},
192        }
193        ControlFlow::Continue(())
194    }
195}

197/// The fields `fields` of `root`, as in `root.a.b`.
198pub(crate) fn field_access(root: SQLExpr, fields: &[Ident]) -> SQLExpr {
199    SQLExpr::CompoundFieldAccess {
200        root: Box::new(root),
201        access_chain: fields
202            .iter()
203            .map(|field| AccessExpr::Dot(SQLExpr::Identifier(field.clone())))
204            .collect(),
205    }
206}

208/// Moves the elements of the lambdas deeper than `above` in `expr` `by` lambdas deeper, for
209/// `expr` placed in the bodies of `by` more lambdas.
210pub(crate) fn shift_lambda_elements(expr: &mut SQLExpr, above: usize, by: usize) {
211    if by == 0 {
212        return;
213    }
214    let _ = dyn_visit::visit_expressions_mut(expr, |expr| {
215        if let SQLExpr::Value(ValueWithSpan {
216            value: SQLValue::Placeholder(p),
217            ..
218        }) = expr
219            && let Some(depth) = element_depth(p)
220            && depth > above
221        {
222            *p = format!("{ELEMENT_MARKER}{}", depth + by);
223        }
224        ControlFlow::<()>::Continue(())
225    });
226}

228/// The lambda nesting depth of the element that `placeholder` stands for.
229pub(crate) fn element_depth(placeholder: &str) -> Option<usize> {
230    placeholder.strip_prefix(ELEMENT_MARKER)?.parse().ok()
231}

233/// The column that stands for the element of the lambda at `depth`, used in the body of the
234/// lambda at `lambda_depth`.
235pub(crate) fn lambda_element(depth: usize, lambda_depth: usize) -> PolarsResult<Expr> {
236    polars_ensure!(
237        depth == lambda_depth,
238        SQLInterface: "a nested lambda cannot use the parameter of an outer lambda"
239    );
240    Ok(col(lambda_element_name(depth)))
241}
242
243pub(crate) fn lambda_element_name(depth: usize) -> PlSmallStr {
244    format_pl_smallstr!("__POLARS_LAMBDA_ELEMENT_{depth}")
245}

247pub(crate) fn check_lambda_body_expr(expr: &SQLExpr) -> PolarsResult<()> {
248    polars_ensure!(
249        !is_subquery_expr(expr),
250        SQLInterface: "subqueries are not supported in a lambda"
251    );
252    Ok(())
