42/// The column `name`. Unlike the DSL `col`, a name such as `*` or `^a$` is read as the column
43/// of that name, not as a selector: a SQL identifier only names a column.
44pub(crate) fn col(name: impl Into<polars_utils::pl_str::PlSmallStr>) -> polars_plan::dsl::Expr {
45    polars_plan::dsl::Expr::Column(name.into())
46}
47
48/// The field `name` of the struct `expr`, read by name like [`col`].
49pub(crate) fn struct_field(
50    expr: polars_plan::dsl::Expr,
51    name: impl Into<polars_utils::pl_str::PlSmallStr>,
52) -> polars_plan::dsl::Expr {
53    use polars_plan::dsl::FunctionExpr;
54    use polars_plan::dsl::function_expr::StructFunction;
55    expr.map_unary(FunctionExpr::StructExpr(StructFunction::FieldByName(
56        name.into(),
57    )))
58}
