1//! `POSITIONAL JOIN`, which sqlparser does not parse. After a table without an alias it parses
2//! as a plain `JOIN` after the alias `positional`, which [`attach_positional_aliases`] turns into
3//! a [`positional_join`]. After another alias it does not parse; then [`extract_positional_joins`]
4//! takes the word `POSITIONAL` out before a `JOIN`, and [`PositionalJoins::attach`] marks the
5//! joins it was taken from, found by position.

18/// The positions of the `POSITIONAL` words taken out of a query. Each must be attached to a
19/// join, or refused.
20#[must_use]
21pub(crate) struct PositionalJoins(Vec<Location>);
22
23/// The operator of a positional join. The parser never gives a `CROSS JOIN` a `NATURAL`
24/// constraint.
25pub(crate) fn positional_join() -> JoinOperator {
26    JoinOperator::CrossJoin(JoinConstraint::Natural)
27}
28
29pub(crate) fn is_positional_join(op: &JoinOperator) -> bool {
30    matches!(op, JoinOperator::CrossJoin(JoinConstraint::Natural))
31}
32
33/// Whether `node` holds a positional join.
34pub(crate) fn has_positional_join(node: &(impl Visit + ?Sized)) -> bool {
35    struct Find;
36
37    impl Visitor for Find {
38        type Break = ();
39
40        fn pre_visit_select(&mut self, select: &Select) -> ControlFlow<()> {
41            find_in(&select.from)
42        }
43
44        fn pre_visit_table_factor(&mut self, factor: &TableFactor) -> ControlFlow<()> {
45            match factor {
46                TableFactor::NestedJoin {
47                    table_with_joins, ..
48                } => find_in(std::slice::from_ref(table_with_joins)),
49                _ => ControlFlow::Continue(()),
50            }
51        }
52    }
53
54    fn find_in(items: &[TableWithJoins]) -> ControlFlow<()> {
55        match items
56            .iter()
57            .flat_map(|item| &item.joins)
58            .any(|join| is_positional_join(&join.join_operator))
59        {
60            true => ControlFlow::Break(()),
61            false => ControlFlow::Continue(()),
62        }
63    }
64
65    dyn_visit::visit(node, &mut Find).is_break()
66}

68impl PositionalJoins {
69    pub(crate) fn is_empty(&self) -> bool {
70        self.0.is_empty()
71    }
72
73    /// Mark each plain `JOIN` without a condition that a `POSITIONAL` word was taken from.
74    pub(crate) fn attach<T: VisitMut>(mut self, stmts: &mut [T]) -> PolarsResult<()> {
75        for stmt in stmts {
76            let _ = dyn_visit::visit_mut(stmt, &mut self);
77        }
78        polars_ensure!(self.0.is_empty(), SQLInterface: "{}", JOIN_CONDITION);
79        Ok(())
80    }
81
82    fn mark(&mut self, item: &mut TableWithJoins) {
83        let mut before = item.relation.span().end;
84        for join in &mut item.joins {
85            let after = join.relation.span().start;
86            if join.join_operator == JoinOperator::Join(JoinConstraint::None)
87                && let Some(i) = self.0.iter().position(|at| before < *at && *at < after)
88            {
89                self.0.swap_remove(i);
90                join.join_operator = positional_join();
91            }
92            before = join.span().end;
93        }
94    }
95}

97impl VisitorMut for PositionalJoins {
98    type Break = ();
99
100    fn pre_visit_select(&mut self, select: &mut Select) -> ControlFlow<()> {
101        for item in &mut select.from {
102            self.mark(item);
103        }
104        ControlFlow::Continue(())
105    }
106
107    fn pre_visit_table_factor(&mut self, factor: &mut TableFactor) -> ControlFlow<()> {
108        if let TableFactor::NestedJoin {
109            table_with_joins, ..
110        } = factor
111        {
112            self.mark(table_with_joins);
113        }
114        ControlFlow::Continue(())
115    }
116}

118const JOIN_CONDITION: &str = "POSITIONAL JOIN does not take a join condition";

120/// Mark each plain `JOIN` after an alias `positional`, written without `AS` or quotes, as a
121/// positional join, and drop the alias.
122pub(crate) fn attach_positional_aliases<T: VisitMut>(stmts: &mut [T]) -> PolarsResult<()> {
123    struct Attach;
124
125    impl VisitorMut for Attach {
126        type Break = PolarsError;
127
128        fn pre_visit_select(&mut self, select: &mut Select) -> ControlFlow<PolarsError> {
129            for item in &mut select.from {
130                if let Err(err) = mark_aliased(item) {
131                    return ControlFlow::Break(err);
132                }
133            }
134            ControlFlow::Continue(())
135        }
136
137        fn pre_visit_table_factor(&mut self, factor: &mut TableFactor) -> ControlFlow<PolarsError> {
138            if let TableFactor::NestedJoin {
139                table_with_joins, ..
140            } = factor
141                && let Err(err) = mark_aliased(table_with_joins)
142            {
143                return ControlFlow::Break(err);
144            }
145            ControlFlow::Continue(())
146        }
147    }
148
149    for stmt in stmts {
150        if let ControlFlow::Break(err) = dyn_visit::visit_mut(stmt, &mut Attach) {
151            return Err(err);
152        }
153    }
154    Ok(())
155}
156
157fn mark_aliased(item: &mut TableWithJoins) -> PolarsResult<()> {
158    for i in 0..item.joins.len() {
159        let has_condition = match &item.joins[i].join_operator {
160            JoinOperator::Join(JoinConstraint::None) => false,
161            JoinOperator::Join(JoinConstraint::On(_) | JoinConstraint::Using(_)) => true,
162            _ => continue,
163        };
164        let left = match i {
165            0 => &mut item.relation,
166            _ if ends_with_relation(&item.joins[i - 1].join_operator) => {
167                &mut item.joins[i - 1].relation
168            },
169            _ => continue,
170        };
171        if take_positional_alias(left) {
172            polars_ensure!(!has_condition, SQLInterface: "{}", JOIN_CONDITION);
173            item.joins[i].join_operator = positional_join();
174        }
175    }
176    Ok(())
177}

179/// Whether nothing is written after the relation of a join with operator `op`.
180fn ends_with_relation(op: &JoinOperator) -> bool {
181    use JoinOperator::*;
182    let (Join(constraint)
183    | Inner(constraint)
184    | Left(constraint)
185    | LeftOuter(constraint)
186    | Right(constraint)
187    | RightOuter(constraint)
188    | FullOuter(constraint)
189    | CrossJoin(constraint)
190    | Semi(constraint)
191    | LeftSemi(constraint)
192    | RightSemi(constraint)
193    | Anti(constraint)
194    | LeftAnti(constraint)
195    | RightAnti(constraint)
196    | StraightJoin(constraint)) = op
197    else {
198        return matches!(op, CrossApply | OuterApply);
199    };
200    matches!(constraint, JoinConstraint::None | JoinConstraint::Natural)
201}

203/// Take an alias `positional`, written without `AS` or quotes, off `factor`.
204fn take_positional_alias(factor: &mut TableFactor) -> bool {
205    let (TableFactor::Table { alias, .. }
206    | TableFactor::Derived { alias, .. }
207    | TableFactor::NestedJoin { alias, .. }
208    | TableFactor::TableFunction { alias, .. }
209    | TableFactor::UNNEST { alias, .. }
210    | TableFactor::Pivot { alias, .. }
211    | TableFactor::Unpivot { alias, .. }) = factor
212    else {
213        return false;
214    };
215    let is_positional = alias.as_ref().is_some_and(|alias| {
216        !alias.explicit
217            && alias.columns.is_empty()
218            && alias.name.quote_style.is_none()
219            && alias.name.value.eq_ignore_ascii_case("POSITIONAL")
220    });
221    if is_positional {
222        *alias = None;
223    }
224    is_positional
225}

227/// Take each unquoted `POSITIONAL` that comes right before `JOIN` out of `tokens`, unless it
228/// stands where a name goes.
229pub(crate) fn extract_positional_joins(tokens: &mut Vec<TokenWithSpan>) -> PositionalJoins {
230    let mut found = Vec::new();
231    let mut i = 0;
232    while i < tokens.len() {
233        let is_positional = matches!(
234            &tokens[i].token,
235            Token::Word(w) if w.quote_style.is_none() && w.value.eq_ignore_ascii_case("POSITIONAL")
236        );
237        if is_positional
238            && tokens[i + 1..]
239                .iter()
240                .find(|t| significant(t))
241                .is_some_and(|t| is_keyword(t, Keyword::JOIN))
242            && !name_follows(&tokens[..i])
243        {
244            found.push(tokens.remove(i).span.start);
245        } else {
246            i += 1;
247        }
248    }
249    PositionalJoins(found)
250}
251
252fn significant(token: &TokenWithSpan) -> bool {
253    !matches!(token.token, Token::Whitespace(_))
254}
255
256fn is_keyword(token: &TokenWithSpan, keyword: Keyword) -> bool {
257    matches!(&token.token, Token::Word(w) if w.keyword == keyword && w.quote_style.is_none())
258}

260/// Whether a name goes right after `tokens` in a FROM: after `.`, `,`, `(`, `AS`, `FROM`,
261/// `JOIN`, `LATERAL` or `USING`. In a run of these keywords, like `FROM using`, every other one
262/// is a name.
263fn name_follows(tokens: &[TokenWithSpan]) -> bool {
264    let mut keywords = 0;
265    for token in tokens.iter().rev().filter(|t| significant(t)) {
266        match &token.token {
267            Token::Word(w)
268                if w.quote_style.is_none()
269                    && matches!(
270                        w.keyword,
271                        Keyword::AS
272                            | Keyword::FROM
273                            | Keyword::JOIN
274                            | Keyword::LATERAL
275                            | Keyword::USING
276                    ) =>
277            {
278                keywords += 1
279            },
280            Token::Period | Token::Comma | Token::LParen => return keywords % 2 == 0,
281            _ => break,
282        }
283    }
284    keywords % 2 == 1
